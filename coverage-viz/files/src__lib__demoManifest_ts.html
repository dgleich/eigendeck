1// Demo network manifest. A demo declares which hosts it needs and WHY, in a JSON
2// block, so we can (a) GATE its internet — no manifest, no network — and (b)
3// SURFACE it in the security panel ("this demo uses api.x for live quotes").
4//
5// <script type="application/eigendeck-manifest+json">
6// { "network": [ { "host": "api.stockdata.example", "purpose": "Live stock quotes" } ] }
7// </script>
8//
9// Declared ≠ granted: the manifest scopes what the demo CAN reach (its declared
10// hosts become the injected connect-src allowlist); a host it didn't declare is
11// blocked. The global master + per-deck switches still sit on top.
12 
13export interface DemoNetEntry { host: string; purpose: string }
14export interface DemoManifest { network: DemoNetEntry[] }
15 
16const MANIFEST_RE =
17 /<script[^>]*type=["']application\/eigendeck-manifest\+json["'][^>]*>([\s\S]*?)<\/script>/i;
18 
19/** Parse a demo's manifest from its HTML. Tolerant: null on absent/invalid JSON. */
20export function parseDemoManifest(html: string | undefined | null): DemoManifest | null {
21 if (!html) return null;
22 const m = html.match(MANIFEST_RE);
23 if (!m) return null;
24 try {
25 const obj = JSON.parse(m[1]) as { network?: unknown };
26 const net = Array.isArray(obj?.network) ? obj.network : [];
27 const network: DemoNetEntry[] = [];
28 for (const e of net as Array<{ host?: unknown; purpose?: unknown }>) {
29 if (e && typeof e.host === 'string' && e.host.trim()) {
30 network.push({ host: e.host.trim(), purpose: typeof e.purpose === 'string' ? e.purpose : '' });
31 }
32 }
33 return { network };
34 } catch { return null; }
35}

37/** Cap on distinct hosts a single demo can declare — bounds the size of the
38 * injected CSP string so a pathological manifest can't bloat it. */
39export const MAX_MANIFEST_HOSTS = 32;
40 
41/** The distinct, VALID hosts a demo declared (capped), or [] if none / no manifest.
42 * Invalid hosts are dropped so the injected CSP and the panel never grant them. */
43export function manifestHosts(html: string | undefined | null): string[] {
44 const m = parseDemoManifest(html);
45 if (!m) return [];
46 return [...new Set(m.network.map((n) => n.host).filter(isValidManifestHost))].slice(0, MAX_MANIFEST_HOSTS);
47}

49// The manifest is ATTACKER-CONTROLLED (it lives in the demo's HTML), and its hosts
50// are interpolated into the injected CSP `<meta content="...">`. So a host must be a
51// strict hostname (optionally `*.sub`, optional :port) or a full ws(s)/http(s)
52// origin — nothing containing a space, `;`, `"`, `'`, `,`, `<`, `>`, or a bare `*`,
53// which would otherwise inject extra CSP directives or break out of the attribute
54// and defeat the gate entirely. Anything else is rejected.
55const HOSTNAME_RE = /^(\*\.)?[a-z0-9]([a-z0-9-]*[a-z0-9])?(\.[a-z0-9]([a-z0-9-]*[a-z0-9])?)*(:\d{1,5})?$/i;
56const ORIGIN_RE = /^(?:https?|wss?):\/\/(\*\.)?[a-z0-9]([a-z0-9-]*[a-z0-9])?(\.[a-z0-9]([a-z0-9-]*[a-z0-9])?)*(:\d{1,5})?$/i;
57 
58/** True iff `h` is a safe declared host (strict hostname or ws(s)/http(s) origin).
59 * Used to filter what the manifest can grant AND what the panel shows as granted. */
60export function isValidManifestHost(h: string): boolean {
61 // Length-guard BEFORE the regex: the host is attacker-controlled, and a bounded
62 // length keeps the regex linear (no pathological backtracking on a huge input).
63 // 253 = DNS max name; +scheme/port headroom.
64 return typeof h === 'string' && h.length > 0 && h.length <= 260
65 && (ORIGIN_RE.test(h) || HOSTNAME_RE.test(h));
66}

68/** Map declared hosts to a CSP source list. A bare host → https + secure-ws for it;
69 * a full origin (has a scheme) is used verbatim. Invalid/unsafe hosts are dropped
70 * so they can never inject into the CSP. */
71export function hostsToCspSources(hosts: string[]): string {
72 const out: string[] = [];
73 for (const h of hosts) {
74 if (!isValidManifestHost(h)) continue; // reject injection / wildcards / junk
75 if (/^[a-z][a-z0-9+.-]*:\/\//i.test(h)) out.push(h); // full origin, verbatim
76 else out.push(`https://${h}`, `wss://${h}`); // bare host → https + wss
77 }
78 return out.join(' ');
79}
