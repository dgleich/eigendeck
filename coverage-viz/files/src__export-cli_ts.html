27/**
28 * #85 — notebook / video / PDF-image elements can't be rendered headlessly, so
29 * the static export embeds their cached preview PNG (asset_cache). The app export
30 * passes this callback (fileOps.getElementPreviewDataUrl); the CLI used to omit
31 * it, so those elements showed "NB"/"PDF" placeholders. Read-only cache lookup
32 * (a deck opened in the editor has previews cached): pick the LARGEST cached
33 * render for the element's (source_id, variant) and return a data: URL. Returns
34 * null on a miss → exportCore falls back to the placeholder, same as before.
35 */
36// Signature matches the exportCore contract `(element, slide)`; `slide` is
37// unused here (the cache key is element-only) but accepted for honesty.
38// eslint-disable-next-line @typescript-eslint/no-explicit-any
39async function getElementPreview(el: any, _slide?: any): Promise<string | null> {
40 try {
41 const key = previewLookupKey(el);
42 if (!key) return null;
43 const variants = await invoke<CacheVariant[]>('db_list_asset_cache_variants', { sourceId: key.sourceId });
44 const best = pickLargestVariant(variants, key.variant);
45 if (!best) return null;
46 const buf = await invoke<ArrayBuffer>('db_get_asset_cache_bytes', {
47 sourceId: key.sourceId, variant: key.variant, width: best.width, height: best.height,
48 });
49 const bytes = new Uint8Array(buf);
50 if (!bytes.length) return null;
51 return pngBytesToDataUrl(bytes);
52 } catch (e) {
53 console.warn('getElementPreview (cli) failed:', e);
54 return null;
55 }
56}

58/**
59 * Build a renderMath function that consults the SQLite math_cache for each
60 * expression. Falls back to the singleton renderer for cache misses.
61 *
62 * Signature matches what exportCore expects:
63 * renderMath(html, bundleId) → Promise<string>
64 *
65 * Walks $..$ and $$..$$ markers in the input html, replaces each with the
66 * SVG from the cache (or singleton-rendered SVG), returns the patched html.
67 */
68function makeCachingRenderMath(preamble: string) {
69 return async (html: string, bundleId?: string): Promise<string> => {
70 if (!/\$\$[\s\S]+?\$\$|\$[^\$\n]+?\$/.test(html)) return html;
71 // Use bundleId 'ptsans' as default if not provided (singleton fallback).
72 const bundle = bundleId || 'ptsans';
73 
74 // Walk the HTML similar to renderMathInHtml — skip tags, find $..$/$$..$$.
75 const parts: string[] = [];
76 let i = 0;
77 while (i < html.length) {
78 if (html[i] === '<') {
79 const end = html.indexOf('>', i);
80 if (end !== -1) { parts.push(html.slice(i, end + 1)); i = end + 1; continue; }
81 }
82 // Display $$..$$
83 if (html[i] === '$' && html[i + 1] === '$') {
84 const end = html.indexOf('$$', i + 2);
85 if (end !== -1) {
86 const tex = html.slice(i + 2, end);
87 const svg = await renderOne(tex, bundle, true, preamble);
88 parts.push(`<div style="text-align:center;">${svg}</div>`);
89 i = end + 2;
90 continue;
91 }
92 }
93 // Inline $..$
94 if (html[i] === '$') {
95 const end = html.indexOf('$', i + 1);
96 if (end !== -1 && !html.slice(i + 1, end).includes('\n')) {
97 const tex = html.slice(i + 1, end);
98 const svg = await renderOne(tex, bundle, false, preamble);
99 parts.push(svg);
100 i = end + 1;
101 continue;
102 }
103 }
104 parts.push(html[i]);
105 i++;
106 }
107 return parts.join('');
108 };
109}

111/**
112 * Resolve one math expression: cache lookup, then singleton fallback.
113 * Returns the SVG markup (with inline display/valign style for inline math).
114 */
115async function renderOne(tex: string, bundleId: string, display: boolean, preamble: string): Promise<string> {
116 const key = mathCacheKey(tex, bundleId, display, preamble);
117 try {
118 const row = await invoke<CachedMathRow | null>('db_get_math_svg', { key });
119 if (row) {
120 // Match the inline-math vertical-align tweak the iframe-pool render adds.
121 if (display) return row.svg;
122 const valign = row.valign || '-0.025ex';
123 return row.svg.replace(/^<svg/, `<svg overflow="visible" style="display:inline;vertical-align:${valign};overflow:visible"`);
124 }
125 } catch { /* fall through to singleton */ }
126 // Cache miss: use the singleton renderer (will be in body font only).
127 const wrapped = display ? `$$${tex}$$` : `$${tex}$`;
128 try {
129 const out = await renderMathInHtml(wrapped);
130 return out;
131 } catch {
132 return wrapped; // give up — leave source visible
133 }
134}
