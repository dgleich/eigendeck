17async function stripOne(input: string): Promise<StripHistoryFileReport> {
18 const start = performance.now();
19 try {
20 await invoke('db_open', { path: input });
21 // keepAll: true is misnamed in the Rust API — it means "delete ALL history"
22 // (vs the default exponential thinning). That's what we want here.
23 const json = await invoke<string>('db_compact', { keepAll: true });
24 const r = JSON.parse(json) as CompactResult;
25 // db_close runs PRAGMA wal_checkpoint(TRUNCATE) and drops the
26 // connection — without this the .eigendeck-{wal,shm} sidecars
27 // stay alongside the stripped file. (The next iteration's
28 // db_open replaces the connection in DB.lock() and rusqlite's
29 // Drop closes it, but Drop alone doesn't checkpoint.)
30 await invoke('db_close');
31 return {
32 input, ok: true,
33 sizeBeforeBytes: r.beforeBytes,
34 sizeAfterBytes: r.afterBytes,
35 savedBytes: r.savedBytes,
36 elapsedMs: performance.now() - start,
37 };
38 } catch (e) {
39 // Best-effort close so a half-failed strip still checkpoints if
40 // db_open succeeded. Ignore close errors (the original one is
41 // the useful signal).
42 try { await invoke('db_close'); } catch { /* swallow */ }
43 return {
44 input, ok: false, error: e instanceof Error ? e.message : String(e),
45 sizeBeforeBytes: 0, sizeAfterBytes: 0, savedBytes: 0,
46 elapsedMs: performance.now() - start,
47 };
48 }
49}
