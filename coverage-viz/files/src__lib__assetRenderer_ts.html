2// size, persisting the result to the SQLite asset_cache. Cache-first:
3// existing entries return without re-render.
4//

134async function withPdfRenderSlot<T>(fn: () => Promise<T>): Promise<T> {
135 // FIFO ordering: a fresh caller must also queue if anyone is already
136 // waiting, otherwise microtask ordering can let it jump ahead of a
137 // resumed waiter (between the previous holder's pdfActive-- and the
138 // waiter's pdfActive++, a new caller sees pdfActive=0 and proceeds).
139 // pdfium's thread_safe Mutex serializes inside Rust either way, but
140 // the correctness of the in-slot putAssetCache + post-queue re-probe
141 // depends on slot ownership being unambiguous.
142 if (pdfActive >= PDF_RENDER_CONCURRENCY || pdfWaiters.length > 0) {
143 await new Promise<void>((resolve) => pdfWaiters.push(resolve));
144 }
145 pdfActive++;
146 try {
147 return await fn();
148 } finally {
149 pdfActive--;
150 const next = pdfWaiters.shift();
151 if (next) next();
152 }
153}

539 'g',
540 );
541 modified = modified.replace(re, `$1${dataUri}$2`);
542 }
543 
544 return {
545 bytes: new TextEncoder().encode(modified),
546 inlined: replacements.length,
547 failed,
548 httpUnsupported,
549 };
550}

665/**
666 * Pre-process SVG bytes before handing them to <img src=blob:>. WebKit
667 * (and to a lesser extent other browsers) loads SVG-as-image in "secure
668 * static mode" — embedded `<image xlink:href="data:...">` subresources
669 * are often silently dropped, while plain `href` on the same element is
670 * honored. Adobe Illustrator exports overwhelmingly use the legacy
671 * `xlink:href`, so without this shim every Illustrator SVG with an
672 * embedded placed-photo renders without the photo.
673 *
674 * Rewrites `xlink:href=` -> `href=` only inside <image> tags (avoiding
675 * <use xlink:href=...> and other SVG idioms that may genuinely depend on
676 * the xlink namespace). Idempotent and fast: regex on the text, no DOM
677 * parse. Returns the input unchanged when no <image xlink:href> match.
678 */
679function normalizeSvgForImg(bytes: Uint8Array): Uint8Array {
680 // Only inspect the leading window — large SVGs almost never have
681 // <image> tags in the first few KB if not at all; a full UTF-8 decode
682 // for a 5 MB icon is wasteful otherwise. If the head shows xlink:href
683 // inside an <image>, fall through to the full rewrite.
684 const head = new TextDecoder('utf-8', { fatal: false }).decode(bytes.subarray(0, Math.min(bytes.length, 16384)));
685 if (!/<image\b[^>]*\bxlink:href\b/i.test(head) && bytes.length <= 16384) {
686 return bytes;
687 }
688 const text = new TextDecoder('utf-8', { fatal: false }).decode(bytes);
689 const rewritten = text.replace(/(<image\b[^>]*?\s)xlink:href(=)/gi, '$1href$2');
690 if (rewritten === text) return bytes;
691 return new TextEncoder().encode(rewritten);
692}

694/**
695 * SVG fast-path threshold (bytes). SVG sources under this size are served as
696 * raw blob URLs so the browser renders them natively at any CSS size
697 * (vector-perfect anti-aliasing, no pixelation in scaled-down sidebar
698 * thumbs). Above this we rasterize to the asset_cache because re-parsing
699 * a million-point path on every paint is expensive — accepting some sidebar
700 * pixelation in exchange for stable per-paint cost.
701 *
702 * 200 KB is empirically the right cutoff for our corpus: every icon / spec
703 * test / simple wikimedia diagram lands below, the only fixture above is
704 * the Inkscape About splash (~400 KB) which legitimately benefits from
705 * caching.
706 */
707const SVG_NATIVE_THRESHOLD_BYTES = 200_000;
708 
709/**
710 * React hook: lazy cache-or-render an asset at the requested tier, returning
711 * a blob URL the consumer can drop into <img src>. URL is revoked when the
712 * component unmounts or the inputs change.
713 *
714 * - SVG (<= SVG_NATIVE_THRESHOLD_BYTES): native browser SVG render, no cache.
715 * Crisp at any size, ideal for icons and small diagrams.
716 * - SVG (> threshold): rasterized to cache to avoid per-paint parse cost on
717 * complex/large files.
718 * - raster: always cached at requested tier (cheap thumbs vs. re-decoding
719 * a multi-MB JPEG every paint).
720 * - pdf: returns undefined permanently until the pdfium path lands.
721 *
722 * Returns undefined while the first render is in flight.
723 */
724export function useRenderedAsset(
725 assetId: string | undefined,
726 kind: AssetKind | undefined,
727 maxWidth: number,
728 maxHeight: number,
729 variant: string = '_',
730): string | undefined {
731 const [url, setUrl] = useState<string | undefined>(undefined);
732 // Bumped by invalidateRenderedAsset() so this hook refetches after the
733 // underlying asset bytes change (e.g. embedding external SVG refs).
734 const [refreshKey, setRefreshKey] = useState(0);
735 
736 useEffect(() => {
737 if (!assetId) return;
738 const handler = (e: Event) => {
739 const detail = (e as CustomEvent).detail as { assetId?: string } | undefined;
740 if (detail?.assetId === assetId) setRefreshKey((k) => k + 1);
741 };
742 window.addEventListener('eigendeck:asset-changed', handler);
743 return () => window.removeEventListener('eigendeck:asset-changed', handler);
744 }, [assetId]);
745 
746 useEffect(() => {
747 if (!assetId || !kind) { setUrl(undefined); return; }
748 let cancelled = false;
749 let current: string | undefined;
750 
751 const fetchBytes = async (): Promise<Uint8Array> =>
752 new Uint8Array(await invoke<ArrayBuffer>('db_get_asset_by_id', { assetId }));
753 
754 if (kind === 'svg') {
755 // Fetch source bytes once; decide native vs cache by size.
756 fetchBytes()
757 .then(async (data) => {
758 if (cancelled) return;
759 const bytes = normalizeSvgForImg(new Uint8Array(data));
760 if (bytes.length <= SVG_NATIVE_THRESHOLD_BYTES) {
761 // Fast path: hand raw SVG to <img>; browser scales it perfectly.
762 const blob = new Blob([bytes as BlobPart], { type: 'image/svg+xml' });
763 current = URL.createObjectURL(blob);
764 setUrl(current);
765 } else {
766 // Slow path: rasterize once into asset_cache, reuse forever.
767 // preFetchedBytes avoids the second db_get_asset_by_id round-trip.
768 try {
769 const u = await renderAsset({ assetId, kind, variant, maxWidth, maxHeight, preFetchedBytes: bytes });
770 if (cancelled) { URL.revokeObjectURL(u); return; }
771 current = u;
772 setUrl(u);
773 } catch {
774 if (!cancelled) setUrl(undefined);
775 }
776 }
777 })
778 .catch(() => { if (!cancelled) setUrl(undefined); });
779 } else {
780 renderAsset({ assetId, kind, variant, maxWidth, maxHeight })
781 .then((u) => {
782 if (cancelled) { URL.revokeObjectURL(u); return; }
783 current = u;
784 setUrl(u);
785 })
786 .catch(() => { if (!cancelled) setUrl(undefined); });
787 }
788 
789 return () => {
790 cancelled = true;
791 if (current) URL.revokeObjectURL(current);
792 };
793 }, [assetId, kind, variant, maxWidth, maxHeight, refreshKey]);
794 
795 return url;
796}
