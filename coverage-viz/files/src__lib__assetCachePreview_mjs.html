8/** PNG bytes → base64 data: URL. Inlined into exported HTML, where a blob: URL
9 * wouldn't survive in the written file. Chunked to avoid blowing the argument
10 * limit of String.fromCharCode on large images. */
11export function pngBytesToDataUrl(bytes) {
12 let binary = '';
13 for (let k = 0; k < bytes.length; k += 8192) {
14 binary += String.fromCharCode(...bytes.slice(k, k + 8192));
15 }
16 return `data:image/png;base64,${btoa(binary)}`;
17}

19/**
20 * Map an element to its asset_cache lookup key, or null if the element type has
21 * no cached preview. Mirrors fileOps.getElementPreviewDataUrl / previewKey:
22 * - image kind:'pdf' → the pdfium raster, keyed by assetId + snapshotVariant
23 * (default '_');
24 * - notebook / video → the proactively-cached 'preview', keyed by the
25 * element's sync identity (syncId, falling back to id).
26 */
27export function previewLookupKey(el) {
28 if (el.type === 'image' && el.kind === 'pdf') {
29 return { sourceId: el.assetId, variant: el.snapshotVariant ?? '_' };
30 }
31 if (el.type === 'notebook' || el.type === 'video') {
32 return { sourceId: el.syncId ?? el.id, variant: 'preview' };
33 }
34 return null;
35}

37/**
38 * From the asset_cache variant rows for a source, pick the largest render
39 * (most pixels) whose variant matches. Returns null if none match. Numeric
40 * compare on width*height — not a string sort — so e.g. a 1024² tier beats 512².
41 */
42export function pickLargestVariant(variants, variant) {
43 const matches = (variants || []).filter((v) => v.variant === variant);
44 if (!matches.length) return null;
45 matches.sort((a, b) => b.width * b.height - a.width * a.height);
46 return matches[0];
47}
