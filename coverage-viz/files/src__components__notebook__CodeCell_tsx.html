1// Single code cell — source area + outputs + Run button.
2//
3// `liveOutputs`: when present, replaces persisted outputs (the result
4// of a fresh run this session). Reverts to `cell.outputs` when null —
5// e.g. after the file watcher reloads.
6//
7// `liveExecutionCount`: replaces the persisted `[N]` prompt;
8// '*' means currently running.
9//
10// `source`: the EFFECTIVE source to display + run. NotebookContent
11// passes the cellEdits overlay value when one exists, else the cell's
12// own source. (CodeCell never reads cell.source directly for
13// display — always this prop — so edits show consistently.)
14//
15// `editable`: when true, the source area becomes a CodeMirror editor
16// (the notebook is being interacted with). When false, a static
17// highlighted <pre>.
18//
19// `language`: kernel language for highlighting + the editor grammar.
20//
21// Run button is disabled when `onRun` is undefined (display-only mode
22// when no kernel layer is wired) or when the cell is already running.

71export function CodeCell({
72 cell, source, liveOutputs, liveExecutionCount, running, onRun,
73 language, highlight = true, dark = false,
74 editable = false, showLineNumbers = false,
75 fontSize = 32, onEdit, onCommit, edited, added, onRevert,
76 highlightedHtml,
77}: CodeCellProps) {
78 const effectiveSource = source ?? cell.source;
79 // Strip ONE trailing newline for DISPLAY (static + editor) so a cell that
80 // ends in "\n" doesn't render a spurious blank last line — which the
81 // wrapped view hid but the line-number / no-wrap view exposes.
82 const displaySource = effectiveSource.replace(/\n$/, '');
83 const outputs = liveOutputs ?? cell.outputs;
84 const promptCount = running
85 ? '*'
86 : liveExecutionCount === '*' ? '*'
87 : liveExecutionCount != null ? String(liveExecutionCount)
88 : cell.executionCount != null ? String(cell.executionCount)
89 : ' ';
90 
91 // Highlighted HTML — null until the lazy load resolves. Recomputed
92 // when the effective source changes (so edits re-highlight when the
93 // cell goes back to static).
94 const [highlighted, setHighlighted] = useState<string | null>(null);
95 useEffect(() => {
96 // A pre-resolved highlight (export path) means we don't run the async
97 // highlighter — the static markup is supplied directly.
98 if (highlightedHtml != null) return;
99 if (!highlight || editable) { setHighlighted(null); return; }
100 let cancelled = false;
101 highlightCode(displaySource, language ?? null).then((html) => {
102 if (!cancelled) setHighlighted(html);
103 });
104 return () => { cancelled = true; };
105 }, [displaySource, language, highlight, editable, highlightedHtml]);
106 
107 // Effective highlighted markup: the pre-resolved prop wins (export),
108 // else the async-resolved state (live).
109 const effectiveHighlighted = highlightedHtml ?? highlighted;
110 
111 // Visual distinction: appended (live-authored) > edited (overlay) >
112 // pristine. Drives the left accent + tag.
113 const markClass = added ? ' nb-cell-added' : edited ? ' nb-cell-edited' : '';
114 
115 return (
116 <div className={`nb-cell nb-cell-code${markClass}`}>
117 <div className={`nb-cell-prompt${running ? ' is-running' : ''}`}>[{promptCount}]</div>
118 <div className="nb-cell-body">
119 <div className="nb-cell-source-row">
120 {editable ? (
121 <div className="nb-cell-source nb-cell-source-editing">
122 <NotebookCellEditor
123 value={displaySource}
124 language={language ?? null}
125 fontSize={fontSize}
126 highlight={highlight}
127 dark={dark}
128 showLineNumbers={showLineNumbers}
129 onChange={(next) => onEdit?.(next)}
130 onRun={() => onRun?.()}
131 onBlur={onCommit}
132 />
133 </div>
134 ) : (
135 <pre className={`nb-cell-source${showLineNumbers ? ' nb-has-linenos' : ''}`}>
136 {showLineNumbers && (
137 <span className="nb-linenos" aria-hidden="true">
138 {Array.from(
139 { length: displaySource.split('\n').length },
140 (_, i) => i + 1,
141 ).join('\n')}
142 </span>
143 )}
144 {displaySource === ''
145 /* Phantom space so an EMPTY cell still has a full line box —
146 keeps it the same height as a one-line code cell instead of
147 collapsing to padding. */
148 ? <code>{' '}</code>
149 : highlight && effectiveHighlighted != null
150 ? <code className="hljs" dangerouslySetInnerHTML={{ __html: effectiveHighlighted }} />
151 : <code>{displaySource}</code>}
152 </pre>
153 )}
154 <div className="nb-cell-actions">
155 {/* No text tag: the colored left accent + the action button are
156 indicator enough — ⟲ revert = edited, ✕ = added. */}
157 {added && onRevert && (
158 <button className="nb-cell-delete"
159 onClick={(e) => { e.stopPropagation(); onRevert(); }}
160 title="Delete this added cell">
161 ✕
162 </button>
163 )}
164 {!added && edited && onRevert && (
165 <button className="nb-cell-revert"
166 onClick={(e) => { e.stopPropagation(); onRevert(); }}
167 title="Revert this cell to the notebook's saved source">
168 ⟲
169 </button>
170 )}
171 {onRun && (
172 <button className="nb-cell-run" disabled={running}
173 onClick={(e) => { e.stopPropagation(); onRun(); }}
174 title="Run this cell (Shift-Enter)">
175 {running ? '…' : '▶'}
176 </button>
177 )}
178 </div>
179 </div>
180 {outputs.length > 0 && (
181 <div className="nb-cell-outputs">
182 {outputs.map((o, i) => <CellOutput key={i} output={o} />)}
183 </div>
184 )}
185 </div>
186 </div>
187 );
188}
