1/**
2 * macOS-style unsaved-changes confirmation dialog (Keynote-modeled).
3 *
4 * Rendered as an in-app modal (not a native sheet — Tauri 2's dialog plugin
5 * doesn't expose 3-button confirms cleanly, and an in-app modal lets us
6 * guarantee Esc=cancel, Enter=save, and consistent layout cross-platform).
7 *
8 * Two cases:
9 * - hasFile: "Do you want to save the changes you made to {title}?"
10 * [Don't Save] [Cancel] [Save]
11 * - !hasFile: "Do you want to keep this new document {title}?"
12 * [Delete] [Cancel] [Save…]
13 *
14 * Save is the primary/default action (Enter activates it). Esc cancels.
15 * The destructive button (Don't Save / Delete) is leftmost per macOS HIG.
16 */
17import { useEffect, useRef } from 'react';
18 
19interface Props {
20 /** Document title shown in the message body */
21 title: string;
22 /** True if the document has been saved to disk before */
23 hasFile: boolean;
24 onSave: () => void;
25 onDiscard: () => void;
26 onCancel: () => void;
27}

29export function UnsavedChangesDialog({ title, hasFile, onSave, onDiscard, onCancel }: Props) {
30 const saveBtnRef = useRef<HTMLButtonElement>(null);
31 
32 // Focus Save (the default action) on open. Auto-focus also enables
33 // Enter-to-activate via the browser's button focus behavior.
34 useEffect(() => {
35 saveBtnRef.current?.focus();
36 }, []);
37 
38 // Esc cancels, regardless of focus.
39 useEffect(() => {
40 const onKey = (e: KeyboardEvent) => {
41 if (e.key === 'Escape') { e.preventDefault(); onCancel(); }
42 if (e.key === 'Enter' && document.activeElement === saveBtnRef.current) {
43 // Default behavior of Enter on a focused button — let it through;
44 // we keep this handler around for Esc only.
45 }
46 };
47 window.addEventListener('keydown', onKey, true);
48 return () => window.removeEventListener('keydown', onKey, true);
49 }, [onCancel]);
50 
51 const message = hasFile
52 ? `Do you want to save the changes you made to "${title}"? Your changes will be lost if you don't save them.`
53 : `Do you want to keep this new document "${title}"? You can choose to save your changes, or delete this document immediately. You can't undo this action.`;
54 
55 const destructiveLabel = hasFile ? "Don't Save" : 'Delete';
56 const saveLabel = hasFile ? 'Save' : 'Save…';
57 
58 return (
59 <div
60 role="dialog" aria-modal="true" aria-labelledby="unsaved-msg"
61 style={{
62 position: 'fixed', top: 0, left: 0, right: 0, bottom: 0,
63 background: 'rgba(0,0,0,0.18)',
64 display: 'flex', alignItems: 'flex-start', justifyContent: 'center',
65 zIndex: 100000, paddingTop: 60,
66 }}
67 onClick={(e) => { if (e.target === e.currentTarget) onCancel(); }}
68 >
69 <div style={{
70 background: '#fff',
71 borderRadius: 8,
72 boxShadow: '0 12px 48px rgba(0,0,0,0.25)',
73 padding: '24px 28px 18px',
74 width: 480,
75 font: '13px/1.4 -apple-system, BlinkMacSystemFont, "Segoe UI", system-ui, sans-serif',
76 }}>
77 <div id="unsaved-msg" style={{ marginBottom: 18, color: '#1f2937' }}>
78 {message}
79 </div>
80 <div style={{
81 display: 'flex', justifyContent: 'flex-end', alignItems: 'center', gap: 8,
82 }}>
83 {/* Destructive — leftmost (macOS convention). Visually subdued. */}
84 <button onClick={onDiscard} style={dialogButton}>{destructiveLabel}</button>
85 {/* Spacer pushes Cancel/Save to the right */}
86 <div style={{ flex: 1 }} />
87 <button onClick={onCancel} style={dialogButton}>Cancel</button>
88 <button ref={saveBtnRef} onClick={onSave} style={primaryButton}>{saveLabel}</button>
89 </div>
90 </div>
91 </div>
92 );
93}
