1/**
2 * MathJax integration for Eigendeck.
3 *
4 * Loads one MathJax bundle at a time (window.MathJax is a singleton). The
5 * bundle id is chosen per render call based on the surrounding font. When a
6 * different bundle is requested, the loader swaps: unloads the current one
7 * and reloads the new bundle's script. Loading takes ~1-2s on cold cache.
8 *
9 * NOTE on per-preset math fonts: ideally math in a title element would use
10 * the title's math font. With the singleton constraint, achieving that
11 * requires either (a) bundle-switching per-element with caching of rendered
12 * SVGs, or (b) iframe isolation for each font. Neither is implemented yet —
13 * for now, all math on a slide uses the bundle currently loaded (effectively
14 * the body font). Per-preset rendering tracked in a future issue.
15 */

52function unloadCurrentBundle(): void {

67/**
68 * Load a MathJax bundle. If `bundleId` differs from the currently loaded
69 * bundle, the previous one is unloaded and the new one is loaded.
70 *
71 * @param bundleId Font package id (see src/lib/fonts.ts). If omitted, uses
72 * the currently-loaded bundle, or 'ptsans' if none is loaded.
73 */
74export function loadMathJax(bundleId?: string): Promise<any> {
75 const desired = bundleId ?? activeBundleId ?? 'ptsans';
76 
77 if (mathjaxReady && activeBundleId === desired) {
78 return Promise.resolve((window as any).MathJax);
79 }
80 if (mathjaxPromise && activeBundleId === desired) return mathjaxPromise;
81 
82 // Either no bundle loaded yet, or a different bundle is loaded — (re)load.
83 if (activeBundleId && activeBundleId !== desired) {
84 unloadCurrentBundle();
85 }
86 
87 installWorkerStub();
88 activeBundleId = desired;
89 
90 const pkg = resolveFontPackage(desired);
91 const bundleUrl = `/mathjax/${pkg.mathjaxBundle}`;

93 mathjaxPromise = new Promise((resolve, reject) => {
94 (window as any).MathJax = {
95 tex: {
96 inlineMath: [['$', '$']],
97 displayMath: [['$$', '$$']],
98 },
99 svg: { fontCache: 'none' },
100 startup: {
101 typeset: false,
102 ready: () => {
103 const MJ = (window as any).MathJax;
104 MJ.startup.defaultReady();
105 MJ.startup.promise.then(() => {
106 mathjaxReady = true;
107 resolve(MJ);
108 });
109 },
110 },
111 };
112 
113 const script = document.createElement('script');
114 script.src = bundleUrl;
115 script.async = true;
116 script.setAttribute('data-mathjax-bundle', desired);
117 script.onerror = () => reject(new Error(`Failed to load MathJax bundle: ${bundleUrl}`));
118 document.head.appendChild(script);
119 });
120 
121 return mathjaxPromise;
122}

143export async function applyMathPreamble(preamble: string, bundleId?: string): Promise<void> {
144 if (!preamble || preamble === appliedPreamble) return;
145 const MJ = await loadMathJax(bundleId);
146 try {
147 MJ.texReset();
148 // Render preamble to register \newcommand, \def, etc.
149 await Promise.race([
150 MJ.tex2svgPromise(`{${preamble}}`, { display: false }),
151 new Promise((_, reject) => setTimeout(() => reject(new Error('preamble timeout')), 3000)),
152 ]);
153 appliedPreamble = preamble;
154 } catch (e) {
155 console.warn('MathJax preamble error:', e);
156 }
157}

159export async function renderMathInHtml(html: string, bundleId?: string): Promise<string> {
160 if (!containsMath(html)) return html;
161 
162 const MJ = await loadMathJax(bundleId);
163 const parts: string[] = [];
164 let i = 0;
165 
166 while (i < html.length) {
167 // Skip HTML tags
168 if (html[i] === '<') {
169 const tagEnd = html.indexOf('>', i);
170 if (tagEnd !== -1) {
171 parts.push(html.slice(i, tagEnd + 1));
172 i = tagEnd + 1;
173 continue;
174 }
175 }
176 
177 // Display math $$...$$
178 if (html[i] === '$' && html[i + 1] === '$') {
179 const end = html.indexOf('$$', i + 2);
180 if (end !== -1) {
181 const tex = texFromHtml(html.slice(i + 2, end));
182 try {
183 MJ.texReset();
184 const container = await Promise.race([
185 MJ.tex2svgPromise(`{${tex}}`, { display: true }),
186 new Promise((_, reject) => setTimeout(() => reject(new Error('timeout')), 2000)),
187 ]);
188 const container2 = container as HTMLElement;
189 // Check for MathJax errors (undefined commands, syntax errors)
190 const errNode = container2.querySelector('mjx-container [data-mjx-error]') || container2.querySelector('[data-mml-node="merror"]');
191 if (errNode) {
192 const errMsg = errNode.getAttribute('data-mjx-error') || errNode.textContent || 'LaTeX error';
193 console.warn(`MathJax error in display math: ${errMsg}\nTeX: ${tex}`);
194 parts.push(`<div style="text-align:center;color:#dc2626;background:#fef2f2;border:1px solid #fca5a5;border-radius:4px;padding:8px;font-size:16px;font-family:monospace;">MathJax: ${errMsg}</div>`);
195 } else {
196 const svg = container2.querySelector('svg');
197 if (svg) {
198 const svgHeight = svg.getAttribute('height') || '';
199 if (svgHeight) displayMathHeights.set(tex, svgHeight);
200 parts.push(`<div style="text-align:center;">${svg.outerHTML}</div>`);
201 } else {
202 parts.push(`$$${tex}$$`);
203 }
204 }
205 } catch (e) {
206 console.warn('MathJax render failed for display math:', tex, e);
207 parts.push(`<div style="color:#dc2626;font-size:14px;font-family:monospace;">$$${tex}$$ (render failed)</div>`);
208 }
209 i = end + 2;
210 continue;
211 }
212 }
213 
214 // Inline math $...$
215 if (html[i] === '$') {
216 const end = html.indexOf('$', i + 1);
217 if (end !== -1 && !html.slice(i + 1, end).includes('\n')) {
218 const tex = texFromHtml(html.slice(i + 1, end));
219 try {
220 MJ.texReset();
221 const container = await Promise.race([
222 MJ.tex2svgPromise(`{${tex}}`, { display: false }),
223 new Promise((_, reject) => setTimeout(() => reject(new Error('timeout')), 2000)),
224 ]);
225 const container2 = container as HTMLElement;
226 const errNode = container2.querySelector('mjx-container [data-mjx-error]') || container2.querySelector('[data-mml-node="merror"]');
227 if (errNode) {
228 const errMsg = errNode.getAttribute('data-mjx-error') || errNode.textContent || 'LaTeX error';
229 console.warn(`MathJax error in inline math: ${errMsg}\nTeX: ${tex}`);
230 parts.push(`<span style="color:#dc2626;background:#fef2f2;border:1px solid #fca5a5;border-radius:2px;padding:1px 4px;font-size:0.8em;font-family:monospace;" title="${tex}">MathJax: ${errMsg}</span>`);
231 } else {
232 const svg = container2.querySelector('svg');
233 if (svg) {
234 const vAlign = svg.style.verticalAlign || '-0.025ex';
235 svg.style.display = 'inline';
236 svg.style.verticalAlign = vAlign;
237 parts.push(svg.outerHTML);
238 } else {
239 parts.push(`$${tex}$`);
240 }
241 }
242 } catch (e) {
243 console.warn('MathJax render failed for inline math:', tex, e);
244 parts.push(`<span style="color:#dc2626;font-family:monospace;">$${tex}$ (error)</span>`);
245 }
246 i = end + 1;
247 continue;
248 }
249 }
250 
251 parts.push(html[i]);
252 i++;
253 }
254 
255 return parts.join('');
256}

259 if (preamble) await applyMathPreamble(preamble, bundleId);
