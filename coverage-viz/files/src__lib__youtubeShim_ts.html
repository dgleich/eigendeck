1// Frontend side of the YouTube loopback shim (docs/youtube-embed-shim.md).
2//
3// In the PACKAGED app the frontend origin is the custom scheme `tauri://localhost`,
4// which YouTube's embed player rejects. There we route YouTube embeds through the
5// Rust loopback shim (youtube_shim.rs): the iframe points at
6// `http://127.0.0.1:<port>/yt/<token>/<id>?<flags>`, whose http origin YouTube
7// accepts. In DEV (`http://localhost:1420`) and on Windows the origin already
8// works, so we embed YouTube directly. Vimeo/PeerTube always embed directly (they
9// play fine from `tauri://`).
10import { useEffect, useState } from 'react';

16// The shim is only needed on the custom scheme (packaged macOS/Linux). Dev is
17// `http:`; Windows serves `http(s)://tauri.localhost` — both accepted by YouTube.
18function shimApplies(): boolean {
19 try {
20 return location.protocol === 'tauri:';
21 } catch {
22 return false;
23 }
24}

26// Cached shim base: `null` = not fetched yet, `''` = fetched-but-unavailable/NA.
27let cachedBase: string | null = null;
28let inflight: Promise<string> | null = null;
29 
30function fetchShimBase(): Promise<string> {
31 if (cachedBase !== null) return Promise.resolve(cachedBase);
32 if (!shimApplies()) {
33 cachedBase = '';
34 return Promise.resolve('');
35 }
36 if (!inflight) {
37 inflight = invoke<string>('youtube_shim_base')
38 .then((b) => (cachedBase = b || ''))
39 .catch(() => (cachedBase = ''));
40 }
41 return inflight;
42}

44/** Hook: the loopback shim base (`http://127.0.0.1:<port>/yt/<token>`), or `''`
45 * when the shim doesn't apply / isn't available. Resolves once, then cached. */
46export function useYoutubeShimBase(): string {
47 const [base, setBase] = useState<string>(cachedBase ?? '');
48 useEffect(() => {
49 let alive = true;
50 fetchShimBase().then((b) => {
51 if (alive) setBase(b);
52 });
53 return () => {
54 alive = false;
55 };
56 }, []);
57 return base;
58}

60/** Build the LIVE (editor/present) iframe src for a video embed. YouTube is routed
61 * through the loopback shim when `shimBase` is set; everything else (Vimeo,
62 * PeerTube, and YouTube in dev/Windows) uses the direct `buildEmbedSrc` URL. */
63export function liveEmbedSrc(el: EmbedEl, shimBase: string): string | null {
64 if (shimBase && el && el.url) {
65 const parsed = detectVideoProvider(el.url);
66 if (parsed && parsed.provider === 'youtube') {
67 const flags = new URLSearchParams();
68 // Raw element flags; the Rust shim derives showControls = controls || !autoplay,
69 // mirroring buildEmbedSrc's YouTube branch.
70 if (el.autoplay) flags.set('autoplay', '1');
71 if (el.muted) flags.set('mute', '1');
72 if (el.loop) flags.set('loop', '1');
73 if (el.controls) flags.set('controls', '1');
74 if (el.captions) flags.set('captions', '1');
75 const q = flags.toString();
76 return `${shimBase}/${parsed.id}${q ? `?${q}` : ''}`;
77 }
78 }
79 return buildEmbedSrc(el);
80}
