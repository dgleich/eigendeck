1// Per-project file-watcher registry.
2//
3// Keyed by `project_id` (the stable UUID in _meta), NOT by project path
4// — projectPath can change underneath us (macOS in-place rename keeps
5// the file handle alive but mutates the filename; Save As writes to a
6// new path). Keying on the stable id means the registry survives
7// renames cleanly; only Save As (which generates a fresh project_id)
8// gets a fresh registry.
9//
10// Within a registry: one `fs.watch` per external_path even if multiple
11// assets reference it (e.g. two ImageElements both pointing at the
12// same source SVG). The Set fan-out means a single disk event causes
13// db_store_asset for each subscribed asset_id; hash dedup at the
14// storage layer makes redundant writes silent.
15//
16// Lifecycle:
17// - getWatcherRegistry(projectId) -> get-or-create
18// - addRef / removeRef from useAssetFileWatcher mount/cleanup
19// - closeWatcherRegistry(projectId) on project switch or window close

28// --- Rust file-watch glue (replaces the JS fs-plugin `watch`) ----------------
29// watch_path returns an id; the Rust side emits `fs-watch-event` {id,path} per
30// change. One process-wide listener fans each event to the registered callback by
31// id. There is no delayMs anymore — Rust emits raw notify events and the per-path
32// COALESCE_MS window in handleChange collapses a save burst (as before).
33const watchCbs = new Map<number, () => void>();
34let watchListenerStarted = false;
35async function ensureWatchListener(): Promise<void> {
36 if (watchListenerStarted) return;
37 watchListenerStarted = true;
38 try {
39 await listen<{ id: number; path: string }>('fs-watch-event', (e) => {
40 watchCbs.get(e.payload.id)?.();
41 });
42 } catch { watchListenerStarted = false; } // non-Tauri context
43}
44/** Start watching `absPath`; returns an unwatch fn. */
45async function startWatch(absPath: string, cb: () => void): Promise<() => void> {
46 await ensureWatchListener();
47 const id = await invoke<number>('watch_path', { path: absPath });
48 watchCbs.set(id, cb);
49 return () => { watchCbs.delete(id); void invoke('unwatch_path', { id }).catch(() => {}); };
50}

80/**
81 * The asset-security read gate for external files (docs/ASSETS-SECURITY.md).
82 * Every disk read of a watched/linked asset goes through this. It:
83 * - reads the CURRENTLY-OPEN deck's token from the store; no token → 'gated';
84 * - requires the deck to be TRUSTED (a received/untrusted deck performs ZERO disk
85 * reads → 'gated', the embedded snapshot stays);
86 * - resolves via resolveAndGate so realpath + the asset-type allowlist apply even
87 * to trusted decks (a symlink to a non-asset / wrong-type file → 'gated'), and a
88 * genuinely missing/erroring file → 'unreadable'.
89 * Never throws. Dynamic imports avoid a static cycle with the store.
90 */
91export async function gatedExternalRead(absPath: string): Promise<GatedRead> {
92 try {
93 const { getDeckToken } = await import('../store/presentation');→ presentation.ts
94 const token = getDeckToken();
95 if (!token) return { status: 'gated' };
96 const { isTrusted, isPathApproved } = await import('./trustStore');→ trustStore.ts
97 if (!(await isTrusted(token))) return { status: 'gated' };
98 const { resolveAndGate } = await import('./assetGate');→ assetGate.ts
99 const gate = await resolveAndGate(absPath);
100 if (!(gate.ok && gate.bytes && gate.canonicalPath)) {
101 return { status: gate.reason === 'unreadable' ? 'unreadable' : 'gated' };
102 }
103 // Per-path approval: a trusted deck reads only paths approved in the ledger
104 // (keyed by the RESOLVED target). Unapproved → gated (snapshot stays) until the
105 // user approves it in the Security panel. See docs/ASSETS-SECURITY.md.
106 if (!(await isPathApproved(token, gate.canonicalPath))) return { status: 'gated' };
107 return { status: 'ok', bytes: gate.bytes };
108 } catch {
109 return { status: 'unreadable' };
110 }

113interface SubscribedAsset {
114 /** Real stable asset_id from the assets table — what db_store_asset
115 * needs to version the right row (NOT a path-derived placeholder). */
116 assetId: string;
117 /** Original path label stored on the asset row (e.g. "images/foo.svg") —
118 * preserved so re-stores keep the same path metadata. */
119 path: string;
120 /** Element ids currently subscribed to this asset. Multiple elements
121 * may be bound to the same asset_id (e.g. user drags the same image
122 * onto 3 slides). Tracking per-element lets removeRef be ref-counted:
123 * the asset entry is dropped only when the LAST element using it
124 * unsubscribes. Without this, slide 2's removeRef would wipe the
125 * whole entry and silently kill the watcher for slides 1+3. */
126 subscribers: Set<string>;
127}

129interface WatchEntry {
130 unwatch: () => void;
131 /** Subscribed assets at this external_path, keyed by asset_id. Each
132 * asset entry tracks its own element subscribers (see SubscribedAsset). */
133 assets: Map<string, SubscribedAsset>;
134 /** MIME type to use when re-storing the asset (cached per first subscriber) */
135 mimeType: string;
136 /** Timestamp of the last handleChange for this path. macOS emits 3–7
137 * raw fs events for one save (write + truncate + close + rename-from
138 * + rename-to + ...); plugin-fs's `delayMs` doesn't actually coalesce
139 * them. We dedup here: skip handleChange if a previous one ran less
140 * than COALESCE_MS ago. Real edits that come in quick succession are
141 * picked up on the next event after the window expires. */
142 lastHandledAt: number;
143}

163 /**
164 * Subscribe an asset to changes of `externalRelPath` (relative to the
165 * project dir). The (real, stable) `assetId` and the asset row's
166 * `path` are stored so the disk-event handler can call db_store_asset
167 * with the correct identity — passing a placeholder asset_id would
168 * create orphan rows.
169 *
170 * First subscriber for a path lazily registers fs.watch; subsequent
171 * subscribers just join the Map.
172 */
173 async addRef(externalRelPath: string, assetId: string, elementId: string, path: string, mimeType: string): Promise<void> {
174 const absPath = resolvePosixPath(this.projectDir, externalRelPath);
175 const existing = this.watchers.get(absPath);
176 if (existing) {
177 const assetEntry = existing.assets.get(assetId);
178 if (assetEntry) {
179 // Asset already in registry — just add this element as another
180 // subscriber. Set semantics keep idempotent re-mounts safe.
181 assetEntry.subscribers.add(elementId);
182 wlog(`addRef join asset=${assetId.slice(0, 8)} element=${elementId.slice(0, 8)} path="${path}" (now ${assetEntry.subscribers.size} elements on this asset, ${existing.assets.size} assets on this path)`);
183 } else {
184 // First subscriber for this asset on this already-watched path.
185 existing.assets.set(assetId, { assetId, path, subscribers: new Set([elementId]) });
186 wlog(`addRef join+ asset=${assetId.slice(0, 8)} element=${elementId.slice(0, 8)} path="${path}" (first element on this asset, ${existing.assets.size} assets on this path)`);
187 }
188 return;
189 }
190 // First reference for the path — start watching.
191 wlog(`addRef new asset=${assetId.slice(0, 8)} element=${elementId.slice(0, 8)} path="${path}" abs="${absPath}" — registering fs.watch...`);
192 const placeholder: WatchEntry = {
193 unwatch: () => {},
194 assets: new Map([[assetId, { assetId, path, subscribers: new Set([elementId]) }]]),
195 mimeType,
196 lastHandledAt: 0,
197 };
198 this.watchers.set(absPath, placeholder);
199 try {
200 placeholder.unwatch = await startWatch(absPath, () => {
201 wlog(`fs.watch FIRED for "${absPath}"`);
202 void this.handleChange(absPath, externalRelPath);
203 });
204 wlog(`fs.watch REGISTERED for "${absPath}"`);
205 } catch (e) {
206 console.warn(`[watcher] watch "${absPath}" FAILED:`, e);
207 }
208 }

210 removeRef(externalRelPath: string, assetId: string, elementId: string): void {
211 const absPath = resolvePosixPath(this.projectDir, externalRelPath);
212 const entry = this.watchers.get(absPath);
213 if (!entry) return;
214 const assetEntry = entry.assets.get(assetId);
215 if (!assetEntry) return;
216 assetEntry.subscribers.delete(elementId);
217 if (assetEntry.subscribers.size > 0) {
218 wlog(`removeRef asset=${assetId.slice(0, 8)} element=${elementId.slice(0, 8)} (${assetEntry.subscribers.size} elements left on this asset)`);
219 return;
220 }
221 // Last element for this asset on this path — drop the asset entry.
222 entry.assets.delete(assetId);
223 if (entry.assets.size === 0) {
224 try { entry.unwatch(); } catch { /* ignore */ }
225 this.watchers.delete(absPath);
226 wlog(`removeRef last asset=${assetId.slice(0, 8)} element=${elementId.slice(0, 8)} — unwatched "${absPath}"`);
227 } else {
228 wlog(`removeRef drop asset=${assetId.slice(0, 8)} element=${elementId.slice(0, 8)} (${entry.assets.size} assets left on this path)`);

232 /**
233 * Disk event for `absPath`: re-read once, fan out to every subscribed
234 * asset. db_store_asset hashes and dedup-skips if bytes haven't
235 * actually changed.
236 */
237 private async handleChange(absPath: string, externalRelPath: string): Promise<void> {
238 const entry = this.watchers.get(absPath);
239 if (!entry || entry.assets.size === 0) { wlog(`handleChange no subscribers for "${absPath}"`); return; }
240 // Coalesce: macOS emits a burst of events for one save (write +
241 // truncate + close + rename). Skip if we just handled this path.
242 const sinceLast = Date.now() - entry.lastHandledAt;
243 if (sinceLast < COALESCE_MS) { wlog(`handleChange coalesced (${sinceLast}ms since last) for "${absPath}"`); return; }
244 entry.lastHandledAt = Date.now();
245 // Asset-security gate: untrusted deck or a blocked target → skip silently
246 // (snapshot stays). Only a genuinely unreadable/missing source flags #74.
247 const read = await gatedExternalRead(absPath);
248 if (read.status === 'gated') { wlog(`handleChange gated (untrusted/blocked) for "${absPath}"`); return; }
249 if (read.status === 'unreadable') {
250 wlog(`handleChange unreadable "${absPath}" → mark ${entry.assets.size} missing (#74)`);
251 for (const { assetId, path } of entry.assets.values()) markAssetMissing(assetId, path);
252 return;
253 }
254 const bytes = read.bytes;
255 try {
256 const st = await pathStat(absPath).catch(() => null);
257 const mtime = statMtimeIso(st);
258 wlog(`handleChange read ${bytes.length} bytes from "${absPath}" mtime=${mtime} → fanout to ${entry.assets.size} asset(s)`);
259 for (const { assetId, path } of entry.assets.values()) {
260 markAssetFound(assetId); // a successful read means the source is back
261 try {
262 const writtenId = await invoke<string>('db_store_asset', {
263 path,
264 data: Array.from(bytes),
265 mimeType: entry.mimeType,
266 externalPath: externalRelPath,
267 externalMtime: mtime,
268 assetId,
269 autoReload: null,
270 });
271 wlog(` db_store_asset ok → assetId=${writtenId.slice(0, 8)} (expected ${assetId.slice(0, 8)})`);
272 await invalidateRenderedAsset(assetId);
273 wlog(` invalidateRenderedAsset(${assetId.slice(0, 8)}) fired`);
274 } catch (e) {
275 console.warn(`[watcher] db_store_asset for ${assetId.slice(0, 8)} FAILED:`, e);
276 }
277 }
278 } catch (e) {
279 console.warn(`[watcher] store "${absPath}" FAILED:`, e);
280 for (const { assetId, path } of entry.assets.values()) {
281 markAssetMissing(assetId, path);
282 }
283 }
284 }

303export function getWatcherRegistry(projectId: string, projectDir: string): WatcherRegistry {
304 let r = registries.get(projectId);
305 if (!r) {
306 r = new WatcherRegistry(projectId, projectDir);
307 registries.set(projectId, r);
308 } else if (r.projectDir !== projectDir) {
309 // Project file was renamed / moved -> update the dir used for
310 // resolution. Existing watchers were bound to absolute paths
311 // resolved against the OLD dir; tear them down so subsequent
312 // mounts re-register against the new dir.
313 r.closeAll();
314 r.projectDir = projectDir;
315 }
316 return r;

330/** POSIX-style: join an absolute dir with a possibly-../-prefixed relative
331 * path. Mac is POSIX; not used on Windows. */
332export function resolvePosixPath(absDir: string, rel: string): string {
333 if (rel.startsWith('/')) return rel;
334 const parts = (absDir + '/' + rel).split('/');
335 const out: string[] = [];
336 for (const p of parts) {
337 if (p === '..') out.pop();
338 else if (p !== '.' && p !== '') out.push(p);
339 }
340 return '/' + out.join('/');
341}

364/**
365 * After opening a project, walk every asset with a source link, stat its
366 * source file, and reload any whose mtime moved since `external_mtime`.
367 * Catches edits made while Eigendeck wasn't running. Returns the count
368 * of assets actually reloaded.
369 */
370export async function scanForChangedAssets(
371 projectDir: string,
372 presOverride: string | null,
373): Promise<{ checked: number; reloaded: number }> {
374 let reloaded = 0;
375 const linked = await invoke<LinkedAssetRow[]>('db_list_linked_assets').catch(() => [] as LinkedAssetRow[]);
376 const globalDefault = getPreference('autoReloadAssets');
377 wlog(`scanForChangedAssets: ${linked.length} linked asset(s), presOverride=${presOverride ?? 'default'}, globalAutoReload=${globalDefault}`);
378 if (linked.length === 0) return { checked: 0, reloaded: 0 };
379 // Asset-security: an untrusted / received deck performs ZERO disk access on open
380 // — not even the #74 existence stat. Skip the whole scan; the embedded snapshots
381 // render. (docs/ASSETS-SECURITY.md.) Trusted decks scan as before, with each
382 // byte-read routed through the gate for realpath + type safety.
383 {
384 const { getDeckToken } = await import('../store/presentation');→ presentation.ts
385 const token = getDeckToken();
386 if (!token) return { checked: 0, reloaded: 0 };
387 const { isTrusted } = await import('./trustStore');→ trustStore.ts
388 if (!(await isTrusted(token))) { wlog('scanForChangedAssets skipped — deck untrusted'); return { checked: 0, reloaded: 0 }; }
389 }
390 for (const a of linked) {
391 const absPath = resolvePosixPath(projectDir, a.external_path);
392 // Existence check is UNGATED by the auto-reload toggle: a missing source
393 // is worth flagging even when you've opted out of live reloads (#74).
394 let st: PathStat;
395 try {
396 st = await pathStat(absPath);
397 } catch (e) {
398 markAssetMissing(a.asset_id, a.path ?? a.external_path);
399 wlog(` source-missing ${a.asset_id.slice(0, 8)} path="${a.path}" abs="${absPath}": ${e}`);
400 continue;
401 }
402 markAssetFound(a.asset_id); // present on disk — clear any stale flag
403 // The RELOAD is gated: don't pull new bytes for assets you've opted out of.
404 if (!effectiveAutoReload(a.auto_reload, presOverride, globalDefault)) {
405 wlog(` skip ${a.asset_id.slice(0, 8)} — auto_reload resolves to OFF (per-asset='${a.auto_reload ?? 'default'}', per-pres='${presOverride ?? 'default'}')`);
406 continue;
407 }
408 try {
409 const diskMtime = statMtimeIso(st);
410 if (diskMtime && diskMtime !== a.external_mtime) {
411 // mtime moved — read bytes and compare hash before deciding
412 // whether to invalidate the rendered cache. Common case after
413 // a fresh insertion (stored mtime is null) or a touch / save
414 // roundtrip: mtime drifted but bytes didn't change. db_store_
415 // asset's short-circuit path updates the stored mtime even
416 // when bytes match, so future scans won't loop.
417 // Gated read (deck already established trusted above): resolveAndGate
418 // applies realpath + the asset-type check. A blocked target → skip; a
419 // vanished/erroring file → flag missing (#74).
420 const read = await gatedExternalRead(absPath);
421 if (read.status === 'gated') { wlog(` gated ${a.asset_id.slice(0, 8)} path="${a.path}"`); continue; }
422 if (read.status === 'unreadable') { markAssetMissing(a.asset_id, a.path ?? a.external_path); continue; }
423 const bytes = read.bytes;
424 const diskHash = await sha256Hex(bytes);
425 await invoke('db_store_asset', {
426 path: a.path ?? absPath.split('/').pop() ?? a.asset_id,
427 data: Array.from(bytes),
428 mimeType: a.mime_type ?? 'application/octet-stream',
429 externalPath: a.external_path,
430 externalMtime: diskMtime,
431 assetId: a.asset_id,
432 autoReload: null,
433 });
434 if (diskHash !== a.hash) {
435 // Bytes actually changed — invalidate rendered cache so the
436 // sidebar + canvas re-render from the new bytes.
437 wlog(` reload ${a.asset_id.slice(0, 8)} path="${a.path}" disk=${diskMtime} stored=${a.external_mtime} (bytes changed)`);
438 await invalidateRenderedAsset(a.asset_id);
439 reloaded++;
440 } else {
441 // Just an mtime drift — bytes are identical. db_store_asset's
442 // short-circuit recorded the new mtime; nothing else to do.
443 wlog(` mtime-only ${a.asset_id.slice(0, 8)} path="${a.path}" disk=${diskMtime} stored=${a.external_mtime} (bytes unchanged)`);
444 }
445 } else {
446 wlog(` unchanged ${a.asset_id.slice(0, 8)} path="${a.path}" mtime=${diskMtime}`);
447 }
448 } catch (e) {
449 // The file passed stat() but vanished/erred during read — treat as missing.
450 markAssetMissing(a.asset_id, a.path ?? a.external_path);
451 wlog(` source-missing ${a.asset_id.slice(0, 8)} path="${a.path}" abs="${absPath}": ${e}`);
452 }
453 }
454 return { checked: linked.length, reloaded };
455}

457/**
458 * Compute the directory remap between a missing source's old absolute path and
459 * the new one the user just picked (#74 follow-up). Strips the common trailing
460 * path (usually the filename, but any shared suffix) and returns the differing
461 * directory prefixes. Returns null when there's nothing to infer (no shared
462 * suffix — the file was also renamed — or identical prefixes — no move).
463 * Exported for unit testing.
464 */
465export function deriveRelocateOffset(oldAbs: string, newAbs: string): { oldPrefix: string; newPrefix: string } | null {
466 const o = oldAbs.split('/');
467 const n = newAbs.split('/');
468 let k = 0;
469 while (k < o.length && k < n.length && o[o.length - 1 - k] === n[n.length - 1 - k]) k++;
470 if (k === 0) return null;
471 const oldPrefix = o.slice(0, o.length - k).join('/');
472 const newPrefix = n.slice(0, n.length - k).join('/');
473 if (oldPrefix === newPrefix) return null;
474 return { oldPrefix, newPrefix };
475}

479 * OTHER currently-missing assets by applying the SAME directory move — i.e. if a
480 * whole folder moved, one relocate reveals where, and the rest follow. Only
481 * touches assets flagged missing; only re-points ones whose remapped file
482 * actually exists on disk. Returns how many additional assets were relocated.

488 newAbs: string,

492 const { oldPrefix, newPrefix } = offset;

496 if (a.asset_id === skipAssetId || !isAssetMissing(a.asset_id)) continue;

500 checked++;
501 try {
502 // The user relocated the anchor file, revealing this whole folder moved; the
503 // siblings within it inherit that consent. Approve each (keyed by its asset id,
504 // so it replaces that asset's old approval in place) on the trusted deck — same
505 // single approval path as add/relocate — so the gated read accepts it.

508 // Gated read: untrusted deck or a blocked/wrong-type relocation target → skip.

510 if (read.status !== 'ok') continue;
