6export type SelectAllResult =
7 | { kind: 'field' } // focus is in a text field → select its text
8 | { kind: 'multi'; ids: string[] } // >1 element on the slide → select them all

12/**
13 * Given whether a text field is focused and the current slide's element ids,

17export function selectAllTarget(inField: boolean, elementIds: readonly string[]): SelectAllResult {
18 if (inField) return { kind: 'field' };
19 if (elementIds.length > 1) return { kind: 'multi', ids: [...elementIds] };
20 if (elementIds.length === 1) return { kind: 'element', id: elementIds[0] };
21 return { kind: 'slide' };
