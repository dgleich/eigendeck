28// Module-level in-session cache of overlays, keyed by element id. The
29// editor and PresentMode mount SEPARATE NotebookContent instances
30// (App returns <PresentMode/> instead of the editor while presenting),
31// so a per-instance overlay would lose state across the present↔edit
32// transition — and a DB reload on remount races the outgoing flush.
33// This cache is the in-session source of truth: any instance's
34// mutations land here, and a remounted instance reads it synchronously
35// (no flicker, no race). The asset is still flushed for durability.
36// Cleared per-element on reload-from-disk and wholesale on deck load.
37const overlayCache = new Map<string, Overlay>();

45// Synced notebooks (the same notebook shown on multiple slides) share ONE
46// overlay — they are the SAME thing. That sharing is achieved by keying the
47// overlay on the element's sync identity (syncId ?? id) at the call site
48// (NotebookContent), so no per-duplicate clone is needed.

54/** Read an overlay by its element key — cache first, then DB. Empty if none. */
55export async function loadOverlayFor(key: string): Promise<Overlay> {
56 const cached = overlayCache.get(key);
57 if (cached) return cached;
58 try {
59 const id = await invoke<string | null>('db_get_owned_asset_id', { ownerElementId: key });
60 if (id) {
61 const buf = await invoke<ArrayBuffer>('db_get_asset_by_id', { assetId: id });
62 return parseOverlay(new Uint8Array(buf));
63 }
64 } catch (e) { console.warn('loadOverlayFor failed:', e); }
65 return emptyOverlay();
66}

68/** Persist `overlay` as owned by `key` (deterministic asset id, reusing an
69 * existing owned asset for that key if present) and seed the cache. Empty
70 * overlays are not persisted (no stray assets). */
71export async function writeOverlayFor(key: string, overlay: Overlay): Promise<void> {
72 overlayCache.set(key, overlay);
73 const bytes = serializeOverlay(overlay);
74 if (isOverlayEmpty(overlay)) return;
75 let id: string | null = null;
76 try {
77 id = await invoke<string | null>('db_get_owned_asset_id', { ownerElementId: key });
78 } catch { id = null; }
79 id = id ?? `overlay-${key}`;
80 try {
81 await invoke('db_store_asset', {
82 path: `overlay:${key}`,
83 data: Array.from(new TextEncoder().encode(bytes)),
84 mimeType: OVERLAY_MIME,
85 externalPath: null, externalMtime: null,
86 assetId: id, autoReload: 'off', ownerElementId: key,
87 });
88 } catch (e) { console.warn('writeOverlayFor failed:', e); }
89}

91/** Copy the overlay at `fromKey` into a private overlay owned by `toKey`.
92 * Used when a synced notebook is freed (unsynced): the freed instance keeps
93 * its own copy of the recording while the still-synced instances keep the
94 * original. When `fromKey` is already cached, the cache seed happens before
95 * the first await, so a caller that fires this immediately before flipping
96 * syncId sees the clone on the very next render (no empty flash). No-op when
97 * the source is empty or the keys coincide. */
98export async function cloneOverlay(fromKey: string, toKey: string): Promise<void> {
99 if (!fromKey || !toKey || fromKey === toKey) return;
100 const cached = overlayCache.get(fromKey);
101 if (cached) {
102 if (isOverlayEmpty(cached)) return;
103 const clone: Overlay = structuredClone(cached);
104 overlayCache.set(toKey, clone); // sync seed, pre-await
105 await writeOverlayFor(toKey, clone);
106 return;
107 }
108 const src = await loadOverlayFor(fromKey);
109 if (isOverlayEmpty(src)) return;
110 await writeOverlayFor(toKey, structuredClone(src));
111}

122/** Reconcile recordings when two notebooks merge under one syncId (the Link /
123 * Time-Machine action). The surviving overlay — the one owned by `keepKey`,
124 * or none — becomes owned by `newKey` (the shared syncId), and BOTH pre-merge
125 * keys are discarded so nothing lingers to be revived later. The caller
126 * decides `keepKey` (auto when 0/1 recordings exist; user-chosen on a real
127 * conflict). Load the winner BEFORE discarding so its bytes survive. */
128export async function applyLinkOverlay(
129 keys: { sourceKey: string; targetKey: string; newKey: string },
130 keepKey: string | null,
131): Promise<void> {
132 const { sourceKey, targetKey, newKey } = keys;
133 const winner = keepKey ? await loadOverlayFor(keepKey) : null;
134 for (const k of new Set([sourceKey, targetKey])) {
135 if (k !== newKey) await discardOverlay(k);
136 }
137 if (winner && !isOverlayEmpty(winner)) {
138 await writeOverlayFor(newKey, structuredClone(winner));
139 } else {
140 await discardOverlay(newKey);
141 }
142}

165export function useOverlay(elementId: string): UseOverlayResult {
166 // Seed from the in-session cache so a remount (e.g. exiting
167 // PresentMode) shows recorded outputs immediately.
168 const [overlay, setOverlay] = useState<Overlay>(() => overlayCache.get(elementId) ?? emptyOverlay());
169 // The overlay asset's id once known (loaded or created). null until
170 // the first flush creates it.
171 const assetIdRef = useRef<string | null>(null);
172 // Last-flushed serialized bytes, to skip no-op flushes. Seeded with the
173 // initial overlay's bytes so an UNTOUCHED overlay never flushes (a ''
174 // seed let a flush racing the load write an empty overlay — one source
175 // of the duplicate-asset bug).
176 const lastFlushedRef = useRef<string>(
177 serializeOverlay(overlayCache.get(elementId) ?? emptyOverlay()),
178 );
179 const flushTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
180 // Latest overlay for the debounced flush closure.
181 const overlayRef = useRef<Overlay>(overlay);
182 overlayRef.current = overlay;
183 
184 // --- load on mount / element change -------------------------------
185 useEffect(() => {
186 // If this element's overlay is already cached this session, trust
187 // it (it holds in-flight edits/outputs that may not be flushed yet)
188 // and skip the DB load — avoids the present→edit reload race.
189 if (overlayCache.has(elementId)) {
190 const cached = overlayCache.get(elementId)!;
191 // We don't know the asset id from the cache alone; fetch it lazily
192 // (cheap) so a subsequent flush updates the existing asset rather
193 // than minting a second one.
194 if (assetIdRef.current === null) {
195 void invoke<string | null>('db_get_owned_asset_id', { ownerElementId: elementId })
196 .then((id) => { if (id) assetIdRef.current = id; })
197 .catch(() => {});
198 }
199 lastFlushedRef.current = serializeOverlay(cached);
200 setOverlay(cached);
201 return;
202 }
203 let cancelled = false;
204 (async () => {
205 try {
206 const id = await invoke<string | null>('db_get_owned_asset_id', {
207 ownerElementId: elementId,
208 });
209 if (cancelled) return;
210 if (id) {
211 assetIdRef.current = id;
212 const buf = await invoke<ArrayBuffer>('db_get_asset_by_id', { assetId: id });
213 if (cancelled) return;
214 const parsed = parseOverlay(new Uint8Array(buf));
215 lastFlushedRef.current = serializeOverlay(parsed);
216 overlayCache.set(elementId, parsed);
217 setOverlay(parsed);
218 } else {
219 assetIdRef.current = null;
220 lastFlushedRef.current = serializeOverlay(emptyOverlay());
221 setOverlay(emptyOverlay());
222 }
223 } catch (e) {
224 console.warn('useOverlay load failed:', e);
225 }
226 })();
227 return () => { cancelled = true; };
228 }, [elementId]);
229 
230 // --- flush (debounced, only-when-changed) -------------------------
231 const flushNow = useCallback(async () => {
232 // Flush the SHARED cache — it's the source of truth across remounts and
233 // the present<->edit transition. Serializing per-instance state here let
234 // a stale/empty instance overwrite a richer overlay. Fall back to the
235 // live overlay if the cache was cleared (reload-from-disk).
236 const current = overlayCache.get(elementId) ?? overlayRef.current;
237 const bytes = serializeOverlay(current);
238 if (bytes === lastFlushedRef.current) return; // nothing changed
239 
240 // Resolve the asset id WITHOUT minting a fresh random one per flush —
241 // that created a new overlay asset on every remount, and the loader
242 // (db_get_owned_asset_id, latest-first) could then pick an empty
243 // duplicate. Reuse this element's existing owned overlay if any;
244 // otherwise use a DETERMINISTIC id so concurrent/remount first-flushes
245 // all converge on ONE asset (also subsumes the old anti-hash-collapse
246 // trick — B3). Never create an asset for an empty overlay.
247 let id = assetIdRef.current;
248 if (id == null) {
249 if (bytes === serializeOverlay(emptyOverlay())) {
250 lastFlushedRef.current = bytes; // nothing to persist
251 return;
252 }
253 try {
254 id = await invoke<string | null>('db_get_owned_asset_id', {
255 ownerElementId: elementId,
256 });
257 } catch { id = null; }
258 id = id ?? `overlay-${elementId}`;
259 assetIdRef.current = id;
260 }
261 try {
262 const data = Array.from(new TextEncoder().encode(bytes));
263 await invoke('db_store_asset', {
264 path: `overlay:${elementId}`,
265 data,
266 mimeType: OVERLAY_MIME,
267 externalPath: null,
268 externalMtime: null,
269 assetId: id,
270 autoReload: 'off',
271 ownerElementId: elementId,
272 });
273 lastFlushedRef.current = bytes;
274 } catch (e) {
275 console.warn('useOverlay flush failed:', e);
276 }
277 }, [elementId]);
278 
279 // Schedule a debounced flush whenever the overlay changes.
280 useEffect(() => {
281 if (flushTimer.current) clearTimeout(flushTimer.current);
282 flushTimer.current = setTimeout(() => { void flushNow(); }, FLUSH_DEBOUNCE_MS);
283 return () => { if (flushTimer.current) clearTimeout(flushTimer.current); };
284 }, [overlay, flushNow]);
285 
286 // Flush any pending change on unmount (best-effort).
287 useEffect(() => {
288 return () => { void flushNow(); };
289 }, [flushNow]);
290 
291 // Register with the global registry so the deck save / close-quit path can
292 // FORCE this overlay to disk without waiting on the debounce (#123).
293 useEffect(() => registerOverlayFlush(flushNow), [flushNow]);

295 // --- mutators -----------------------------------------------------
296 // All mutations go through applyOverlay so the in-session cache stays
297 // in lock-step with component state — that's what survives the
298 // present↔edit remount.
299 const applyOverlay = useCallback((updater: (o: Overlay) => Overlay) => {
300 setOverlay((prev) => {
301 const next = updater(prev);
302 overlayCache.set(elementId, next);
303 return next;
304 });
305 }, [elementId]);
