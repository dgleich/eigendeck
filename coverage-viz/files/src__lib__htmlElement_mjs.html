1// Raw-HTML element (#137) — the ONE place the sandbox + CSP for the general HTML
2// escape hatch are defined, shared by every render path and the HTML export so
3// the isolation can never drift. The element renders arbitrary markup inside a
4// sandboxed <iframe srcdoc>; safety comes from the browser, not a sanitizer:
5//
6// • NO `allow-scripts` in the sandbox → zero JavaScript runs (inline handlers
7// like onerror can't fire either), and the framed doc can't drop its own
8// sandbox. This is the whole no-script guarantee.
9// • An injected CSP blocks ALL network — only data: URIs for images/media/fonts
10// and inline styles are allowed. No egress, so decks stay offline-portable
11// and can't leak IPs / load trackers.
12//
13// The editor uses the `allow-same-origin` sandbox (still WITHOUT allow-scripts,
14// which is the safe combination) so the parent can reach into contentDocument and
15// toggle contentEditable for best-effort in-canvas editing. Every non-editing
16// target (present, export, thumbnail) uses the fully-locked empty sandbox.

28/** CSP injected into every srcdoc: block all network, allow only inline styles and

30export const HTML_ELEMENT_CSP =

33/** Wrap raw HTML into a full srcdoc document carrying the CSP. Transparent body so
34 * the element composites onto the slide; a per-element `background` can override.
35 * When the html declares variables (#138), `vars` (stored values) + `theme` (for
36 * tint resolution) are spliced in: `:root{--name:…}` custom properties for the CSS
37 * side and `{{name}}` tokens for the HTML side; the manifest is stripped. Omitting
38 * vars/theme still applies the manifest DEFAULTS (so every path stays consistent). */
39export function htmlElementSrcdoc(rawHtml, background, vars, theme, opts) {
40 // `raw` mode skips the splice — used by in-canvas editing so the contentEditable
41 // DOM (and the markup read back out of it) keeps the manifest + literal {{tokens}},
42 // instead of the derived, spliced body that would lose them.
43 const { html: body, rootCss } = opts && opts.raw
44 ? { html: typeof rawHtml === 'string' ? rawHtml : '', rootCss: '' }
45 : spliceHtmlVars(rawHtml, vars, theme);
46 const bg = background ? String(background).replace(/[<>"]/g, '') : 'transparent';
47 return '<!doctype html><html><head><meta charset="utf-8">'
48 + `<meta http-equiv="Content-Security-Policy" content="${HTML_ELEMENT_CSP}">`
49 // print-color-adjust:exact so the iframe's backgrounds/gradients/fills survive
50 // print/PDF — the parent's print-color-adjust does NOT cascade into a sandboxed
51 // iframe, so without this the element loses its background + colours in print (#137).
52 + `<style>html,body{margin:0;padding:0;height:100%;background:${bg};box-sizing:border-box;`
53 + `-webkit-print-color-adjust:exact;print-color-adjust:exact;}*{box-sizing:border-box;}${rootCss}</style>`
54 + `</head><body>${body}</body></html>`;
55}

57/** The `<iframe>` HTML string for string-render targets (HTML export, PDF inline).

59export function htmlElementIframeHtml(el, styleStr, sandbox = HTML_SANDBOX_LOCKED, theme) {

64/** Scale-mode iframe for string-render targets (HTML export, PDF/print). `boxStyleStr`
65 * positions the (clipping) wrapper in the caller's units; `L` carries the design
66 * size + centering offsets ALREADY in `unit` (px for export, in for print) and the
67 * unitless `scale`. Mirrors the DOM paths' wrapper+transform. */
68export function htmlElementScaledIframeHtml(el, boxStyleStr, L, unit = 'px', sandbox = HTML_SANDBOX_LOCKED, theme) {
69 const srcdoc = htmlEscapeForSrcdoc(htmlElementSrcdoc(el.html, el.background, el.vars, theme));
70 const iframeStyle =
71 `position:absolute;left:0;top:0;width:${L.designW}${unit};height:${L.designH}${unit};`
72 + 'border:none;background:transparent;'
73 + `transform:translate(${L.offsetX}${unit},${L.offsetY}${unit}) scale(${L.scale});transform-origin:top left;`;
74 return `<div style="${boxStyleStr};overflow:hidden;">`
75 + `<iframe srcdoc="${srcdoc}" style="${iframeStyle}" sandbox="${sandbox}"></iframe></div>`;
76}

78/** Whether the element opts into contain-scaling (needs both the flag and a design

84/** Contain-scale layout for a scale-mode html element (#137). Given the live box
85 * (bw×bh) and the captured design size (sw×sh), returns the UNIFORM scale factor
86 * (aspect preserved) and the centering offset that fits the design-size content
87 * inside the box, letterboxed. `scale` is a pure ratio; `designW/H` and
88 * `offsetX/Y` are lengths in the SAME unit as the box — so a caller working in
89 * other units (print inches) converts those lengths and leaves `scale` as-is.
90 * Missing/degenerate design size falls back to the box (scale 1 = no-op). */
91export function htmlScaleLayout(bw, bh, sw, sh) {
92 const designW = sw > 0 ? sw : bw;
93 const designH = sh > 0 ? sh : bh;
94 const scaleRaw = designW > 0 && designH > 0 ? Math.min(bw / designW, bh / designH) : 1;
95 // Round for clean CSS output (avoid float noise like -3.5e-15px / 0.5499999…):
96 // offsets to whole px (sub-px centering is imperceptible), scale to 4 decimals.
97 return {
98 designW, designH,
99 scale: Math.round(scaleRaw * 1e4) / 1e4,
100 offsetX: Math.round((bw - designW * scaleRaw) / 2),
101 offsetY: Math.round((bh - designH * scaleRaw) / 2),
102 };
