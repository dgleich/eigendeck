11/** Snap a display dimension up to the next render tier.
12 *
13 * Tier (not exact element dims) so the cache stays sticky during a
14 * resize-handle drag — every animation-frame size would otherwise be
15 * a fresh cache key + fresh pdfium render.
16 *
17 * 2x for retina sharpness — on an Apple-Silicon display a 1200px
18 * slide-space element is ~2400 actual pixels at 100% editor zoom,
19 * so rendering at slide-space dims means immediate blur. 256px
20 * step granularity keeps typical element sizes from over-shooting
21 * the next tier (PNG encode cost is roughly quadratic). Cap at
22 * slide width — past 1920 we'd be rendering more pixels than the
23 * slide is ever shown at. */
24function pdfRenderTier(displayWidth: number, displayHeight: number): number {
25 const max = Math.max(displayWidth, displayHeight);
26 const stepped = Math.ceil((max * 2) / 256) * 256;
27 return Math.min(stepped, ASSET_TIER.full);
28}

30/**
31 * Resolve an image element to a renderable URL. Branching:
32 *
33 * - `kind === 'pdf'` → goes through useRenderedAsset (pdfium-rasterized
34 * PNG cached in asset_cache). PDFs can't render as `<img src="blob:application/pdf">`
35 * — WebKit doesn't natively rasterize PDF inline.
36 * - everything else (raster/svg/undefined) → useAssetUrl returns a raw
37 * blob URL the browser renders natively.
38 *
39 * `displayWidth/Height` size the PDF render. SVG/raster ignore them
40 * (browser scales those for free). Pass element.position.{width,height}.
41 */
42export function useImageSrc(
43 assetId: string,
44 kind?: 'raster' | 'svg' | 'pdf',
45 opts?: { displayWidth?: number; displayHeight?: number; snapshotVariant?: string },
46): string | undefined {
47 const isPdf = kind === 'pdf';
48 // Always call both hooks (rules-of-hooks) but feed one of them an
49 // undefined assetId so it no-ops. The unused branch returns undefined.
50 const blobUrl = useAssetUrl(isPdf ? undefined : assetId);
51 const tier = isPdf
52 ? pdfRenderTier(opts?.displayWidth ?? ASSET_TIER.full, opts?.displayHeight ?? ASSET_TIER.full)
53 : ASSET_TIER.full;
54 const pdfUrl = useRenderedAsset(
55 isPdf ? assetId : undefined,
56 isPdf ? 'pdf' : undefined,
57 tier, tier,
58 opts?.snapshotVariant,
59 );
60 return isPdf ? pdfUrl : blobUrl;
61}
