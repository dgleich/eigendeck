1// The asset-type allowlist — the 0th-order gate for what Eigendeck may treat as a
2// watchable/embeddable asset. See docs/ASSETS-SECURITY.md ("Watchable asset types").
3//
4// Two independent checks, both applied to the RESOLVED target's bytes/name:
5// 1. isAllowedExtension(path) — is the extension on the allowlist?
6// 2. contentMatchesExtension(...) — do the bytes actually match that type?

8// the eigendeck-demo marker for our .html)

10// Pure + framework-free (Uint8Array/ArrayBuffer/string in; no DOM, runs in Node too)
11// so the same rules serve the app, the CLI, and build tooling — one source of truth.

30/** Lowercased extension of a path/filename, without the dot; '' if none. */
31export function extensionOf(path) {
32 const base = String(path).split(/[\\/]/).pop() || '';
33 const dot = base.lastIndexOf('.');
34 return dot > 0 ? base.slice(dot + 1).toLowerCase() : '';
35}

49/** First `n` bytes of the input (Uint8Array | ArrayBuffer | string) as a byte array. */
50function toBytes(input, n) {
51 if (typeof input === 'string') {
52 const out = new Uint8Array(Math.min(input.length, n));
53 for (let i = 0; i < out.length; i++) out[i] = input.charCodeAt(i) & 0xff;
54 return out;
55 }
56 const u8 = input instanceof Uint8Array ? input : new Uint8Array(input);
57 return u8.subarray(0, n);
58}

60/** A latin1 (byte-preserving) string of the prefix, BOM stripped, for text sniffs. */
61function prefixString(input, n = 512) {
62 if (typeof input === 'string') {
63 const s = input.charCodeAt(0) === 0xfeff ? input.slice(1) : input; // BOM char
64 return s.slice(0, n);
65 }
66 let b = toBytes(input, n + 3);
67 if (b.length >= 3 && b[0] === 0xef && b[1] === 0xbb && b[2] === 0xbf) b = b.subarray(3); // UTF-8 BOM
68 let s = '';
69 for (let i = 0; i < b.length && i < n; i++) s += String.fromCharCode(b[i]);
70 return s;
71}

73function startsWithBytes(input, sig, offset = 0) {
74 const b = toBytes(input, offset + sig.length);
75 if (b.length < offset + sig.length) return false;
76 for (let i = 0; i < sig.length; i++) if (b[offset + i] !== sig[i]) return false;
77 return true;
78}

82/**
83 * Sniff the eigendeck-demo marker on a demo `.html` file's bytes.
84 * Tolerates a leading BOM, whitespace, and an optional `<!DOCTYPE html>` before
85 * the marker (marker sits right after the DOCTYPE to keep standards mode).
86 * Returns { ok, version, supported }. `ok` = a well-formed marker was found;
87 * `supported` = its version is one this build runs (fail-closed on unknown).
88 */
89export function isEigendeckDemo(input) {
90 let s = prefixString(input, 512).replace(/^\s+/, '');
91 const doctype = s.match(/^<!doctype\s+html\s*>/i);
92 if (doctype) s = s.slice(doctype[0].length).replace(/^\s+/, '');
93 // Canonical version only — no leading zeros ("exact bytes" contract). v01 is not
94 // a valid marker, so it fails closed rather than aliasing to v1.
95 const m = s.match(/^<!--eigendeck-demo-v([1-9]\d*)-->/);
96 if (!m) return { ok: false, version: null, supported: false };
97 const version = Number(m[1]);
98 return { ok: true, version, supported: SUPPORTED_DEMO_VERSIONS.has(version) };
99}

112function isWebp(input) {
113 const b = toBytes(input, 12);
114 return b.length >= 12 &&
115 b[0] === 0x52 && b[1] === 0x49 && b[2] === 0x46 && b[3] === 0x46 && // RIFF
116 b[8] === 0x57 && b[9] === 0x45 && b[10] === 0x42 && b[11] === 0x50; // WEBP
117}

119function isMp4(input) {
120 // ....ftyp at offset 4 (ISO base media / QuickTime)
121 return startsWithBytes(input, [0x66, 0x74, 0x79, 0x70], 4); // "ftyp"
122}

124function isVtt(input) {
125 // WebVTT: the file must start with "WEBVTT" followed by a space, tab, newline, or EOF
126 // (per the spec — so "WEBVTTX" is NOT a cue file), after an optional BOM.
127 return /^WEBVTT(?:[ \t\r\n]|$)/.test(prefixString(input, 16));
128}

130function isSvg(input) {
131 // Text/XML: a *leading* <svg root (the spec's "<svg> root", not merely an <svg
132 // somewhere in the file). Tolerate any prolog of xml-decl / <!DOCTYPE svg …> /
133 // comments in any order — Illustrator + Inkscape emit an SVG DOCTYPE before <svg>.
134 return /^\s*(?:(?:<\?xml[^>]*>|<!doctype\s+svg[^>]*>|<!--[\s\S]*?-->)\s*)*<svg[\s>]/i
135 .test(prefixString(input, 512));
136}

162/** Does this extension's content check need the WHOLE file rather than a short prefix?
163 * Every other type is decided from magic bytes / a <=512B prolog / the demo marker, so
164 * a bounded "sniff" read gives the same verdict as a full read. Only .ipynb parses the
165 * entire JSON. Callers doing a bounded read must re-read these in full. Keeping the fact
166 * HERE (beside isNotebookJson) is the single source of truth, so the read-size choice
167 * can't drift from what the gate actually inspects. */
168export function needsFullContent(ext) {
169 return String(ext).toLowerCase() === 'ipynb';
170}

172// Notebooks need the whole file to JSON.parse, not just a prefix.
173function prefixStringFull(input) {
174 if (typeof input === 'string') return input;
175 const u8 = input instanceof Uint8Array ? input : new Uint8Array(input);
176 return new TextDecoder('utf-8').decode(u8);

179/** Interchange (non-demo) content identity by native magic / structural parse. */
180function matchesInterchangeMagic(input, e) {
181 switch (e) {
182 case 'png': return startsWithBytes(input, MAGIC.png);
183 case 'jpg': case 'jpeg': return startsWithBytes(input, MAGIC.jpg);
184 case 'gif': return startsWithBytes(input, MAGIC.gif);
185 case 'webp': return isWebp(input);
186 case 'pdf': return startsWithBytes(input, MAGIC.pdf);
187 case 'webm': return startsWithBytes(input, MAGIC.webm);
188 case 'ogg': case 'ogv': return startsWithBytes(input, MAGIC.ogg);
189 case 'mp4': case 'mov': case 'm4v': return isMp4(input);
190 case 'svg': return isSvg(input);
191 case 'ipynb': return isNotebookJson(input);
192 case 'vtt': return isVtt(input);
193 default: return false;
194 }
195}

197/**
198 * Content-identity check → `{ ok, reason }`. THE single place the per-type content
199 * rules live: our demo `.html` via the eigendeck marker (with the distinct
200 * `unsupported-demo-version` reason), everything else via native magic. Both
201 * `contentMatchesExtension()` and `assetTypeGate()` route through this, so the
202 * rules can never diverge. `reason` is null on success, else `content-mismatch` |
203 * `unsupported-demo-version`.
204 */
205export function checkContent(input, ext) {
206 const e = String(ext).toLowerCase();
207 if (e === 'html') {
208 const d = isEigendeckDemo(input);
209 if (!d.ok) return { ok: false, reason: 'content-mismatch' };
210 if (!d.supported) return { ok: false, reason: 'unsupported-demo-version' };
211 return { ok: true, reason: null };
212 }
213 return matchesInterchangeMagic(input, e)
214 ? { ok: true, reason: null }
215 : { ok: false, reason: 'content-mismatch' };
216}

218/** Boolean form of {@link checkContent}: do the bytes match the type its
219 * extension claims? (Interchange → native magic; demo `.html` → the marker.) */
220export function contentMatchesExtension(input, ext) {
221 return checkContent(input, ext).ok;
222}

224/**
225 * The full 0th-order gate: allowed extension AND content that matches it.
226 * Returns { ok, kind, reason }; `reason` is a short machine tag for the UI
227 * ("bad-extension" | "content-mismatch" | "unsupported-demo-version").
228 *
229 * SECURITY CONTRACT — `resolvedRealPath` MUST be the fully `realpath`-resolved
230 * target, never a deck-supplied reference name. The extension gate is judged on
231 * this path, so the entire symlink defense (`a.png → id_rsa` must be judged as
232 * `id_rsa`) depends on the caller resolving symlinks first. `input` must be the
233 * bytes actually read from that resolved target. Passing an unresolved reference
234 * is a security bug, not a convenience.
235 */
236export function assetTypeGate(input, resolvedRealPath) {
237 const ext = extensionOf(resolvedRealPath);
238 const kind = ASSET_EXTENSIONS[ext] ?? null;
239 if (!kind) return { ok: false, kind: null, reason: 'bad-extension' };
240 // Size cap. The read primitive (resolve_and_read) already rejects oversized EXTERNAL
241 // files before reading; this catches the add path, which gates on already-read bytes.
242 // (A bounded "sniff" read is under the cap and won't false-trip — the Rust side reports
243 // the true too-large for that path.)
244 if (byteLength(input) > MAX_ASSET_BYTES) return { ok: false, kind, reason: 'too-large' };
245 const c = checkContent(input, ext); // one code path; no per-site re-implementation
246 return { ok: c.ok, kind, reason: c.reason };
