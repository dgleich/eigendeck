26async function exportOne(input: string, outputDir: string): Promise<ExportFileReport> {
27 const start = performance.now();
28 const output = `${outputDir}/${basename(input).replace(/\.eigendeck$/, '.html')}`;
29 try {
30 await invoke('db_open', { path: input });
31 // Warm pool cache from this file's math_cache so renderTextElement hits.
32 resetMathCacheWarmupFlag();
33 await warmMathCacheFromSqlite();
34 
35 const json = await invoke<string>('db_export_json');
36 const presentation = JSON.parse(json);
37 
38 const fontFacesCss = await buildEmbeddedFontFacesCSS(presentation);
39 const html = await buildExportHtml({
40 presentation,
41 readFile: async (path: string) => {
42 const data = await invoke<number[]>('db_get_asset', { path });
43 return new Uint8Array(data);
44 },
45 readTextFile: async (path: string) => {
46 const data = await invoke<number[]>('db_get_asset', { path });
47 return new TextDecoder().decode(new Uint8Array(data));
48 },
49 renderTextElement: makeTextElementRenderer(presentation),
50 fontFacesCss,
51 });
52 await writeTextFileNative(output, html);

55 const fontFamilies = [...new Set([...html.matchAll(/font-family: '([^']+)'/g)].map((m) => m[1]))].sort();
56 const cdnLeaks = (html.match(/fonts\.googleapis\.com/g) || []).length;

85export async function runBatchExportHtml(): Promise<void> {
86 try {
87 const picked = await pickDirectoryWithEigendecks('Pick a directory of .eigendeck files to export');
88 if (!picked) return;
89 const { dir, files } = picked;
90 if (files.length === 0) {
91 await showBatchError('batch-html-export', 'No .eigendeck files in the chosen directory.');
92 return;
93 }
94 
95 // Snapshot current project so we can restore it after; batch swaps the
96 // SQLite handle through every file.
97 const originalPath = usePresentationStore.getState().projectPath;
98 
99 const outDir = `${dir}/_debug-export`;
100 try { await mkdirNative(outDir); } catch { /* exists */ }
101 
102 const startWall = performance.now();
103 const reports: ExportFileReport[] = [];
104 for (const f of files) reports.push(await exportOne(f, outDir));
105 const elapsedSeconds = (performance.now() - startWall) / 1000;
106 
107 const meta: RunMeta = {
108 action: 'batch-html-export',
109 startedAt: new Date().toISOString(),
110 directory: dir,
111 totalFiles: files.length,
112 elapsedSeconds,
113 passed: reports.filter((r) => r.ok).length,
114 failed: reports.filter((r) => !r.ok).length,
115 };
116 const report: ExportReport = { meta, files: reports };
117 await writeReportAndAlert(dir, meta, report);
118 
119 // Restore the user's original project (or close).
120 if (originalPath) {
121 try { await openSqliteProject(originalPath); } catch { /* user can reopen manually */ }
122 } else {
123 try { await invoke('db_close'); } catch { /* nothing open */ }
124 }
125 } catch (e) {
126 await showBatchError('batch-html-export', e);
127 }
