1// The positioned HTML for ONE text element — the 3-div structure (outer box →
2// vertical-align wrapper → styled content) shared by the HTML export (exportCore,
3// px units) and the PDF export (printSlideHtml, inches/points). The ONLY
4// difference between those two was the unit, so units are injected as formatters
5// (`len`: px→CSS length, `fsize`: px→CSS font-size) and everything else — which
6// styles, their order, footnote-tight line-height, padding overrides, fill /
7// box-shadow / border-radius / rotation, font weight/style — lives here once so
8// the two exports can't drift.
9//
10// The caller pre-computes the parts that genuinely differ per target and passes
11// them in: the resolved `color` and `fontFamily`, the px `fontSize`, and the
12// inner `content` HTML (the HTML export math-renders + code-fonts it; the PDF
13// export marks it as eigendeck-origin — neither concern belongs here).

16import { textBackgroundResolved, textBoxShadowCss, textShadowCss } from './textStyle.mjs';→ textStyle.mjs

19/**
20 * @param el text element ({ preset, position, verticalAlign?, padding?,
21 * borderRadius?, rotation?, backgroundColor?, backgroundOpacity?,
22 * boxShadow?, textEffect? })
23 * @param o.color resolved text color
24 * @param o.fontFamily resolved font family
25 * @param o.fontSize px font size (already resolved via effectiveFontSize)
26 * @param o.content inner HTML (already math-rendered / code-fonted / marked)
27 * @param o.len (px:number) => string length formatter, e.g. n=>`${n}px` or px2in
28 * @param o.fsize (px:number) => string font-size formatter, e.g. n=>`${n}px` or px2pt
29 */
30export function textElementHtml(el, { color, fontFamily, fontSize, content, len, fsize, theme }) {
31 const ps = TEXT_PRESET_STYLES[el.preset] || TEXT_PRESET_STYLES.body;
32 const p = el.position;
33 const box = textPresetBoxCss(el.preset);
34 
35 const valign = el.verticalAlign || (el.preset === 'title' || el.preset === 'footnote' ? 'bottom' : undefined);
36 const valignStyle = valign === 'middle' ? 'display:flex;flex-direction:column;justify-content:center;' :
37 valign === 'bottom' ? 'display:flex;flex-direction:column;justify-content:flex-end;' : '';
38 
39 // Escape every dynamic value spliced into the style/attribute strings — neutralize
40 // " < > & so a crafted property (padding/rotation/borderRadius/color/fontFamily/
41 // geometry, or a size from an unvalidated config.textSizes) can't break out of the
42 // quoted attribute and inject markup. This builder feeds BOTH the HTML export (a
43 // possibly-hosted artifact) and the PDF path (audit C-2). `content` is the already-
44 // sanitized inner HTML and is intentionally NOT escaped. Legit values have none of
45 // these chars, so output is byte-identical (WYSIWYG). Mirrors escAttr in TextElementSvg.
46 const e = (v) => String(v).replace(/&/g, '&amp;').replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
47 
48 const pad = el.padding
49 ? `${e(len(el.padding.top))} ${e(len(el.padding.right))} ${e(len(el.padding.bottom))} ${e(len(el.padding.left))}`
50 : `${e(len(box.padY))} ${e(len(box.padX))}`;
51 
52 const bg = textBackgroundResolved(el, theme); // theme-aware (boxTint) fill; falls back to the fixed color
53 const sh = textBoxShadowCss(el);
54 const fx = textShadowCss(el, color);
55 const rot = el.rotation ? `transform:rotate(${e(el.rotation)}deg);` : '';
56 const rad = el.borderRadius ? `border-radius:${e(len(el.borderRadius))};` : '';
57 
58 return `<div style="position:absolute;left:${e(len(p.x))};top:${e(len(p.y))};width:${e(len(p.width))};height:${e(len(p.height))};overflow:hidden;${bg ? `background:${e(bg)};` : ''}${sh ? `box-shadow:${e(sh)};` : ''}${rad}${rot}">` +
59 `<div style="width:100%;height:100%;${valignStyle}">` +
60 `<div style="font-family:${e(fontFamily)};font-weight:${e(ps.fontWeight)};font-style:${e(ps.fontStyle)};font-size:${e(fsize(fontSize))};color:${e(color)};line-height:${e(box.lineHeight)};padding:${pad};${fx ? `text-shadow:${e(fx)};` : ''}">${content}</div>` +
61 `</div></div>`;
62}
