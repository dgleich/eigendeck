28 if (valign === 'middle') return 'display:flex;flex-direction:column;justify-content:center';

33/**
34 * Strip HTML tags but keep math source ($..$ / $$..$$) — produces readable
35 * alt-text for accessibility. The result preserves "text $\\alpha + 1$ more"
36 * as "text $\alpha + 1$ more" so screen readers can announce the equation
37 * source intelligibly.
38 */
39function altTextFromHtml(html: string): string {
40 if (!html) return '';
41 // Replace block tags with newlines, then strip remaining tags. Decode
42 // common entities. Collapse whitespace.
43 return html
44 .replace(/<br\s*\/?>/gi, '\n')
45 .replace(/<\/(p|div|li|h[1-6])>/gi, '\n')
46 .replace(/<[^>]+>/g, '')
47 .replace(/&nbsp;/g, ' ')
48 .replace(/&amp;/g, '&')
49 .replace(/&lt;/g, '<')
50 .replace(/&gt;/g, '>')
51 .replace(/&quot;/g, '"')
52 .replace(/&#39;/g, "'")
53 .replace(/[ \t]+/g, ' ')
54 .replace(/\n{3,}/g, '\n\n')
55 .trim();
56}

68/**
69 * Build the SVG/foreignObject markup string for a text element.
70 * Pure function — no React, no DOM. The iframe pool must already have
71 * pre-rendered math (passed in as `renderedHtml`).
72 *
73 * Used by:
74 * - TextElementSvg (React display component)
75 * - HTML export (fileOps.ts, via exportCore renderTextElement callback)
76 *
77 * The output is fully self-contained — math SVGs have inlined glyph paths
78 * (fontCache:'none') so they can be moved across documents.
79 *
80 * Accessibility: the SVG carries role="img" + aria-label with the source
81 * text (math marker syntax preserved), and an inner <title> element. Screen
82 * readers pick up either; sighted users don't see them as tooltips because
83 * we use aria-label instead of relying on title-as-tooltip behavior.
84 */
85// applyCodeFont (mono family for <code> runs) lives in lib/textStyle.mjs, shared
86// with the HTML export; imported above.

88export function buildTextElementSvgMarkup(
89 element: TextElement,
90 renderedHtml: string,
91 ctx: { fontFamily: string; fontSize: number; fontWeight: string; fontStyle: string; color: string; valign?: string; mono?: string },
92): string {
93 const textShadow = textShadowCss(element, ctx.color);
94 const box = textPresetBoxCss(element.preset);
95 const w = element.position.width;
96 const h = element.position.height;
97 // Alt text comes from the SOURCE element.html (with $..$), not the
98 // post-render html (which would have lost the math source to inline SVGs).
99 const alt = altTextFromHtml(element.html);
100 // Clip the OUTER box to its bounds (overflow="hidden" attribute, not just CSS —
101 // WebKit's UA stylesheet already forces overflow:hidden on <svg>/<foreignObject>,
102 // and only the presentation attribute can change it). Content that overflows the
103 // box is cut off, matching edit mode (SlideElementRenderer's edit wrapper) and
104 // the editor — true WYSIWYG. This also kills the present-mode "pop" (#79): with
105 // nothing overflowing, the opacity-fade's box-clipped compositor buffer has no
106 // out-of-box ink to reveal late. Italic/integral GLYPH ink overhang is preserved
107 // by the INNER MathJax <svg> keeping overflow:visible (see mathjaxRenderer.ts) —
108 // that ink stays within the box, so this outer clip doesn't touch it.
109 // Escape EVERY dynamic value spliced into an attribute or the style="" string:
110 // neutralize " < > & — the characters that break out of the quoted attribute. A
111 // crafted property (padding/fontFamily/color/geometry, or a size that came from an
112 // unvalidated config.textSizes) can't then inject markup into this privileged
113 // dangerouslySetInnerHTML output (audit C-2). escAttr leaves single-quotes alone, so
114 // legit CSS (single-quoted font stacks, #hex, px numbers) is byte-identical — WYSIWYG
115 // preserved. This is the sink defense that backs up the ingress normalizer and also
116 // protects the headless export, which shares this builder.
117 const e = (v: unknown): string => escAttr(String(v));
118 const eW = e(w), eH = e(h);
119 const outerStyle = `width:${eW}px;height:${eH}px;${e(valignToCss(ctx.valign))};overflow:hidden;box-sizing:border-box;`;
120 const innerStyle =
121 `width:100%;font-family:${e(ctx.fontFamily)};font-size:${e(ctx.fontSize)}px;` +
122 `font-weight:${e(ctx.fontWeight)};font-style:${e(ctx.fontStyle)};color:${e(ctx.color)};` +
123 `line-height:${e(box.lineHeight)};padding:${e(textPaddingCss(element, element.preset))};` +
124 `${textShadow ? `text-shadow:${e(textShadow)};` : ''}`;
125 return (
126 `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${eW} ${eH}" preserveAspectRatio="none" width="${eW}" height="${eH}" overflow="hidden" role="img" aria-label="${escAttr(alt)}" style="display:block;overflow:hidden;">` +
127 `<title>${escText(alt)}</title>` +
128 `<foreignObject x="0" y="0" width="${eW}" height="${eH}" overflow="hidden">` +
129 `<div xmlns="http://www.w3.org/1999/xhtml" style="${outerStyle}">` +
130 `<div style="${innerStyle}">` +
131 applyCodeFont(renderedHtml || '', ctx.mono) +
132 `</div>` +
133 `</div>` +
134 `</foreignObject>` +
135 `</svg>`
136 );
137}

152export function TextElementSvg({
153 element, slide, presentationTheme, presentationConfig,
154 className, zIndex, styleOverride,
155}: Props) {
156 const presetStyle = TEXT_PRESET_STYLES[element.preset];
157 const theme = resolveTheme(presentationTheme, slide.theme);
158 const themeColor = themeColorForPreset(theme, element.preset);
159 const presetFontPkg = fontForPreset(element.preset, slide, presentationConfig);
160 const presetFontFamily = fontFamilyForPreset(presetFontPkg, element.preset);
161 
162 const fontFamily = element.fontFamily || presetFontFamily;
163 const fontSize = effectiveFontSize(element, presentationConfig);
164 const fontWeight = presetStyle.fontWeight;
165 const fontStyle = presetStyle.fontStyle;
166 const color = resolveColor(element.color, theme, themeColor);
167 const mathBundleId = presetFontPkg.id;
168 
169 const valign = element.verticalAlign || (element.preset === 'title' || element.preset === 'footnote' ? 'bottom' : undefined);
170 
171 // Pre-render math via per-preset iframe pool. The INITIAL value pulls any
172 // already-cached SVGs synchronously (renderMathInHtmlSync) so warmed math
173 // paints in the FIRST frame — no "slide appears, then math fades in" flash
174 // (the projector warms its cache from SQLite, so this is a full hit there).
175 // Uncached expressions stay raw and the async effect below fills them in.
176 const [renderedHtml, setRenderedHtml] = useState<string>(() => {
177 if (!containsMathExpr(element.html)) return element.html || '';
178 return renderMathInHtmlSync(element.html, mathBundleId, presentationConfig.mathPreamble) ?? (element.html || '');
179 });
180 useEffect(() => {
181 let cancelled = false;
182 if (!containsMathExpr(element.html)) {
183 setRenderedHtml(element.html || '');
184 return () => { cancelled = true; };
185 }
186 renderMathInIframe(element.html, mathBundleId, presentationConfig.mathPreamble).then((html) => {
187 if (!cancelled) setRenderedHtml(html);
188 }).catch(() => {
189 if (!cancelled) setRenderedHtml(element.html || '');
190 });
191 return () => { cancelled = true; };
192 }, [element.html, mathBundleId]);
193 
194 const svgMarkup = buildTextElementSvgMarkup(element, renderedHtml, {
195 fontFamily, fontSize, fontWeight, fontStyle, color, valign,
196 mono: resolveMonoFontPackage(presentationConfig.defaultMonoFont).family,
197 });
198 
199 return (
200 <div
201 className={className}
202 style={{
203 position: 'absolute', left: element.position.x, top: element.position.y,
204 width: element.position.width, height: element.position.height,
205 backgroundColor: textBackgroundResolved(element, theme),
206 boxShadow: textBoxShadowCss(element),
207 ...(element.borderRadius ? { borderRadius: element.borderRadius } : {}),
208 ...(element.rotation ? { transform: `rotate(${element.rotation}deg)` } : {}),
209 ...(zIndex !== undefined ? { zIndex } : {}),
210 ...styleOverride,
211 }}
212 dangerouslySetInnerHTML={{ __html: svgMarkup }}
213 />
214 );
215}
