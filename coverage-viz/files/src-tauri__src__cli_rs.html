152fn strip_html(html: &str) -> String {
153 let mut result = String::new();
154 let mut in_tag = false;
155 for c in html.chars() {
156 if c == '<' { in_tag = true; }
157 else if c == '>' { in_tag = false; }
158 else if !in_tag { result.push(c); }
159 }
160 result.replace("&nbsp;", " ").replace("&amp;", "&").replace("&lt;", "<").replace("&gt;", ">")
161}

374 } else {
375 println!(" Slide {}, {} ({}): ...{}...", i + 1, &id[..8.min(id.len())], preset, &text[start..end]);
376 }

380 }
381 }
382 }

681fn cmd_import(db_path: &str, args: &[String]) -> Result<(), String> {
682 let format = args.first().map(|s| s.as_str()).ok_or("Usage: import json <input.json>")?;
683 if format != "json" { return Err("Usage: import json <input.json>".to_string()); }
684 let input = args.get(1).ok_or("Usage: import json <input.json>")?;
685 let content = std::fs::read_to_string(input).map_err(|e| format!("Failed to read {}: {}", input, e))?;
686 // Validate JSON
687 let _: Value = serde_json::from_str(&content).map_err(|e| format!("Invalid JSON: {}", e))?;
688 // Build the deck in a FRESH in-memory DB, then atomically replace the
689 // target file — same model as the GUI's New Project. db_import_json
690 // preserves assets, so we must NOT import in place over the existing
691 // file (its old assets would survive and collide by path, the dca9005
692 // bug); building in empty memory + atomic save gives a true clean slate.
693 // main() already opened db_path, and open_memory_db is a no-op while a
694 // DB is open, so close first to guarantee a fresh empty in-memory DB.
695 storage::close_db().map_err(|e| e.to_string())?;
696 storage::open_memory_db().map_err(|e| e.to_string())?;
697 storage::db_import_json(content)?;
698 storage::save_to_file(db_path)?;
699 println!("Imported from {}", input);
700 Ok(())
701}

753fn cmd_unpack(db_path: &str, args: &[String]) -> Result<(), String> {
754 let demos_only = args.iter().any(|a| a == "--demos");
755 let images_only = args.iter().any(|a| a == "--images");
756 let output_dir = args.iter()
757 .position(|a| a == "--output")
758 .and_then(|i| args.get(i + 1))
759 .map(|s| s.as_str());
760 
761 let base_dir = output_dir.unwrap_or_else(|| {
762 db_path.strip_suffix(".eigendeck").unwrap_or(db_path)
763 });
764 
765 if std::path::Path::new(base_dir).exists() && output_dir.is_none() {
766 return Err(format!("{} already exists. Use --output <dir>.", base_dir));
767 }
768 
769 // Get assets via raw SQL (db_get_asset only gets one at a time)
770 // This is acceptable since unpack is a bulk operation
771 let json_str = storage::db_export_json()?;
772 let p: Value = serde_json::from_str(&json_str).map_err(|e| e.to_string())?;
773 
774 // Collect (asset_id, path) for every asset referenced by a current
775 // element. Phase 4: paths live on the asset, not the element — go
776 // through assetId and look up the path via storage.
777 let mut asset_ids = std::collections::HashSet::new();
778 if let Some(slides) = p.get("slides").and_then(|v| v.as_array()) {
779 for s in slides {
780 if let Some(els) = s.get("elements").and_then(|v| v.as_array()) {
781 for el in els {
782 if let Some(id) = el.get("assetId").and_then(|v| v.as_str()) {
783 asset_ids.insert(id.to_string());
784 }
785 }
786 }
787 }
788 }
789 
790 let mut count = 0;
791 for asset_id in &asset_ids {
792 let meta = match storage::db_get_asset_meta_by_id(asset_id.clone()) {
793 Ok(Some(m)) => m,
794 Ok(None) => { eprintln!(" Warning: asset {} not found in DB", &asset_id[..8.min(asset_id.len())]); continue; }
795 Err(e) => { eprintln!(" Warning: asset meta lookup failed: {}", e); continue; }
796 };
797 let path = match meta.path.as_deref() {
798 Some(p) => p,
799 None => { eprintln!(" Warning: asset {} has no path label — skipping", &asset_id[..8.min(asset_id.len())]); continue; }
800 };
801 let is_demo = path.starts_with("demos/");
802 let is_image = path.starts_with("images/");
803 if demos_only && !is_demo { continue; }
804 if images_only && !is_image { continue; }
805 match storage::db_get_asset_bytes_by_id(asset_id.clone()) {
806 Ok(data) => {
807 let full_path = format!("{}/{}", base_dir, path);
808 let dir = std::path::Path::new(&full_path).parent().unwrap();
809 std::fs::create_dir_all(dir).map_err(|e| e.to_string())?;
810 std::fs::write(&full_path, &data).map_err(|e| e.to_string())?;
811 count += 1;
812 }
813 Err(_) => eprintln!(" Warning: asset {} bytes not found in DB", path),
814 }
815 }
816 println!("Unpacked {} files to {}", count, base_dir);
817 Ok(())
818}

834// UUID helper (minimal, no external crate)
835mod uuid {
836 use std::fmt;
837 pub struct Uuid;
838 impl Uuid {
839 pub fn new_v4() -> UuidResult {
840 use std::time::{SystemTime, UNIX_EPOCH};
841 let d = SystemTime::now().duration_since(UNIX_EPOCH).unwrap();
842 let a = d.as_nanos();
843 use std::sync::atomic::{AtomicU64, Ordering};
844 static CTR: AtomicU64 = AtomicU64::new(0);
845 let c = CTR.fetch_add(1, Ordering::Relaxed);
846 UuidResult(format!("{:08x}-{:04x}-4{:03x}-{:04x}-{:012x}",
847 ((a >> 96) as u32).wrapping_add(c as u32),
848 ((a >> 80) as u16) & 0xfff,
849 ((a >> 64) as u16) & 0xfff,
850 0x8000 | (((a >> 48) as u16) & 0x3fff),
851 (a as u64) & 0xffffffffffff))
852 }
853 }
854 pub struct UuidResult(String);
855 impl fmt::Display for UuidResult {
856 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
857 write!(f, "{}", self.0)
858 }
859 }
