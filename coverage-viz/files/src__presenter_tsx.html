4 * #3 — same viewer as the main window: this renders the REAL <PresentMode>
5 * (controlled by events), not a parallel renderer. It receives the

30function PresenterApp() {
31 const [ready, setReady] = useState(false);
32 const [index, setIndex] = useState(0);
33 const [windowed, setWindowed] = useState(false);
34 
35 useEffect(() => {
36 const unsubs: (() => void)[] = [];
37 (async () => {
38 unsubs.push(await listen<{ presentation: Presentation; currentIndex: number; projectPath: string | null; windowed?: boolean }>(
39 'presenter:init', async (event) => {
40 // Warm the math cache from the shared DB so PresentMode's text
41 // renders from cached SVGs (cold projector would otherwise re-render
42 // and time out on complex math — the raw-LaTeX spillover).
43 try { await warmMathCacheFromSqlite(); } catch { /* best effort */ }
44 // Populate THIS window's store so PresentMode reads the deck.
45 usePresentationStore.setState({
46 presentation: event.payload.presentation,
47 currentSlideIndex: event.payload.currentIndex,
48 projectPath: event.payload.projectPath,
49 });
50 setIndex(event.payload.currentIndex);
51 setWindowed(!!event.payload.windowed);
52 setReady(true);
53 }));
54 unsubs.push(await listen<{ index: number }>('presenter:goto', (event) => {
55 setIndex(event.payload.index);
56 }));
57 unsubs.push(await listen<{ presentation: Presentation }>('presenter:update', (event) => {
58 usePresentationStore.setState({ presentation: event.payload.presentation });
59 }));
60 await emitTo('main', 'presenter:ready', {});
61 })();
62 return () => { unsubs.forEach((fn) => fn()); };
63 }, []);
64 
65 const onExit = async () => {
66 await emitTo('main', 'presenter:closed', {});
67 const { getCurrentWindow } = await import('@tauri-apps/api/window');
68 await getCurrentWindow().close();

71 // Keyboard/clicker focused on the projector window: forward the requested
