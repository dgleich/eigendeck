1// Minimal in-memory representation of a Jupyter notebook (.ipynb).
2// We parse only the fields we render — full nbformat spec is huge,
3// and the renderer doesn't care about most of it.
4//
5// Reference: https://nbformat.readthedocs.io/en/latest/format_description.html
6//
7// One quirk worth knowing: source/text fields are EITHER a string OR
8// an array of strings (Jupyter writes one-line-per-array-entry to keep
9// diffs small). The parser normalizes both forms to a single string.
10 
11export type MimeBundle = Record<string, string | string[]>;
12 
13export interface CodeCell {
14 kind: 'code';
15 /** Stable index in the original cells[] array — used as a key and
16 * to address the cell when writing outputs/saving. */
17 index: number;
18 /** Cell source as a single string (newlines preserved). */
19 source: string;
20 /** Execution count shown in the `In [N]:` prompt. null if never run. */
21 executionCount: number | null;
22 outputs: CellOutput[];
23}
24 
25export interface MarkdownCell {
26 kind: 'markdown';
27 index: number;
28 source: string;
29}

31export interface RawCell {
32 kind: 'raw';

37export type Cell = CodeCell | MarkdownCell | RawCell;

39export type CellOutput =
40 | { kind: 'stream'; name: 'stdout' | 'stderr'; text: string }
41 | { kind: 'display_data'; data: MimeBundle }
42 | { kind: 'execute_result'; data: MimeBundle; executionCount: number | null }
43 | { kind: 'error'; ename: string; evalue: string; traceback: string[] };

45export interface Notebook {
46 cells: Cell[];
47 /** Kernelspec name as advertised in metadata, e.g. 'python3', 'julia-1.10'.
48 * null when the .ipynb lacks metadata.kernelspec.name. The element-level
49 * kernel field overrides this; we keep it only for sensible defaults. */
50 kernelspecName: string | null;
51 /** Display name as written by the authoring tool, for UI hints only. */
52 kernelDisplayName: string | null;
53 /** Language for syntax highlighting — from metadata.language_info.name
54 * when present, otherwise inferred from kernelspecName, otherwise null. */
55 language: string | null;
56}

58/** Coerce nbformat's "string or array of strings" multiline fields to a
59 * single string. */
60export function joinMultiline(v: unknown): string {
61 if (Array.isArray(v)) return v.join('');
62 if (typeof v === 'string') return v;
63 return '';
