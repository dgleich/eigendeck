11/** Open (or focus) the Settings window. `tab` deep-links to a section — a fresh
12 * window carries it in the URL hash; an already-open window is told via the
13 * `settings-tab` event (settings.tsx bridges it to the panel). */
14export async function openSettingsWindow(tab?: SettingsTab): Promise<void> {
15 const existing = await WebviewWindow.getByLabel('settings');
16 if (existing) {
17 // Raise it (needs core:window:allow-{unminimize,show,set-focus} in the
18 // capability — same perms the security window relies on).
19 await existing.unminimize();
20 await existing.show();
21 await existing.setFocus();
22 if (tab) await existing.emit('settings-tab', tab);
23 return;
24 }
