4// dylib is loaded on first call from the Tauri resource bundle. With

29/// Locate the bundled pdfium dylib. In `cargo tauri dev` this resolves
30/// to `src-tauri/resources/pdfium/libpdfium.dylib`; in a packaged .app
31/// it lands inside `Contents/Resources/resources/pdfium/`.
32///
33/// Tauri's `path().resource_dir()` is the right anchor for both, with
34/// per-platform filename branches (handled in build.rs for the download
35/// half — the runtime filename is hard-coded here to match what we ship).
36fn resolve_pdfium_dylib(app: &tauri::AppHandle) -> Result<PathBuf, String> {
37 let resource_dir = app.path().resource_dir()
38 .map_err(|e| format!("resource_dir() failed: {}", e))?;
39 
40 // bundle.resources = ["resources/pdfium/*"] mirrors the on-disk
41 // layout, so the relative path inside resource_dir matches.
42 let filename = if cfg!(target_os = "windows") {
43 "pdfium.dll"
44 } else if cfg!(target_os = "macos") {
45 "libpdfium.dylib"
46 } else {
47 "libpdfium.so"
48 };
49 let candidate = resource_dir.join("resources").join("pdfium").join(filename);
50 if candidate.exists() {
51 return Ok(candidate);
52 }
53 
54 // Fallback: some tauri-dev modes don't prefix `resources/` — check
55 // the flat form too.
56 let flat = resource_dir.join(filename);
57 if flat.exists() {
58 return Ok(flat);
59 }
60 
61 Err(format!(
62 "pdfium dylib not found. Checked: {} and {}. \
63 Make sure build.rs ran (downloads bblanchon prebuilt).",
64 candidate.display(), flat.display(),
65 ))
66}

77fn get_pdfium(app: &tauri::AppHandle) -> Result<&'static Pdfium, String> {
78 let init = PDFIUM.get_or_init(|| {
79 let lib_path = resolve_pdfium_dylib(app)?;
80 let bindings = Pdfium::bind_to_library(&lib_path)
81 .map_err(|e| format!("Pdfium::bind_to_library({}) failed: {}", lib_path.display(), e))?;
82 Ok(Pdfium::new(bindings))
83 });
84 init.as_ref().map_err(|e| e.clone())
85}

87/// Pure render: given a Pdfium instance and PDF bytes, rasterize the
88/// requested page to PNG bytes aspect-fit into (max_width, max_height).
89/// Extracted from db_render_pdf_page so it can be tested without a
90/// Tauri AppHandle (the smoke test below does exactly that).
91fn render_page_inner(
92 pdfium: &Pdfium,
93 pdf_bytes: &[u8],
94 page: u32,
95 max_width: u32,
96 max_height: u32,
97) -> Result<Vec<u8>, String> {
98 let t_load = std::time::Instant::now();
99 let document = pdfium
100 .load_pdf_from_byte_slice(pdf_bytes, None)
101 .map_err(|e| format!("load_pdf_from_byte_slice: {}", e))?;
102 plog!("[pdf] load_pdf_from_byte_slice ({}KB): {}ms",
103 pdf_bytes.len() / 1024, t_load.elapsed().as_millis());
104 
105 let pdf_page = document.pages().get(page as PdfPageIndex)
106 .map_err(|e| format!("page {} get: {}", page, e))?;
107 
108 // Aspect-fit into (max_width, max_height) WITHOUT auto-rotating
109 // landscape pages — embedded slide images should keep their source
110 // orientation. set_target_width + the two maximums together produce
111 // "fit inside the box, never upscale past natural" without
112 // scale_page_to_display_size's implicit 90° landscape rotation.
113 //
114 // Transparent clear color (alpha=0) preserves the PDF's actual
115 // background. Default pdfium-render clears to opaque white, which
116 // turns transparent-bg figure exports (Illustrator, Matplotlib,
117 // Inkscape "save selection as PDF") into white-backgrounded blocks
118 // on the slide. Pdfs with explicit white backgrounds still render
119 // white — only the implicit fill changes.
120 let config = PdfRenderConfig::new()
121 .set_target_width(max_width as Pixels)
122 .set_maximum_width(max_width as Pixels)
123 .set_maximum_height(max_height as Pixels)
124 .set_clear_color(PdfColor::new(0, 0, 0, 0));
125 
126 let t_render = std::time::Instant::now();
127 let bitmap = pdf_page.render_with_config(&config)
128 .map_err(|e| format!("render page {}: {}", page, e))?;
129 plog!("[pdf] render_with_config ({}x{}): {}ms",
130 max_width, max_height, t_render.elapsed().as_millis());
131 
132 // Skip the as_image() detour — pdfium already gives us RGBA bytes
133 // directly. Going through DynamicImage adds a 14MB-class clone for
134 // a 1920² page that we don't need.
135 let t_bytes = std::time::Instant::now();
136 let (w, h) = (bitmap.width() as u32, bitmap.height() as u32);
137 let rgba = bitmap.as_rgba_bytes();
138 plog!("[pdf] as_rgba_bytes ({}KB): {}ms",
139 rgba.len() / 1024, t_bytes.elapsed().as_millis());
140 
141 // Default zlib compression. Fast-mode produced 10MB outputs (50x
142 // bigger), which slowed the downstream SQLite write more than the
143 // encoder saved. Default (level 6) is the right balance for rendered
144 // PDF pages: ~200KB output, ~200ms encode at 1920².
145 let t_png = std::time::Instant::now();
146 let mut png_bytes: Vec<u8> = Vec::new();
147 {
148 use image::codecs::png::{CompressionType, FilterType, PngEncoder};
149 use image::ImageEncoder;
150 let encoder = PngEncoder::new_with_quality(
151 &mut png_bytes,
152 CompressionType::Default,
153 FilterType::NoFilter,
154 );
155 encoder.write_image(&rgba, w, h, image::ExtendedColorType::Rgba8)
156 .map_err(|e| format!("encode png: {}", e))?;
157 }
158 plog!("[pdf] encode_png ({}KB out): {}ms",
159 png_bytes.len() / 1024, t_png.elapsed().as_millis());
160 
161 Ok(png_bytes)
162}

164/// Render one page of a stored PDF asset to PNG bytes, aspect-fit into
165/// (max_width, max_height). `page` is 0-indexed.
166///
167/// Bytes come from `storage::db_get_asset_by_id` so this respects the
168/// usual asset lifecycle (current version, no path lookup, watcher /
169/// restore-aware). Caller (assetRenderer.ts) writes the result into
170/// asset_cache under the assetId key.
171// async fn + spawn_blocking — pdfium can take 40+ seconds on pathological
172// PDFs (Form-XObject-heavy vector exports). Tauri 2 sync `pub fn` commands
173// run on the WebView main thread, which blocks the compositor — even
174// though JS keeps running (setTimeouts fire), the screen can't repaint,
175// producing a beachball + queued visual updates. spawn_blocking moves
176// the pdfium parse/render onto tokio's blocking thread pool, leaving the
177// main thread free for paint commits and IPC dispatch.
178#[tauri::command]
179pub async fn db_render_pdf_page(
180 app: tauri::AppHandle,
181 asset_id: String,
182 page: u32,
183 max_width: u32,
184 max_height: u32,
185) -> Result<tauri::ipc::Response, String> {
186 let png = tauri::async_runtime::spawn_blocking(move || -> Result<Vec<u8>, String> {
187 let t_total = std::time::Instant::now();
188 plog!("[pdf] db_render_pdf_page asset={} page={} {}x{}",
189 &asset_id[..8.min(asset_id.len())], page, max_width, max_height);
190 
191 let t_bind = std::time::Instant::now();
192 let pdfium = get_pdfium(&app)?;
193 plog!("[pdf] get_pdfium (bind): {}ms", t_bind.elapsed().as_millis());
194 
195 let t_fetch = std::time::Instant::now();
196 let bytes = storage::db_get_asset_bytes_by_id(asset_id.clone())?;
197 plog!("[pdf] db_get_asset_by_id ({}KB): {}ms",
198 bytes.len() / 1024, t_fetch.elapsed().as_millis());
199 
200 let png = render_page_inner(pdfium, &bytes, page, max_width, max_height)
201 .map_err(|e| format!("{}: {}", asset_id, e))?;
202 plog!("[pdf] db_render_pdf_page TOTAL: {}ms ({}KB PNG)",
203 t_total.elapsed().as_millis(), png.len() / 1024);
204 Ok(png)
205 })
206 .await
207 .map_err(|e| format!("spawn_blocking join: {}", e))??;
208 Ok(tauri::ipc::Response::new(png))
209}

211/// Number of pages in a stored PDF asset. Cheap (parses header, doesn't
212/// rasterize). Used by the inspector "page 1 of N" hint.
213#[tauri::command]
214pub fn db_pdf_page_count(
215 app: tauri::AppHandle,
216 asset_id: String,
217) -> Result<u32, String> {

220 let document = pdfium

222 .map_err(|e| format!("load_pdf_from_byte_slice for {}: {}", asset_id, e))?;

224}
