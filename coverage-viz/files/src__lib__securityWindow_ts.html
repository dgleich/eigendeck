12export async function openSecurityWindow(): Promise<void> {
13 const store = usePresentationStore.getState();
14 const payload = { presentation: store.presentation, projectPath: store.projectPath };
15 
16 const existing = await WebviewWindow.getByLabel('security');
17 if (existing) {
18 // Raise it to the top: unminimize (if minimized) + show (if hidden) + focus.
19 // These require core:window:allow-{unminimize,show,set-focus} in the capability
20 // (default.json) — without them the calls are DENIED by the ACL, which is what
21 // made this silently do nothing before. Don't swallow errors: a rejected call
22 // means a missing permission, and we want that to surface, not hide.
23 await existing.unminimize();
24 await existing.show();
25 await existing.setFocus();
26 await emitTo('security', 'security:init', payload);
27 return;
28 }

30 // Register the ready-handshake listener BEFORE creating the window. The child
31 // emits `security:ready` on mount; if we only start listening after creating it,
32 // a fast child can fire first and we miss it — leaving the window stuck on
33 // "Loading…" until a retrigger (the transient blank).
34 //
35 // Send `security:init` EXACTLY ONCE. Each init bumps the child's initKey and
36 // REMOUNTS the app, so a second (redundant) init makes the window visibly blink.
37 // The 1500ms timer is a FALLBACK for a missed ready event only — it no-ops once
38 // the handshake has already sent.
39 let sent = false;
40 const sendInit = () => { if (sent) return; sent = true; void emitTo('security', 'security:init', payload).catch(() => {}); };
41 const unlisten = await listen('security:ready', () => sendInit());
42 setTimeout(() => { unlisten(); }, 15000);
43 
44 new WebviewWindow('security', {
45 url: '/security.html',
46 title: 'Security & linked files',
47 width: 760,
48 height: 660,
49 resizable: true,
50 focus: true,
51 });
