1//! Loopback HTTP shim that hosts ONLY a YouTube `<iframe>`.
2//!
3//! The packaged app runs at origin `tauri://localhost`, which YouTube's embedded
4//! player rejects (it requires a valid http(s) origin/Referer — see
5//! docs/youtube-embed-shim.md). This server lets the app present YouTube by
6//! framing `http://127.0.0.1:<port>/yt/<token>/<id>`, a page served here whose
7//! http origin YouTube accepts. The main app window stays on `tauri://localhost`,
8//! so Tauri IPC/capabilities are untouched.
9//!
10//! Hardening (all enforced below): binds `127.0.0.1` only; a single exact route;
11//! a 256-bit unguessable per-launch token in the path; a Host-header allowlist
12//! (anti-DNS-rebinding); a strict 11-char YouTube-id allowlist; GET only; NO CORS
13//! header (cross-site JS gets an opaque body); a restrictive response CSP +
14//! `nosniff` + `no-store`; no filesystem access; a panic-safe handler. The only
15//! attacker-influenced input, the id, is validated to `[A-Za-z0-9_-]{11}` before
16//! it ever reaches HTML, and the option flags are parsed as strict booleans.
17

31/// A real YouTube video id is exactly 11 chars from `[A-Za-z0-9_-]`.
32fn valid_youtube_id(id: &str) -> bool {
33 id.len() == 11
34 && id
35 .bytes()
36 .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'_')
37}

43/// `?a=1&b=0` → is `key` set to exactly "1"? Anything else is false.
44fn flag(query: &str, key: &str) -> bool {
45 query.split('&').any(|kv| {
46 let mut it = kv.splitn(2, '=');
47 it.next() == Some(key) && it.next() == Some("1")
48 })
49}

51/// Build the real YouTube embed URL from a (validated) id + option flags. Mirrors
52/// the YouTube branch of `buildEmbedSrc` (src/lib/videoEmbedParse.mjs), minus
53/// `enablejsapi` — that needs a valid http origin the packaged app lacks and is
54/// part of why the embed failed in the first place.
55fn youtube_embed_url(id: &str, query: &str) -> String {
56 let autoplay = flag(query, "autoplay");
57 let show_controls = flag(query, "controls") || !autoplay;
58 let mut p: Vec<String> = Vec::new();
59 if autoplay {
60 p.push("autoplay=1".into());
61 }
62 if flag(query, "mute") {
63 p.push("mute=1".into());
64 }
65 if flag(query, "loop") {
66 p.push("loop=1".into());
67 p.push(format!("playlist={id}")); // single-video loop needs playlist=id
68 }
69 p.push(format!("controls={}", if show_controls { 1 } else { 0 }));
70 if flag(query, "captions") {
71 p.push("cc_load_policy=1".into());
72 }
73 p.push("rel=0".into());
74 format!(
75 "https://www.youtube-nocookie.com/embed/{id}?{}",
76 p.join("&")
77 )
78}

80/// The one page this server serves: a full-bleed YouTube iframe. `embed_url`
81/// contains only the validated 11-char id + our own fixed params, so nothing
82/// attacker-controlled reaches the HTML.
83fn page(embed_url: &str) -> String {
84 format!(
85 "<!doctype html><html><head><meta charset=\"utf-8\">\
86<meta name=\"referrer\" content=\"strict-origin-when-cross-origin\">\
87<style>html,body{{margin:0;height:100%;background:#000}}iframe{{display:block;border:0;width:100%;height:100%}}</style>\
88</head><body><iframe src=\"{embed_url}\" \
89allow=\"autoplay; encrypted-media; fullscreen; picture-in-picture\" \
90allowfullscreen referrerpolicy=\"strict-origin-when-cross-origin\"></iframe></body></html>"
91 )
92}

94/// Start the shim on `127.0.0.1:<ephemeral>`. Returns the [`ShimState`] carrying
95/// the base URL with a fresh per-launch token. Spawns a detached handler thread;
96/// the server is torn down when the process exits.
97pub fn start() -> std::io::Result<ShimState> {
98 let server = Server::http("127.0.0.1:0")
99 .map_err(|e| std::io::Error::other(e.to_string()))?;
100 let port = server
101 .server_addr()
102 .to_ip()
103 .map(|a| a.port())
104 .unwrap_or(0);
105 // 256-bit token (two v4 UUIDs), comfortably beyond the 128-bit floor. Held in
106 // memory only — never logged, persisted, or placed in argv.
107 let token = format!("{}{}", Uuid::new_v4().simple(), Uuid::new_v4().simple());
108 let allowed_hosts: Arc<[String]> =
109 Arc::from(vec![format!("127.0.0.1:{port}"), format!("localhost:{port}")]);
110 let route_prefix = format!("/yt/{token}/");
111 let base = format!("http://127.0.0.1:{port}/yt/{token}");
112 
113 std::thread::spawn(move || {
114 for request in server.incoming_requests() {
115 let hosts = allowed_hosts.clone();
116 let prefix = route_prefix.clone();
117 // Never let a malformed request panic the handler thread / app.
118 let _ = std::panic::catch_unwind(std::panic::AssertUnwindSafe(move || {
119 handle(request, &hosts, &prefix);
120 }));
121 }
122 });
123 Ok(ShimState { base })
124}

126fn deny(request: tiny_http::Request, code: u16) {
127 let _ = request.respond(Response::empty(code));
128}

130fn handle(request: tiny_http::Request, allowed_hosts: &[String], route_prefix: &str) {
131 // GET only.
132 if request.method() != &Method::Get {
133 return deny(request, 405);
134 }
135 // Host-header allowlist (anti-DNS-rebinding).
136 let host_ok = request.headers().iter().any(|h| {
137 h.field.equiv("Host") && allowed_hosts.iter().any(|a| a.as_str() == h.value.as_str())
138 });
139 if !host_ok {
140 return deny(request, 403);
141 }
142 // Exact route: /yt/<token>/<id>, with an optional ?flags query.
143 let url = request.url().to_string();
144 let (path, query) = url.split_once('?').unwrap_or((url.as_str(), ""));
145 let id = match path.strip_prefix(route_prefix) {
146 Some(rest) => rest,
147 None => return deny(request, 404),
148 };
149 if !valid_youtube_id(id) {
150 return deny(request, 400);
151 }
152 
153 let body = page(&youtube_embed_url(id, query));
154 let response = Response::from_string(body)
155 .with_header(header("Content-Type", "text/html; charset=utf-8"))
156 // The shim page's own tight policy: it may frame only YouTube, nothing else.
157 .with_header(header(
158 "Content-Security-Policy",
159 "default-src 'none'; frame-src https://www.youtube-nocookie.com https://www.youtube.com; style-src 'unsafe-inline'; img-src data:",
160 ))
161 .with_header(header("X-Content-Type-Options", "nosniff"))
162 .with_header(header("Cache-Control", "no-store"));
163 // Deliberately NO Access-Control-Allow-Origin: a cross-site page that guesses
164 // the token still cannot READ the response body.
165 let _ = request.respond(response);
166}
