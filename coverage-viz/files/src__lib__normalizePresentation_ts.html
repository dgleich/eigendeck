1// Fail-closed normalization for UNTRUSTED presentation content (audit H-1 + C-2).
2//
3// A .eigendeck is attacker-controllable data, and element `data` is parsed as
4// arbitrary JSON with no runtime schema check — TypeScript types are compile-time
5// only. Two sinks make that dangerous:
6// - element.html is rendered via dangerouslySetInnerHTML (the rich-text sanitizer
7// handles this — see sanitizeRichText); and
8// - element PROPERTIES (fontFamily, color, geometry) are string-interpolated into
9// the SVG/markup that buildTextElementSvgMarkup produces and then
10// dangerouslySetInnerHTML'd in the PRIVILEGED frame. A crafted fontFamily/color
11// containing a `"`/`<`/`>` breaks out of the style attribute and injects a tag
12// (e.g. <img onerror>), so a payload in a property bypasses the html sanitizer
13// and fires automatically on display (C-2).
14//
15// This is the ONE boundary to run at EVERY untrusted ingress (deck open/import,
16// undo-seed, history restore, clipboard paste). It sanitizes text html and validates
17// each property against its known-safe shape, DROPPING any element whose values fall
18// outside it (fail-closed). It is transparent to legitimate decks: real fonts/colors/
19// numbers all pass, so nothing a normal deck contains is altered or dropped.

24// Characters that let a value escape the CSS declaration / HTML style attribute /
25// SVG attribute it's interpolated into: close the attribute ("), end/inject a CSS
26// declaration (; { }), open/close a tag (< >), or use an escape (\ `). Legit CSS
27// font stacks ('PT Sans', sans-serif) and colors (#0af, rgb(0,0,0), red) contain
28// NONE of these — they use only letters, digits, spaces, commas, dots, %, ()
29// (rgb/hsl), hyphens and single-quotes. Control chars (< 0x20, 0x7f) are rejected too.
30const CSS_BREAKOUT = new Set(['<', '>', '"', '`', ';', '{', '}', '\\']);
31 
32/** A string safe to interpolate into a CSS value / style / SVG attribute (no breakout). */
33export function isSafeCssValue(s: unknown): boolean {
34 if (typeof s !== 'string') return false;
35 for (let i = 0; i < s.length; i++) {
36 const code = s.charCodeAt(i);
37 if (code < 0x20 || code === 0x7f) return false; // control chars
38 if (CSS_BREAKOUT.has(s[i])) return false; // breakout chars
39 }
40 return true;
41}

43// A REAL CSS color: hex, an rgb/rgba/hsl/hsla function with numeric args, or a bare
44// keyword (named colors, `transparent`, `currentColor`, theme tokens like `accent`).
45// Deliberately NOT breakout-char filtering — `isSafeCssValue` accepts `url(...)`, which
46// in a `color`/`background` field loads a network resource (a beacon in an exported
47// artifact). This rejects `url(`/`@import`/expressions and only permits real colors.
48const COLOR_HEX = /^#[0-9a-fA-F]{3,8}$/;
49const COLOR_FUNC = /^(?:rgb|rgba|hsl|hsla)\([0-9.,%\s/]*\)$/i;
50const COLOR_KEYWORD = /^[a-zA-Z][a-zA-Z0-9-]*$/;
51 
52/** True when `v` is a genuine CSS color (not `url()`/`@import`/injection). */
53export function isSafeColor(v: unknown): boolean {
54 if (typeof v !== 'string') return false;
55 const s = v.trim();
56 if (s === '') return false;
57 return COLOR_HEX.test(s) || COLOR_FUNC.test(s) || COLOR_KEYWORD.test(s);
58}

64/**
65 * Validate ONE untrusted element. Returns the element (possibly with its html
66 * sanitized in place) if it is safe to render, or `null` if it must be dropped
67 * (a value outside its known-safe shape).
68 */
69export function normalizeUntrustedElement(el: SlideElement): SlideElement | null {
70 // Never THROW on malformed structure (null / non-object / a string where an object
71 // is expected) — drop it. A throw would abort the whole normalize pass mid-slide and
72 // (per the non-fatal deck-open catch) leave EARLIER unsafe elements installed.
73 if (!el || typeof el !== 'object') return null;
74 
75 // Geometry is string-concatenated into text SVG (viewBox/width/height) and used
76 // as layout everywhere; require finite numbers for every element type.
77 const p = (el as { position?: unknown }).position as { x?: unknown; y?: unknown; width?: unknown; height?: unknown } | undefined;
78 if (!p || typeof p !== 'object' || !isFiniteNum(p.x) || !isFiniteNum(p.y) || !isFiniteNum(p.width) || !isFiniteNum(p.height)) return null;
79 
80 // `padding`, when present, is spliced into the SVG style (textPaddingCss) — every
81 // side must be a finite number.
82 const pad = (el as { padding?: unknown }).padding;
83 if (pad != null) {
84 const pp = pad as { top?: unknown; right?: unknown; bottom?: unknown; left?: unknown };
85 if (typeof pad !== 'object' || !isFiniteNum(pp.top) || !isFiniteNum(pp.right) || !isFiniteNum(pp.bottom) || !isFiniteNum(pp.left)) return null;
86 }
87 
88 // `color` and `backgroundColor` reach CSS `color:`/`background:` — they must be REAL
89 // colors, not `url()`/`@import` (which would load a network resource in an export
90 // artifact) and not a breakout. Validate wherever present (a non-empty string).
91 const anyEl = el as unknown as Record<string, unknown> & { fontFamily?: unknown; fontSize?: unknown; html?: unknown; type?: string };
92 if (typeof anyEl.color === 'string' && (anyEl.color as string).trim() !== '' && !isSafeColor(anyEl.color)) return null;
93 if (typeof anyEl.backgroundColor === 'string' && (anyEl.backgroundColor as string).trim() !== '' && !isSafeColor(anyEl.backgroundColor)) return null;
94 
95 // Optional numeric visual fields that reach a CSS/SVG numeric (opacity, rotate(),
96 // border-radius, rgba() alpha, arrow stroke/head) — finite if present, else drop.
97 for (const k of ['rotation', 'borderRadius', 'opacity', 'backgroundOpacity', 'strokeWidth', 'headSize']) {
98 if (anyEl[k] != null && !isFiniteNum(anyEl[k])) return null;
99 }
100 
101 if (anyEl.type === 'text') {
102 if (anyEl.fontFamily != null && !isSafeCssValue(anyEl.fontFamily)) return null;
103 if (anyEl.fontSize != null && !isFiniteNum(anyEl.fontSize)) return null;
104 if (typeof anyEl.html === 'string') anyEl.html = sanitizeRichText(anyEl.html);
105 }
106 return el;
107}

109/**
110 * Normalize a whole untrusted presentation IN PLACE: sanitize text html and drop
111 * every element whose properties fall outside their known-safe shape. Idempotent.
112 * Returns the number of elements dropped (0 for a clean deck).
113 */
114export function normalizeUntrustedPresentation(presentation: {
115 slides?: unknown;
116 config?: { textSizes?: Record<string, unknown> };
117}): number {
118 let dropped = 0;
119 const slides = Array.isArray(presentation.slides) ? (presentation.slides as Array<{ elements?: unknown }>) : [];
120 for (const slide of slides) {
121 if (!slide || !Array.isArray(slide.elements)) continue;
122 const els = slide.elements as SlideElement[];
123 const kept = els.filter((el) => normalizeUntrustedElement(el) !== null);
124 if (kept.length !== els.length) {
125 dropped += els.length - kept.length;
126 (slide as { elements: SlideElement[] }).elements = kept;
127 }
128 }
129 // Deck-level config.textSizes feeds effectiveFontSize() → the same SVG style, so a
130 // named size (element.fontSizeName / a preset's size) can resolve to an UNTRUSTED
131 // value here. Drop any non-finite entry so it can never reach the sink as a string;
132 // effectiveFontSize then falls back to the built-in default for that name.
133 const ts = presentation.config?.textSizes;
134 if (ts && typeof ts === 'object') {
135 for (const k of Object.keys(ts)) if (!isFiniteNum((ts as Record<string, unknown>)[k])) delete (ts as Record<string, unknown>)[k];
136 }
137 return dropped;
138}
