1//! Native macOS NSToolbar for the main window. Uses the Expanded toolbar style,
2//! so the window's native title + proxy icon render CENTERED on their own row
3//! (the BBEdit/Keynote look) — giving the proxy path popover, drag-to-share, and
4//! the edited dot for free — with the toolbar buttons/fields in the row below:
5//! Add Slide / Add Build / Save on the left, Author / Venue / Export / Present on
6//! the right. The title/proxy are driven from lib.rs `set_window_document`
7//! (setRepresentedURL + setTitle + setDocumentEdited).
8//!
9//! - BUTTON items post a Rust→JS `toolbar:action` event; the frontend runs the
10//! same action as the HTML button (dispatchToolbarAction).
11//! - The Author/Venue TEXT FIELDS post `toolbar:field` {id,value} on edit; the
12//! frontend writes them to config. set_fields() pushes current values back.
13//!
14//! SPIKE — behind the `mac-toolbar` cargo feature (off by default).

39/// Per-glyph vertical nudge (points) so off-balance SF Symbols read as centered
40/// against the boxy neighbors (Jupyter, Present). POSITIVE = up, NEGATIVE = down.
41/// Separate values for normal vs compact since the glyph is smaller in compact.
42/// Export (square.and.arrow.up): arrow above the box → box sinks → nudge UP.
43/// Save (square.and.arrow.down): arrow below the box → box rides high → nudge DOWN.
44const EXPORT_NUDGE_REGULAR: f64 = 8.0;
45const EXPORT_NUDGE_COMPACT: f64 = 4.0;

49fn nudge_for(symbol: &str, compact: bool) -> f64 {
50 match symbol {
51 "square.and.arrow.up" => if compact { EXPORT_NUDGE_COMPACT } else { EXPORT_NUDGE_REGULAR },
52 "square.and.arrow.down" => if compact { SAVE_NUDGE_COMPACT } else { SAVE_NUDGE_REGULAR },
53 _ => 0.0,
54 }

128/// Return a copy of `src` grown by |dy| points of transparent space so the glyph
129/// shifts when the toolbar centers it. dy > 0 pads the BOTTOM (glyph moves UP);
130/// dy < 0 pads the TOP (glyph moves DOWN). Preserves template rendering (so it
131/// still tints for light/dark). Main thread (lockFocus).
132#[allow(deprecated)] // lockFocus/unlockFocus: fine for this small compositing use
133fn nudge_image(src: &NSImage, dy: f64) -> Retained<NSImage> {
134 let s = src.size();
135 let extra = dy.abs();
136 let out = NSImage::initWithSize(
137 NSImage::alloc(),
138 NSSize { width: s.width, height: s.height + extra },
139 );
140 out.setTemplate(src.isTemplate());
141 // Non-flipped image space (origin bottom-left). Up: draw high (pad at bottom).
142 // Down: draw at 0 (pad at top).
143 let draw_y = if dy >= 0.0 { extra } else { 0.0 };
144 out.lockFocus();
145 src.drawAtPoint_fromRect_operation_fraction(
146 NSPoint { x: 0.0, y: draw_y },
147 NSRect::new(NSPoint::new(0.0, 0.0), s),
148 NSCompositingOperation::SourceOver,
149 1.0,
150 );
151 out.unlockFocus();
152 out
153}

174 fn drawing_rect_for_bounds(&self, bounds: NSRect) -> NSRect {
175 let mut r: NSRect = unsafe { msg_send![super(self), drawingRectForBounds: bounds] };

242/// Apply the visible chrome for a BUTTON item. Shared by the initial build
243/// (item_for) and the live toggles (set_compact / set_save_dirty) so all three
244/// produce identical styling.
245///
246/// - `compact`: hide the visible label and shrink the icon (paletteLabel + toolTip
247/// keep the name available for a11y/hover).
248/// - `dirty`: tint the glyph with the system accent color — used on the Save item
249/// to flag unsaved changes (mirrors the title-bar edited dot).
250fn style_button(item: &NSToolbarItem, label: &str, symbol: &str, compact: bool, dirty: bool) {
251 let ns_label = NSString::from_str(label);
252 // Normal mode shows the label under the icon; compact clears the text (icon
253 // only). paletteLabel + toolTip keep the name available regardless.
254 item.setLabel(&NSString::from_str(if compact { "" } else { label }));
255 item.setPaletteLabel(&ns_label);
256 item.setToolTip(Some(&ns_label));
257 if let Some(image) = NSImage::imageWithSystemSymbolName_accessibilityDescription(
258 &NSString::from_str(symbol),
259 Some(&ns_label),
260 ) {
261 let point = if compact { ICON_POINT_SIZE_COMPACT } else { ICON_POINT_SIZE_REGULAR };
262 let cfg = unsafe {
263 NSImageSymbolConfiguration::configurationWithPointSize_weight_scale(
264 point,
265 NSFontWeightRegular,
266 // Large — the label row is controlled by the toolbar displayMode, not
267 // image size (#125), so keep the bigger glyph.
268 NSImageSymbolScale::Large,
269 )
270 };
271 // Accent-tint the dirty Save icon; otherwise leave it template (adapts to
272 // light/dark). Combine the size + color configs.
273 let cfg = if dirty {
274 let color = NSColor::controlAccentColor();
275 let color_cfg = NSImageSymbolConfiguration::configurationWithHierarchicalColor(&color);
276 cfg.configurationByApplyingConfiguration(&color_cfg)
277 } else {
278 cfg
279 };
280 let sized = image.imageWithSymbolConfiguration(&cfg).unwrap_or(image);
281 // Nudge off-balance glyphs (Export up, Save down) so their boxes align.
282 let dy = nudge_for(symbol, compact);
283 let sized = if dy != 0.0 { nudge_image(&sized, dy) } else { sized };
284 item.setImage(Some(&sized));
285 }
286 // Always borderless: a BORDERED plain toolbar item renders icon-only (the label
287 // is dropped for the capsule chrome). Borderless is what shows icon+label.
288 item.setBordered(false);
289}

630/// Toggle compact view live, IN PLACE (no item rebuild). Compact = labels off +
631/// smaller icons via a shorter toolbar style (UnifiedCompact vs Expanded). We
632/// restyle the existing items (label/image) and switch the window toolbar style;
633/// setLabel/setImage relayout the items, and validateVisibleItems nudges a
634/// re-measure. Main thread.
635pub fn set_compact(compact: bool) {
636 DELEGATE.with(|d| {
637 let Some(del) = d.borrow().as_ref().cloned() else { return };
638 del.ivars().compact.set(compact);
639 
640 // Switch the window toolbar style (Expanded = tall w/ centered title row;
641 // UnifiedCompact = short single row).
642 if let Some(win) = del.ivars().app.get_webview_window("main") {
643 if let Ok(ptr) = win.ns_window() {
644 // Safety: Tauri owns a valid NSWindow for "main" for its lifetime.
645 let ns_win: &NSWindow = unsafe { &*(ptr as *const NSWindow) };
646 let style = if compact {
647 NSWindowToolbarStyle::UnifiedCompact
648 } else {
649 NSWindowToolbarStyle::Expanded
650 };
651 ns_win.setToolbarStyle(style);
652 }
653 }
654 
655 // Restyle the existing items in place (label text + image size).
656 restyle_buttons(&del);
657 
658 // THE label-row toggle: displayMode. UnifiedCompact forces the toolbar to
659 // IconOnly and switching back to Expanded does NOT restore it — so set it
660 // explicitly. IconAndLabel turns the label row back on (#125).
661 TOOLBAR.with(|t| {
662 if let Some(tb) = t.borrow().as_ref() {
663 tb.setDisplayMode(if compact {
664 NSToolbarDisplayMode::IconOnly
665 } else {
666 NSToolbarDisplayMode::IconAndLabel
667 });
668 tb.validateVisibleItems();
669 }
670 });
671 });
672}

690/// Update the Jupyter server-status icon. status "gray" = the deck has no live
691/// notebooks → the item is REMOVED entirely (mirrors the HTML pill hiding). Any
692/// other status inserts it (before Export) if absent and tints it. Main thread.
693pub fn set_jupyter(status: &str, tooltip: &str) {
694 DELEGATE.with(|d| {
695 let Some(del) = d.borrow().as_ref().cloned() else { return };
696 *del.ivars().jupyter_status.borrow_mut() = status.to_string();
697 *del.ivars().jupyter_tooltip.borrow_mut() = tooltip.to_string();
698 let visible = status != "gray";
699 TOOLBAR.with(|t| {
700 let Some(tb) = t.borrow().as_ref().cloned() else { return };
701 match (visible, item_index(&tb, JUPYTER_ID)) {
702 // Show: insert before Export (item_for styles it from the status
703 // we just stored). Fall back to the end if Export isn't found.
704 (true, None) => {
705 let idx = item_index(&tb, "export").unwrap_or_else(|| tb.items().count());
706 tb.insertItemWithItemIdentifier_atIndex(&NSString::from_str(JUPYTER_ID), idx as isize);
707 }
708 // Already shown: just re-tint.
709 (true, Some(_)) => restyle_jupyter(&del),
710 // Hide: remove it and drop the stale ref.
711 (false, Some(i)) => {
712 tb.removeItemAtIndex(i as isize);
713 *del.ivars().jupyter_item.borrow_mut() = None;
714 }
715 (false, None) => {}
716 }
717 });
718 });
719}

736/// Install the native toolbar on the main window. Call from the Tauri setup hook.
737pub fn install(app: &AppHandle) {
738 let Some(mtm) = MainThreadMarker::new() else { return };
739 let Some(window) = app.get_webview_window("main") else { return };
740 let Ok(ns_win_ptr) = window.ns_window() else { return };
741 // Safety: Tauri owns a valid NSWindow for "main" for its lifetime.
742 let ns_win: &NSWindow = unsafe { &*(ns_win_ptr as *const NSWindow) };
743 
744 let delegate = ToolbarDelegate::new(mtm, app.clone());
745 let toolbar = NSToolbar::initWithIdentifier(
746 NSToolbar::alloc(mtm),
747 &NSString::from_str("EigendeckMainToolbar"),
748 );
749 let proto: &ProtocolObject<dyn NSToolbarDelegate> = ProtocolObject::from_ref(&*delegate);
750 toolbar.setDelegate(Some(proto));
751 // (Icon-only is achieved per-item via empty labels — Expanded style ignores
752 // displayMode, so we don't set it.)
753 // Center the editable presentation-title item in the toolbar row. (Singular
754 // API is deprecated on 13+ but works on 11+; the native title lives on the
755 // separate Expanded row, so this only centers our title field.)
756 #[allow(deprecated)]
757 toolbar.setCenteredItemIdentifier(Some(&NSString::from_str(TITLE_ID)));
758 ns_win.setToolbar(Some(&toolbar));
759 // Expanded → native title + proxy centered on their own row (BBEdit/Keynote),
760 // toolbar items on the row below. The native title provides the proxy path
761 // popover + drag + edited dot for free (driven by setRepresentedURL in lib.rs).
762 ns_win.setToolbarStyle(NSWindowToolbarStyle::Expanded);
763 // Label row on by default (the label-row toggle set_compact drives). See #125.
764 toolbar.setDisplayMode(NSToolbarDisplayMode::IconAndLabel);
765 // Start hidden. The window boots into the welcome/startup screen (no project),
766 // which has no toolbar; the frontend calls set_toolbar_visible(true) once a
767 // deck is open. Installing visible flashed the empty toolbar for a frame
768 // before React mounted and hid it.
769 toolbar.setVisible(false);
770 
771 TOOLBAR.with(|t| *t.borrow_mut() = Some(toolbar));
772 DELEGATE.with(|d| *d.borrow_mut() = Some(delegate));
