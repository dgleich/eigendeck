1// Enumerate the OPEN deck's demos and surface which ones reach the internet and
2// why, for the security window's Internet tab. A demo declares its hosts + purpose
3// in a manifest (see demoManifest.ts); this walks the deck, reads each distinct
4// demo asset once, and returns the declared network for the ones that use it.
5//
6// A demo and its pieces share ONE HTML asset (same assetId), so we group by
7// assetId — one row per demo document, listing the slides it appears on. Demos
8// with no manifest get no internet, so they're not listed here (nothing to show).

30/** Distinct hosts, first purpose wins, in declared order. */
31function distinctHosts(network: DemoNetEntry[]): DemoNetEntry[] {
32 const seen = new Set<string>();
33 const out: DemoNetEntry[] = [];
34 for (const n of network) {
35 if (seen.has(n.host)) continue;
36 seen.add(n.host);
37 out.push(n);
38 }
39 return out;
40}

42/** Build the per-demo internet report for the deck currently in the store. */
43export async function buildDemoNetReport(): Promise<DemoNetReportEntry[]> {
44 const slides = usePresentationStore.getState().presentation.slides;
45 // assetId -> set of 1-based slide numbers it appears on.
46 const bySlide = new Map<string, Set<number>>();
47 slides.forEach((slide, i) => {
48 for (const el of slide.elements) {
49 if ((el.type === 'demo' || el.type === 'demo-piece') && el.assetId) {
50 if (!bySlide.has(el.assetId)) bySlide.set(el.assetId, new Set());
51 bySlide.get(el.assetId)!.add(i + 1);
52 }
53 }
54 });
55 
56 const entries = await Promise.all(
57 [...bySlide.entries()].map(async ([assetId, slideSet]): Promise<DemoNetReportEntry | null> => {
58 const html = await readAssetText(assetId);
59 const manifest = parseDemoManifest(html);
60 if (!manifest || manifest.network.length === 0) return null;
61 return { assetId, slides: [...slideSet].sort((a, b) => a - b), hosts: distinctHosts(manifest.network) };
62 }),
63 );
64 return entries.filter((e): e is DemoNetReportEntry => e !== null)
