57function usageLabel(assetId: string): string {
58 const pres = usePresentationStore.getState().presentation;
59 const u = computeAssetUsage(pres, assetId);
60 if (u.elementCount === 0) return 'unused';
61 if (u.slideCount === 1) return u.elementCount === 1 ? 'used on 1 slide' : `used ${u.elementCount}× on 1 slide`;

65/** Build the report for the currently-open deck. Never throws. */
66export async function buildDeckSecurityReport(): Promise<DeckSecurityReport> {
67 const store = usePresentationStore.getState();
68 const token = getDeckToken();
69 const trusted = token ? await isTrusted(token) : false;
70 const projectDir = store.projectPath ? dirname(store.projectPath) : '';
71 
72 let linked: LinkedRow[] = [];
73 try { linked = await invoke<LinkedRow[]>('db_list_linked_assets'); } catch { linked = []; }
74 
75 const rows: ExternalPathRow[] = [];
76 for (const a of linked) {
77 if (!a.external_path) continue;
78 // Decision-only (bounded) read: the report classifies each linked file but never
79 // uses its bytes, so it must not read a large video/PDF in full just to build a row.
80 const gate = await resolveAndGateDecision(resolvePosixPath(projectDir, a.external_path));
81 let state: RowState;
82 let reason: string | null = null;
83 let approvedAt: number | null = null;
84 let approvedHow: string | null = null;
85 if (!gate.ok) {
86 if (gate.reason === 'unreadable') { state = 'missing'; reason = 'source not found on disk'; }
87 else { state = 'forbidden'; reason = gate.reason; }
88 } else {
89 const approved = token ? await isPathApproved(token, gate.canonicalPath!) : false;
90 state = approved ? 'approved' : 'eligible';
91 if (approved && token) {
92 const detail = await approvalDetail(token, gate.canonicalPath!);
93 approvedAt = detail?.at ?? null;
94 approvedHow = detail?.reason ?? null;
95 }
96 }
97 rows.push({
98 assetId: a.asset_id,
99 referencePath: a.external_path,
100 resolvedPath: gate.canonicalPath ?? null,
101 resolvedDir: gate.canonicalPath ? dirname(gate.canonicalPath) : null,
102 state,
103 reason,
104 usage: usageLabel(a.asset_id),
105 approvedAt,
106 approvedHow,
107 });
108 }
109 
110 const ds = token ? await deckState(token) : null;
111 return {
112 trusted,
113 status: ds?.status ?? 'untrusted-new',
114 trustedAt: ds?.trustedAt ?? null,
115 trustReason: ds?.trustReason ?? null,
116 watch: { global: getPreference('autoReloadAssets'), deck: store.presentation.config.autoReloadAssets ?? null },
117 projectDir,
118 counts: countRows(rows),
119 rows,
120 };
121}

123/** Tally the per-state counts for a row set. Shared by buildDeckSecurityReport and the
124 * in-place update in approveDirectory (which flips rows without a second full build). */
125function countRows(rows: ExternalPathRow[]): DeckSecurityReport['counts'] {
126 return {
127 total: rows.length,
128 approved: rows.filter((r) => r.state === 'approved').length,
129 eligible: rows.filter((r) => r.state === 'eligible').length,
130 blocked: rows.filter((r) => r.state === 'forbidden').length,
131 missing: rows.filter((r) => r.state === 'missing').length,
132 };
133}

165/** Approve one eligible asset. REQUIRES the deck to be trusted already — trust is a
166 * separate, explicit act (trustThisDeck); this never trusts on your behalf. A no-op if
167 * the deck isn't trusted (approvePath guards it) or the target no longer resolves. */
168export async function approveOne(assetId: string, referencePath: string): Promise<DeckSecurityReport> {
169 const store = usePresentationStore.getState();
170 const token = getDeckToken();
171 if (token) {
172 const projectDir = store.projectPath ? dirname(store.projectPath) : '';
173 const gate = await resolveAndGate(resolvePosixPath(projectDir, referencePath));
174 if (gate.ok && gate.canonicalPath) {
175 const { approvePath } = await import('./trustStore'); // approvePath no-ops if untrusted→ trustStore.ts
176 await approvePath(token, assetId, gate.canonicalPath, 'approve');
177 }
178 }
179 return buildDeckSecurityReport();
180}

190/** Approve ALL currently-eligible files whose resolved target sits in `resolvedDir`
191 * — the "approve everything in this folder" bulk action (still a per-path decision,
192 * just batched by directory). REQUIRES a trusted deck; no-op otherwise. This is NOT a
193 * trust action — the deck must already be trusted (trustThisDeck), keeping deck-trust
194 * and file-approval two distinct steps. Returns the fresh report. */
195export async function approveDirectory(resolvedDir: string): Promise<DeckSecurityReport> {
196 const token = getDeckToken();
197 const report = await buildDeckSecurityReport(); // built ONCE
198 if (!token) return report;
199 const { approvePath } = await import('./trustStore');→ trustStore.ts
200 const now = Date.now();
201 // Approve the eligible rows in this folder and flip them IN the already-built report
202 // (state + provenance), instead of a second full buildDeckSecurityReport() that would
203 // re-resolve and re-read every linked file off disk.
204 let changed = false;
205 for (const r of report.rows) {
206 if (r.state === 'eligible' && r.resolvedPath && r.resolvedDir === resolvedDir) {
207 if (await approvePath(token, r.assetId, r.resolvedPath, 'approve-folder')) {
208 r.state = 'approved'; r.approvedAt = now; r.approvedHow = 'approve-folder'; changed = true;
209 }
210 }
211 }
212 if (changed) report.counts = countRows(report.rows);
213 return report;
