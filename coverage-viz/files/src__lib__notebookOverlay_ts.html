1// The "overlay" — eigendeck's record of a live notebook session,
2// stored as a SEPARATE eigendeck-owned asset (mime
3// application/x-eigendeck-overlay+json) tagged with the owning
4// element via assets.owner_element_id (discovered by query, not
5// referenced from element.data). The user's .ipynb asset is NEVER
6// mutated; the overlay holds everything that happened in eigendeck.
7//
8// Why an owned asset (not inline element.data, not a second
9// element→asset slot):
10// - free versioning + Restore via the temporal asset history
11// - keeps potentially-large outputs (base64 PNGs) out of the
12// temporal `elements` row, which is rewritten on any element edit
13// - respects 0/1/∞: the element still references ONE asset (.ipynb)
14//
15// "eigendeck is not a notebook editor, but it is a recorder":
16// - source authored in JupyterLab → the .ipynb (pristine)
17// - the live session (edits, outputs, live-authored cells) → here
18//
19// See .claude/notes/notebook-recording-decisions.md for the why.

25/** Signature of a notebook's SOURCE, for change detection. Two parses of
26 * the same .ipynb bytes produce equal signatures, so an incidental
27 * re-fetch (preview/cache invalidation, watcher re-eval) is a no-op; a
28 * deliberate reload-from-disk / version-restore parses to different
29 * content and yields a different signature. */
30export function notebookSourceSignature(nb: Notebook): string {

34/** Decide whether an element's overlay should be RESET because its .ipynb
35 * source genuinely changed. True ONLY when, for the SAME element, the new
36 * signature differs from a previously-seen one — never on first load, and
37 * never when a component instance is reused for a different element. This
38 * is what stops incidental `eigendeck:asset-changed` events from wiping
39 * recorded outputs/edits during plain editing (the wipe-during-edit bug). */
40export function overlaySourceChanged(
41 prev: { id: string; sig: string } | null,
42 nextId: string,
43 nextSig: string,
44): boolean {
45 return prev !== null && prev.id === nextId && prev.sig !== nextSig;
46}

85/** Human one-line summary of what a recording holds, for the link-reconcile
86 * chooser ("which recording do you want to keep?"). */
87export function summarizeOverlay(r: Overlay): string {
88 const edits = Object.keys(r.cellEdits).length;
89 const outs = Object.keys(r.cellOutputs).length;
90 const added = r.appendedCells.length;
91 const parts: string[] = [];
92 if (edits) parts.push(`${edits} edited cell${edits > 1 ? 's' : ''}`);
93 if (outs) parts.push(`${outs} recorded output${outs > 1 ? 's' : ''}`);
94 if (added) parts.push(`${added} added cell${added > 1 ? 's' : ''}`);
95 return parts.length ? parts.join(' · ') : 'empty recording';
96}

98/** Parse recording-asset bytes. Tolerant: malformed → empty. */
99export function parseOverlay(bytes: Uint8Array | ArrayBuffer | string): Overlay {
100 try {
101 const text = typeof bytes === 'string'
102 ? bytes
103 : new TextDecoder('utf-8').decode(bytes instanceof Uint8Array ? bytes : new Uint8Array(bytes));
104 const o = JSON.parse(text);
105 if (!o || typeof o !== 'object') return emptyOverlay();
106 return {
107 version: 1,
108 cellEdits: numKeyedStrings(o.cellEdits),
109 cellOutputs: numKeyedOutputs(o.cellOutputs),
110 cellCounts: numKeyedCounts(o.cellCounts),
111 appendedCells: Array.isArray(o.appendedCells)
112 ? o.appendedCells.filter(isAppended)
113 : [],
114 };
115 } catch {
116 return emptyOverlay();
117 }
118}

126/** A cell in the merged render list. `origin` drives the visual
127 * distinction between pristine notebook cells and recording content. */
128export type MergedCell =
129 | { origin: 'ipynb'; cell: Cell; source: string; outputs: CellOutput[];
130 executionCount: number | null; edited: boolean; outputRecorded: boolean }
131 | { origin: 'appended'; appended: AppendedCell };

133/** Merge a parsed notebook with a recording into the ordered render
134 * list. Source precedence: cellEdits → cell.source. Output
135 * precedence: cellOutputs → baked-in cell.outputs → []. Appended
136 * cells splice in after their anchor index. */
137export function mergeNotebook(notebook: Notebook | null, rec: Overlay): MergedCell[] {
138 const out: MergedCell[] = [];
139 const appendedByAnchor = new Map<number | null, AppendedCell[]>();
140 for (const a of rec.appendedCells) {
141 const k = a.afterIndex;
142 if (!appendedByAnchor.has(k)) appendedByAnchor.set(k, []);
143 appendedByAnchor.get(k)!.push(a);
144 }
145 // Appended cells anchored to top (afterIndex null) render first.
146 for (const a of appendedByAnchor.get(null) ?? []) {
147 out.push({ origin: 'appended', appended: a });
148 }
149 const cells = notebook?.cells ?? [];
150 for (const cell of cells) {
151 if (cell.kind === 'code') {
152 const editKey = cell.index;
153 const editedSource = rec.cellEdits[editKey];
154 const edited = editedSource !== undefined && editedSource !== cell.source;
155 const recOut = rec.cellOutputs[editKey];
156 const outputRecorded = recOut !== undefined;
157 const codeCell = cell as CodeCell;
158 out.push({
159 origin: 'ipynb', cell,
160 source: editedSource ?? cell.source,
161 outputs: recOut ?? codeCell.outputs,
162 executionCount: outputRecorded
163 ? (rec.cellCounts[editKey] ?? null)
164 : codeCell.executionCount,
165 edited,
166 outputRecorded,
167 });
168 } else {
169 // markdown / raw — source edits apply, no outputs.
170 const editedSource = rec.cellEdits[cell.index];
171 const edited = editedSource !== undefined && editedSource !== cell.source;
172 out.push({
173 origin: 'ipynb', cell,
174 source: editedSource ?? cell.source,
175 outputs: [],
176 executionCount: null,
177 edited,
178 outputRecorded: false,
179 });
180 }
181 // Appended cells anchored after this index.
182 for (const a of appendedByAnchor.get(cell.index) ?? []) {
183 out.push({ origin: 'appended', appended: a });
184 }
185 }
186 return out;
187}

191function numKeyedStrings(v: unknown): Record<number, string> {
192 const out: Record<number, string> = {};
193 if (v && typeof v === 'object') {
194 for (const [k, val] of Object.entries(v as Record<string, unknown>)) {
195 const n = Number(k);
196 if (Number.isInteger(n) && typeof val === 'string') out[n] = val;
197 }
198 }
199 return out;
200}

202function numKeyedOutputs(v: unknown): Record<number, CellOutput[]> {

205 for (const [k, val] of Object.entries(v as Record<string, unknown>)) {
206 const n = Number(k);

213function numKeyedCounts(v: unknown): Record<number, number | null> {

224function isAppended(v: unknown): v is AppendedCell {
225 if (!v || typeof v !== 'object') return false;
226 const o = v as Record<string, unknown>;
227 return typeof o.id === 'string'
228 && (o.afterIndex === null || typeof o.afterIndex === 'number')
229 && (o.cellType === 'code' || o.cellType === 'markdown')
230 && typeof o.source === 'string';
