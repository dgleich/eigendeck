1/**
2 * Multi-bundle MathJax renderer using hidden iframes.
3 *
4 * Each font package's MathJax bundle is loaded in its own iframe. The iframe
5 * has its own window.MathJax (no singleton conflict). The main page sends
6 * {tex, bundle} via postMessage; the iframe replies with the rendered SVG.
7 *
8 * This sidesteps the "only one window.MathJax at a time" limitation in
9 * src/lib/mathjax.ts, allowing different math fonts in different elements
10 * on the same slide.
11 *
12 * Each tex render is cached by (bundle, tex, display) so repeated renders
13 * are ~free. First load of each bundle takes ~1-2s.
14 *
15 * # Debugging
16 *
17 * The iframe (public/mathjax-renderer.html) has a dbg(msg) function that
18 * routes log lines back to the parent via postMessage {type:'log'}. We
19 * surface those here as console.warn entries prefixed
20 * `[mathjaxRenderer/iframe <bundle>]` — so iframe-side issues are visible
21 * in the main page's devtools without context-switching.
22 *
23 * Currently dbg() is only invoked for unexpected events (errors, missing
24 * SVG, preamble failures). When debugging a regression, temporarily add
25 * dbg() calls in the success paths inside mathjax-renderer.html (look for
26 * the comments next to the existing calls).
27 */

32/**
33 * Stable cache key for (tex, bundle, display, preamble). The same string is
34 * used both in-memory (pool.cache) and in the SQLite math_cache table, so
35 * the CLI exporter can look up SVGs the editor already produced.
36 */
37export function mathCacheKey(tex: string, bundle: string, display: boolean, preamble: string): string {
38 // FNV-1a hash — small, deterministic, good enough for cache keys (collisions
39 // are functionally fine because we'd just re-render).
40 let h = 0x811c9dc5;
41 const s = `${bundle}\x1f${display ? 'd' : 'i'}\x1f${preamble}\x1f${tex}`;
42 for (let i = 0; i < s.length; i++) {
43 h ^= s.charCodeAt(i);
44 h = Math.imul(h, 0x01000193);
45 }
46 return (h >>> 0).toString(16).padStart(8, '0');
47}

135function installMessageListener() {
136 if (messageListenerInstalled) return;
137 messageListenerInstalled = true;
138 window.addEventListener('message', (ev) => {
139 const msg = ev.data;
140 if (!msg || typeof msg !== 'object') return;
141 if (msg.type === 'log') {
142 // Demo/notebook frames are intentionally script-capable and can post to
143 // their parent. Treat the claimed bundle as data: only the actual hidden
144 // renderer iframe for that bundle may speak the renderer protocol.
145 const sourcePool = typeof msg.bundle === 'string' ? pools.get(msg.bundle) : undefined;
146 if (!sourcePool || ev.source !== sourcePool.iframe.contentWindow) return;
147 // Echo iframe-side dbg() messages. Iframe only sends these for
148 // unexpected events (errors, missing SVG, preamble failures); they
149 // surface here as console warnings so they're visible without opening
150 // the iframe context in devtools.
151 console.warn(`[mathjaxRenderer/iframe ${msg.bundle}]`, msg.msg);
152 return;
153 }
154 if (msg.type !== 'rendered' && msg.type !== 'error') return;
155 const id = msg.id as string | undefined;
156 if (!id) return;
157 // Find which pool owns this request id
158 for (const pool of pools.values()) {
159 const pending = pool.pending.get(id);
160 if (!pending) continue;
161 // A request id is not an authentication token (ids are deliberately small
162 // and predictable). Bind the reply to the window that owns the request so
163 // a sandboxed demo cannot forge a MathJax result into the privileged DOM.
164 if (ev.source !== pool.iframe.contentWindow) continue;
165 pool.pending.delete(id);
166 if (msg.type === 'rendered') {
167 pending.resolve({
168 svg: msg.svg, width: msg.width || '', height: msg.height || '', valign: msg.valign || '',
169 });
170 } else {
171 pending.reject(new Error(msg.message || 'render failed'));
172 }
173 return;
174 }
175 });
176}

196 const ready = new Promise<void>((resolve, reject) => {
197 const readyHandler = (ev: MessageEvent) => {
198 const msg = ev.data;
199 if (!msg || typeof msg !== 'object') return;
200 if (ev.source !== iframe.contentWindow) return;
201 if (msg.type === 'ready' && msg.bundle === bundleId) {
202 window.removeEventListener('message', readyHandler);
203 resolve();
204 } else if (msg.type === 'error' && !msg.id) {
205 console.warn('[mathjaxRenderer] bundle load failed:', bundleId, msg.message);
206 window.removeEventListener('message', readyHandler);
207 reject(new Error(msg.message || 'bundle load failed'));
208 }
209 };
210 window.addEventListener('message', readyHandler);
211 iframe.addEventListener('error', (e) => {
212 console.warn('[mathjaxRenderer] iframe load error:', bundleId, e);
213 });
214 setTimeout(() => {
215 window.removeEventListener('message', readyHandler);
216 reject(new Error(`MathJax bundle '${bundleId}' load timed out`));
217 }, 30000);
218 });

220 pool = { iframe, ready, pending: new Map(), cache: new Map(), appliedPreamble: '' };
221 pools.set(bundleId, pool);
222 document.body.appendChild(iframe);

230/**
231 * Set the math preamble for a bundle's iframe. Idempotent — only forwards
232 * if it differs from what was last applied.
233 */
234export async function setMathPreamble(preamble: string, bundleId: string): Promise<void> {
235 const pool = getOrCreatePool(bundleId);
236 // Already applied (or in-flight) for this exact preamble: reuse the same
237 // readiness promise so every caller waits for the SAME registration.
238 if (pool.appliedPreamble === preamble && pool.preambleReady) return pool.preambleReady;
239 pool.appliedPreamble = preamble;
240 const p = (async () => {
241 await pool.ready;
242 // Wait for the iframe to ACK that the preamble is registered. Without this,
243 // renders fire before the macros exist (a fresh present/projector window) →
244 // undefined control sequence → raw-LaTeX cruft. A safety timeout keeps a
245 // missed ack from hanging renders forever.
246 await new Promise<void>((resolve) => {
247 const handler = (ev: MessageEvent) => {
248 const m = ev.data;
249 if (ev.source !== pool.iframe.contentWindow) return;
250 if (m && m.type === 'preamble-applied' && m.bundle === bundleId) {
251 window.removeEventListener('message', handler);
252 resolve();
253 }
254 };
255 window.addEventListener('message', handler);
256 setTimeout(() => { window.removeEventListener('message', handler); resolve(); }, 6000);
257 pool.iframe.contentWindow?.postMessage({ type: 'preamble', tex: preamble }, '*');
258 });
259 })();
260 pool.preambleReady = p;
261 return p;
262 // NOTE: we do NOT clear pool.cache here. The cache key already includes the
263 // preamble (mathCacheKey(tex, bundle, display, preamble)), so entries are
264 // namespaced by preamble and can never be served for a different one.
265}

267export async function renderMath(
268 tex: string,
269 bundleId: string,
270 display: boolean = false,
271 preamble?: string,
272): Promise<RenderResult> {
273 const pool = getOrCreatePool(bundleId);
274 // Apply preamble AND wait for the iframe to register it before rendering.
275 // Idempotent: returns immediately once registered for this preamble.
276 if (preamble) {
277 await setMathPreamble(preamble, bundleId);
278 }
279 const effectivePreamble = preamble || '';
280 const cacheKey = mathCacheKey(tex, bundleId, display, effectivePreamble);
281 const hit = pool.cache.get(cacheKey);
282 if (hit) return hit;
283 
284 await pool.ready;
285 
286 const id = `r${++nextRequestId}`;
287 const result = await new Promise<RenderResult>((resolve, reject) => {
288 pool.pending.set(id, { resolve, reject });
289 pool.iframe.contentWindow?.postMessage({ id, type: 'render', tex, display }, '*');
290 setTimeout(() => {
291 if (pool.pending.has(id)) {
292 pool.pending.delete(id);
293 reject(new Error(`render timeout (${bundleId}): ${tex.slice(0, 40)}`));
294 }
295 }, 5000);
296 });
297 
298 pool.cache.set(cacheKey, result);
299 // Write-through to SQLite so headless tools (CLI export) can find it.
300 void persistToSqlite(cacheKey, tex, bundleId, display, effectivePreamble, result);
301 return result;
302}

309/**
310 * Turn the raw HTML slice between `$…$` delimiters into clean LaTeX.
311 *
312 * Math is authored as plain text inside the element's HTML, but two things
313 * pollute that slice:
314 * 1. Formatting the surrounding text — e.g. applying a colour across part of
315 * an equation — makes contentEditable wrap a sub-range in `<font>`/`<span
316 * style="color:…">`, and those tags can land INSIDE the `$…$`. So
317 * `$x^2$` coloured mid-expression becomes `$x<font color="#dc2626">^2</font>$`,
318 * and the naive slice hands MathJax `x<font …>^2</font>` → render failure.
319 * 2. Literal `<`, `>`, `&` (valid LaTeX: relations, matrix/align `&`) are
320 * stored HTML-escaped (`&lt;`, `&amp;`), which MathJax can't parse either.
321 *
322 * Stripping tags first, THEN decoding entities (with `&amp;` last so
323 * `&amp;lt;` decodes to `&lt;`, not `<`) yields the LaTeX the author typed.
324 * Pure tex (`\lambda_1`, `\sum_{i=1}^n`) passes through unchanged.
325 */
326export function texFromHtml(raw: string): string {
327 return raw
328 .replace(/<[^>]*>/g, '')
329 .replace(/&nbsp;/g, ' ')
330 .replace(/ /g, ' ')
331 .replace(/&lt;/g, '<')
332 .replace(/&gt;/g, '>')
333 .replace(/&quot;/g, '"')
334 .replace(/&#39;/g, "'")
335 .replace(/&amp;/g, '&');
336}

338/**
339 * Contain a math expression that FAILED to render. We splice the raw source
340 * back so the author can see what broke — but clipped to the element's width on
341 * a single line (with an ellipsis) and flagged red, so a failure can NEVER spill
342 * its raw LaTeX across the slide. (Successful math keeps overflow:visible for
343 * italic ink overhang — #61 — so the clip is scoped to the fallback only.)
344 * `block` picks a block wrapper for display math vs inline for inline math.
345 */
346function failedMathFallback(raw: string, block: boolean): string {
347 const tag = block ? 'div' : 'span';
348 const disp = block ? 'block' : 'inline-block';
349 const safe = raw.replace(/</g, '&lt;').replace(/>/g, '&gt;');
350 return `<${tag} style="display:${disp};max-width:100%;overflow:hidden;white-space:nowrap;text-overflow:ellipsis;vertical-align:bottom;color:#c0392b;" title="This math failed to render — check the expression and the deck's LaTeX preamble.">${safe}</${tag}>`;
351}

353/**
354 * Walk an HTML string, find $..$ and $$..$$ math expressions, render each
355 * with the given bundle, and splice the resulting SVG markup back in.
356 * Skips inside HTML tags (matching renderMathInHtml in src/lib/mathjax.ts).
357 */
358export async function renderMathInHtml(html: string, bundleId: string, preamble?: string): Promise<string> {
359 if (!containsMath(html)) return html;
360 if (preamble) {
361 try { await setMathPreamble(preamble, bundleId); } catch (e) { console.warn('preamble apply failed:', e); }
362 }
363 
364 const parts: string[] = [];
365 let i = 0;
366 while (i < html.length) {
367 if (html[i] === '<') {
368 const tagEnd = html.indexOf('>', i);
369 if (tagEnd !== -1) {
370 parts.push(html.slice(i, tagEnd + 1));
371 i = tagEnd + 1;
372 continue;
373 }
374 }
375 
376 // Display math $$..$$
377 if (html[i] === '$' && html[i + 1] === '$') {
378 const end = html.indexOf('$$', i + 2);
379 if (end !== -1) {
380 const tex = texFromHtml(html.slice(i + 2, end));
381 try {
382 // Pass `preamble` so the cache is keyed by the SAME preamble the read
383 // path (renderMathInHtmlSync / warm-from-SQLite) uses. Without it the
384 // write keyed under "" and every present-mode lookup missed → endless
385 // live re-rendering of already-cached math.
386 const { svg } = await renderMath(tex, bundleId, true, preamble);
387 parts.push(`<div style="text-align:center;">${svg}</div>`);
388 } catch (e) {
389 console.warn('Display math render failed:', e);
390 parts.push(failedMathFallback(`$$${tex}$$`, true));
391 }
392 i = end + 2;
393 continue;
394 }
395 }
396 
397 // Inline math $..$
398 if (html[i] === '$') {
399 const end = html.indexOf('$', i + 1);
400 if (end !== -1 && !html.slice(i + 1, end).includes('\n')) {
401 const tex = texFromHtml(html.slice(i + 1, end));
402 try {
403 const r = await renderMath(tex, bundleId, false, preamble);
404 // Match the inline-math styling from the existing renderer
405 // (vertical-align baseline tweak so it sits on the text line)
406 const valign = r.valign || '-0.025ex';
407 // Inject inline display styles + overflow="visible" attribute so
408 // italic-glyph ink (e.g. \gamma) isn't clipped at the SVG box.
409 const svg = r.svg.replace(
410 /^<svg/,
411 `<svg overflow="visible" style="display:inline;vertical-align:${valign};overflow:visible"`
412 );
413 parts.push(svg);
414 } catch (e) {
415 console.warn('Inline math render failed:', e);
416 parts.push(failedMathFallback(`$${tex}$`, false));
417 }
418 i = end + 1;
419 continue;
420 }
421 }
422 
423 parts.push(html[i]);
424 i++;
425 }
426 
427 return parts.join('');
428}

430/**
431 * Synchronous, cache-ONLY version of renderMathInHtml. Returns HTML with every
432 * ALREADY-CACHED expression spliced to its SVG and any uncached ones left as
433 * raw source (the async renderMathInHtml fills those in afterward). Returns
434 * null if the bundle's pool doesn't exist yet (nothing to hit).
435 *
436 * This lets TextElementSvg show warmed math IMMEDIATELY on mount instead of
437 * flashing raw-source → SVG (renderMath is async even on a cache hit). On a
438 * warm window (e.g. the projector after warmMathCacheFromSqlite), everything is
439 * a hit, so there's no flash at all.
440 */
441export function renderMathInHtmlSync(html: string, bundleId: string, preamble?: string): string | null {
442 if (!containsMath(html)) return html;
443 const pool = pools.get(bundleId);
444 if (!pool) return null;
445 const eff = preamble || '';
446 const parts: string[] = [];
447 let i = 0;
448 while (i < html.length) {
449 if (html[i] === '<') {
450 const tagEnd = html.indexOf('>', i);
451 if (tagEnd !== -1) { parts.push(html.slice(i, tagEnd + 1)); i = tagEnd + 1; continue; }
452 }
453 if (html[i] === '$' && html[i + 1] === '$') {
454 const end = html.indexOf('$$', i + 2);
455 if (end !== -1) {
456 const tex = texFromHtml(html.slice(i + 2, end));
457 const hit = pool.cache.get(mathCacheKey(tex, bundleId, true, eff));
458 parts.push(hit ? `<div style="text-align:center;">${hit.svg}</div>` : `$$${tex}$$`);
459 i = end + 2; continue;
460 }
461 }
462 if (html[i] === '$') {
463 const end = html.indexOf('$', i + 1);
464 if (end !== -1 && !html.slice(i + 1, end).includes('\n')) {
465 const tex = texFromHtml(html.slice(i + 1, end));
466 const hit = pool.cache.get(mathCacheKey(tex, bundleId, false, eff));
467 if (hit) {
468 const valign = hit.valign || '-0.025ex';
469 parts.push(hit.svg.replace(/^<svg/, `<svg overflow="visible" style="display:inline;vertical-align:${valign};overflow:visible"`));
470 } else {
471 parts.push(`$${tex}$`);
472 }
473 i = end + 1; continue;
474 }
475 }
476 parts.push(html[i]); i++;
477 }
478 return parts.join('');
