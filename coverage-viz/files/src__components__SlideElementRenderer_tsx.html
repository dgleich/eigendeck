214 assetPathCache.set(assetId, v);

420 // otherwise render the spliced, variable-driven output.
421 const srcDoc = editing

437 const doc = editDocRef.current;
438 // Commit only from the editable doc AND only while it's still the frame's current
439 // document; otherwise there was no real edit to save (e.g. cancel before load).
440 const html = doc && doc === iframeRef.current?.contentDocument ? doc.body?.innerHTML : undefined;
441 if (html != null && html !== element.html) onUpdate({ html } as any);
442 setEditing(false);

445 // Scale-to-fit: measure the CONTENT's natural size once (when scale mode is on
446 // but not yet measured) and store it as the design size. The editor iframe is
447 // same-origin, so we can read the framed document: shrink the body to fit its
448 // content, read the scroll extent (synchronous — no visible flicker), restore.
449 // Falls back to the box when the content has no intrinsic size (e.g. width:100%).

451 if (!element.scaleMode || (element.scaleW && element.scaleH)) return;

462 else onUpdate({ scaleW: Math.round(element.position.width), scaleH: Math.round(element.position.height) } as any);

471 // switches srcDoc to the raw source (for var elements), which RELOADS the iframe —

502 }, [editing, hasVars]);

1567 * shared by DraggableBox and ArrowRenderer so arrows get the same link authoring
1568 * chrome as every other element. Self-contained: needs only the element's id + link
1569 * fields; all actions go through the store / window events. */
1570export function ElementLinkBadges({ element, isSelected }: {
1571 element: { id: string; linkId?: string; syncId?: string; _linkId?: string; _syncId?: string };
1572 isSelected: boolean;
1573}) {
1574 if (!isSelected) return null;
1575 const elementId = element.id;
1576 const { linkId, syncId, _linkId, _syncId } = element;
1577 return (
1578 <div className="el-link-badges" onClick={(e) => e.stopPropagation()} onPointerDown={(e) => e.stopPropagation()}>
1579 {/* Sync badge. Green = synced (click to free). Grey = either freed
1580 (click to re-sync) OR animation-linked-but-not-synced (click to

1582 {(syncId || _syncId || linkId) && (() => {
