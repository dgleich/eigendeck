1// Pure planner for the present-mode slide transition. Given the previous and
2// current slide, it classifies every CURRENT element with its TRUE z-order and
3// transition role, plus the elements that are LEAVING. PresentMode renders
4// straight from this plan — one stable list, keyed by element id, in z-order —
5// which is what keeps iframes from remounting and z-order correct across the
6// transition (see docs/presenter-architecture.md).
7//
8// No React, no DOM, no store — trivially unit-tested (presentTransition.test.ts).

12export type PresentRole =
13 | 'static' // cover mask, or an element carried over from the prev slide (same
14 // id) — appears instantly, no fade (fading would flicker it).
15 | 'fade' // genuinely new on this slide — fades in.
16 | 'linked'; // has a linkId matched on the prev slide — animates position/size
17 // from its partner (`from`).
18 
19export interface PresentPlanItem {
20 element: SlideElement;
21 /** True z-order = index in currentSlide.elements. z-index MUST come from here,
22 * never a per-role counter, or stacking is wrong mid-transition (#z-jump). */
23 z: number;
24 role: PresentRole;
25 /** The matched partner on the previous slide (role === 'linked' only). */
26 from?: SlideElement;
27}
28 
29export interface PresentPlan {
30 /** Current-slide elements, in z-order. */
31 items: PresentPlanItem[];
32 /** Elements only on the previous slide (linked, no match here) — fade out. */
33 fadeOut: { element: SlideElement; z: number }[];
34}
35 
36export function planPresentTransition(
37 prevSlide: Slide | null,
38 currentSlide: Slide,
39): PresentPlan {
40 const prevByLinkId = new Map<string, SlideElement>();
41 const prevIds = new Set<string>();
42 if (prevSlide) {
43 for (const el of prevSlide.elements) {
44 prevIds.add(el.id);
45 if (el.linkId) prevByLinkId.set(el.linkId, el);
46 }
47 }
48 
49 const matchedPrevLinkIds = new Set<string>();
50 const items: PresentPlanItem[] = currentSlide.elements.map((element, z) => {
51 const from = element.linkId ? prevByLinkId.get(element.linkId) : undefined;
52 if (from) {
53 matchedPrevLinkIds.add(element.linkId!);
54 return { element, z, role: 'linked', from };
55 }
56 // Cover masks are instant; an element carried over from the prev slide (same
57 // id — a synced element across build steps) stays put. Both → static.
58 if (element.type === 'cover' || prevIds.has(element.id)) {
59 return { element, z, role: 'static' };
60 }
61 return { element, z, role: 'fade' };
62 });
63 
64 const fadeOut = !prevSlide ? [] : prevSlide.elements
65 .map((element, z) => ({ element, z }))
66 .filter(({ element }) => element.linkId && !matchedPrevLinkIds.has(element.linkId));
67 
68 return { items, fadeOut };
69}
