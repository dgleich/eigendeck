1// "Which recording do you want to keep?" chooser for promoting an animation
2// link to a sync. Promote is destructive — the linked copies collapse into one
3// element — so when they hold different notebook recordings we ask which one
4// survives. The picked copy becomes the master (its recording is kept; the
5// others are discarded). Shown only on a real conflict (2+ different
6// recordings); a single/no recording promotes without asking.
7 
8interface Candidate {
9 elementId: string;
10 slideNo: number; // 1-based, for display
11 summary: string; // summarizeOverlay() of this copy's recording
12}
13 
14export function PromoteChooser({ candidates, onPick, onCancel }: {
15 candidates: Candidate[];
16 onPick: (elementId: string) => void;
17 onCancel: () => void;
18}) {
19 return (
20 <div className="link-overlay" onClick={onCancel}>
21 <div className="link-overlay-content" onClick={(e) => e.stopPropagation()}>
22 <div className="link-overlay-header">
23 <span>These linked notebooks have different recordings — keep which one?
24 The others are discarded.</span>
25 <button className="link-overlay-close" onClick={onCancel}>Cancel</button>
26 </div>
27 <div className="overlay-conflict-choices">
28 {candidates.map((c) => (
29 <button key={c.elementId} className="overlay-conflict-card"
30 onClick={() => onPick(c.elementId)}>
31 <strong>Slide {c.slideNo}</strong>
32 <span>{c.summary}</span>
33 <em>Keep this recording</em>
34 </button>
35 ))}
36 </div>
37 </div>
38 </div>
39 );
40}
