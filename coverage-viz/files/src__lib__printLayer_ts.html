38/** Gather slideId:elementId → screenshot data-URL for every live element:
39 * cached previews first, then (if liveCapture) a flip-through capture of the misses. */
40async function gatherScreenshots(
41 presentation: Presentation, liveCapture: boolean, onNeedsLiveCapture?: () => void,
42): Promise<Map<string, string>> {
43 const shots = new Map<string, string>();
44 const hasLive = presentation.slides.some((s) => s.elements.some((e) => isLivePrintElement(e.type)));
45 if (!hasLive) return shots;
46 for (const slide of presentation.slides) {
47 for (const el of slide.elements) {
48 if (!isLivePrintElement(el.type)) continue;
49 // A notebook's cached preview goes theme-stale for OTHER slides on a theme
50 // switch — skip a stale one so it re-captures live below (#140).
51 if (el.type === 'notebook') {
52 const salt = previewThemeSalt(resolveTheme(presentation.theme, slide.theme));
53 if (await isPreviewThemeStale(previewKey(el), salt)) continue;
54 }
55 const cached = await loadPreviewDataUrl(previewKey(el));
56 if (cached) shots.set(`${slide.id}:${el.id}`, cached);
57 }
58 }
59 const needsLive = presentation.slides.some((s) =>
60 s.elements.some((e) => isLivePrintElement(e.type) && !shots.has(`${s.id}:${e.id}`)));
61 if (!needsLive) return shots;
62 await onNeedsLiveCapture?.();
63 if (!liveCapture) return shots; // CLI: leave the misses to placeholder rendering

88 usePresentationStore.getState().selectSlide(originalSlideIndex);

122/**
123 * Build the print layer: the per-slide inch-based `<div class="slide">` strings.
124 * Deterministic given its async inputs; the caller owns fonts + the document
125 * wrapper (see printPageCss in printSlideHtml.ts for the matching @page CSS).
126 */
127export async function preparePrintLayer(
128 presentation: Presentation, opts: PreparePrintLayerOpts = {},
129): Promise<{ slideHtmls: string[] }> {
130 const renderMath = opts.renderMath ?? renderMathInHtml;
131 const demoScreenshots = await gatherScreenshots(presentation, !!opts.liveCapture, opts.onNeedsLiveCapture);
132 const imageCache = await gatherImageCache(presentation);
133 
134 // Pre-render math per text element to inline SVG (the print path emits plain
135 // HTML, not the live render). Key by slideId:elementId — a shared element can
136 // appear on multiple slides with a different preset font.
137 const mathHtmlByKey = new Map<string, string>();
138 for (const slide of presentation.slides) {
139 for (const el of slide.elements as SlideElement[]) {
140 if (el.type === 'text' && el.html && containsMath(el.html)) {
141 const bundleId = fontForPreset(el.preset, slide as Slide, presentation.config).id;
142 const rendered = await renderMath(el.html, bundleId, presentation.config.mathPreamble || '').catch(() => el.html as string);
143 mathHtmlByKey.set(`${slide.id}:${el.id}`, rendered);
144 }
145 }
146 }
147 
148 const slideHtmls = presentation.slides.map((slide, i) =>
149 buildPrintSlideHtml(slide, presentation, imageCache, demoScreenshots, mathHtmlByKey, i + 1));
150 return { slideHtmls };
