18// --- round-trip source embed (File > Import from HTML) ------------------------
19// The exported HTML carries the full presentation JSON in an
20// `<!-- eigendeck-source: BASE64 -->` comment; import reads it back. The base64
21// is UTF-8-safe (a deck is full of non-ASCII: math λ Σ ∫ −, curly quotes, …), so
22// ENCODE and DECODE must be exact inverses. They used to live inline in two files
23// (export here, import in store/fileOps.ts) and DRIFTED — export UTF-8-encoded
24// while import used a plain `atob`, so every deck with non-ASCII imported as
25// mojibake or failed to parse (#164). Both sides now share this ONE pair, and the
26// round-trip test calls parseEigendeckSource (not a re-implementation) so they
27// can't drift again.
28export const EIGENDECK_SOURCE_RE = /<!-- eigendeck-source: (.+?) -->/;

30/** Base64 (UTF-8-safe) of the presentation JSON, for the export comment. */
31export function encodeEigendeckSource(presentation) {
32 const json = JSON.stringify(presentation);
33 return (typeof btoa !== 'undefined')
34 ? btoa(unescape(encodeURIComponent(json)))
35 : Buffer.from(json, 'utf-8').toString('base64');
36}

38/** Parse the embedded presentation out of exported HTML. Returns null when the
39 * comment is absent (not an Eigendeck export); throws only if the JSON is
40 * corrupt. Exact inverse of encodeEigendeckSource. */
41export function parseEigendeckSource(html) {
42 const m = html.match(EIGENDECK_SOURCE_RE);
43 if (!m) return null;
44 const b64 = m[1];
45 const json = (typeof atob !== 'undefined')
46 ? decodeURIComponent(escape(atob(b64)))
47 : Buffer.from(b64, 'base64').toString('utf-8');
48 return JSON.parse(json);
49}

115/**
116 * Inject role/piece hash AND a unique channel key into a demo HTML.
117 * In srcdoc iframes, location.pathname is empty, so demos that derive their
118 * BroadcastChannel name from pathname would all collide. We override the
119 * BroadcastChannel constructor to inject a unique prefix per slide+demo.
120 */
121export function injectDemoBootstrap(html, hash, channelKey) {
122 // Parse the hash to extract params (e.g. "#piece=lattice" -> {piece: "lattice"})
123 const hashParams = {};
124 if (hash) {
125 const qs = hash.startsWith('#') ? hash.slice(1) : hash;
126 for (const part of qs.split('&')) {
127 const [k, v] = part.split('=');
128 if (k) hashParams[k] = v || '';
129 }
130 }
131 const bootstrap = `<script>
132(function(){
133 var __ch = ${JSON.stringify(channelKey)};
134 var __hp = ${JSON.stringify(hashParams)};
135 // Set location.hash (works in normal iframes, may fail in srcdoc)
136 try { window.location.hash = ${JSON.stringify(hash || '')}; } catch(e) {}
137 // Patch URLSearchParams so demos reading location.hash get injected values
138 var _USP = window.URLSearchParams;
139 window.URLSearchParams = function(init) {
140 var inst = new _USP(init);
141 if (!init || init === '' || init === '#') {
142 for (var k in __hp) inst.set(k, __hp[k]);
143 }
144 return inst;
145 };
146 window.URLSearchParams.prototype = _USP.prototype;
147 // Replace BroadcastChannel with postMessage relay via parent.
148 // srcdoc iframes may have opaque origins where BroadcastChannel won't work.
149 window.BroadcastChannel = function(name) {
150 this._name = __ch + ':' + name;
151 this.onmessage = null;
152 var self = this;
153 window.addEventListener('message', function(e) {
154 if (!e.data || !self.onmessage) return;
155 // Enveloped message from relay
156 if (e.data.__bc === self._name) {
157 self.onmessage({ data: e.data.payload });
158 }
159 // Raw request-state from parent (slide navigation re-request)
160 if (e.data.type === 'request-state' && !e.data.__bc) {
161 self.onmessage({ data: e.data });
162 }
163 });
164 };
165 window.BroadcastChannel.prototype.postMessage = function(msg) {
166 // Send to parent, which relays to all sibling iframes
167 try {
168 window.parent.postMessage({ __bc: this._name, payload: msg }, '*');
169 } catch(e) {}
170 };
171 window.BroadcastChannel.prototype.close = function() { this.onmessage = null; };
172})();
173</script>`;
174 if (html.includes('<head>')) {
175 return html.replace('<head>', '<head>' + bootstrap);
176 }
177 return bootstrap + html;
178}

180/**
181 * Convert bytes to a base64 data URL.
182 * Encodes in chunks to avoid stack overflow on large images.
183 */
184export function bytesToDataUrl(bytes, ext) {
185 const mime = ext === 'svg' ? 'image/svg+xml' : `image/${ext === 'jpg' ? 'jpeg' : ext}`;
186 // Universal base64 encoding (works in browser and Node)
187 let binary = '';
188 const chunkSize = 8192;
189 for (let i = 0; i < bytes.length; i += chunkSize) {
190 binary += String.fromCharCode.apply(null, Array.from(bytes.slice(i, i + chunkSize)));
191 }
192 // btoa exists in browsers and modern Node
193 const base64 = (typeof btoa !== 'undefined')
194 ? btoa(binary)
195 : Buffer.from(binary, 'binary').toString('base64');
196 return `data:${mime};base64,${base64}`;
197}
