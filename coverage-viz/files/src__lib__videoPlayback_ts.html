7/** Toggle a file <video>'s play/pause. Used as the click handler when controls
8 * are off, so a chrome-free video is still startable by clicking it (otherwise
9 * there's no affordance — in present mode it would just sit on frame 0). */
10export function togglePlay(v: HTMLVideoElement | null): void {
11 if (!v) return;
12 if (v.paused) void v.play().catch(() => {});
13 else v.pause();
14}

16/** Apply playback speed to an EMBED via the provider's postMessage API. Posts a
17 * few times after the iframe (re)loads since there's no player-ready callback
18 * wired — best-effort. No-op at rate 1 (provider default). */
19export function useEmbedSpeed(
20 ref: RefObject<HTMLIFrameElement | null>, provider: string | undefined, rate: number, srcKey: unknown,
21): void {
22 useEffect(() => {
23 if (!provider || !rate || rate === 1) return;
24 const post = () => postEmbedSpeed(ref.current?.contentWindow, provider, rate);
25 const timers = [600, 1500, 3000].map((ms) => setTimeout(post, ms));
26 return () => timers.forEach(clearTimeout);
27 }, [ref, provider, rate, srcKey]);
28}

40/** Ping-pong reverse loop (file only, BEST-EFFORT). When enabled, the video
41 * plays forward, then reverse-seeks back to the start, repeating. Reverse has
42 * no native support — we step currentTime backwards via rAF, which is smooth
43 * only for short clips (depends on keyframe spacing). Disable native loop when
44 * this is on. No-op when disabled or the video isn't ready. */
45export function usePingPong(
46 ref: RefObject<HTMLVideoElement | null>, enabled: boolean, rate: number, srcKey: unknown,
47): void {
48 useEffect(() => {
49 const v = ref.current;
50 if (!v || !enabled) return;
51 let raf = 0;
52 let last = 0;
53 const step = (now: number) => {
54 const el = ref.current;
55 if (!el) return;
56 const dt = (now - last) / 1000;
57 last = now;
58 el.currentTime = Math.max(0, el.currentTime - dt * (rate || 1));
59 if (el.currentTime <= 0.001) { void el.play().catch(() => {}); return; } // back at start → forward
60 raf = requestAnimationFrame(step);
61 };
62 const onEnded = () => { last = performance.now(); raf = requestAnimationFrame(step); };
63 v.addEventListener('ended', onEnded);
64 return () => { v.removeEventListener('ended', onEnded); cancelAnimationFrame(raf); };
65 }, [ref, enabled, rate, srcKey]);
66}
