16/** Best-effort playback-speed control for an embed, via each provider's
17 * postMessage player API (YouTube IFrame API / Vimeo player.js / PeerTube
18 * PlayerAPI). No-op/silent if the player isn't ready or the provider rejects
19 * it — speed is "best-effort" for embeds. */
20export function postEmbedSpeed(win: Window | null | undefined, provider: string | undefined, rate: number): void {
21 if (!win || !provider) return;
22 try {
23 if (provider === 'youtube') {
24 win.postMessage(JSON.stringify({ event: 'command', func: 'setPlaybackRate', args: [rate] }), '*');
25 } else if (provider === 'vimeo') {
26 win.postMessage({ method: 'setPlaybackRate', value: rate }, '*');
27 } else if (provider === 'peertube') {
28 win.postMessage({ method: 'setPlaybackRate', params: [rate] }, '*');
29 }
30 } catch { /* cross-origin / not ready — ignore */ }
31}

38/** Resolve a poster thumbnail URL for any provider. YouTube is synchronous (a
39 * CDN URL); Vimeo/PeerTube need an oEmbed fetch (CORS-permitting; null on
40 * failure/offline). Returns a URL for <img src> display — not bytes. */
41export async function fetchEmbedThumbnail(el: { provider?: string; url?: string }): Promise<string | null> {
42 if (!el.url) return null;
43 const parsed = detectVideoProvider(el.url);
44 if (!parsed) return null;
45 if (parsed.provider === 'youtube') return youtubeThumb(parsed.id);
46 try {
47 const oembed = parsed.provider === 'vimeo'
48 ? `https://vimeo.com/api/oembed.json?url=${encodeURIComponent(el.url)}`
49 : `${parsed.origin}/services/oembed?format=json&url=${encodeURIComponent(el.url)}`;
50 const res = await fetch(oembed);
51 if (!res.ok) return null;
52 const j = await res.json();
53 return (j && typeof j.thumbnail_url === 'string') ? j.thumbnail_url : null;
54 } catch {
55 return null;
56 }
