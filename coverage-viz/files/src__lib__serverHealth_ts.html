1// Aggregate Jupyter server health for the open deck — the same computation the
2// topbar ServerStatusPill does, factored out so the native macOS toolbar can
3// mirror it (issue #128). Returns a coarse status + a human tooltip.
4//
5// green — every kernel the deck needs has a server reached recently (≤30 min)
6// yellow — kernels matched but the server is stale (>30 min) or never tested
7// red — at least one needed kernel has no registered server
8// gray — nothing to report (no notebooks in the deck, or presenting)

20// Unique external kernel names the deck's notebooks need. Lite/pyodide
21// notebooks need no server and are skipped; a notebook with no kernelName
22// falls back to the deck default, then 'python3'. Shared by ServerStatusPill
23// (the HTML topbar) and useAggregateServerHealth (the native toolbar icon) so
24// the two surfaces can't drift.
25export function deckExternalKernels(slides: Slide[], config: PresentationConfig | undefined): string[] {
26 const names = new Set<string>();
27 for (const slide of slides) {
28 for (const el of slide.elements) {
29 if (el.type !== 'notebook') continue;
30 if (el.kernel?.kind === 'lite') continue;
31 const name =
32 (el.kernel?.kind === 'external' ? el.kernel.kernelName : undefined)
33 ?? (config?.notebookKernel?.kind === 'external' ? config.notebookKernel.kernelName : undefined)
34 ?? 'python3';
35 names.add(name);
36 }
37 }
38 return [...names];
39}

41export function useAggregateServerHealth(): { status: ServerHealth; tooltip: string } {
42 const isPresenting = usePresentationStore((s) => s.isPresenting);
43 const slides = usePresentationStore((s) => s.presentation?.slides ?? []);
44 const config = usePresentationStore((s) => s.presentation?.config);
45 const [servers] = usePreference('jupyterServers');
46 
47 return useMemo(() => {
48 const deckKernels = deckExternalKernels(slides, config);
49 
50 if (isPresenting || deckKernels.length === 0) {
51 return { status: 'gray' as const, tooltip: 'Jupyter: no live notebooks in this deck' };
52 }
53 
54 const requirements = deckKernels.map((k) => ({ kernelName: k, matched: findServerForKernel(k, servers) }));
55 const missing = requirements.filter((r) => r.matched == null);
56 const matched = requirements.map((r) => r.matched).filter(Boolean) as { lastSeenAt?: number }[];
57 const now = Date.now();
58 const allFresh = matched.every((m) => m.lastSeenAt != null && now - m.lastSeenAt < STALE_MS);
59 const freshCount = matched.filter((m) => m.lastSeenAt != null && now - m.lastSeenAt < STALE_MS).length;
60 
61 if (missing.length > 0) {
62 return {
63 status: 'red' as const,
64 tooltip: `Jupyter: no server for ${missing.map((m) => m.kernelName).join(', ')} (click to configure)`,
65 };
66 }
67 if (allFresh) {
68 return { status: 'green' as const, tooltip: `Jupyter: all ${matched.length} server(s) reachable` };
69 }
70 return {
71 status: 'yellow' as const,
72 tooltip: `Jupyter: ${freshCount}/${matched.length} server(s) reachable recently (click to refresh)`,
73 };
74 }, [slides, config, servers, isPresenting]);
