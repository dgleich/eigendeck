1// Demo theme inheritance (#86) — single source of truth for the CSS that lets a
2// demo iframe match the deck's fonts + theme.
3//
4// Demos render in isolated iframes (blob: in editor/present, srcdoc in export),
5// so they don't see the app's @font-face or theme. We inject:
6// 1. @font-face for the deck's fonts (data: URLs) so font-family:'PT Sans'
7// resolves inside the iframe — this fixes existing demos automatically,
8// since they already name the deck fonts.
9// 2. The resolved theme as CSS custom properties (--eigendeck-*). VARS ONLY —
10// we do NOT set body{} defaults, so existing demos are untouched; a demo
11// opts in by using var(--eigendeck-bg) etc.
12//
13// This module is plain .mjs (no React/DOM-only deps in the pure parts) so the
14// app (TS) AND the node HTML exporter share ONE source — the value CSS can't
15// drift between editor, present, and export. Only the *delivery* differs:
16// in-app injects into the live contentDocument; export splices a <style> into
17// the static srcdoc (no live DOM there).
18 
19export const DEMO_FONTS_STYLE_ID = 'eigendeck-demo-fonts';
20export const DEMO_VARS_STYLE_ID = 'eigendeck-demo-vars';

22/**
23 * Build the `:root { --eigendeck-* }` custom-property block for a resolved
24 * theme. `colors` is a ThemeColors ({background,text,heading,accent,muted});
25 * the font names are bare CSS family names (e.g. 'PT Sans').
26 */
27export function demoThemeVarsCss(colors, opts = {}) {
28 const { font, narrow, mono, baseSize } = opts;
29 const lines = [
30 `--eigendeck-bg: ${colors.background};`,
31 `--eigendeck-fg: ${colors.text};`,
32 `--eigendeck-heading: ${colors.heading};`,
33 `--eigendeck-accent: ${colors.accent};`,
34 `--eigendeck-muted: ${colors.muted};`,
35 ];
36 if (font) lines.push(`--eigendeck-font: '${font}';`);
37 if (narrow) lines.push(`--eigendeck-narrow: '${narrow}';`);
38 if (mono) lines.push(`--eigendeck-mono: '${mono}';`);
39 if (baseSize) lines.push(`--eigendeck-base-size: ${baseSize}px;`);
40 return `:root{\n ${lines.join('\n ')}\n}`;
41}

43/**
44 * Whether a demo's HTML actually names one of the deck fonts — either the
45 * `--eigendeck-font/narrow/mono` custom properties, or a `font-family` name that
46 * the embedded @font-face block declares. A demo that never names a deck font
47 * can't render with one, so there's no reason to splice the (large, base64)
48 * @font-face faces into its document — that's pure per-mount parse cost (a
49 * default deck embeds ~1.7MB of PT Sans, re-parsed on every iframe mount /
50 * slide switch). Fonts still embed for any demo that DOES name them (the #86
51 * "existing demos already name the deck fonts" case is preserved).
52 */
53export function demoReferencesFonts(html, fontFacesCss) {
54 if (!fontFacesCss || !html) return false;
55 if (/--eigendeck-(?:font|narrow|mono)\b/.test(html)) return true;
56 const re = /font-family:\s*(['"])([^'"]+)\1/gi;
57 let m;
58 while ((m = re.exec(fontFacesCss)) !== null) {
59 if (html.includes(m[2])) return true;
60 }
61 return false;
62}

64/**
65 * A self-contained `<style>` string carrying the font faces + theme vars, for
66 * splicing into a static demo srcdoc (export). Empty parts are omitted.
67 */
68export function demoThemeStyleTag(fontFacesCss, varsCss) {
69 const css = [fontFacesCss, varsCss].filter(Boolean).join('\n');
70 if (!css) return '';
71 return `<style id="${DEMO_VARS_STYLE_ID}">\n${css}\n</style>`;
72}

74/**
75 * Splice the demo-theme <style> into a demo HTML string (export path). Inserted
76 * right after <head> when present, else prepended so it still applies.
77 */
78export function injectDemoThemeIntoHtml(html, fontFacesCss, varsCss) {
79 const tag = demoThemeStyleTag(fontFacesCss, varsCss);
80 if (!tag) return html;
81 const headOpen = html.match(/<head[^>]*>/i);
82 if (headOpen) {
83 const at = headOpen.index + headOpen[0].length;
84 return html.slice(0, at) + '\n' + tag + html.slice(at);
85 }
86 return tag + '\n' + html;
87}

89/**
90 * Inject (or update) the demo-theme styles in a LIVE iframe document
91 * (editor/present). Idempotent and split so the heavy font-face block is set
92 * once while the cheap vars block updates on theme change with no reflow churn.
93 * Silently no-ops if the document isn't reachable/ready (cross-origin or not
94 * yet loaded) — mirrors previewCache.ts's contentDocument guard.
95 *
96 * @param doc the iframe's contentDocument
97 * @param fontFacesCss @font-face block (data: URLs); may be '' to skip
98 * @param varsCss the :root{--eigendeck-*} block
99 */
100export function injectDemoThemeIntoDoc(doc, fontFacesCss, varsCss) {
101 if (!doc) return;
102 const head = doc.head || doc.documentElement;
103 if (!head) return;
104 // Fonts: set once (or when the css changes) — keyed by content length+head
105 // so we don't re-parse a megabyte of base64 on every theme switch.
106 if (fontFacesCss) {
107 let fs = doc.getElementById(DEMO_FONTS_STYLE_ID);
108 if (!fs) {
109 fs = doc.createElement('style');
110 fs.id = DEMO_FONTS_STYLE_ID;
111 head.appendChild(fs);
112 }
113 if (fs.textContent !== fontFacesCss) fs.textContent = fontFacesCss;
114 }
115 // Vars: cheap, update freely.
116 let vs = doc.getElementById(DEMO_VARS_STYLE_ID);
117 if (!vs) {
118 vs = doc.createElement('style');
119 vs.id = DEMO_VARS_STYLE_ID;
120 head.appendChild(vs);
121 }
122 if (vs.textContent !== varsCss) vs.textContent = varsCss;
123}
