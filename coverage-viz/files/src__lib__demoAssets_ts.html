7import { useState, useEffect } from 'react';
8import { invoke } from '@tauri-apps/api/core';

10// Cache: assetId -> blob URL (without hash).
11const blobCache = new Map<string, string>();
12// Cache: assetId -> MIME type, learned from db_get_asset_meta_by_id on
13// first fetch. Used so blob URLs have the right `type` and the browser
14// dispatches the right rendering pipeline.
15const mimeCache = new Map<string, string>();
16 
17async function fetchMime(assetId: string): Promise<string> {
18 const cached = mimeCache.get(assetId);
19 if (cached) return cached;
20 try {
21 const meta = await invoke<{ mime_type: string | null } | null>(
22 'db_get_asset_meta_by_id', { assetId },
23 );
24 const mime = meta?.mime_type || 'application/octet-stream';
25 mimeCache.set(assetId, mime);
26 return mime;
27 } catch {
28 return 'application/octet-stream';
29 }
30}

32/** Load an asset from SQLite and return a blob URL. Uses a cache. */
33export async function getAssetUrl(
34 assetId: string | undefined,
35 hash?: string,
36): Promise<string | undefined> {
37 if (!assetId) return undefined;
38 let blobUrl = blobCache.get(assetId);
39 if (!blobUrl) {
40 try {
41 const [data, mime] = await Promise.all([
42 invoke<ArrayBuffer>('db_get_asset_by_id', { assetId }),
43 fetchMime(assetId),
44 ]);
45 const blob = new Blob([new Uint8Array(data)], { type: mime });
46 blobUrl = URL.createObjectURL(blob);
47 blobCache.set(assetId, blobUrl);
48 } catch {
49 return undefined;
50 }
51 }
52 return hash ? `${blobUrl}#${hash}` : blobUrl;
53}

55/** React hook: load an asset from SQLite as a blob URL.
56 *
57 * Listens for `eigendeck:asset-changed` (fired by invalidateRenderedAsset
58 * after the file watcher reloads an asset). Drops the cached blob URL
59 * on assetId match and re-fetches so the live `<img>` picks up new bytes.
60 */
61export function useAssetUrl(
62 assetId: string | undefined,
63 hash?: string,
64): string | undefined {
65 const [url, setUrl] = useState<string | undefined>(() => {
66 if (!assetId) return undefined;
67 const cached = blobCache.get(assetId);
68 return cached ? (hash ? `${cached}#${hash}` : cached) : undefined;
69 });
70 const [refreshKey, setRefreshKey] = useState(0);
71 
72 useEffect(() => {
73 if (!assetId) return;
74 const handler = (e: Event) => {
75 const detail = (e as CustomEvent).detail as { assetId?: string } | undefined;
76 if (detail?.assetId === assetId) {
77 invalidateAsset(assetId);
78 setRefreshKey((k) => k + 1);
79 }
80 };
81 window.addEventListener('eigendeck:asset-changed', handler);
82 return () => window.removeEventListener('eigendeck:asset-changed', handler);
83 }, [assetId]);
84 
85 useEffect(() => {
86 if (!assetId) { setUrl(undefined); return; }
87 getAssetUrl(assetId, hash).then(setUrl);
88 }, [assetId, hash, refreshKey]);
89 
90 return url;
91}

98/** Invalidate a specific cached asset (e.g. after re-import). */
99export function invalidateAsset(assetId: string) {
100 const old = blobCache.get(assetId);
101 if (old) {
102 URL.revokeObjectURL(old);
103 blobCache.delete(assetId);
104 }
105 mimeCache.delete(assetId);
106}

108/** Clean up all cached blob URLs (call on project close) */
109export function clearAssetCache() {

113 blobCache.clear();
114 mimeCache.clear();
