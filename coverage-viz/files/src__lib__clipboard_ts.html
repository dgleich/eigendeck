1// Eigendeck clipboard interop — shared marker for round-tripping
2// inline text markup (bold / italic / sub / sup / color spans)
3// through copy-paste back into eigendeck text elements without
4// accepting arbitrary external HTML.
5//
6// Producers (places that emit eigendeck HTML the user might copy):
7// - onCopy in SlideElementRenderer's contentEditable: prepends
8// the marker when the user copies a selection inside a text
9// element being edited.
10// - HTML export (renderSlideForPrint, etc.): prepends the marker
11// per text element so a full-element copy from a browser
12// viewing an exported page also trips the trusted-paste path.
13//
14// Consumers:
15// - onPaste in SlideElementRenderer: checks for marker in
16// clipboard's text/html; present → trust the HTML (strip

19/** Legacy magic-comment marker for eigendeck-origin HTML. Kept for back-compat
20 * (and it works in-webview, where clipboardData round-trips verbatim), but a
21 * leading HTML comment does NOT reliably survive the macOS NSPasteboard's
22 * text/html re-serialization — so it can't be the only signal. Versioned. */
23export const EIGENDECK_PASTE_MARKER = '<!--eigendeck-copy:v1-->';
24 
25/** Robust marker: a data ATTRIBUTE on a real wrapper element. Unlike a comment,
26 * a DOM attribute survives the OS clipboard's HTML re-serialization (macOS), so
27 * an eigendeck→eigendeck paste is still recognized after a native pasteboard
28 * round-trip. `div` wrapper (valid around block or inline content). */
29const EIGENDECK_ATTR_MARKER = 'data-eigendeck-copy';
30 
31/** Prepend the comment marker only (no wrapper element — no layout impact).
32 * Used where the marked HTML is RENDERED (HTML export / print), so it must not
33 * alter structure; the comment is enough there because that HTML isn't put on
34 * the OS clipboard until a user copies it (and the data-attribute form is used
35 * for the in-app clipboard producers). */
36export function markAsEigendeck(html: string): string {
37 return EIGENDECK_PASTE_MARKER + html;
38}
39 
40/** Wrap with BOTH markers for content going onto the CLIPBOARD. The data
41 * attribute survives the macOS NSPasteboard's text/html re-serialization (a
42 * leading comment does not), so an eigendeck→eigendeck paste is still routed
43 * through the trusted/element path after a native pasteboard round-trip. */
44export function markAsEigendeckForClipboard(html: string): string {
45 return `${EIGENDECK_PASTE_MARKER}<div ${EIGENDECK_ATTR_MARKER}="v1">${html}</div>`;
46}
47 
48/** True iff the HTML carries either marker anywhere. */
49export function hasEigendeckMarker(html: string): boolean {
50 return html.includes(EIGENDECK_ATTR_MARKER) || html.includes(EIGENDECK_PASTE_MARKER);
51}
52 
53/** Remove both markers (the comment, and the data attribute — leaving the plain
54 * wrapper div, which the sanitizer keeps) before inserting trusted HTML. */
55export function stripEigendeckMarker(html: string): string {
56 return html
57 .split(EIGENDECK_PASTE_MARKER).join('')
58 .replace(new RegExp(`\\s*${EIGENDECK_ATTR_MARKER}="v1"`, 'gi'), '');
59}
