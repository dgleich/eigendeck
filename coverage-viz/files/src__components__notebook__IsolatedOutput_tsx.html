1// Render a script-bearing notebook output (Plotly, bokeh, any self-contained
2// interactive text/html) in an OPAQUE-ORIGIN sandboxed iframe
3// (docs/NOTEBOOK-ISOLATION.md). It stays interactive but has no line to Tauri, so
4// a crafted deck's "output" can't reach the filesystem. Static output is
5// sanitized inline instead (CellOutput); only executable output reaches here.
6//
7// The iframe carries `el-demo-frame`, so the parent rAF pump + relay armed by
8// useDemoHost (SlideEditor / PresentMode) drive it at full framerate, and it
9// reports its content height back so the box grows to fit.

42 useEffect(() => {
43 const onMsg = (e: MessageEvent) => {
44 const d = e.data as { __eigendeck?: number; type?: string; h?: number } | undefined;
45 if (!d || d.__eigendeck !== 1 || d.type !== 'iso-size') return;
46 if (e.source !== iframeRef.current?.contentWindow) return;
