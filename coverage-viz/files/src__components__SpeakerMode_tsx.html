18export function SpeakerMode() {
19 const { presentation, setPresenting } = usePresentationStore();
20 const [currentIndex, setCurrentIndex] = useState(
21 usePresentationStore.getState().currentSlideIndex
22 );
23 const [elapsed, setElapsed] = useState(0);
24 const [timerRunning, setTimerRunning] = useState(true);
25 // #29 — zoom the AUDIENCE (projector) slide from here; the audience slide
26 // itself has no zoom chrome. Toggle = center zoom; then move over the Current
27 // Slide preview to pan the focal point.
28 const ZOOM = 2.2;
29 const [zoomed, setZoomed] = useState(false);
30 // Swap Displays only makes sense with a real second monitor (the dual-screen
31 // projector path), not the single-screen screen-share window.
32 const [canSwap, setCanSwap] = useState(false);
33 useEffect(() => {
34 availableMonitors().then((m) => setCanSwap(m.length >= 2)).catch(() => setCanSwap(false));
35 }, []);
36 const startTime = useRef(Date.now());
37 const timerRef = useRef<number | null>(null);
38 
39 const totalSlides = presentation.slides.length;
40 
41 // Timer
42 useEffect(() => {
43 if (timerRunning) {
44 timerRef.current = window.setInterval(() => {
45 setElapsed(Math.floor((Date.now() - startTime.current) / 1000));
46 }, 1000);
47 }
48 return () => { if (timerRef.current) clearInterval(timerRef.current); };
49 }, [timerRunning]);
50 
51 const goTo = useCallback((index: number) => {
52 if (index < 0 || index >= totalSlides) return;
53 setCurrentIndex(index);
54 setZoomed(false); // a new slide starts un-zoomed (the projector resets too)
55 navigatePresenter(index);
56 usePresentationStore.getState().selectSlide(index);
57 }, [totalSlides]);
58 
59 const goNext = useCallback(() => goTo(currentIndex + 1), [currentIndex, goTo]);
60 const goPrev = useCallback(() => goTo(currentIndex - 1), [currentIndex, goTo]);
61 
62 // Toggle audience zoom (center). Pan happens via the preview move handler.
63 const toggleZoom = useCallback(() => {
64 setZoomed((on) => { const next = !on; void zoomPresenter(next ? ZOOM : 1, 0.5, 0.5); return next; });
65 }, []);
66 // While zoomed, moving over the Current-Slide preview steers the projector's
67 // focal point (normalized [0,1] over the preview, which shows the whole slide).
68 const panFromPreview = useCallback((e: React.MouseEvent) => {
69 if (!zoomed) return;
70 const r = (e.currentTarget as HTMLElement).getBoundingClientRect();
71 void zoomPresenter(ZOOM, clamp01((e.clientX - r.left) / r.width), clamp01((e.clientY - r.top) / r.height));
72 }, [zoomed]);
73 
74 // Keyboard navigation
75 useEffect(() => {
76 const handleKeyDown = (e: KeyboardEvent) => {
77 switch (e.key) {
78 case 'Escape':
79 closePresenterWindow();
80 setPresenting(false);
81 break;
82 case 'ArrowRight': case 'ArrowDown': case ' ': case 'PageDown':
83 e.preventDefault(); goNext(); break;
84 case 'ArrowLeft': case 'ArrowUp': case 'PageUp':
85 e.preventDefault(); goPrev(); break;
86 case 'Home': e.preventDefault(); goTo(0); break;
87 case 'End': e.preventDefault(); goTo(totalSlides - 1); break;
88 }
89 };
90 window.addEventListener('keydown', handleKeyDown);
91 return () => window.removeEventListener('keydown', handleKeyDown);
92 }, [goNext, goPrev, goTo, totalSlides, setPresenting]);
93 
94 // The projector window forwards its own keyboard/clicker presses here (it
95 // doesn't own the index). Drive the same goTo so both windows stay in sync.
96 useEffect(() => {
97 const unlistenP = listen<{ index: number }>('presenter:nav', (e) => goTo(e.payload.index));
98 return () => { unlistenP.then((fn) => fn()); };
99 }, [goTo]);
100 
101 const slide = presentation.slides[currentIndex];
102 const nextSlide = currentIndex < totalSlides - 1 ? presentation.slides[currentIndex + 1] : null;
103 
104 const formatTime = (s: number) => {
105 const m = Math.floor(s / 60);
106 const sec = s % 60;
107 return `${m}:${sec.toString().padStart(2, '0')}`;
108 };

110 return (
111 <div className="speaker-mode">
112 <div className="speaker-header">
113 <div className="speaker-timer">
114 <span className="speaker-time">{formatTime(elapsed)}</span>
115 <button onClick={() => setTimerRunning(!timerRunning)}>
116 {timerRunning ? 'Pause' : 'Resume'}
117 </button>
118 <button onClick={() => { startTime.current = Date.now(); setElapsed(0); }}>
119 Reset
120 </button>
121 </div>
122 <div className="speaker-slide-count">
123 Slide {currentIndex + 1} / {totalSlides}
124 {' '}(#{getSlideNumber(presentation.slides, currentIndex)})
125 </div>
126 <div style={{ display: 'flex', gap: 8 }}>
127 {canSwap && (
128 <button className="speaker-swap" onClick={() => { void swapPresenterDisplay(); }} title="Swap which display shows the slides vs the speaker view">
129 Swap Displays
130 </button>
131 )}
132 <button className="speaker-exit" onClick={() => { closePresenterWindow(); setPresenting(false); }}>
133 End Presentation
134 </button>
135 </div>
136 </div>
137 
138 <div className="speaker-body">
139 {/* Current slide preview */}
140 <div className="speaker-current">
141 <div className="speaker-preview-label" style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
142 <span>Current Slide{zoomed && <span className="speaker-zoom-hint"> — move to pan the audience view</span>}</span>
143 <button className="speaker-zoom-btn" onClick={toggleZoom} aria-pressed={zoomed}
144 title="Zoom the audience slide (then move over this preview to pan)">
145 {zoomed ? 'Zoom out' : 'Zoom in'}
146 </button>
147 </div>
148 <div className="speaker-preview" style={zoomed ? { cursor: 'crosshair' } : undefined} onMouseMove={panFromPreview}>
149 {slide && <SlideThumbnail presentation={presentation} slide={slide} imageTier={ASSET_TIER.full} />}
150 </div>
151 {/* Notes */}
152 <div className="speaker-notes">
153 <div className="speaker-notes-label">Notes</div>
154 <div className="speaker-notes-text">
155 {slide?.notes || <span style={{ color: '#999', fontStyle: 'italic' }}>No notes for this slide</span>}
156 </div>
157 </div>
158 </div>
159 
160 {/* Next slide preview */}
161 <div className="speaker-next">
162 <div className="speaker-preview-label">Next Slide</div>
163 {nextSlide ? (
164 <div className="speaker-preview speaker-preview-small">
165 <SlideThumbnail presentation={presentation} slide={nextSlide} imageTier={ASSET_TIER.full} />
166 </div>
167 ) : (
168 <div className="speaker-preview-empty">End of presentation</div>
169 )}
170 </div>
171 </div>
172 
173 <div className="speaker-nav">
174 <button onClick={goPrev} disabled={currentIndex === 0}>&larr; Previous</button>
175 <button onClick={goNext} disabled={currentIndex >= totalSlides - 1}>Next &rarr;</button>
176 </div>
177 </div>
