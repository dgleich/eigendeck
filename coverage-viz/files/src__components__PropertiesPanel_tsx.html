19/** Strip per-run inline text colors (the format toolbar's foreColor produces
20 * `<span style="color:…">` / `<font color>`) so the element-level Text Color
21 * governs the whole element uniformly. DOM-based so it handles nested markup. */
22function stripInlineTextColors(html: string): string {
23 if (typeof document === 'undefined' || !html) return html;
24 const div = document.createElement('div');
25 div.innerHTML = html;
26 div.querySelectorAll('*').forEach((node) => {
27 const el = node as HTMLElement;
28 el.style?.removeProperty('color');
29 el.removeAttribute('color'); // legacy <font color="…">
30 if (el.getAttribute('style') === '') el.removeAttribute('style');
31 });
32 return div.innerHTML;
33}
