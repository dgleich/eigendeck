1// Registry of pending notebook-overlay flushers.
2//
3// Each `useOverlay()` instance persists its overlay (the sidecar holding
4// cellEdits + cellOutputs) via its OWN 800ms debounced flush. That flush is
5// independent of the deck save, so a deck saved or closed within ~800ms of an
6// edit/run could drop the overlay (#123). Each hook registers its flush here so
7// the save path (flushToSqlite) and the close/quit path can FORCE any pending
8// overlay to disk immediately, bypassing the debounce.
9//
10// Plain module (no React / no store import) so both the hook and the store can
11// depend on it without a cycle.
12 
13type Flusher = () => Promise<void>;
14 
15const flushers = new Set<Flusher>();
16 
17/** Register a flush fn; returns an unregister fn for effect cleanup. */
18export function registerOverlayFlush(fn: Flusher): () => void {
19 flushers.add(fn);
20 return () => { flushers.delete(fn); };
21}
22 
23/**
24 * Force every registered overlay to flush now. Each flusher is a no-op when
25 * nothing changed, so this is cheap to call on every save/autosave. Per-flusher
26 * errors are swallowed so one bad overlay can't block the deck save.
27 */
28export async function flushAllOverlays(): Promise<void> {
29 await Promise.all([...flushers].map((f) => f().catch((e) => {
30 console.warn('overlay flush failed during forced flush:', e);
31 })));
