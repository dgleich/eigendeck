1// The one place external-file reads and the asset-type gate are combined.
2//
3// resolveAndGate() is what every read site (watcher, reload-from-disk, add-asset,
4// scan-on-load) should call — never `assetTypeGate` with a raw reference path.
5// It resolves the reference to its REAL target via the Rust `resolve_and_read`
6// primitive (following symlinks + normalizing), then runs the type gate on the
7// RESOLVED path + bytes. This makes the realpath contract structural: a caller
8// cannot accidentally gate an unresolved deck-supplied reference, and a symlink
9// `a.png -> ~/.ssh/id_rsa` is judged as `id_rsa`. See docs/ASSETS-SECURITY.md.
10//
11// It does NOT consult deck-trust or approval — that's a separate layer the caller
12// applies (a read only happens for a trusted deck + approved path). This module is
13// strictly the resolve + type-gate step.
14//
15// Two entry points over ONE internal `gateRead`, so the decision logic can't fork:
16// - resolveAndGate: full read, returns the bytes (for the actual read path).
17// - resolveAndGateDecision: bounded "sniff" read, returns only the verdict + resolved
18// path, never bytes (for the report/inspector, which classify but don't use the
19// bytes). The verdict is identical to a full read because the type gate inspects at
20// most a ~512B prefix — except .ipynb, which is re-read in full (needsFullContent).

32// Bytes read for a TYPE DECISION. Every content sniff inspects at most a ~512B prefix
33// (magic bytes / XML prolog / demo marker); 4 KB is generous headroom. .ipynb is the one
34// type that parses the whole file and is re-read in full below.
35const SNIFF_BYTES = 4096;

37export interface GateResult {
38 ok: boolean;
39 /** asset kind when ok; null otherwise */
40 kind: AssetKind | null;
41 /** why it failed: the type-gate reason (incl. 'too-large'), or 'unreadable' when the
42 * file couldn't be resolved/read at all (missing, not a regular file, denied). */
43 reason: GateReason | 'unreadable' | null;
44 /** the fully-resolved real target (null if it couldn't be resolved) */
45 canonicalPath: string | null;
46 /** the resolved bytes — only when ok, and only from resolveAndGate (the full read);
47 * resolveAndGateDecision never returns bytes (they may be a partial sniff). */
48 bytes: Uint8Array | null;
49 /** underlying error string when reason === 'unreadable' */
50 error?: string;
51}

53/** The single resolve + read + gate. `maxBytes` undefined = full read. Returns the gate
54 * result plus whether the read was a truncated prefix of the file. Never throws. */
55async function gateRead(referencePath: string, maxBytes?: number): Promise<{ result: GateResult; truncated: boolean }> {
56 let res: ResolvedRead;
57 try {
58 res = await invoke<ResolvedRead>('resolve_and_read', { path: referencePath, maxBytes: maxBytes ?? null });
59 } catch (e) {
60 // resolve_and_read rejects oversized files ("file too large (N bytes)") BEFORE reading;
61 // surface that distinctly so the UI can say so (vs a generic missing/unreadable file).
62 const err = String(e);
63 const reason = /file too large/i.test(err) ? 'too-large' : 'unreadable';
64 return { result: { ok: false, kind: null, reason, canonicalPath: null, bytes: null, error: err }, truncated: false };
65 }
66 const bytes = new Uint8Array(res.bytes);
67 const gate = assetTypeGate(bytes, res.canonicalPath);
68 return {
69 result: { ok: gate.ok, kind: gate.kind, reason: gate.reason, canonicalPath: res.canonicalPath, bytes: gate.ok ? bytes : null },
70 truncated: res.bytes.length < res.size,
71 };
72}

74/**
75 * Resolve `referencePath` to its real target, read it IN FULL, and gate on the resolved
76 * path+bytes. Use this when you need the bytes (the actual read path). Never throws.
77 */
78export async function resolveAndGate(referencePath: string): Promise<GateResult> {
79 return (await gateRead(referencePath)).result;
80}

82/**
83 * Same resolve + gate, but reads only a bounded prefix — for callers that classify a
84 * linked asset (report / inspector) but do NOT use its bytes. Avoids reading a 100 MB
85 * video off disk just to learn it's a watchable video. The verdict equals a full read
86 * because the type gate is prefix-bounded; a truncated full-content type (.ipynb) is
87 * re-read in full so it can't disagree with the real read gate. Returns bytes: null.
88 */
89export async function resolveAndGateDecision(referencePath: string): Promise<GateResult> {
90 const first = await gateRead(referencePath, SNIFF_BYTES);
91 let result = first.result;
92 if (result.canonicalPath && first.truncated && needsFullContent(extensionOf(result.canonicalPath))) {
93 result = (await gateRead(referencePath)).result; // .ipynb: the sniff was too short to parse
94 }
95 return { ...result, bytes: null };
