3/**
4 * Shift an element by (dx, dy) IN PLACE — arrows move their four endpoint
5 * coords, every other type moves its position box. Used when dropping a
6 * duplicate/paste on the same slide so the copy doesn't stack exactly on top.
7 */
8export function offsetElement(el: SlideElement, dx: number, dy: number): void {
9 // eslint-disable-next-line @typescript-eslint/no-explicit-any
10 const e = el as any;
11 if (e.type === 'arrow') {
12 e.x1 += dx; e.y1 += dy; e.x2 += dx; e.y2 += dy;
13 } else {
14 e.position = { ...e.position, x: e.position.x + dx, y: e.position.y + dy };
15 }
