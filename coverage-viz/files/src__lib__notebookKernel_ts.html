1// Resolve the kernel + server for a NotebookElement.
2//
3// Two-step resolution:
4// 1. The CASCADE picks the requested kernel name. Default-setting
5// flavor per DESIGN_DECISIONS.md "Preferences cascade":
6// element.kernel.kernelName
7// ?? config.notebookKernel.kernelName
8// ?? notebook.metadata.kernelspec.name
9// ?? 'python3' (final fallback)
10// Kernel kind ('external' | 'lite') walks the same cascade.
11//
12// 2. The REGISTRY picks the server. Notebook elements never carry
13// baseUrl + token (those are auth artifacts; decks are
14// git-committable). Instead we scan PrefSchema.jupyterServers
15// for the first entry whose `availableKernels` contains the
16// requested kernel name. No match → ResolvedExternal.server is
17// null and the status pill (Phase 3) shows red.

29export interface ResolvedExternal {
30 kind: 'external';
31 /** The kernel name to start (e.g., 'python3', 'julia-1.10'). */
32 kernelName: string;
33 /** The registered server we'll dial, or null if no entry in the
34 * registry advertises the requested kernel. Consumers should
35 * render a "no matching server" state when null rather than
36 * attempting to connect. */
37 server: JupyterServerEntry | null;
38}

46/** Pick the first server in `registry` advertising `kernelName`. The
47 * registry order is the user's preference order; reorder in Settings
48 * to change the default match. A server with no `availableKernels`
49 * field (never tested) is skipped — discovery hasn't confirmed it
50 * has the kernel, so we don't gamble during a talk. */
51export function findServerForKernel(
52 kernelName: string,
53 registry: JupyterServerEntry[],
54): JupyterServerEntry | null {
55 for (const s of registry) {
56 if (s.availableKernels?.includes(kernelName)) return s;
57 }
58 return null;
59}

61export function resolveNotebookKernel(
62 element: NotebookElement,
63 config: PresentationConfig | undefined,
64 notebook: Notebook | null,
65 registry: JupyterServerEntry[],

67 // Step 1a — pick the kind via the standard cascade.
68 const kind =
69 element.kernel?.kind
70 ?? config?.notebookKernel?.kind
71 ?? 'external';
72 
73 if (kind === 'lite') return { kind: 'lite' };
74 
75 // Step 1b — pick the kernel name via the cascade.
76 const elemExt = element.kernel?.kind === 'external' ? element.kernel : undefined;
77 const deckExt = config?.notebookKernel?.kind === 'external'
78 ? config.notebookKernel : undefined;
79 const kernelName =
80 elemExt?.kernelName
81 ?? deckExt?.kernelName
82 ?? notebook?.kernelspecName
83 ?? DEFAULT_KERNEL_NAME;
84 
85 // Step 2 — registry lookup. Decoupled from the cascade so server
86 // selection lives entirely on the local machine; deck files never
87 // know the URL or token.
88 return {
89 kind: 'external',
90 kernelName,
91 server: findServerForKernel(kernelName, registry),
92 };
