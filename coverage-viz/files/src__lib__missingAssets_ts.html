1// Tracks asset source files that have gone missing from disk (moved or
2// deleted out from under a linked asset). Issue #74.
3//
4// Detection fires from three places, all of which already stat/read the
5// source file:
6// • scanForChangedAssets() — on project open (the common case: the file
7// moved while the deck was closed). UNGATED by the auto-reload toggle —
8// a missing source matters even if you're not auto-reloading it.
9// • WatcherRegistry.handleChange() — live, while the deck is open (covers
10// a delete/rename that the OS reports as a change event).
11// • AssetSection "Reload from disk now" — manual retry.
12//
13// IMPORTANT: a missing source never loses content. The last-loaded bytes stay
14// in the assets table (the snapshot), so the slide keeps rendering. This module
15// only SURFACES the situation (inspector banner + count) and powers Relocate.
16//
17// Observable three ways so any layer can react without a store dependency:
18// • subscribeMissing(cb) — low-level listener (set of missing asset ids)
19// • useAssetMissing(id) / useMissingCount() — React hooks
20// • window 'eigendeck:asset-missing' / 'eigendeck:asset-found' CustomEvents
21// (matches the existing 'eigendeck:asset-changed' convention)
22

25export interface MissingAsset {
26 assetId: string;
27 /** Human-readable path label (the asset row's `path`, else external_path). */
28 path: string;
29}
30 
31// assetId -> path label
32const missing = new Map<string, string>();
33 
34type Listener = (ids: ReadonlySet<string>) => void;
35const listeners = new Set<Listener>();
36 
37function notify(): void {
38 const ids = new Set(missing.keys());
39 for (const l of listeners) l(ids);
40}
41 
42function dispatch(name: string, detail: unknown): void {
43 if (typeof window !== 'undefined' && typeof window.dispatchEvent === 'function') {
44 window.dispatchEvent(new CustomEvent(name, { detail }));
45 }
46}

48/** Mark an asset's source file as missing. No-op if already marked the same. */
49export function markAssetMissing(assetId: string, path: string): void {
50 if (missing.get(assetId) === path) return;
51 missing.set(assetId, path);
52 notify();
53 dispatch('eigendeck:asset-missing', { assetId, path });
54}
55 
56/** Clear the missing flag — the source reappeared / was relocated. */
57export function markAssetFound(assetId: string): void {
58 if (!missing.has(assetId)) return;
59 missing.delete(assetId);
60 notify();
61 dispatch('eigendeck:asset-found', { assetId });
62}
63 
64export function isAssetMissing(assetId: string): boolean {
65 return missing.has(assetId);
66}
67 
68export function getMissingAssets(): MissingAsset[] {
69 return Array.from(missing, ([assetId, path]) => ({ assetId, path }));
70}
71 
72/** Drop all missing flags — called on project close so stale flags from a
73 * previous deck don't leak into the next one. */
74export function clearAllMissing(): void {
75 if (missing.size === 0) return;
76 missing.clear();
77 notify();
78}
79 
80export function subscribeMissing(l: Listener): () => void {
81 listeners.add(l);
82 l(new Set(missing.keys()));
83 return () => { listeners.delete(l); };
84}

86// --- React hooks (getSnapshot returns a primitive → stable, no extra renders).
87 
88export function useAssetMissing(assetId: string): boolean {
89 return useSyncExternalStore(
90 (cb) => subscribeMissing(() => cb()),
91 () => missing.has(assetId),
92 );
