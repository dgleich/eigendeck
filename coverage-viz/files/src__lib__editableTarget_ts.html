1// Is a clipboard/keyboard event happening INSIDE a text editor (a
2// contentEditable text element, or an <input>/<textarea>)?
3//
4// The obvious check — `e.target.closest('[contenteditable="true"]')` — is not
5// enough. WebKit (Tauri's engine) can dispatch a KEYBOARD-initiated paste/copy
6// (Cmd+V / Cmd+C) with `event.target` set to <body>, NOT the focused editable,
7// even while the caret is sitting in a contentEditable. When that happens an
8// `e.target`-only guard fails to bail, so a window-level canvas paste handler
9// ALSO runs — the user gets BOTH a new element on the canvas AND text inserted
10// at the caret (the browser's default paste still targets the focused editable).
11// Same failure on copy: an in-edit copy gets treated as a canvas element copy
12// and clobbers the clipboard.
13//
14// The fix: also consult the FOCUS (`document.activeElement`) and the current
15// selection anchor, which stay correct regardless of the event target.
16 
17/** True iff `node` is, or is inside, a contentEditable / input / textarea. */
18export function isEditableNode(node: Node | EventTarget | null | undefined): boolean {
19 const el = asElement(node);
20 if (!el) return false;
21 const tag = el.tagName;
22 if (tag === 'INPUT' || tag === 'TEXTAREA') return true;
23 // `contenteditable="true"` on the element or any ancestor.
24 return !!el.closest?.('[contenteditable="true"]');
25}

27function asElement(node: Node | EventTarget | null | undefined): Element | null {
28 if (!node) return null;
29 // Element node.
30 if (typeof (node as Element).closest === 'function') return node as Element;
31 // Text (or other) node → its parent element.
32 const parent = (node as Node).parentElement ?? (node as Node).parentNode;
33 return parent && typeof (parent as Element).closest === 'function' ? (parent as Element) : null;
34}

36/** Minimal shape of the document facts we consult — injectable for tests. */
37export interface EditorFocusDoc {
38 activeElement: Element | null;
39 getSelection?: () => { focusNode: Node | null } | null;
40}

42/**
43 * True when a clipboard/keyboard event should be treated as happening inside a
44 * text editor. Checks the event target AND the focused element AND the current
45 * selection anchor, so it's correct even when WebKit mis-targets the event to
46 * <body> (see file header). `doc` is injectable for testing.
47 */
48export function eventInTextEditor(
49 e: { target?: EventTarget | null },
50 doc?: EditorFocusDoc | null,
51): boolean {
52 if (isEditableNode(e.target ?? null)) return true;
53 const d = doc ?? (typeof document !== 'undefined' ? (document as unknown as EditorFocusDoc) : null);
54 if (!d) return false;
55 if (isEditableNode(d.activeElement)) return true;
56 const focusNode = d.getSelection?.()?.focusNode ?? null;
57 return isEditableNode(focusNode);
58}
