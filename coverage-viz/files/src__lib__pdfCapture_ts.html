1// PDF (Screenshots) export capture pass (#176). Mirrors captureAllSnapshots
2// (snapshotAll.ts): step through every slide, let it paint, and screenshot its
3// `.slide-canvas` to a JPEG byte array — reporting per-slide progress so the
4// caller can drive the same fade/counter busy overlay the Snapshot commands use.
5// The actual PDF assembly (buildPdf) + native file write stay with the caller;
6// this is just the "walk the deck and grab a picture of each slide" half, split
7// out so the progress cadence is unit-testable (jsdom can't rasterize, but the
8// onProgress calls + slide restoration are exactly what we want to pin).

19/**
20 * Screenshot every slide to a JPEG byte array (one per slide, in order). Adds
21 * `body.pdf-capturing` for the duration (hides editor chrome — resize handles,
22 * the cut-off badge, etc.), restores the original slide + selection when done
23 * (even on error), and fires `onProgress({current,total})` before each slide so
24 * the caller can update the busy counter. A slide whose canvas can't be captured
25 * yields an empty byte array (the caller renders a blank page) rather than
26 * aborting the whole export. Best-effort; never throws for a single bad slide.
27 */
28export async function captureSlideJpegs(
29 presentation: Presentation,
30 opts: {
31 onProgress?: (p: PdfCaptureProgress) => void;
32 dwellMs?: number;
33 width?: number;
34 height?: number;
35 } = {},
36): Promise<Uint8Array[]> {
37 const { onProgress, dwellMs = DWELL_MS, width: W = 1920, height: H = 1080 } = opts;
38 const { domToDataUrl } = await import('modern-screenshot');
39 const store = usePresentationStore.getState();
40 const originalIdx = store.currentSlideIndex;
41 const originalSel = store.selectedObject;
42 
43 usePresentationStore.getState().selectObject({ type: 'slide' });
44 document.body.classList.add('pdf-capturing');
45 
46 const jpegImages: Uint8Array[] = [];
47 const total = presentation.slides.length;
48 try {
49 for (let i = 0; i < total; i++) {
50 onProgress?.({ current: i + 1, total });
51 usePresentationStore.getState().selectSlide(i);
52 await new Promise((r) => setTimeout(r, dwellMs));
53 
54 const canvas = document.querySelector('.slide-canvas') as HTMLElement | null;
55 let bytes = new Uint8Array(0);
56 if (canvas) {
57 try {
58 const dataUrl = await domToDataUrl(canvas, {
59 width: W, height: H, scale: 1,
60 style: { transform: 'none', transformOrigin: 'top left' },
61 });
62 // Flatten onto an opaque white background and re-encode as JPEG (the PDF
63 // embeds JPEG frames; the source PNG may be transparent).
64 const img = new Image();
65 await new Promise<void>((resolve) => { img.onload = () => resolve(); img.onerror = () => resolve(); img.src = dataUrl; });
66 const cvs = document.createElement('canvas');
67 cvs.width = W; cvs.height = H;
68 const ctx = cvs.getContext('2d')!;
69 ctx.fillStyle = '#ffffff';
70 ctx.fillRect(0, 0, W, H);
71 ctx.drawImage(img, 0, 0, W, H);
72 const jpegUrl = cvs.toDataURL('image/jpeg', 0.92);
73 const b64 = jpegUrl.split(',')[1];
74 const binary = atob(b64);
75 bytes = new Uint8Array(binary.length);
76 for (let j = 0; j < binary.length; j++) bytes[j] = binary.charCodeAt(j);
77 } catch (e) {
78 console.warn(`Failed to capture slide ${i + 1}:`, e);
79 }
80 }
81 jpegImages.push(bytes);
82 }
83 } finally {
84 document.body.classList.remove('pdf-capturing');
85 usePresentationStore.getState().selectSlide(originalIdx);
86 usePresentationStore.getState().selectObject(originalSel);
87 }
88 return jpegImages;
89}
