5// slide as an `html` element (#137). This is the single decision point, kept
6// pure so it can be unit-tested without the Tauri drag-drop plumbing. The

14export type DroppedHtmlRoute =
15 /** A marked, multi-piece demo — one demo-piece element per referenced name. */
16 | { kind: 'demo-pieces'; pieces: string[] }
17 /** A marked single-frame demo. */
18 | { kind: 'demo' }
19 /** Not a demo, but a usable raw-HTML snippet → insert an `html` element. */
20 | { kind: 'html-element'; html: string; interactive: boolean }
21 /** Not a demo and not a usable snippet → reject with reasons for a toast. */

24/**
25 * Classify the text of a dropped `.html` file. The Eigendeck-demo marker wins:
26 * only a non-demo file falls back to the `html`-element path. A multi-piece demo
27 * is one that both references `piece === '…'` checks AND uses BroadcastChannel
28 * (the piece-coordination transport) — matching the original insertion logic.
29 */
30export function classifyDroppedHtml(html: string): DroppedHtmlRoute {
31 if (isEigendeckDemo(html).ok) {
32 const pieces = extractDemoPieceNames(html);
33 if (pieces.length > 0 && html.includes('BroadcastChannel')) {
34 return { kind: 'demo-pieces', pieces };
35 }
36 return { kind: 'demo' };
37 }
38 const v = validateHtmlSnippet(html);
39 if (v.ok) return { kind: 'html-element', html: v.html, interactive: v.interactive };
40 return { kind: 'reject', problems: v.problems };
41}
