1// "Paste as…" — let the user pick which representation on the clipboard to
2// paste, instead of the automatic ladder (docs/copy-and-paste.md Stage 4).
3//
4// Two pure-ish pieces here:
5// • clipboardRepresentations(types) — map a raw pasteboard type/MIME list to
6// the offered choices (Image / SVG / PDF / HTML / Text). PURE + unit-tested.
7// • gatherClipboardTypes() / readRepresentation(kind) — talk to the OS
8// clipboard (native NSPasteboard on macOS, the async Clipboard API
9// elsewhere). The actual element INSERT is done by SlideEditor's existing
10// paste helpers (reused via a CustomEvent), so nothing is duplicated here.
11 
12// 'html-image' = rasterize the clipboard HTML to a STATIC image (the screenshot
13// path), offered only when both HTML and text are present. Unlike the automatic
14// ⌘V screenshot it's an explicit choice, and the capture never touches the
15// network (see captureHtmlToPng) so it can't hang on a remote resource.
16export type PasteKind = 'image' | 'svg' | 'pdf' | 'html-image' | 'html' | 'text';

58/** Which representations does this raw clipboard type/UTI list offer? PURE. */
59export function clipboardRepresentations(types: readonly string[]): PasteRep[] {
60 const set = new Set(types.map((t) => t.toLowerCase()));
61 const has = (toks: string[]) => toks.some((t) => set.has(t.toLowerCase()));
62 // "Simple Image" (rasterize the HTML) is offered only when BOTH HTML and text
63 // are present — i.e. a rich copy (browser/Word/Docs) where rasterizing is a
64 // meaningful alternative to pasting the text.
65 const htmlImage = has(HTML_TYPES) && has(TEXT_TYPES);
66 return KINDS
67 .filter((k) => (k.kind === 'html-image' ? htmlImage : has(k.types)))
68 .map((k) => ({ kind: k.kind, label: k.label }));
69}

80/** Union of the native pasteboard UTIs (macOS) and the async Clipboard API MIME
81 * types (Linux/Windows). Used to build the chooser. */
82export async function gatherClipboardTypes(): Promise<string[]> {
83 const out = new Set<string>();
84 const native = await invokeSafe<string[]>('pasteboard_list_types', {});
85 native?.forEach((t) => out.add(t));
86 if (typeof navigator !== 'undefined' && navigator.clipboard?.read) {
87 try {
88 const items = await navigator.clipboard.read();
89 items.forEach((it) => it.types.forEach((t) => out.add(t)));
90 } catch { /* permission / not focused */ }
91 }
92 return [...out];
93}

95async function readNativeBytes(uti: string): Promise<Uint8Array | null> {
96 const arr = await invokeSafe<number[] | null>('pasteboard_read_type', { uti });

100async function readAsyncBytes(mimes: string[]): Promise<Uint8Array | null> {
101 if (typeof navigator === 'undefined' || !navigator.clipboard?.read) return null;
102 try {
103 const items = await navigator.clipboard.read();
104 for (const it of items) {
105 for (const m of mimes) {
106 if (it.types.includes(m)) {
107 const blob = await it.getType(m);
108 return new Uint8Array(await blob.arrayBuffer());
109 }
110 }
111 }
112 } catch { /* ignore */ }
113 return null;
114}

125/** Read the bytes/text for the chosen representation off the clipboard, native
126 * first (macOS) then the async Clipboard API. Returns null if it's gone. */
127export async function readRepresentation(kind: PasteKind): Promise<RepData | null> {
128 const spec = KINDS.find((k) => k.kind === kind);
129 if (!spec) return null;
130 
131 if (kind === 'text') {
132 const nb = await readNativeBytes('public.utf8-plain-text') ?? await readNativeBytes('public.text');
133 if (nb) return { kind, text: new TextDecoder('utf-8').decode(nb) };
134 try {
135 const t = await navigator.clipboard?.readText?.();
136 if (t) return { kind, text: t };
137 } catch { /* ignore */ }
138 return null;
139 }
140 
141 if (kind === 'html' || kind === 'html-image') {
142 // Both read the clipboard HTML; the caller rasterizes it for 'html-image'
143 // and creates a raw HTML element for 'html'.
144 const nb = await readNativeBytes('public.html');
145 if (nb) return { kind, html: new TextDecoder('utf-8').decode(nb) };
146 const ab = await readAsyncBytes(['text/html']);
147 if (ab) return { kind, html: new TextDecoder('utf-8').decode(ab) };
148 return null;
149 }
150 
151 // image / svg / pdf → raw bytes
152 for (const t of spec.types) {
153 const nb = await readNativeBytes(t);
154 if (nb) return { kind, bytes: nb, mime: spec.mime, ext: spec.ext };
155 }
156 const ab = await readAsyncBytes(spec.types);
157 if (ab) return { kind, bytes: ab, mime: spec.mime, ext: spec.ext };
158 return null;
