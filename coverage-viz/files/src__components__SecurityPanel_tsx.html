37// Which band to show, in precedence order: nothing linked → watching off (global wins
38// over per-deck) → trust state.
39function deckCase(r: DeckSecurityReport): DeckCase {
40 if (r.rows.length === 0) return 'A';
41 if (!r.watch.global) return 'B1';
42 if (r.watch.deck === 'off') return 'B2';
43 if (r.status === 'untrusted-ttl') return 'F';
44 if (r.status === 'untrusted-new') return 'E';
45 return r.trustReason === 'file-new' ? 'C' : 'D';
46}

48// Relative for the last week, absolute after.
49function fmtWhen(at: number | null): string {
50 if (!at) return '';
51 const ms = Date.now() - at, day = 86_400_000;
52 if (ms < 3_600_000) return 'just now';
53 if (ms < day) { const h = Math.round(ms / 3_600_000); return h <= 1 ? '1 hour ago' : `${h} hours ago`; }
54 if (ms < 7 * day) { const d = Math.round(ms / day); return d <= 1 ? 'yesterday' : `${d} days ago`; }
55 return new Date(at).toLocaleDateString();
56}
57// Plain, accurate copy for a blocked row, keyed by WHY the gate rejected it. Most
58// blocked files are benign (an older or hand-made file that isn't a recognized type),

60function blockedText(referencePath: string, reason: string | null): string {
61 const ext = (referencePath.split('.').pop() || '').toUpperCase();
62 const isHtml = ext === 'HTML';
63 switch (reason) {
64 case 'unsupported-demo-version':
65 return 'This demo was built with a newer Eigendeck. Update Eigendeck to run it.';
66 case 'bad-extension':
67 return `Eigendeck doesn't load ${ext || 'this kind of'} files.`;
68 case 'content-mismatch':
69 return isHtml
70 ? "Not a recognized Eigendeck demo (it may be an older or hand-made HTML file), so it won't run live."
71 : `This file's contents don't match a ${ext || 'file'} of that type, so Eigendeck won't load it.`;
72 default:
73 return "Eigendeck can't use this file, so it won't be loaded.";
74 }

474// Render a path so it wraps at '/' separators (a <wbr> after each slash) rather
475// than mid-segment. Pair with `overflow-wrap: anywhere` so an over-long single
476// segment can still break as a last resort.
477function PathText({ path }: { path: string }): React.ReactElement {
478 const segs = path.split('/');
479 return (
480 <>
481 {segs.map((seg, i) => (
482 <Fragment key={i}>{i > 0 ? <>/<wbr /></> : null}{seg}</Fragment>
483 ))}
484 </>

488function GroupedRows({ report, canAct, busy, onApprove, onApproveDir, onRevokeApproval }: {
489 report: DeckSecurityReport; canAct: boolean; busy: boolean;
490 onApprove: (assetId: string, ref: string) => void; onApproveDir: (dir: string) => void; onRevokeApproval: (assetId: string) => void;
491}): React.ReactElement {
492 // Group by resolved folder; unresolved rows (blocked / missing) share one group.
493 const groups = new Map<string, ExternalPathRow[]>();
494 for (const r of report.rows) {
495 const key = r.resolvedDir ?? '';
496 if (!groups.has(key)) groups.set(key, []);
497 groups.get(key)!.push(r);
498 }
499 // Trusted-but-can't-act == watching is off (global or per-deck) — the per-file
500 // controls are OVERRIDDEN by that higher-priority switch, so dim the list (motif).
501 const watchingOff = !!report.trusted && !canAct;
502 return (
503 <div style={{ display: 'flex', flexDirection: 'column', gap: 12, opacity: watchingOff ? OVERRIDDEN_DIM : 1 }}>
504 {[...groups.entries()].map(([dir, rows]) => {
505 const outside = dir !== '' && !isInsideDeck(dir, report.projectDir);
506 const eligibleHere = rows.filter((r) => r.state === 'eligible');
507 return (
508 <div key={dir || '(unresolved)'} style={{
509 border: `1px solid ${outside ? '#fde68a' : '#eee'}`,
510 background: outside ? '#fffbeb' : 'transparent',
511 borderRadius: 6, padding: 8,
512 }}>
513 <div style={{ display: 'flex', alignItems: 'center', gap: 8, marginBottom: 6 }}>
514 {/* The folder is the headline (bigger). Its files are listed under
515 it as descendants (↳ + filename) so the full path isn't repeated
516 per row. */}
517 <span style={{ fontSize: 13, fontWeight: 600, color: outside ? '#92400e' : '#374151', overflowWrap: 'anywhere' }}>
518 <span style={{ fontFamily: 'monospace' }}><PathText path={dir || 'Unresolved links'} /></span>
519 {outside && (
520 <span style={{ fontWeight: 400, fontStyle: 'italic', fontSize: 12, whiteSpace: 'nowrap' }}> · outside the deck folder</span>
521 )}
522 </span>
523 <span style={{ flex: 1 }} />
524 {/* Batch approve — only when there's more than one eligible file
525 (a single file's own Approve button already covers it). Kept
526 deliberately subtle so it blends into the folder band (amber on
527 the outside-the-deck warning band, gray otherwise) rather than
528 competing with the per-file Approve chips. */}
529 {canAct && eligibleHere.length > 1 && dir && (
530 <button onClick={() => onApproveDir(dir)} disabled={busy}
531 title={`Approve all ${eligibleHere.length} eligible files in this folder at once`}
532 style={{
533 padding: '2px 8px', fontSize: 11, cursor: 'pointer', borderRadius: 4,
534 background: 'transparent',
535 border: '1px solid ' + (outside ? '#fcd34d' : '#e5e7eb'),
536 color: outside ? '#92400e' : '#6b7280',
537 }}>
538 Approve all {eligibleHere.length} files
539 </button>
540 )}
541 </div>
542 <div style={{ display: 'flex', flexDirection: 'column', gap: 6 }}>
543 {rows.map((r) => <Row key={r.assetId} r={r} outside={outside} canAct={canAct} trusted={!!report.trusted} busy={busy} onApprove={onApprove} onRevokeApproval={onRevokeApproval} />)}
544 </div>
545 </div>
546 );
547 })}
548 </div>
549 );
550}

552function Row({ r, outside, canAct, trusted, busy, onApprove, onRevokeApproval }: {
553 r: ExternalPathRow; outside: boolean; canAct: boolean; trusted: boolean; busy: boolean;
554 onApprove: (assetId: string, ref: string) => void; onRevokeApproval: (assetId: string) => void;
555}): React.ReactElement {
556 const st = STATE_STYLE[r.state];
557 return (
558 // No stark white card — a subtle tint that sits gently on the folder band
559 // (a touch warmer than the amber warning band, near-transparent otherwise)
560 // so the list reads calmly instead of as high-contrast white boxes.
561 <div style={{
562 border: '1px solid ' + (outside ? '#fbe6b3' : '#eef0f2'),
563 borderRadius: 5, padding: '8px 10px',
564 background: outside ? 'rgba(255,255,255,0.45)' : 'rgba(0,0,0,0.015)',
565 }}>
566 <div style={{ display: 'flex', alignItems: 'center', gap: 8, justifyContent: 'space-between' }}>
567 {/* Just the file name, as a child of the folder headline above (↳). The
568 folder already carries the full real path, so we don't repeat it; the
569 authored ../-relative referencePath is never surfaced. Unresolved rows
570 (no resolved path) fall back to the reference. */}
571 <span style={{ fontFamily: 'monospace', fontSize: 11.5, overflowWrap: 'anywhere', color: '#374151' }}>
572 <span style={{ color: '#9ca3af', marginRight: 5, fontFamily: 'system-ui' }}>↳</span>
573 <PathText path={r.resolvedPath ? (r.resolvedPath.split('/').pop() || r.resolvedPath) : r.referencePath} />
574 </span>
575 <span style={{ flexShrink: 0, fontSize: 11, padding: '2px 8px', borderRadius: 10, color: st.color, background: st.bg }}>{st.label}</span>
576 </div>
577 <div style={{ display: 'flex', alignItems: 'center', gap: 10, marginTop: 5 }}>
578 <span style={{ fontSize: 11, color: '#999', display: 'inline-flex', alignItems: 'center' }}>
579 {r.usage}
580 {r.usage === 'unused' && <InfoTip text={UNUSED_TOOLTIP} />}
581 </span>
582 {r.state === 'approved' && r.approvedAt && (
583 <span style={{ fontSize: 11, color: st.color }}>Approved {fmtWhen(r.approvedAt)} · {howLabel(r.approvedHow)}</span>
584 )}
585 {r.state === 'forbidden' && <span style={{ fontSize: 11, color: st.color }}>{blockedText(r.referencePath, r.reason)}</span>}
586 {r.state === 'missing' && <span style={{ fontSize: 11, color: st.color }}>Source file not found on disk. Eigendeck is showing the last saved copy.</span>}
587 <span style={{ flex: 1 }} />
588 {r.state === 'approved' && canAct && (
589 <button onClick={() => onRevokeApproval(r.assetId)} disabled={busy} className="chip-btn danger" style={ghostBtn}>Revoke approval</button>
590 )}
591 {r.state === 'eligible' && (
592 canAct
593 ? <button onClick={() => onApprove(r.assetId, r.referencePath)} disabled={busy} className="chip-btn primary" style={smallBtn}>Approve</button>
594 : trusted
595 ? <span style={{ fontSize: 11, ...overriddenLabel }}>watching is off</span>
596 : <span style={{ fontSize: 11, color: '#999' }}>trust the deck first</span>
597 )}
598 </div>
599 </div>
600 );
601}
