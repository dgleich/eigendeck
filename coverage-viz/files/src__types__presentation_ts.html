117/**
118 * Parse a pasted color palette (#2) into a normalized list of #rrggbb hex
119 * strings. Accepts colors separated by commas / whitespace / newlines /
120 * semicolons, with or without a leading '#', in 3- or 6-digit hex. Expands
121 * shorthand (#abc → #aabbcc), lowercases, dedupes (order-preserving), and
122 * caps the count so a giant paste can't blow up the toolbar.
123 */
124export function parsePalette(input: string, max = 24): string[] {
125 if (!input) return [];
126 const out: string[] = [];
127 const seen = new Set<string>();
128 for (const raw of input.split(/[\s,;]+/)) {
129 const tok = raw.trim().replace(/^#/, '');
130 let hex: string | null = null;
131 if (/^[0-9a-fA-F]{6}$/.test(tok)) hex = tok.toLowerCase();
132 else if (/^[0-9a-fA-F]{3}$/.test(tok)) hex = tok.toLowerCase().split('').map((c) => c + c).join('');
133 if (!hex) continue;
134 const norm = '#' + hex;
135 if (seen.has(norm)) continue;
136 seen.add(norm);
137 out.push(norm);
138 if (out.length >= max) break;
139 }
140 return out;
141}

485export function createTextElement(preset: TextPreset, overrides?: Partial<ElementPosition>): TextElement {
486 // All positions are multiples of 30 — the default alignment-grid spacing — with
487 // a 60px (2-cell) outer margin, so freshly-inserted elements sit on the grid
488 // and the standard template breathes evenly from the slide edges.
489 // Body starts flush at the title's bottom (no gap) and grows down to the
490 // footnote. The footnote renders TIGHT (no padding, single line-height — see
491 // textPresetBoxCss), so its 24px text fits a slim 30px box flush on the 60px
492 // bottom margin.
493 const defaults: Record<TextPreset, ElementPosition> = {
494 title: { x: 60, y: 60, width: 1800, height: 180 },
495 body: { x: 60, y: 240, width: 1800, height: 750 },
496 textbox: { x: 210, y: 330, width: 810, height: 330 },
497 annotation: { x: 210, y: 720, width: 600, height: 150 },
498 footnote: { x: 60, y: 990, width: 1020, height: 30 },
499 hype: { x: 720, y: 360, width: 570, height: 360 },
500 };
501 
502 const defaultText: Record<TextPreset, string> = {
503 title: 'Title',
504 body: '',
505 textbox: 'Text',
506 annotation: 'Annotation',
507 footnote: 'Footnote',
508 hype: 'HYPE!',
509 };
510 
511 return {
512 id: crypto.randomUUID(),
513 type: 'text',
514 preset,
515 html: defaultText[preset],
516 position: { ...defaults[preset], ...overrides },

518 // the preset / hype font role) + a jaunty tilt so it reads as a tacked-on
519 // callout (#8). Per-element rotation is editable in the inspector.
520 ...(preset === 'hype' ? { backgroundColor: '#fde047', rotation: -4 } : {}),
521 };
522}

568/** Get the display slide number for a given slide index (groups share a number) */
569export function getSlideNumber(slides: Slide[], index: number): number {
570 let num = 0;
571 for (let i = 0; i <= index; i++) {
572 const slide = slides[i];
573 const prev = i > 0 ? slides[i - 1] : null;
574 // Increment number if this slide starts a new group or has no group
575 if (!slide.groupId || !prev || prev.groupId !== slide.groupId) {
576 num++;
577 }
578 }
579 return num;
580}

582/** Check if a slide is a child (not the first) in its group */
583export function isGroupChild(slides: Slide[], index: number): boolean {
584 const slide = slides[index];
585 if (!slide.groupId) return false;
586 if (index === 0) return false;
587 return slides[index - 1].groupId === slide.groupId;
588}

590/** Get all slide indices in the same group */
591export function getGroupIndices(slides: Slide[], index: number): number[] {
592 const slide = slides[index];
593 if (!slide.groupId) return [index];
594 return slides.reduce<number[]>((acc, s, i) => {
595 if (s.groupId === slide.groupId) acc.push(i);
596 return acc;
597 }, []);
