10// nbformat is loosely typed; using `unknown` + narrowing rather than `any`.
11function asObj(v: unknown): Record<string, unknown> | null {
12 return (typeof v === 'object' && v !== null && !Array.isArray(v))
13 ? v as Record<string, unknown>
14 : null;
15}

17function parseOutput(raw: unknown): CellOutput | null {
18 const o = asObj(raw);
19 if (!o) return null;
20 const type = String(o.output_type ?? '');
21 switch (type) {
22 case 'stream': {
23 const name = (o.name === 'stderr') ? 'stderr' : 'stdout';
24 return { kind: 'stream', name, text: joinMultiline(o.text) };
25 }
26 case 'display_data':
27 return { kind: 'display_data', data: parseMimeBundle(o.data) };
28 case 'execute_result':
29 return {
30 kind: 'execute_result',
31 data: parseMimeBundle(o.data),
32 executionCount: typeof o.execution_count === 'number' ? o.execution_count : null,
33 };
34 case 'error':
35 return {
36 kind: 'error',
37 ename: String(o.ename ?? 'Error'),
38 evalue: String(o.evalue ?? ''),
39 traceback: Array.isArray(o.traceback) ? o.traceback.map(String) : [],
40 };
41 default:
42 return null;
43 }
44}

46function parseMimeBundle(raw: unknown): Record<string, string | string[]> {
47 const o = asObj(raw);
48 if (!o) return {};
49 const out: Record<string, string | string[]> = {};
50 for (const [mime, value] of Object.entries(o)) {
51 // Jupyter stores text mimes as string or string[]; binary mimes (image/png,
52 // application/pdf, etc.) as base64 strings. We keep both shapes — renderers
53 // call joinMultiline() or use the string directly per mime type.
54 if (Array.isArray(value)) out[mime] = value.map(String);
55 else if (typeof value === 'string') out[mime] = value;
56 }
57 return out;
58}

60function parseCell(raw: unknown, index: number): Cell | null {
61 const c = asObj(raw);
62 if (!c) return null;
63 const cellType = String(c.cell_type ?? '');
64 const source = joinMultiline(c.source);
65 switch (cellType) {
66 case 'code': {
67 const cell: CodeCell = {
68 kind: 'code',
69 index,
70 source,
71 executionCount: typeof c.execution_count === 'number' ? c.execution_count : null,
72 outputs: Array.isArray(c.outputs)
73 ? c.outputs.map(parseOutput).filter((o): o is CellOutput => o !== null)
74 : [],
75 };
76 return cell;
77 }
78 case 'markdown': {
79 const cell: MarkdownCell = { kind: 'markdown', index, source };
80 return cell;
81 }
82 case 'raw': {
83 const cell: RawCell = { kind: 'raw', index, source };
84 return cell;
85 }
86 default:
87 return null;
88 }
89}

91/** Parse a .ipynb JSON document into our in-memory shape. Throws on
92 * unparseable JSON; returns a best-effort Notebook for malformed-but-
93 * JSON content (e.g. drops unknown cell types). */
94export function parseNotebook(json: string | object): Notebook {
95 const root = typeof json === 'string' ? JSON.parse(json) : json;
96 const o = asObj(root);
97 if (!o) throw new Error('notebook: top-level JSON is not an object');
98 
99 const cells = Array.isArray(o.cells)
100 ? o.cells.map((c, i) => parseCell(c, i)).filter((c): c is Cell => c !== null)
101 : [];
102 
103 const meta = asObj(o.metadata) ?? {};
104 const kernelspec = asObj(meta.kernelspec) ?? {};
105 const langInfo = asObj(meta.language_info) ?? {};
106 const kernelspecName = typeof kernelspec.name === 'string' ? kernelspec.name : null;
107 const kernelDisplayName = typeof kernelspec.display_name === 'string' ? kernelspec.display_name : null;
108 const language =
109 (typeof langInfo.name === 'string' && langInfo.name) ||
110 (kernelspecName && kernelspecName.startsWith('python') ? 'python' : null) ||
111 (kernelspecName === 'ir' ? 'r' : null) ||
112 (kernelspecName?.startsWith('julia') ? 'julia' : null) ||
113 null;
114 
115 return { cells, kernelspecName, kernelDisplayName, language };
116}

119export function parseNotebookBytes(bytes: Uint8Array | ArrayBuffer): Notebook {
120 const buf = bytes instanceof Uint8Array ? bytes : new Uint8Array(bytes);
121 const text = new TextDecoder('utf-8').decode(buf);
