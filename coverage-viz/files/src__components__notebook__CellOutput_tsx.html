18// Static-HTML-export mode: the export renders via renderToStaticMarkup (no
19// effects, no live blob iframes) inside a srcdoc iframe that has no `allow-scripts`.
20// So in export we render ALL output sanitized-inline SYNCHRONOUSLY (DOMPurify is
21// preloaded by notebookExport) — no async SanitizedBlock, no IsolatedOutput.
22export const StaticExportContext = createContext(false);

24export function CellOutput({ output }: { output: CellOutputT }) {
25 switch (output.kind) {

27 return (

47function MimeRender({ bundle }: { bundle: MimeBundle }) {
48 const staticExport = useContext(StaticExportContext);
49 const html = bundle['text/html'] ? joinMultiline(bundle['text/html']) : null;
50 // The routing decision parses the whole output; memoize it per html string.
51 const htmlExecutable = useMemo(() => (html != null ? outputHasExecutable(html) : false), [html]);
52 if (bundle['image/png']) {
53 const b64 = typeof bundle['image/png'] === 'string'
54 ? bundle['image/png']
55 : bundle['image/png'].join('');
56 return <img src={`data:image/png;base64,${b64}`} className="nb-output nb-image" alt="" />;
57 }
58 if (bundle['image/svg+xml']) {
59 const svg = joinMultiline(bundle['image/svg+xml']);
60 return staticExport
61 ? <div className="nb-output nb-image" dangerouslySetInnerHTML={{ __html: sanitizeSvgSync(svg) }} />
62 : <SanitizedBlock raw={svg} kind="svg" className="nb-output nb-image" />;
63 }
64 if (html != null) {
65 // Export: no scripts can run in the srcdoc, so render sanitized-inline (static)
66 // for everything — no blob iframe (it would be a dead src in the exported file).
67 if (staticExport) {
68 return <div className="nb-output nb-html" dangerouslySetInnerHTML={{ __html: sanitizeHtmlSync(html) }} />;
69 }
70 // Live: executable output (Plotly etc.) → contained-but-interactive iframe;
71 // static output (pandas tables, styled divs) → sanitized inline.
72 return htmlExecutable
73 ? <IsolatedOutput html={html} />
74 : <SanitizedBlock raw={html} kind="html" className="nb-output nb-html" />;
75 }
76 if (bundle['text/plain']) {
77 return <pre className="nb-output nb-plain">{joinMultiline(bundle['text/plain'])}</pre>;
78 }
79 return null;
80}

82/** Render an untrusted HTML/SVG output string, DOMPurify'd. Sanitization is async
83 * (DOMPurify is lazy-loaded), so this shows nothing for a frame then the clean
84 * markup — same async shape the notebook already has. */
85function SanitizedBlock({ raw, kind, className }: { raw: string; kind: 'html' | 'svg'; className: string }) {
86 const [clean, setClean] = useState('');
87 useEffect(() => {
88 let alive = true;
89 (kind === 'svg' ? sanitizeSvg : sanitizeHtml)(raw).then((c) => { if (alive) setClean(c); });
90 return () => { alive = false; };
91 }, [raw, kind]);
92 return <div className={className} dangerouslySetInnerHTML={{ __html: clean }} />;
93}

95// Strip ANSI color codes from tracebacks — Jupyter's tracebacks come
96// with ANSI escape sequences (\x1b[...m). Renderers that want color
97// can parse these; for v1 we just remove them.
98function stripAnsi(s: string): string {
99 // eslint-disable-next-line no-control-regex
100 return s.replace(/\x1b\[[0-9;]*m/g, '');
