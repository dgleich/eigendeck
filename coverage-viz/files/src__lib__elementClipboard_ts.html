1// Asset-aware copy/paste for slide elements (image / SVG), layered on the Rust
2// `clip_*` commands (see src-tauri/src/clip.rs).
3//
4// Copy of an image element: the bytes go to a process-global "internal clip"
5// (shared across windows) AND to the system clipboard (so it pastes into other
6// apps). Paste prefers the internal clip — it stores the bytes into the CURRENT
7// deck and rebuilds the element with a fresh id/assetId (so cross-deck paste
8// works and never carries a stale syncId). The internal clip is staleness-
9// checked in Rust (clipboard generation), so a foreign copy after an eigendeck
10// copy correctly wins.

50function extFromMime(mime: string, path?: string): string {
51 if (path && path.includes('.')) return path.split('.').pop()!.toLowerCase();
52 switch (mime) {
53 case 'image/svg+xml': return 'svg';
54 case 'application/pdf': return 'pdf';
55 case 'image/jpeg': return 'jpg';
56 case 'image/gif': return 'gif';
57 case 'image/webp': return 'webp';
58 default: return 'png';
59 }
60}

62/** Copy a single image/SVG element to the internal clip + system clipboard.
63 * Returns true if it handled the element. */
64export async function copyAssetElement(
65 el: SlideElement,
66 ctx?: { fromSlideId: string; fromSlideIndex: number },
67): Promise<boolean> {
68 if (!isCopyableAsset(el)) return false;
69 const assetId = (el as { assetId: string }).assetId;
70 try {
71 const meta = await invoke<AssetMeta | null>('db_get_asset_meta_by_id', { assetId });
72 const kind = (el as { kind?: string }).kind;
73 // Fall back from kind when asset meta is missing — must distinguish pdf, or
74 // build_reps would send it down the raster (PNG) branch instead of the PDF one.
75 const mime = meta?.mime_type
76 || (kind === 'svg' ? 'image/svg+xml' : kind === 'pdf' ? 'application/pdf' : 'image/png');
77 let sourceDeckId: string | null = null;
78 try { sourceDeckId = await invoke<string>('db_get_project_id'); } catch { /* unsaved deck */ }
79 const payload: AssetClipPayload = {
80 v: PAYLOAD_V, sourceDeckId, element: detachedFields(el), ext: extFromMime(mime, meta?.path),
81 fromSlideId: ctx?.fromSlideId, fromSlideIndex: ctx?.fromSlideIndex,
82 sourceId: el.id, sourceSyncId: (el as { syncId?: string }).syncId,
83 };
84 await invoke('clip_copy_asset', { assetId, payload: JSON.stringify(payload), mime });
85 return true;
86 } catch (e) {
87 console.warn('[clip] copyAssetElement failed:', e);
88 return false;
89 }
90}

98function plainTextFromHtml(html: string): string {
99 return html
100 .replace(/<br\s*\/?>/gi, '\n')
101 .replace(/<\/(p|div|li|h[1-6])>/gi, '\n')
102 .replace(/<[^>]+>/g, '')
103 .replace(/&nbsp;/g, ' ').replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>')
104 .replace(/[ \t]+/g, ' ').replace(/\n{3,}/g, '\n\n').trim();
105}

107/** The visible (foreign-app-facing) styled HTML + plain text for a copied text
108 * element. Returns the RAW styled markup — the caller wraps it with the marker
109 * + private JSON flavor via encodeClipHtml (clipboardModel). */
110export function textElementClipboardHtml(
111 el: TextElement, slide: Slide, config: PresentationConfig, theme: string,
112): { styledHtml: string; plain: string } {
113 const preset = TEXT_PRESET_STYLES[el.preset];
114 const pkg = fontForPreset(el.preset, slide, config);
115 const fontFamily = el.fontFamily || fontFamilyForPreset(pkg, el.preset);
116 const fontSize = effectiveFontSize(el, config);
117 const clipTheme = resolveTheme(theme, slide.theme);
118 const color = resolveColor(el.color, clipTheme, themeColorForPreset(clipTheme, el.preset));
119 const rendered = containsMath(el.html)
120 ? (renderMathInHtmlSync(el.html, pkg.id, config.mathPreamble) ?? el.html ?? '')
121 : (el.html || '');
122 const styled =
123 `<div style="font-family:${fontFamily};font-size:${fontSize}px;font-weight:${preset.fontWeight};` +
124 `font-style:${preset.fontStyle};color:${color};line-height:1.3;">${rendered}</div>`;
125 return { styledHtml: styled, plain: plainTextFromHtml(el.html || '') };
126}

128/** Is there a FRESH internal asset clip right now? (staleness-checked in Rust).
129 * The image-paste handler uses this to defer to the internal-clip paste. */
130export async function hasFreshInternalAsset(): Promise<boolean> {
131 try {
132 const m = await invoke<PeekResult | null>('clip_peek_internal');

147export async function pasteAssetElement(): Promise<PastedAsset | null> {
148 let meta: PeekResult | null = null;
149 try { meta = await invoke<PeekResult | null>('clip_peek_internal'); } catch { return null; }
150 if (!meta || !meta.has_bytes) return null;
151 let payload: AssetClipPayload;
152 try { payload = JSON.parse(meta.payload) as AssetClipPayload; } catch { return null; }
153 const ext = payload.ext || 'png';
154 const path = `images/pasted-${Date.now()}.${ext}`;
155 let res: { asset_id: string; payload: string } | null = null;
156 try { res = await invoke<{ asset_id: string; payload: string } | null>('clip_paste_asset', { path }); }
157 catch (e) { console.warn('[clip] clip_paste_asset failed:', e); return null; }
158 if (!res) return null;
159 const element = {
160 ...payload.element,
161 id: crypto.randomUUID(),
162 assetId: res.asset_id,
163 } as unknown as SlideElement;
164 return { element, link: { fromSlideId: payload.fromSlideId, sourceId: payload.sourceId, sourceSyncId: payload.sourceSyncId } };
