14// Sanitize each history element's html once and cache by the element object
15// (snapshot elements are stable across the panel's re-renders / slide stepping),
16// so we don't re-parse+sanitize every text box on every render.
17const _richCache = new WeakMap<object, string>();
18function cachedSanitizeRichText(el: { html?: string }): string {
19 if (!el.html) return '';
20 let v = _richCache.get(el);
21 if (v === undefined) { v = sanitizeRichText(el.html); _richCache.set(el, v); }
22 return v;
23}

25/** Format a timestamp like "2026-04-15T10:30:45.123Z-00000042" for display */
26function formatTime(ts: string): string {
27 // Strip the sequence suffix
28 const iso = ts.replace(/-\d{8}$/, '');
29 try {
30 const d = new Date(iso);
31 return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
32 } catch {
33 return ts.slice(11, 19);
34 }
35}

37function formatDate(ts: string): string {
38 const iso = ts.replace(/-\d{8}$/, '');
39 try {
40 const d = new Date(iso);
41 return d.toLocaleDateString([], { month: 'short', day: 'numeric' });
42 } catch {
43 return ts.slice(0, 10);
44 }
45}
