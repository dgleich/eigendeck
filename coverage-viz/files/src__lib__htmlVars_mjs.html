1// HTML-element variables (#138) — the ONE place the `html` element's typed
2// variables are declared, parsed and validated. Shared by the inspector (typed
3// controls) and, later, the render builder (splice), so the two never drift.
4//
5// A snippet declares variables in a JSON data-island:
6// <script type="application/eigendeck-vars+json">
7// { "value": { "type":"float", "default":62, "min":0, "max":100, "step":0.5,
8// "label":"Value", "help":"Needle position", "width":72 } }
9// </script>
10// The element stores the current VALUES in `el.vars` (a name→value map); the
11// DECLARATION (types/defaults/ranges/help) lives here in the html. There is NO
12// logic — a variable is a flat typed value spliced in two ways (CSS var + {{token}}).
13//
14// The manifest <script> never executes (the sandbox has no allow-scripts) and is
15// stripped from the rendered body by stripVarsManifest().
16import { textBackgroundResolved } from './textStyle.mjs';→ textStyle.mjs

21/** The four variable types. */
22export const VAR_TYPES = ['float', 'int', 'color', 'string'];

24// Match the manifest <script> block (type in either quote style, any attr order).
25// Global so ALL manifest blocks are parsed and stripped, not just the first.
26const MANIFEST_RE =
27 /<script\b[^>]*\btype\s*=\s*["']application\/eigendeck-vars\+json["'][^>]*>([\s\S]*?)<\/script\s*>/gi;

29// A {{name}} token in the HTML body (name = a safe identifier).
30const TOKEN_RE = /\{\{([A-Za-z_][\w-]*)\}\}/g;

32// A var NAME must be a safe identifier: it becomes a CSS custom prop (`--name`)
33// and an HTML token (`{{name}}`), so restrict to letters/digits/_/- (no leading digit).
34const NAME_RE = /^[A-Za-z_][\w-]*$/;

36// Pragmatic CSS-color check for the `color` type: hex (3/4/6/8), rgb()/rgba()/
37// hsl()/hsla(), or a common named color. Lenient enough for authoring, strict
38// enough to reject garbage that would break the injected `:root{--k:…}`.
39const HEX_RE = /^#(?:[0-9a-f]{3,4}|[0-9a-f]{6}|[0-9a-f]{8})$/i;
40const FUNC_COLOR_RE = /^(?:rgb|rgba|hsl|hsla)\(\s*[0-9.,%\s/deg]+\)$/i;
41const NAMED_COLORS = new Set([
42 'transparent', 'currentcolor', 'black', 'white', 'red', 'green', 'blue',
43 'yellow', 'orange', 'purple', 'pink', 'gray', 'grey', 'brown', 'cyan',
44 'magenta', 'lime', 'teal', 'navy', 'maroon', 'olive', 'silver', 'gold',
45 'indigo', 'violet', 'coral', 'salmon', 'crimson', 'tomato', 'khaki',
46]);

48/** Is `v` a valid CSS color for the `color` type? */
49export function isValidColor(v) {
50 if (typeof v !== 'string') return false;
51 const s = v.trim();
52 return HEX_RE.test(s) || FUNC_COLOR_RE.test(s) || NAMED_COLORS.has(s.toLowerCase());
53}

55/** A `color` value can be a literal CSS color OR a theme-relative TINT token,
56 * `tint:<base>` — where `<base>` is `accent` (follows the slide theme) or a
57 * semantic hex (e.g. `tint:#dc2626`). The token stores a tint; the real color is
58 * resolved per slide theme at render time (`textBackgroundResolved`), so the same
59 * variable stays colored + contrasting on any theme. Keeps `vars` flat (a string). */
60export const TINT_PREFIX = 'tint:';
61 
62/** The base of a tint token (`tint:accent` → `accent`), or '' if `v` isn't one. */
63export function tintBase(v) {
64 return typeof v === 'string' && v.startsWith(TINT_PREFIX) ? v.slice(TINT_PREFIX.length) : '';
65}

67/** Is `v` a valid tint token (`tint:accent` or `tint:<hex>`)? */
68export function isTintToken(v) {
69 const base = tintBase(v);
70 return base === 'accent' || isValidColor(base);
71}

73/** Is `v` a valid value for a `color` variable — a literal color OR a tint token? */
74export function isColorValue(v) {
75 return isValidColor(v) || isTintToken(v);

83/** Parse the variable manifest out of `html` → an ORDERED list of specs.
84 * Malformed entries (bad name/type) are dropped; a missing/unparseable
85 * manifest yields []. Never throws. */
86export function parseHtmlVars(html) {
87 if (typeof html !== 'string' || html.indexOf(VARS_SCRIPT_TYPE) === -1) return [];
88 
89 const specs = [];
90 const seen = new Set(); // first declaration of a name wins (across all manifests)
91 for (const m of html.matchAll(MANIFEST_RE)) {
92 let obj;
93 try {
94 obj = JSON.parse(m[1].trim());
95 } catch {
96 continue;
97 }
98 if (!obj || typeof obj !== 'object' || Array.isArray(obj)) continue;
99 
100 for (const [name, raw] of Object.entries(obj)) {
101 if (seen.has(name) || !NAME_RE.test(name) || !raw || typeof raw !== 'object') continue;
102 const type = VAR_TYPES.includes(raw.type) ? raw.type : null;
103 if (!type) continue;
104 
105 const spec = { name, type };
106 // Default (synthesised per-type when absent/invalid so a control always has a value).
107 if (type === 'int' || type === 'float') {
108 let d = num(raw.default);
109 if (d === undefined) d = 0;
110 spec.default = type === 'int' ? Math.round(d) : d;
111 const mn = num(raw.min); if (mn !== undefined) spec.min = mn;
112 const mx = num(raw.max); if (mx !== undefined) spec.max = mx;
113 const st = num(raw.step); if (st !== undefined && st > 0) spec.step = st;
114 } else if (type === 'color') {
115 spec.default = isColorValue(raw.default) ? String(raw.default).trim() : '#000000';
116 } else {
117 spec.default = raw.default == null ? '' : String(raw.default);
118 if (raw.multiline === true) spec.multiline = true; // render a textarea
119 }
120 
121 if (typeof raw.label === 'string' && raw.label.trim()) spec.label = raw.label.trim();
122 if (typeof raw.help === 'string' && raw.help.trim()) spec.help = raw.help.trim();
123 const w = num(raw.width); if (w !== undefined && w > 0) spec.width = Math.round(w);
124 seen.add(name);
125 specs.push(spec);
126 }
127 }
128 return specs;
129}

131/** Validate a raw value against a spec. Returns `{ ok, value }` where `value` is
132 * the coerced value when ok (number for int/float, trimmed string otherwise).
133 * Used for the inspector's red-✕ flag and (later) render-time coercion. */
134export function validateVarValue(spec, raw) {
135 if (spec.type === 'int' || spec.type === 'float') {
136 const s = typeof raw === 'string' ? raw.trim() : raw;
137 if (s === '' || s == null) return { ok: false };
138 const n = Number(s);
139 if (!Number.isFinite(n)) return { ok: false };
140 if (spec.type === 'int' && !Number.isInteger(n)) return { ok: false };
141 if (spec.min !== undefined && n < spec.min) return { ok: false };
142 if (spec.max !== undefined && n > spec.max) return { ok: false };
143 return { ok: true, value: n };
144 }
145 if (spec.type === 'color') {
146 return isColorValue(raw) ? { ok: true, value: String(raw).trim() } : { ok: false };
147 }
148 // string — any value is valid.
149 return { ok: true, value: raw == null ? '' : String(raw) };
150}

152/** Resolve a specs list against stored `vars` → a name→value map, falling back to
153 * each spec's default when the stored value is missing or invalid. */
154export function resolveVars(specs, vars) {
155 const out = {};
156 for (const spec of specs) {
157 const stored = vars ? vars[spec.name] : undefined;
158 if (stored == null) { out[spec.name] = spec.default; continue; }
159 const v = validateVarValue(spec, stored);
160 out[spec.name] = v.ok ? v.value : spec.default;
161 }
162 return out;
163}

178/** Resolve a color VALUE (literal or `tint:<base>` token) to a real CSS color for
179 * the given slide theme. Tints resolve like card backgrounds (a wash relative to
180 * the theme background) so they stay colored + contrasting on any theme. */
181export function resolveColorVar(value, theme) {
182 if (isTintToken(value)) {
183 return textBackgroundResolved({ boxTint: tintBase(value) }, theme || DEFAULT_THEME);
184 }
185 return typeof value === 'string' ? value : '';
186}

188// Sanitize a resolved value for the CSS `:root{--k:…}` context, by type. Returns
189// null to omit the declaration (never emit something that could break out of the
190// style). Numbers → bare number; color → validated color; string → a CSS string.
191function cssVarValue(type, value) {
192 if (type === 'int' || type === 'float') {
193 const n = Number(value);
194 return Number.isFinite(n) ? String(n) : null;
195 }
196 if (type === 'color') {
197 return isValidColor(value) ? String(value).trim() : null;
198 }
199 const s = String(value).replace(/[\\"<{}]/g, '').replace(/[\r\n]+/g, ' ');
200 return `"${s}"`;
201}

203// Escape a value for HTML text context (the `{{token}}` side). Escapes `'` too so a
204// token placed inside a single-quoted attribute can't break out (defense-in-depth —
205// the no-script sandbox already neutralises any injected markup).
206function htmlEscapeText(s) {
207 return String(s)
208 .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
209 .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
210}

212/** Splice an html element's variables into its markup. Returns the body with the
213 * manifest stripped and every `{{name}}` replaced (HTML-escaped), plus the
214 * `:root{…}` CSS block declaring `--name` custom properties. No manifest → the
215 * html is returned unchanged and `rootCss` is ''. Never throws. */
216export function spliceHtmlVars(html, vars, theme) {
217 if (typeof html !== 'string') return { html: html == null ? '' : String(html), rootCss: '' };
218 const specs = parseHtmlVars(html);
219 if (specs.length === 0) return { html, rootCss: '' };
220 const resolved = resolveVars(specs, vars);
221 let decls = '';
222 const tokenVals = Object.create(null); // name → HTML-escaped text
223 for (const spec of specs) {
224 let val = resolved[spec.name];
225 if (spec.type === 'color') val = resolveColorVar(val, theme);
226 const css = cssVarValue(spec.type, val);
227 if (css != null) decls += `--${spec.name}:${css};`;
228 tokenVals[spec.name] = htmlEscapeText(val);
229 }
230 // Single left-to-right pass so a value that itself contains `{{other}}` is inserted
231 // verbatim and never re-substituted (no cascade / order dependence). Unknown tokens
232 // are left untouched.
233 const body = stripVarsManifest(html).replace(TOKEN_RE, (m, name) =>
234 name in tokenVals ? tokenVals[name] : m);
235 return { html: body, rootCss: decls ? `:root{${decls}}` : '' };
236}
