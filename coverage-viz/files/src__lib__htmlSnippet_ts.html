1// Validate an HTML-element snippet before inserting it (#137). Snippets are just
2// self-contained HTML fragments (see examples-html-elements/), so "Insert HTML
3// Element from File…" — and, later, a "download from an online repo" flow — need a
4// gate that rejects cruddy input: not-HTML, scripts (won't run in the sandbox),
5// and remote resource references (blocked by the CSP). This is a UX guard, NOT a
6// security boundary — the real containment is the no-`allow-scripts` sandbox + CSP
7// applied to `el.html` in every render path; a snippet that slipped past this still
8// executes nothing. This function is PURE (no Tauri / DOM deps) so it can guard both
9// the local file picker and any remote fetch.
10import { stripVarsManifest } from './htmlVars.mjs';→ htmlVars.mjs

12export interface SnippetCheck {

22// Snippets should be small. 2 MB leaves room for a modest inline data: image while
23// still rejecting "someone handed me a whole web page / a binary".
24const MAX_BYTES = 2_000_000;
25 
26// A remote resource reference the sandbox CSP would block (http/https/protocol-
27// relative) in a load context: src / srcset / CSS url() / @import / <link href>.
28const REMOTE_PATTERNS: RegExp[] = [
29 /\bsrc\s*=\s*["']?\s*(?:https?:)?\/\//i,
30 /\bsrcset\s*=\s*["']?\s*(?:https?:)?\/\//i,
31 /url\(\s*["']?\s*(?:https?:)?\/\//i,
32 /@import\s+(?:url\(\s*)?["']?\s*(?:https?:)?\/\//i,
33 /<link\b[^>]*\bhref\s*=\s*["']?\s*(?:https?:)?\/\//i,
34];

36/** Read the interactive flag from the `<!-- eigendeck-html-element … interactive -->`
37 * metadata comment, if present. */
38export function snippetIsInteractive(raw: string): boolean {
39 const m = raw.match(/<!--\s*eigendeck-html-element\b([\s\S]*?)-->/i);
40 return !!(m && /\binteractive\b/i.test(m[1]));
41}

43export function validateHtmlSnippet(raw: unknown): SnippetCheck {
44 const problems: string[] = [];
45 const html = typeof raw === 'string' ? raw : '';
46 const trimmed = html.trim();
47 
48 if (!trimmed) {
49 problems.push('The file is empty.');
50 return { ok: false, html, interactive: false, problems };
51 }
52 if (html.length > MAX_BYTES) {
53 problems.push(`Too large (${(html.length / 1e6).toFixed(1)} MB, max 2 MB) — snippets should be small; embed a big image as a real image element instead.`);
54 }
55 // Must actually be HTML — at least one element tag.
56 if (!/<([a-z][a-z0-9-]*)\b[^>]*>/i.test(trimmed)) {
57 problems.push('This doesn’t look like HTML — no elements were found.');
58 }
59 // The variables manifest (`<script type="application/eigendeck-vars+json">`, #138)
60 // is a non-executing data island — strip it before the script check so it's exempt
61 // while real scripts still trip the gate.
62 if (/<script[\s>/]/i.test(stripVarsManifest(html))) {
63 problems.push('Contains a <script> — scripts don’t run in the sandbox. Use CSS animation or native form controls instead.');
64 }
65 if (/<[a-z][^>]*\son[a-z]+\s*=/i.test(html)) {
66 problems.push('Contains inline event handlers (onclick, onerror, …) — they don’t run in the sandbox.');
67 }
68 if (REMOTE_PATTERNS.some((re) => re.test(html))) {
69 problems.push('References remote resources (http/https). The sandbox blocks the network — embed images and fonts as data: URIs.');
70 }
71 
72 return { ok: problems.length === 0, html, interactive: snippetIsInteractive(html), problems };
73}
