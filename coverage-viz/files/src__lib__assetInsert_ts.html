19/** User-facing reason an add was refused by the asset-type gate. */
20function addRejectMessage(reason: GateReason | null, path: string): string {
21 const ext = extensionOf(path).toUpperCase() || 'this';
22 switch (reason) {
23 case 'bad-extension':
24 return `“${extensionOf(path) || path}” isn’t a supported asset type. Eigendeck supports images, PDFs, videos, notebooks, captions, and demos.`;
25 case 'unsupported-demo-version':
26 return 'This demo was built for a newer version of Eigendeck and can’t be added here.';
27 case 'content-mismatch':
28 return extensionOf(path) === 'html'
29 ? 'That .html isn’t an Eigendeck demo, so it can’t be added. Only demos authored for Eigendeck can be embedded (see DEMO_AUTHORING) — an arbitrary web page or HTML file can’t.'
30 : `That file isn’t a valid ${ext} (its contents don’t match its name), so it can’t be added.`;
31 case 'too-large':
32 return 'That file is over Eigendeck’s 512 MB per-file limit, so it can’t be added. '
33 + 'If you need larger files (for example long videos), please open an issue at '
34 + 'https://github.com/dgleich/eigendeck describing the file so we can look at raising the limit.';
35 default:
36 return 'That file couldn’t be added.';
37 }
38}

73/** Store asset bytes via the raw-body IPC (memcpy) — bytes in the request
74 * body, metadata in a base64 JSON header. Avoids the ~300ms/MB args path
75 * that froze the UI on large assets (#174). */
76export async function storeAssetRaw(meta: {
77 path: string; mimeType: string;
78 externalPath?: string | null; externalMtime?: string | null;
79 assetId?: string | null; autoReload?: string | null; ownerElementId?: string | null;
80}, data: Uint8Array): Promise<string> {
81 const json = JSON.stringify(meta);
82 // base64(UTF-8(json)); json is small (paths/strings), so this is cheap.
83 const metaB64 = btoa(String.fromCharCode(...new TextEncoder().encode(json)));
84 // The Uint8Array is the PAYLOAD (not nested in an object) → Tauri sends it
85 // as the raw request body (memcpy). Metadata rides in the header.
86 return invoke<string>('db_store_asset_raw', data, { headers: { 'x-asset-meta': metaB64 } });
87}

126 * → db_store_asset, return fresh assetId. No dialog.
127 *
128 * Existing asset, new bytes match the existing asset's ORIGINAL

185 * resolve→realpath via the SAME gate the watcher uses, then record the canonical in
186 * the ledger. No-op for untrusted decks (must be trusted first) or unresolvable /
187 * wrong-type targets. This is the SINGLE approval code path shared by every user act

192export async function approveExternalAbsPath(assetId: string, absPath: string, reason: string): Promise<void> {
193 try {

197 if (!(await isTrusted(token))) return;

199 const gate = await resolveAndGate(absPath);
200 if (gate.ok && gate.canonicalPath) await approvePath(token, assetId, gate.canonicalPath, reason);

218/**
219 * Read a file to add as an asset, rejecting oversized ones BEFORE loading them into
220 * memory. Every add-from-disk site should use this instead of a raw `readFile`: a
221 * multi-GB pick would otherwise be slurped whole (choking the app) only to be rejected
222 * by the gate afterwards. Stats first; over the cap → shows the "too large" toast and
223 * returns null (caller should bail). Returns the bytes otherwise.
224 */
225export async function readAddFileCapped(fullPath: string): Promise<Uint8Array | null> {
226 const { statNative, readFileNative } = await import('./nativeFs');→ nativeFs.ts
227 try {
228 const st = await statNative(fullPath);
229 if (typeof st.size === 'number' && st.size > MAX_ASSET_BYTES) {
230 showToast({ kind: 'error', ttl: 9000, message: addRejectMessage('too-large', fullPath) });
231 return null;
232 }
233 } catch { /* stat failed → let the read surface the real error below */ }
234 // readFileNative goes through resolve_and_read (same MAX_ASSET_BYTES cap); the
235 // stat precheck above just gives the nicer "too large" toast before reading.
236 return readFileNative(fullPath);
237}

271 // Validate EVERY add against the same asset-type gate the watcher uses
272 // (docs/ASSETS-SECURITY.md): the bytes must actually be the type the extension
273 // claims (native magic for interchange types; the eigendeck-demo marker for
274 // .html). This makes "what you can add" == "what the watcher will later accept",
