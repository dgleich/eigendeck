1// E2E coverage beacon (spike). Active ONLY in Istanbul-instrumented builds
2// (COVERAGE_INSTRUMENT=1 → vite-plugin-istanbul → window.__coverage__ exists).
3// It periodically POSTs the coverage map to the e2e collector server
4// (e2e/coverage-server.mjs), so EVERY e2e probe contributes its real-WebKitGTK
5// line hits with zero per-probe changes. A no-op in normal builds (no
6// __coverage__), so calling it unconditionally from main is free.
7export function installCoverageBeacon(): void {
8 const w = window as unknown as { __coverage__?: unknown; __covBeaconInstalled?: boolean };
9 if (!w.__coverage__ || w.__covBeaconInstalled) return;
10 w.__covBeaconInstalled = true;
11 
12 // One stable id per page → the collector overwrites cov-<id>.json each POST,
13 // keeping only the latest cumulative snapshot (so re-posting never double-counts).
14 const id = 'p' + Math.random().toString(36).slice(2) + Date.now().toString(36);
15 const url = `/__coverage__?id=${id}`;
16 
17 const send = (): void => {
18 try {
19 // Plain fetch (NOT keepalive): the coverage body far exceeds the 64KB
20 // keepalive/sendBeacon cap, so a keepalive request would be silently dropped.
21 fetch(url, {
22 method: 'POST',
23 headers: { 'Content-Type': 'application/json' },
24 body: JSON.stringify(w.__coverage__),
25 }).catch(() => {});
26 } catch { /* ignore */ }
27 };
28 
29 // Coverage is cumulative; snapshot early (short probes) then on an interval so we
30 // capture hits even if the rig hard-closes the session without a clean teardown.
31 setTimeout(send, 1000);
32 setInterval(send, 1500);
33 // Best-effort final flush (sendBeacon is >64KB-capped so it usually no-ops here;
34 // the last interval already captured essentially everything).
35 window.addEventListener('pagehide', () => { send(); });
36 window.addEventListener('visibilitychange', () => {
37 if (document.visibilityState === 'hidden') send();
38 });
39}
