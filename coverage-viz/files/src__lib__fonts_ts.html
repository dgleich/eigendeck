40/**
41 * Build embedded @font-face declarations as data: URLs for all fonts used
42 * by a presentation. Fetches font files via fetch() (so works in browser/
43 * Tauri contexts where /fonts/ is served).
44 *
45 * Returns the CSS string ready to drop in <style>.
46 */
47export async function buildEmbeddedFontFacesCSS(presentation: {
48 config?: { defaultTitleFont?: string; defaultBodyFont?: string; defaultHypeFont?: string };
49 slides?: Array<{ titleFont?: string; bodyFont?: string; hypeFont?: string }>;
50}): Promise<string> {
51 const usedFontIds = collectUsedFontIds(presentation);
52 const lines: string[] = [];
53 for (const id of usedFontIds) {
54 const pkg = resolveAnyFontPackage(id);
55 const family = bareFamilyName(pkg);
56 const narrowFamily = bareNarrowFamilyName(pkg);
57 for (const { filename, cssAttrs } of fontFilesForPackage(pkg)) {
58 try {
59 // Prefer the WOFF2 sibling (Brotli, ~40% of the TTF): this CSS is inlined
60 // as base64 into every demo/export document, so the smaller payload is
61 // re-parsed far faster on each opaque-iframe mount (docs/perf-report.md,
62 // "Demos"). Falls back to the raw TTF/OTF when no .woff2 exists (a build
63 // that hasn't run tools/build_font_woff2.py).
64 const base = `/fonts/${pkg.id}/${filename}`;
65 const woff2Url = base.replace(/\.(?:ttf|otf)$/i, '.woff2');
66 let mime = 'font/woff2';
67 let fmt = "format('woff2')";
68 let resp = await fetch(woff2Url);
69 if (!resp.ok) {
70 const ext = filename.split('.').pop() || 'ttf';
71 mime = ext === 'otf' ? 'font/otf' : 'font/ttf';
72 fmt = ext === 'otf' ? "format('opentype')" : "format('truetype')";
73 resp = await fetch(base);
74 if (!resp.ok) continue;
75 }
76 const buf = new Uint8Array(await resp.arrayBuffer());
77 const dataUrl = `data:${mime};base64,${bytesToBase64(buf)}`;
78 const fontFamily = cssAttrs.isNarrow && narrowFamily ? narrowFamily : family;
79 lines.push(
80 `@font-face { font-family: '${fontFamily}'; src: url('${dataUrl}') ${fmt}; ` +
81 `font-weight: ${cssAttrs.weight}; font-style: ${cssAttrs.style}; font-display: swap; }`
82 );
83 } catch (e) {
84 console.warn(`Failed to embed font ${pkg.id}/${filename}:`, e);
85 }
86 }
87 }
88 return lines.join('\n');
89}

91/**
92 * Inject @font-face declarations for all font packages into <head>.
93 * Idempotent: subsequent calls replace the existing block.
94 */
95export function injectFontFaces(): void {
96 if (typeof document === 'undefined') return;
97 const STYLE_ID = 'eigendeck-font-faces';
98 let style = document.getElementById(STYLE_ID) as HTMLStyleElement | null;
99 if (!style) {
100 style = document.createElement('style');
101 style.id = STYLE_ID;
102 document.head.appendChild(style);
103 }
104 style.textContent = allFontFacesCSS();
105}
