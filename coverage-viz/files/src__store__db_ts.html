20type DbEvent =
21 | 'slides-changed'
22 | 'elements-changed'
23 | 'presentation-changed'
24 | `slide-elements-changed:${string}`;

26const listeners = new Map<string, Set<() => void>>();
27 
28function emit(event: DbEvent) {
29 listeners.get(event)?.forEach((fn) => fn());
30 // Also emit a wildcard for components that watch everything
31 listeners.get('*')?.forEach((fn) => fn());
32}
33 
34function subscribe(event: DbEvent | '*', fn: () => void): () => void {
35 if (!listeners.has(event)) listeners.set(event, new Set());
36 listeners.get(event)!.add(fn);
37 return () => { listeners.get(event)?.delete(fn); };
38}

81export function toggleSelectElement(id: string) {
82 const sel = _selectedObject;
83 if (!sel || sel.type === 'slide') {
84 _selectedObject = { type: 'element', id };
85 } else if (sel.type === 'element') {
86 if (sel.id === id) _selectedObject = { type: 'slide' };
87 else _selectedObject = { type: 'multi', ids: [sel.id, id] };
88 } else if (sel.type === 'multi') {
89 const ids = sel.ids.includes(id) ? sel.ids.filter((i) => i !== id) : [...sel.ids, id];
90 if (ids.length === 0) _selectedObject = { type: 'slide' };
91 else if (ids.length === 1) _selectedObject = { type: 'element', id: ids[0] };
92 else _selectedObject = { type: 'multi', ids };
93 }
94 emitUI();
95}

131export function useSlides(): Slide[] {

134 useEffect(() => {
135 let mounted = true;
136 const load = async () => {
137 try {
138 const json = await invoke<string>('db_get_slides');

152/** Elements for a specific slide. Re-fetches when that slide's elements change. */
153export function useSlideElements(slideId: string | null): SlideElement[] {
154 const [elements, setElements] = useState<SlideElement[]>([]);
155 
156 useEffect(() => {
157 if (!slideId) { setElements([]); return; }
158 let mounted = true;
159 const load = async () => {
160 try {
161 const json = await invoke<string>('db_get_slide_elements', { slideId });
162 if (mounted) setElements(JSON.parse(json));
163 } catch (e) {
164 console.error('Failed to load elements:', e);
165 }
166 };
167 load();
168 const unsub1 = subscribe(`slide-elements-changed:${slideId}`, load);
169 const unsub2 = subscribe('elements-changed', load);
170 return () => { mounted = false; unsub1(); unsub2(); };
171 }, [slideId]);
172 
173 return elements;
174}
