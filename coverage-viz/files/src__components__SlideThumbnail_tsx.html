3// live): heavy elements (demo / notebook / video) show cached preview images

28/**
29 * A static, scaled snapshot of one slide.
30 *
31 * - Pass `width` (px) for a FIXED-size thumbnail (the sidebar, whose column is a
32 * known width).
33 * - Omit `width` for a RESPONSIVE thumbnail that fills its parent's width and
34 * takes the deck's aspect ratio (the speaker view). This avoids the
35 * fixed-width-inside-a-flex-box mismatch that clipped the next-slide preview.
36 *
37 * `imageTier` is the max px (long edge) raster images render at; defaults to the
38 * small sidebar thumb tier. The speaker view passes ASSET_TIER.full so the
39 * current/next previews use crisp, slide-native images instead of 256px ones.
40 */
41export function SlideThumbnail({ presentation, slide, width, imageTier = ASSET_TIER.thumb }: {
42 presentation: Presentation; slide: Slide; width?: number; imageTier?: number;
43}) {
44 const slideW = presentation.config.width || 1920;
45 const slideH = presentation.config.height || 1080;
46 
47 // Responsive mode measures the rendered width; fixed mode uses the prop.
48 const ref = useRef<HTMLDivElement>(null);
49 const [measuredW, setMeasuredW] = useState(0);
50 useLayoutEffect(() => {
51 if (width !== undefined) return; // fixed-size: no observer needed
52 const el = ref.current;
53 if (!el) return;
54 setMeasuredW(el.clientWidth);
55 const ro = new ResizeObserver((entries) => {
56 for (const e of entries) setMeasuredW(e.contentRect.width);
57 });
58 ro.observe(el);
59 return () => ro.disconnect();
60 }, [width]);
61 
62 const renderW = width !== undefined ? width : measuredW;
63 const scale = renderW / slideW;
64 // Fixed: explicit box. Responsive: fill width, height from the deck ratio
65 // (set even before measuring so the box doesn't collapse on first paint).
66 const outerStyle: React.CSSProperties = width !== undefined
67 ? { width, height: slideH * scale }
68 : { width: '100%', aspectRatio: `${slideW} / ${slideH}` };
69 
70 return (
71 <div ref={ref} className="slide-thumb-clip" style={{ ...outerStyle, position: 'relative' }}>
72 {renderW > 0 && (
73 <div className="slide-thumb-render" style={{
74 width: slideW, height: slideH,
75 transform: `scale(${scale})`, transformOrigin: 'top left',
76 position: 'absolute', top: 0, left: 0,
77 background: resolveTheme(presentation.theme, slide.theme).background,
78 }}>
79 {slide.elements.map((el) => (
80 <ThumbElement key={el.id} element={el} slide={slide} presentation={presentation} imageTier={imageTier} />
81 ))}
82 </div>
83 )}
84 </div>
85 );
86}

88function ThumbElement({ element: el, slide, presentation, imageTier }: {
89 element: SlideElement; slide: Slide; presentation: Presentation; imageTier: number;
90}) {
91 switch (el.type) {
92 case 'text':
93 return (
94 <TextElementSvg element={el} slide={slide}
95 presentationTheme={presentation.theme} presentationConfig={presentation.config} />
96 );
97 case 'image':
98 return <ThumbImage element={el} imageTier={imageTier} />;
99 case 'arrow': {
100 const a = describeArrow(el, resolveTheme(presentation.theme, slide.theme));
101 return (
102 <svg style={{ position: 'absolute', top: 0, left: 0, width: '100%', height: '100%', pointerEvents: 'none', overflow: 'visible' }}>
103 <ArrowGlyph geo={a.geo} color={a.color} strokeWidth={a.strokeWidth} opacity={a.opacity} />
104 </svg>
105 );
106 }
107 case 'demo':
108 case 'demo-piece':
109 case 'notebook':
110 return <ThumbPreview element={el} />;
111 case 'html': {
112 // Static + locked (no script/network) → render the real thing for a true
113 // mini-preview; it scales with the thumbnail's CSS transform.
114 const p = el.position;
115 const htmlTheme = resolveTheme(presentation.theme, slide.theme);
116 if (htmlIsScaled(el)) {
117 const L = htmlScaleLayout(p.width, p.height, el.scaleW!, el.scaleH!);
118 return (
119 <div style={{ position: 'absolute', left: p.x, top: p.y, width: p.width, height: p.height, overflow: 'hidden' }}>
120 <iframe title="HTML element" srcDoc={htmlElementSrcdoc(el.html, el.background, el.vars, htmlTheme)}
121 sandbox={HTML_SANDBOX_LOCKED} scrolling="no" style={{
122 position: 'absolute', left: 0, top: 0, width: L.designW, height: L.designH,
123 border: 'none', background: 'transparent', pointerEvents: 'none',
124 transform: `translate(${L.offsetX}px, ${L.offsetY}px) scale(${L.scale})`,
125 transformOrigin: 'top left',
126 }} />
127 </div>
128 );
129 }
130 return (
131 <iframe title="HTML element" srcDoc={htmlElementSrcdoc(el.html, el.background, el.vars, htmlTheme)}
132 sandbox={HTML_SANDBOX_LOCKED} scrolling="no" style={{
133 position: 'absolute', left: p.x, top: p.y, width: p.width, height: p.height,
134 border: 'none', background: 'transparent', pointerEvents: 'none',
135 }} />
136 );
137 }
138 case 'video':
139 return <ThumbVideo element={el} />;
140 case 'cover': {
141 // Match the slide background (it's a reveal mask) — no border, so the
142 // static render matches the live slide / speaker view. Explicit color wins.
143 const tt = resolveTheme(presentation.theme, slide.theme);
144 const d = describeCover(el, tt.background, tt);
145 return <CoverView box={d.box} background={d.background} />;
146 }
147 default:
148 return null;
149 }
150}

152function ThumbImage({ element, imageTier }: { element: Extract<SlideElement, { type: 'image' }>; imageTier: number }) {
153 const p = element.position;
154 const kind = element.kind ?? 'raster';
155 const url = useRenderedAsset(element.assetId, kind, imageTier, imageTier, element.snapshotVariant);
156 useAssetFileWatcher(element.assetId, element.id);
157 return (
158 <div style={{
159 position: 'absolute', left: p.x, top: p.y, width: p.width, height: p.height, overflow: 'hidden',
160 background: url ? 'transparent' : '#f0f0f0', border: url ? 'none' : '1px solid #ddd', borderRadius: 2,
161 display: 'flex', alignItems: 'center', justifyContent: 'center',
162 }}>
163 {url ? <img src={url} alt="" style={{ width: '100%', height: '100%', objectFit: 'contain', ...imageVisualStyle(element) }} />
164 : <span style={{ fontSize: 24, color: '#aaa' }}>IMG</span>}
165 </div>
166 );
167}

169// demo / demo-piece / notebook all render as a cached-preview image with a
170// typed placeholder fallback in an identical positioned box. The placeholder
171// identity comes from ELEMENT_PLACEHOLDERS; notebook fills big with no border,
172// demo/demo-piece are smaller dashed boxes; demo-piece shows the piece name.
173function ThumbPreview({ element }: { element: Extract<SlideElement, { type: 'demo' | 'demo-piece' | 'notebook' }> }) {
174 const p = element.position;
175 useAssetFileWatcher(element.assetId, element.id);
176 const spec = PH[element.type];
177 const isNb = element.type === 'notebook';
178 const fontSize = isNb ? 64 : element.type === 'demo' ? 20 : 16;
179 const label = element.type === 'demo-piece' ? element.piece : spec.label;
180 // Demos capture TRANSPARENT now (#111): keep the wrapper transparent so the
181 // slide background (from the thumb-render div) and any elements beneath the demo
182 // show through, matching the live slide. (Notebooks keep their white card.)
183 return (
184 <div style={{ position: 'absolute', left: p.x, top: p.y, width: p.width, height: p.height, overflow: 'hidden', background: isNb ? '#fff' : 'transparent' }}>
185 <ElementPreviewImg cacheKey={element.syncId ?? element.id} fallback={
186 <div style={{ width: '100%', height: '100%', display: 'flex', alignItems: 'center', justifyContent: 'center', fontSize, color: spec.color, background: spec.bg, ...(isNb ? {} : { border: `1px dashed ${spec.borderColor}` }) }}>{label}</div>
187 } />
188 </div>
189 );
190}
