1// Module-level subscribe pattern for the "asset has silently changed
2// since first add" awareness dialog.
3//
4// Fired when a user re-inserts an asset (drag-drop or file picker) whose
5// path already exists in the project AND the existing asset's current
6// bytes differ from its ORIGINAL bytes (oldest version in history). The
7// user-visible surprise the dialog surfaces: file-watching silently
8// updated their existing copy when the source file changed on disk;
9// they may not have noticed.
10//
11// User picks one of two intents:
12// 'accept' : "I understand and want the auto-updating behavior."
13// Existing asset stays as-is (silently-updated bytes);
14// new element is added pointing at it; same outcome as if
15// the dialog had never appeared. The dialog is FYI in this
16// path.
17// 'revert' : "I want to revert the existing copy to its original and
18// add this as a separate new asset; disable auto-updating
19// for this presentation." Existing asset's original bytes
20// are restored (old element shows the originally-added
21// version again); a NEW asset is created with the bytes
22// being inserted now; presentation auto-reload set to OFF.
23// 'cancel' : Esc / outside-click. Abort the insertion entirely. No
24// visible button — paternalism cap; users expect Esc to
25// get out of a modal.
26//
27// See docs/ASSETS.md → "Path collision dialog" for the design.
28 
29export type CollisionChoice = 'accept' | 'revert' | 'cancel';
30 
31export interface CollisionRequest {
32 /** Path label of the asset (e.g. 'images/chart.svg'). */
33 path: string;
34 /** 1-based slide numbers where the existing asset is currently used
35 * (per getSlideNumber). May be a single slide or several. Used to
36 * render the "added on slide X" / "added on slides 2 and 7" copy. */
37 slideNumbers: number[];
38 /** Did the EXISTING embedded copy already change from what was first added
39 * (current hash != original hash)? True = a watcher auto-update already
40 * happened (the "…which has already happened" framing). False = the embedded
41 * copy is still the original (untrusted/unwatched deck), so the divergence is
42 * only in the NEW file being added — the message must NOT claim an update
43 * already occurred. */
44 existingChanged: boolean;
45 /** When `existingChanged` is false, the concrete reason the embedded copy was
46 * never live-updated (untrusted deck / unsaved / auto-reload off / …), so the
47 * message can be accurate instead of vaguely saying "isn't set up". */
48 notLiveReason?: string;
49}

51interface PendingRequest extends CollisionRequest {
52 id: number;
53 resolve: (choice: CollisionChoice) => void;
54}

56let pending: PendingRequest | null = null;
57let nextId = 1;
58const subscribers = new Set<(req: PendingRequest | null) => void>();
59 
60/** Show the dialog. Caller awaits a Promise resolving to the choice. */
61export function showCollisionDialog(req: CollisionRequest): Promise<CollisionChoice> {
62 return new Promise((resolve) => {
63 pending = {
64 ...req,
65 id: nextId++,
66 resolve: (choice) => {
67 pending = null;
68 emit();
69 resolve(choice);
70 },
71 };
72 emit();
73 });
74}
75 
76/** Component subscribes to be notified when the pending request changes. */
77export function subscribeCollisionDialog(cb: (req: PendingRequest | null) => void): () => void {
78 subscribers.add(cb);
79 cb(pending);
80 return () => { subscribers.delete(cb); };
81}
82 
83function emit() {
84 for (const cb of subscribers) cb(pending);
