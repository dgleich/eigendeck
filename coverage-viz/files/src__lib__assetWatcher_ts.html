1// useAssetFileWatcher — thin React wrapper around the per-project
2// WatcherRegistry. Each component that displays an asset (sidebar
3// thumbnail, editor canvas, presenter) mounts this hook for the asset
4// it's showing; the registry dedups so multiple subscribers for the
5// same source path share one fs.watch.
6//
7// All the actual work (resolving paths, kernel watch, reload + invalidate
8// fan-out) lives in src/lib/watcherRegistry.ts. This file is just the
9// React glue: get the registry by project_id on mount, addRef the
10// asset's source path, removeRef on cleanup.

33/**
34 * Watch the source file behind an asset (if any) and auto-reload the
35 * stored bytes when it changes on disk. No-op when:
36 * - the asset has no external_path (e.g. pasted from clipboard,
37 * embedded as snapshot, or restored from history with auto_reload='off')
38 * - the presentation isn't saved yet (no project dir to resolve against)
39 *
40 * Idempotent across re-mounts; safely unwatches on unmount or input change.
41 */
42export function useAssetFileWatcher(
43 assetId: string | undefined,
44 elementId: string,
45): void {
46 const projectPath = usePresentationStore((s) => s.projectPath);
47 const [projectId, setProjectId] = useState<string | null>(null);
48 // 3-layer cascade: per-asset > per-presentation > global default.
49 const [globalAutoReload] = usePreference('autoReloadAssets');
50 const presOverride = usePresentationStore((s) => s.presentation?.config?.autoReloadAssets ?? null);
51 
52 // Bumped when an asset-changed event arrives matching this asset. The
53 // event fires when meta changes (auto_reload toggled, asset restored,
54 // watcher wrote a new version). Triggering a refetch lets us pick up
55 // auto_reload flips: if the user just unchecked Watch, the cascade
56 // re-evaluates to false and the cleanup unsubscribes us. Without this
57 // the subscription persisted across the flip — the file watcher kept
58 // firing despite the user opting out.
59 const [refetchKey, setRefetchKey] = useState(0);
60 
61 // Fetch project_id once the project is loaded.
62 useEffect(() => {
63 if (!projectPath) { setProjectId(null); return; }
64 let cancelled = false;
65 invoke<string>('db_get_project_id')
66 .then((id) => { if (!cancelled) setProjectId(id); })
67 .catch(() => { if (!cancelled) setProjectId(null); });
68 return () => { cancelled = true; };
69 }, [projectPath]);
70 
71 // Listen for asset-changed events that match our asset. On match,
72 // bump refetchKey to re-run the subscription effect below (which
73 // refetches meta and re-evaluates the cascade).
74 useEffect(() => {
75 if (!assetId) return;
76 const handler = (e: Event) => {
77 const detail = (e as CustomEvent).detail as { assetId?: string } | undefined;
78 if (detail?.assetId === assetId) {
79 hlog(`asset-changed event for asset=${assetId.slice(0, 8)} → refetching meta + re-evaluating cascade`);
80 setRefetchKey((k) => k + 1);
81 }
82 };
83 window.addEventListener('eigendeck:asset-changed', handler);
84 return () => window.removeEventListener('eigendeck:asset-changed', handler);
85 }, [assetId]);
86 
87 useEffect(() => {
88 if (!assetId || !projectPath || !projectId) {
89 if (assetId) hlog(`skip mount for asset=${assetId.slice(0, 8)} — projectPath=${!!projectPath} projectId=${!!projectId}`);
90 return;
91 }
92 let cancelled = false;
93 let registeredExternalRel: string | null = null;
94 let registeredAssetId: string | null = null;
95 
96 (async () => {
97 const meta = await invoke<AssetMeta | null>('db_get_asset_meta_by_id', { assetId })
98 .catch((e) => { hlog(`meta-by-id lookup FAILED for ${assetId.slice(0, 8)}:`, e); return null; });
99 if (cancelled) return;
100 if (!meta) {
101 hlog(`no asset meta for asset=${assetId.slice(0, 8)} — nothing to watch`);
102 return;
103 }
104 hlog(`meta for ${assetId.slice(0, 8)}: path="${meta.path}" external_path=${meta.external_path} mtime=${meta.external_mtime} auto_reload=${meta.auto_reload}`);
105 const effective = effectiveAutoReload(meta.auto_reload, presOverride, globalAutoReload);
106 if (!effective) {
107 hlog(`skip — auto_reload resolves to OFF (per-asset='${meta.auto_reload ?? 'default'}', per-pres='${presOverride ?? 'default'}', global=${globalAutoReload})`);
108 return;
109 }
110 if (!meta.external_path) {
111 hlog(`skip — no external_path for asset=${meta.asset_id.slice(0, 8)} (e.g. pasted, snapshot, or never linked)`);
112 return;
113 }
114 
115 const registry = getWatcherRegistry(projectId, dirname(projectPath));
116 const origPath = meta.path ?? meta.asset_id;
117 const effectiveMime = meta.mime_type ?? 'application/octet-stream';
118 await registry.addRef(meta.external_path, meta.asset_id, elementId, origPath, effectiveMime);
119 registeredExternalRel = meta.external_path;
120 registeredAssetId = meta.asset_id;
121 })();
122 
123 return () => {
124 cancelled = true;
125 if (registeredExternalRel && registeredAssetId && projectId) {
126 const registry = getWatcherRegistry(projectId, dirname(projectPath));
127 registry.removeRef(registeredExternalRel, registeredAssetId, elementId);
128 }
129 };
130 }, [assetId, elementId, projectPath, projectId, globalAutoReload, presOverride, refetchKey]);
131}
