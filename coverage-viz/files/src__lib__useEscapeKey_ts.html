3// Close-on-Escape for in-app "fake" modals (not native dialogs, so they need
4// their own key handling — #120). Listens at the window in the CAPTURE phase
5// while `active`, so Escape dismisses the modal regardless of which child has
6// focus (e.g. an autoFocus text input that would otherwise swallow it) and
7// before the editor's own Escape handling runs. Only re-subscribes when
8// `active` toggles; the latest callback is read via a ref.
9export function useEscapeKey(active: boolean, onEscape: () => void): void {
10 const cb = useRef(onEscape);
11 cb.current = onEscape;
12 useEffect(() => {
13 if (!active) return;
14 const handler = (e: KeyboardEvent) => {
15 if (e.key === 'Escape') {
16 e.preventDefault();
17 e.stopPropagation();
18 cb.current();
19 }
20 };
21 window.addEventListener('keydown', handler, true);
22 return () => window.removeEventListener('keydown', handler, true);
23 }, [active]);
24}
