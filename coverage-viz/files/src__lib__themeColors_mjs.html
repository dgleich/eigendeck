1// SINGLE SOURCE OF TRUTH for built-in theme colors. Pure `.mjs` so BOTH the TS
2// theme system (themes.ts → BUILT_IN_THEMES) and the static HTML export
3// (exportCore.mjs, used by the headless eigendeck-cli) share one copy — neither
4// keeps its own. Foreground colors live here too (not just backgrounds), which
5// is what lets the CLI export resolve theme-aware text color and fixes #104
6// (the CLI legacy text path previously used hard-coded preset colors, so a
7// default-color text element on a dark/black theme came out near-invisible).
8//
9// Values mirror BUILT_IN_THEMES in themes.ts (which now consumes this map).

11export const THEME_COLORS = {

13 light: { background: '#f5f0e8', text: '#2c2418', heading: '#2c2418', accent: '#1e5c99', muted: '#8c7e6a' },

18// Text-preset name → which theme color key it draws from.
19export const PRESET_COLOR_MAP = {
20 title: 'heading',
21 body: 'text',
22 textbox: 'text',
23 annotation: 'accent',
24 footnote: 'muted',
25};
26 
27/** Resolve a slide's effective theme colors by name (slide theme wins over deck). */
28export function themeColorsByName(presentationTheme, slideTheme) {
29 const name = slideTheme || presentationTheme || 'white';
30 return THEME_COLORS[name] || THEME_COLORS.white;
31}
32 
33/** Theme-default color for a text preset (falls back to body text color). */
34export function themeColorForPreset(theme, preset) {
35 const key = PRESET_COLOR_MAP[preset];
36 return key ? theme[key] : theme.text;
