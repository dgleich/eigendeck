120function formatSlideList(nums: number[]): string {
121 if (nums.length === 0) return 'no slides'; // shouldn't fire — caller skips dialog in that case
122 if (nums.length === 1) return `slide ${nums[0]}`;
123 if (nums.length === 2) return `slides ${nums[0]} and ${nums[1]}`;
124 const head = nums.slice(0, -1).join(', ');
125 return `slides ${head}, and ${nums[nums.length - 1]}`;
