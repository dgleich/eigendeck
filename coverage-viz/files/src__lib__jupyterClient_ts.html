61function uuid(): string {
62 // RFC 4122 v4 — good enough for client-side correlation
63 return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
64 const r = (Math.random() * 16) | 0;
65 return (c === 'x' ? r : (r & 0x3) | 0x8).toString(16);

131 /** POST /api/kernels + open WS channels. Idempotent: if a kernel is

133 async startKernel(kernelName: string): Promise<void> {
134 if (this.kernel) return;

149 private openChannels(): Promise<void> {
150 if (!this.kernel) return Promise.reject(new Error('no kernel'));
151 const wsBase = this.baseUrl.replace(/^http/, 'ws');
152 const url = `${wsBase}/api/kernels/${this.kernel.id}/channels${this.authQuery()}`;
153 return new Promise((resolve, reject) => {
154 let settled = false;
155 const ws = new WebSocket(url);
156 this.ws = ws;
157 // Resolve connect() only once the kernel is confirmed ready — never on bare
158 // ws.onopen. jupyter_server runs a connection "nudge" (a kernel_info
159 // handshake) on every new channels WebSocket and does not reliably deliver
160 // shell messages that arrive before it completes. Sending execute_request in
161 // onopen therefore races that nudge and the first cell run is silently
162 // dropped (no output). We force the handshake ourselves and gate readiness
163 // on kernel_info_reply.
164 const finish = () => {
165 if (settled) return;
166 settled = true;
167 if (this.readyTimer != null) { clearTimeout(this.readyTimer); this.readyTimer = null; }
168 this.markReady = null;
169 this.kernelInfoMsgId = null;
170 resolve();
171 };
172 this.markReady = finish;
173 ws.onopen = () => {
174 const mid = uuid();
175 this.kernelInfoMsgId = mid;
176 try {
177 ws.send(JSON.stringify({
178 header: this.buildHeader('kernel_info_request', mid),
179 parent_header: {}, metadata: {}, content: {}, channel: 'shell',
180 }));
181 } catch { /* fall through to the timeout fallback */ }
182 // Fallback: proceed anyway if no reply arrives (an older server or a
183 // kernel slow to answer kernel_info) so connect() never hangs.
184 this.readyTimer = setTimeout(finish, 5000);
185 };
186 ws.onerror = () => { if (!settled) { settled = true; reject(new Error('WebSocket error during open')); } };
187 ws.onmessage = (ev) => this.handleMessage(ev.data);
188 ws.onclose = () => {
189 // Notify all pending executions that the kernel went away.
190 for (const [, p] of this.pending) {
191 p.callbacks.onStatus?.('dead');
192 p.resolve('aborted');
193 }
194 this.pending.clear();
195 this.ws = null;
196 if (!settled) {
197 settled = true;
198 if (this.readyTimer != null) { clearTimeout(this.readyTimer); this.readyTimer = null; }
199 reject(new Error('WebSocket closed before the kernel was ready'));
200 }
201 };
202 });
203 }

205 private handleMessage(raw: string | ArrayBuffer | Blob): void {
206 if (typeof raw !== 'string') return; // binary buffers aren't used by our message types
207 let msg: {
208 header?: { msg_type?: string };
209 parent_header?: { msg_id?: string };
210 channel?: string;
211 content?: Record<string, unknown>;
212 };
213 try { msg = JSON.parse(raw); } catch { return; }
214 const t = msg.header?.msg_type;
215 const parentId = msg.parent_header?.msg_id;
216 const c = msg.content ?? {};
217 // Readiness handshake (see openChannels): our kernel_info_request has been
218 // answered → the channel is wired and safe for execute_request.
219 if (t === 'kernel_info_reply' && parentId && parentId === this.kernelInfoMsgId) {
220 this.markReady?.();
221 return;
222 }
223 if (!parentId) return;
224 const pending = this.pending.get(parentId);
225 if (!pending) return;
226 const cb = pending.callbacks;
227 
228 if (t === 'status') {
229 const state = c.execution_state as string | undefined;
230 if (state === 'busy' || state === 'idle' || state === 'dead') {
231 cb.onStatus?.(state);
232 }
233 } else if (t === 'stream') {
234 const name = c.name === 'stderr' ? 'stderr' : 'stdout';
235 cb.onStream?.({ name, text: String(c.text ?? '') });
236 } else if (t === 'display_data') {
237 cb.onDisplayData?.({ data: (c.data ?? {}) as MimeBundle });
238 } else if (t === 'execute_result') {
239 cb.onExecuteResult?.({
240 data: (c.data ?? {}) as MimeBundle,
241 executionCount: typeof c.execution_count === 'number' ? c.execution_count : null,
242 });
243 } else if (t === 'error') {
244 cb.onError?.({
245 ename: String(c.ename ?? 'Error'),
246 evalue: String(c.evalue ?? ''),
247 traceback: Array.isArray(c.traceback) ? (c.traceback as unknown[]).map(String) : [],
248 });
249 } else if (t === 'execute_reply' && msg.channel === 'shell') {
250 const status = c.status as string;
251 const norm: 'ok' | 'error' | 'aborted' =
252 status === 'ok' || status === 'error' || status === 'aborted' ? status : 'aborted';
253 cb.onReply?.({
254 status: norm,
255 executionCount: typeof c.execution_count === 'number' ? c.execution_count : null,
256 });
257 pending.resolve(norm);
258 this.pending.delete(parentId);
259 }
260 }

262 /** Send execute_request. Throws if no kernel is running.
263 * Returns a handle whose `done` resolves on execute_reply. */
264 execute(code: string, callbacks: ExecuteCallbacks = {}): ExecuteHandle {
265 if (!this.ws || this.ws.readyState !== WebSocket.OPEN) {
266 throw new Error('execute: not connected');
267 }
268 const msgId = uuid();
269 const msg = {
270 header: this.buildHeader('execute_request', msgId),
271 parent_header: {},
272 metadata: {},
273 content: {
274 code, silent: false, store_history: true,
275 user_expressions: {}, allow_stdin: false, stop_on_error: true,
276 },
277 buffers: [],
278 channel: 'shell',
279 };
280 let resolve!: (s: 'ok' | 'error' | 'aborted') => void;
281 const done = new Promise<'ok' | 'error' | 'aborted'>((r) => { resolve = r; });
282 this.pending.set(msgId, { resolve, callbacks });
283 this.ws.send(JSON.stringify(msg));
284 return {
285 msgId,
286 done,
287 interrupt: () => this.interrupt(),
288 };
289 }

300 /** DELETE /api/kernels/:id and close the WS. Safe to call any time. */

304 this.kernelInfoMsgId = null;
