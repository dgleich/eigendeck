9/**
10 * Confirm a destructive write-in-place batch action. Returns true if user
11 * accepts. The shared phrasing makes the intent explicit (this rewrites
12 * source files; no backups; revert via git if needed).
13 */
14export async function confirmWriteInPlace(action: string, dir: string, fileCount: number): Promise<boolean> {
15 return ask(
16 `${action} will REWRITE all ${fileCount} .eigendeck file(s) in:\n${dir}\n\n` +
17 `No backups are created — commit or back up first if needed.\n\nContinue?`,
18 { title: `Debug: ${action}`, kind: 'warning', okLabel: 'Rewrite files', cancelLabel: 'Cancel' },
19 );
20}

22/** ISO timestamp safe for filenames. */
23function stampForFilename(): string {
24 const d = new Date();
25 const pad = (n: number) => `${n}`.padStart(2, '0');
26 return `${d.getFullYear()}${pad(d.getMonth() + 1)}${pad(d.getDate())}-${pad(d.getHours())}${pad(d.getMinutes())}${pad(d.getSeconds())}`;
27}

29/**
30 * Write a report file alongside the directory + show a native "done" alert
31 * summarising pass/fail. Returns the report path.
32 */
33export async function writeReportAndAlert(
34 dir: string,
35 meta: RunMeta,
36 report: object,
37): Promise<string> {
38 const path = `${dir}/debug-report-${meta.action}-${stampForFilename()}.json`;
39 await writeTextFileNative(path, JSON.stringify(report, null, 2));
40 await message(
41 `${meta.passed}/${meta.totalFiles} passed (${meta.failed} failed) in ${meta.elapsedSeconds.toFixed(1)}s\n\nReport: ${path}`,
42 { title: `Debug: ${meta.action}`, kind: meta.failed === 0 ? 'info' : 'warning' },
43 );
44 return path;
45}
