1// Filesystem commands that replace the frontend's JS `fs`-plugin usage, so the
2// webview can drop the `fs` capability entirely (no ambient disk access for any
3// injected script). Reads of asset bytes keep going through `resolve_and_read`
4// (the gated realpath+size-cap read in lib.rs); this module adds the rest:
5// writes, stat/exists, mkdir, read_dir, the trust-ledger persistence, and a
6// notify-based file watcher that emits `fs-watch-event` to the frontend.
7//
8// Paths handed to write/stat/read_dir come from native dialogs or deck-relative
9// resolution — user-chosen, not attacker-chosen. Nothing here canonicalizes or
10// gates (that's `resolve_and_read`'s job for the security-sensitive byte reads);
11// instead the write/stat/watch commands require the caller to be the MAIN editor
12// window (`require_main`) — see the caller-authorization note below.

22// --- caller authorization ----------------------------------------------------
23// These commands act on whatever path the caller passes, with NO canonicalization
24// or trust check (that is `resolve_and_read`'s job for the security-sensitive byte
25// reads, which stays open to every window because the presenter renders assets
26// through it). Only the MAIN editor window has any legitimate reason to drive
27// arbitrary-path writes/stat/watch — the presenter/settings/security windows never
28// call them. Refusing from any other window means a stray or secondary webview
29// label can't turn its IPC access into arbitrary-path disk I/O. (audit C-3)
30//
31// The filesystem logic lives in window-free `*_impl` helpers so the unit tests can
32// exercise it without constructing a WebviewWindow; the `#[tauri::command]`
33// wrappers add the guard.
34//
35// These helpers are `pub(crate)` because the same caller boundary applies to the
36// other arbitrary-path commands elsewhere in the crate (storage.rs db_open/save,
37// lib.rs cli_write_and_exit/resolve_and_read, llmtools.rs install_llm_tools).
38 
39/// Refuse unless the calling window's label is in `allowed`.
40pub(crate) fn require_windows(window: &WebviewWindow, allowed: &[&str]) -> Result<(), String> {
41 if allowed.contains(&window.label()) {
42 Ok(())
43 } else {
44 Err(format!(
45 "command not permitted from window '{}'",
46 window.label()
47 ))
48 }
49}

117fn path_stat_impl(path: &str) -> Result<StatInfo, String> {
118 let meta = std::fs::metadata(path).map_err(|e| e.to_string())?;
119 let mtime_ms = meta
120 .modified()
121 .ok()
122 .and_then(|t| t.duration_since(UNIX_EPOCH).ok())
123 .map(|d| d.as_millis() as i64);
124 Ok(StatInfo {
125 mtime_ms,
126 size: meta.len(),
127 is_file: meta.is_file(),
128 is_dir: meta.is_dir(),
129 })
130}

162// Debug-builds-only: arbitrary directory enumeration is dev/batch tooling (its sole
163// caller is the debug `dirPicker`). `#[cfg(debug_assertions)]` compiles it OUT of a
164// release build entirely — a shipped binary has no `read_dir` command at all — matching
165// the rest of the debug subsystem, which is already debug-builds-only (lib.rs). The
166// registration in lib.rs's `generate_handler!` is `#[cfg(debug_assertions)]` to match.
167// (audit C-3)
168#[cfg(debug_assertions)]
169#[tauri::command]
170pub fn read_dir(path: String) -> Result<Vec<DirEntryInfo>, String> {
171 let mut out = Vec::new();
172 for entry in std::fs::read_dir(&path).map_err(|e| e.to_string())? {
173 let entry = entry.map_err(|e| e.to_string())?;
174 let is_dir = entry.file_type().map(|t| t.is_dir()).unwrap_or(false);
175 out.push(DirEntryInfo {
176 name: entry.file_name().to_string_lossy().into_owned(),
177 path: entry.path().to_string_lossy().into_owned(),
178 is_dir,
179 });
180 }
181 Ok(out)
182}

210// --- file watcher (was JS fs-plugin `watch`) ---------------------------------
211// One RecommendedWatcher per subscribed path, keyed by an incrementing id the
212// frontend holds and passes back to unwatch_path. Each raw notify event emits a
213// `fs-watch-event` {id, path}; the frontend's own COALESCE_MS window collapses
214// the burst a single save produces (same as before — the JS `delayMs` never
215// truly coalesced either).
216 
217#[derive(Default)]
218pub struct WatchState {
219 next_id: u32,
220 watchers: HashMap<u32, RecommendedWatcher>,
221}
222 
223#[tauri::command]
224pub fn watch_path(
225 window: WebviewWindow,
226 app: AppHandle,
227 state: State<Mutex<WatchState>>,
228 path: String,
229) -> Result<u32, String> {
230 require_main(&window)?;
231 let id = {
232 let mut s = state.lock().map_err(|e| e.to_string())?;
233 let id = s.next_id;
234 s.next_id = s.next_id.wrapping_add(1);
235 id
236 };
237 let app_for_event = app.clone();
238 let path_for_event = path.clone();
239 let mut watcher = notify::recommended_watcher(move |res: notify::Result<notify::Event>| {
240 if res.is_ok() {
241 let _ = app_for_event.emit(
242 "fs-watch-event",
243 serde_json::json!({ "id": id, "path": path_for_event }),
244 );
245 }
246 })
247 .map_err(|e| e.to_string())?;
248 watcher
249 .watch(std::path::Path::new(&path), RecursiveMode::NonRecursive)
250 .map_err(|e| e.to_string())?;
251 state
252 .lock()
253 .map_err(|e| e.to_string())?
254 .watchers
255 .insert(id, watcher);
256 Ok(id)
257}
258 
259#[tauri::command]
260pub fn unwatch_path(state: State<Mutex<WatchState>>, id: u32) {
261 // Dropping the watcher stops it.
262 if let Ok(mut s) = state.lock() {
263 s.watchers.remove(&id);
264 }
265}
