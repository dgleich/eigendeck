89/** Asset-bytes resolver for the notebook export builder. Handles two
90 * kinds of keys:
91 * - a normal asset id (the .ipynb) → db_get_asset_by_id;
92 * - the `overlay-<elementKey>` convention the builder uses to ask for a

94 * asset id (db_get_owned_asset_id, the same lookup useOverlay does)

591/** Dialog-free core of import-from-HTML: decode the embedded presentation from an
592 * exported HTML string, build a FRESH deck from it, atomic-save to `savePath`,
593 * and open it. Returns the imported deck's title, or null if the HTML carried no
594 * embedded Eigendeck source. Throws on a corrupt payload / IO error. Shared by
595 * importFromHtml (which wraps it in the open + save-as dialogs) and the E2E seam,
596 * so the real read -> decode (#164) -> db_import_json -> save -> open path can be
597 * exercised without native dialogs. */
598export async function importHtmlToDeck(htmlContent: string, savePath: string): Promise<string | null> {
599 // Prefer the modern self-contained deck block (structure + assets[] base64) —
600 // db_import_json restores the assets, so demos/images/notebooks come back
601 // (docs .claude/notes/html-export-plan.md). Fall back to the legacy structure-only
602 // <!-- eigendeck-source --> comment (assets not recoverable). The comment decode
603 // is the exact inverse of the UTF-8-safe encode (#164 — a plain atob mangled
604 // non-ASCII); the deck block is already JSON, passed straight through.
605 const deckJson = extractEigendeckDeckJson(htmlContent) as string | null;
606 let importJson: string;
607 let title: string;
608 if (deckJson) {
609 importJson = deckJson;
610 title = (JSON.parse(deckJson).title as string) || 'Imported Presentation';
611 } else {
612 const presentation = parseEigendeckSource(htmlContent) as Presentation | null;
613 if (!presentation) return null;
614 importJson = JSON.stringify(presentation);
615 title = presentation.title;
616 }
617 // Build in a fresh in-memory DB and atomic-save over `savePath` rather than
618 // opening (possibly an existing file) to clear it in place. See createProject for
619 // why (dca9005 / issue #65). Close any open project first: db_open_memory is a
620 // no-op while a DB is open, so without this we'd import into the CURRENT deck's
621 // DB and carry its assets across.
622 const { closeSqliteProject } = await import('./presentation');→ presentation.ts
623 await closeSqliteProject();
624 await invoke('db_open_memory');
625 await invoke('db_import_json', { json: importJson });
626 await invoke('db_save_to_file', { path: savePath });
627 await openSqliteProject(savePath);
628 addRecentProject(savePath, title);
629 return title;
630}
