1// Thin wrapper around Tauri's `invoke` that lazy-imports the core module, no-ops
2// off Tauri, and swallows errors — collapses the repeated
3// `import('@tauri-apps/api/core').then(({invoke}) => invoke(...)).catch(()=>{})`
4// boilerplate (esp. the native-toolbar bridge effects in App.tsx).
5export async function invokeSafe<T = void>(
6 cmd: string,
7 args?: Record<string, unknown>,
8): Promise<T | undefined> {
9 try {
10 const { invoke } = await import('@tauri-apps/api/core');
11 return await invoke<T>(cmd, args);
12 } catch {
13 return undefined;
14 }
