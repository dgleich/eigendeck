19async function readSchemaVersion(): Promise<number | null> {
20 try {
21 const v = await invoke<string>('db_export_json'); // just used to ensure handle open
22 // db_export_json doesn't return the schema version directly; query via raw SQL.
23 // The storage layer exposes _meta via db_export_json's presentation/config
24 // path, but the schema version lives in the _meta table directly. There's
25 // no dedicated command — fall back to assuming v2 after db_open since
26 // open performs the migration. Return null if we can't determine.
27 void v;
28 return 2; // post-open we're always v2 (migration runs in db_open)
29 } catch {
30 return null;
31 }
32}

41async function warmOne(input: string): Promise<WarmCacheFileReport> {
42 const start = performance.now();
43 const errors: string[] = [];
44 try {
45 await invoke('db_open', { path: input });
46 // db_open runs schema migration as a side effect — we can't easily
47 // distinguish v1-before from v2-before without raw SQL access, but
48 // we KNOW the file is v2 after the open call completes.
49 const schemaAfter = await readSchemaVersion();
50 const cacheRowsBefore = await cacheRowCount();
51 
52 const json = await invoke<string>('db_export_json');
53 const presentation = JSON.parse(json);
54 const preamble = presentation.config?.mathPreamble || '';
55 
56 let expressionsFound = 0;
57 let rendered = 0;
58 let renderFailures = 0;
59 for (const slide of presentation.slides || []) {
60 for (const el of slide.elements || []) {
61 if (el.type !== 'text' || !el.html) continue;
62 const bundle = fontForPreset(el.preset, slide, presentation.config).id;
63 for (const [tex, display] of extractMath(el.html)) {
64 expressionsFound++;
65 try {
66 // renderMath checks pool.cache first; on miss it renders via the
67 // iframe pool and write-throughs to SQLite via persistToSqlite.
68 await renderMath(tex, bundle, display, preamble);
69 rendered++;
70 } catch (e) {
71 renderFailures++;
72 if (errors.length < 5) errors.push(`${bundle} ${display ? 'disp' : 'inline'} ${JSON.stringify(tex.slice(0, 40))}: ${e instanceof Error ? e.message : String(e)}`);
73 }
74 }
75 }
76 }
77 
78 const cacheRowsAfter = await cacheRowCount();
79 return {
80 input, ok: renderFailures === 0,
81 error: renderFailures > 0 ? `${renderFailures}/${expressionsFound} expression(s) failed to render` : undefined,
82 schemaBefore: null, schemaAfter,
83 expressionsFound, cacheRowsBefore, cacheRowsAfter,
84 rendered, renderFailures, errors,
85 elapsedMs: performance.now() - start,
86 };
87 } catch (e) {
88 return {
89 input, ok: false, error: e instanceof Error ? e.message : String(e),
90 schemaBefore: null, schemaAfter: null,
91 expressionsFound: 0, cacheRowsBefore: 0, cacheRowsAfter: 0,
92 rendered: 0, renderFailures: 0, errors,
93 elapsedMs: performance.now() - start,
94 };
95 }
96}

98export async function runBatchWarmCache(): Promise<void> {
99 try {
100 const picked = await pickDirectoryWithEigendecks('Pick a directory of .eigendeck files to warm (rewrites in place)');
101 if (!picked) return;
102 const { dir, files } = picked;
103 if (files.length === 0) {
104 await showBatchError('batch-warm-cache', 'No .eigendeck files in the chosen directory.');
105 return;
106 }
107 if (!(await confirmWriteInPlace('Batch Warm Cache', dir, files.length))) return;
108 
109 const originalPath = usePresentationStore.getState().projectPath;
110 const startWall = performance.now();
111 const reports: WarmCacheFileReport[] = [];
112 for (const f of files) reports.push(await warmOne(f));
113 const elapsedSeconds = (performance.now() - startWall) / 1000;
114 
115 const meta: RunMeta = {
116 action: 'batch-warm-cache',
117 startedAt: new Date().toISOString(),
118 directory: dir,
119 totalFiles: files.length,
120 elapsedSeconds,
121 passed: reports.filter((r) => r.ok).length,
122 failed: reports.filter((r) => !r.ok).length,
123 };
124 const report: WarmCacheReport = { meta, files: reports };
125 await writeReportAndAlert(dir, meta, report);
126 
127 if (originalPath) {
128 try { await openSqliteProject(originalPath); } catch { /* manual reopen */ }
129 } else {
130 try { await invoke('db_close'); } catch { /* nothing open */ }
131 }
132 } catch (e) {
133 await showBatchError('batch-warm-cache', e);
134 }
