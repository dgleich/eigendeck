3// The pure state machine lives in trustLedger.mjs; this wraps it with:
4// - lazy load from a JSON file in Tauri's appDataDir (in-memory cache),
5// - best-effort write-through on every mutation,
6// - graceful degradation to in-memory-only when Tauri APIs are absent

10// keyed by deck-token; it is NOT stored in any deck.

15// Persistence is the Rust read_trust_ledger / write_trust_ledger pair (app-data
16// JSON, dir created before write, None-on-missing). The webview has no fs-plugin
17// access; these commands own the file I/O. In a non-Tauri context (tests/CLI)
18// `invoke` throws → we degrade to an in-memory-only ledger, so callers never
19// special-case it.
20 
21let _ledger: Promise<TrustLedger> | null = null;
22 
23async function loadLedger(): Promise<TrustLedger> {
24 try {
25 const json = await invoke<string | null>('read_trust_ledger');
26 if (!json) return new TrustLedger(); // missing → empty (fail-safe: untrusted)
27 return TrustLedger.deserialize(JSON.parse(json));
28 } catch {
29 // non-Tauri context, or corrupt/unreadable JSON → start clean (untrusted).
30 return new TrustLedger();
31 }
32}
33 
34/** The in-memory ledger, loaded once. */
35function ledger(): Promise<TrustLedger> {
36 if (!_ledger) _ledger = loadLedger();
37 return _ledger;
38}

40/** Best-effort write-through. Never throws — a failed persist must not break the
41 * app; worst case a trust decision doesn't survive a restart (fail-safe: the deck
42 * re-prompts, it never silently gains trust). The Rust command creates the
43 * app-data dir before writing. */
44async function persist(l: TrustLedger): Promise<void> {
45 try {
46 await invoke('write_trust_ledger', { json: JSON.stringify(l.serialize()) });
47 } catch { /* ignore — see doc comment */ }
48}

50// NOTE: load→mutate→persist is not atomic. Fine today — this is a single-window
51// app and all calls are await-serialized on one in-memory ledger. If multi-window
52// (multi-presentation windows are on the roadmap) ever shares this appData file,
53// last-writer-wins could drop an approval; funnel all ledger mutations through one
54// owner (or add file locking / merge-on-write) before that lands.
55async function mutate<T>(fn: (l: TrustLedger, now: number) => T): Promise<T> {
56 const l = await ledger();
57 const out = fn(l, Date.now());
58 await persist(l);
59 return out;
60}
