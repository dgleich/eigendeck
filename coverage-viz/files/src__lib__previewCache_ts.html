2// asset_cache (variant 'preview'), keyed by the element's sync identity
3// (syncId ?? id) — synced instances look identical and share one preview, the

27function dataUrlToBytes(dataUrl: string): Uint8Array {
28 const comma = dataUrl.indexOf(',');
29 const b64 = comma >= 0 ? dataUrl.slice(comma + 1) : dataUrl;
30 const bin = atob(b64);
31 const out = new Uint8Array(bin.length);
32 for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
33 return out;
34}

44/** Dedup + persist a capture: skip if the size/theme signature is unchanged
45 * (in-memory + persisted), else run `produce` for the PNG data URL and store it.
46 * Shared by the demo-iframe path (bridge round-trip) and the in-DOM path
47 * (modern-screenshot) — they differ only in the signature and the producer. */
48async function storeCapture(
49 key: string, width: number, height: number, sig: string,
50 produce: () => Promise<string | null | undefined>,
51): Promise<void> {
52 if (lastHash.get(key) === sig) return;
53 try {
54 const variants = await invoke<CacheVariant[]>('db_list_asset_cache_variants', { sourceId: key });
55 const ex = variants.find((v) => v.variant === 'preview' && v.width === width && v.height === height);
56 if (ex && ex.source_hash === sig) { lastHash.set(key, sig); return; } // persisted + unchanged
57 } catch { /* fall through and (re)capture */ }
58 if (inflight.has(key)) return;
59 inflight.add(key);
60 try {
61 const dataUrl = await produce();
62 if (!dataUrl) return;
63 const bytes = dataUrlToBytes(dataUrl);
64 await invoke('db_put_asset_cache', { sourceId: key, variant: 'preview', width, height, png: Array.from(bytes), sourceHash: sig });
65 lastHash.set(key, sig);
66 bumpPreviewVersion(key);
67 } catch (e) {
68 console.warn('capturePreview failed:', e);
69 } finally {
70 inflight.delete(key);
71 }
72}

74/** Capture the element's rendered content to a PNG and store it as the cached
75 * preview. Captures `[data-element-id=el.id]`, or — when `innerSelector` is
76 * given — that descendant (e.g. '.nb-frame', so authoring chrome on the outer
77 * DraggableBox is excluded). No-op if not mounted / no size. Never throws.
78 *
79 * `cacheSalt` is mixed into the change-detection hash for content that affects
80 * the picture but ISN'T in the captured node's own HTML — e.g. a demo's theme
81 * (#86) is injected as CSS vars in the iframe's <head>, so a theme switch leaves
82 * the captured <body> HTML identical and would otherwise never recapture.
83 *
84 * `backgroundColor` fills the capture's backdrop. A demo iframe is TRANSPARENT
85 * (so the slide shows through live); without this the rasterized PNG keeps that
86 * transparency and reads as the app's grey wherever it's shown standalone
87 * (sidebar / link picker / export). Pass the slide's resolved background so the
88 * thumbnail matches the slide. It's folded into the cache hash too, so changing
89 * the slide theme re-captures with the new backdrop. */
90export async function capturePreview(
91 el: SlideElement, innerSelector?: string, cacheSalt?: string,
92 backgroundColor?: string,
93): Promise<void> {
94 const key = previewKey(el);
95 if (inflight.has(key)) return;
96 const host = document.querySelector(`[data-element-id="${el.id}"]`) as HTMLElement | null;
97 if (!host) return;
98 let node: HTMLElement = (innerSelector
99 ? (host.querySelector(innerSelector) as HTMLElement | null)
100 : host) ?? host;
101 const width = Math.round(el.position.width);
102 const height = Math.round(el.position.height);
103 if (!width || !height) return;
104 
105 // A demo renders in an OPAQUE-ORIGIN iframe (docs/DEMO-PLATFORM.md §8): the
106 // parent can't reach its DOM, so ask the in-demo bridge to rasterize itself and
107 // post the PNG back. Dedup by size + theme salt (the content HTML is unreadable).
108 if (node instanceof HTMLIFrameElement) {
109 const iframe = node;
110 const sig = hashString(`${width}x${height}|${cacheSalt ?? ''}|${backgroundColor ?? ''}`);
111 await storeCapture(key, width, height, sig, async () => {
112 const { requestDemoCapture } = await import('./demoMount');→ demoMount.ts
113 return requestDemoCapture(iframe, { width, height, backgroundColor });
114 });
115 return;
116 }

118 // In-DOM node (notebook / video): rasterize directly. The signature includes the
119 // node's HTML so it changes exactly when the picture's structure does. (Canvas
120 // pixel state isn't in the HTML — acceptable: a thumbnail is a single frame.)
121 // The theme salt is kept as a READABLE prefix (not folded into the hash) so the
122 // PDF/print export can compare it against the current theme WITHOUT the node's
123 // outerHTML — which it can't read for a notebook that isn't on the current slide.
124 // A theme change then reads stale at export → forces a fresh capture (#140).
125 const sig = `${cacheSalt ?? ''}${PREVIEW_SALT_SEP}${hashString(`${width}x${height}|${backgroundColor ?? ''}|${node.outerHTML}`)}`;
126 await storeCapture(key, width, height, sig, async () => {
127 const { domToDataUrl } = await import('modern-screenshot');
128 return domToDataUrl(node, { width, height, scale: 1, backgroundColor });
129 });
130}

133// A monotonic version per key, bumped on (re)capture, so consumers can
134// re-fetch the blob URL when the preview changes (the URL itself is stable
135// bytes; the version is the cache-busting signal).
136const previewVersion = new Map<string, number>();
137const listeners = new Set<() => void>();
138 
139function bumpPreviewVersion(key: string): void {
140 previewVersion.set(key, (previewVersion.get(key) ?? 0) + 1);
141 for (const l of listeners) l();
142}
143/** Subscribe to any preview (re)capture — consumers re-read on change. */
144export function onPreviewChange(fn: () => void): () => void {
145 listeners.add(fn);
146 return () => listeners.delete(fn);
147}

149/** Drop a preview: forget the session hash AND delete the persisted cache row,
150 * so the next capturePreview re-renders it even if the content is unchanged.
151 * Used by "Refresh All Snapshots" to force a full re-render. Best-effort. */
152export async function clearPreview(key: string): Promise<void> {
153 lastHash.delete(key);
154 try { await invoke('db_clear_asset_cache', { sourceId: key }); }
155 catch (e) { console.warn('clearPreview failed:', e); }
156 bumpPreviewVersion(key);
157}

161/** Load the element's cached preview as an object URL, or null on a miss.
162 * Size-robust: discovers the stored `preview` size via list-variants rather
163 * than assuming the element's current size. Caller owns revoking the URL. */
164export async function loadPreviewUrl(key: string): Promise<string | null> {
165 try {
166 const variants = await invoke<CacheVariant[]>('db_list_asset_cache_variants', { sourceId: key });
167 const p = variants.find((v) => v.variant === 'preview');
168 if (!p) return null;
169 const buf = await invoke<ArrayBuffer>('db_get_asset_cache_bytes', {
170 sourceId: key, variant: 'preview', width: p.width, height: p.height,
171 });
172 const bytes = new Uint8Array(buf);
173 if (bytes.length === 0) return null; // empty Response = miss
174 return URL.createObjectURL(new Blob([bytes], { type: 'image/png' }));
175 } catch (e) {
176 console.warn('loadPreviewUrl failed:', e);
177 return null;
178 }
179}

201/** True when a cached DOM-node (notebook) preview exists but was captured under a
202 * DIFFERENT theme salt than `expectedSalt` — so the PDF/print export should
203 * re-capture it instead of baking the stale-theme PNG (#140). False when there's
204 * no cached preview, or its stored signature predates the salt-prefix format
205 * (can't tell → treat as fresh, preserving prior behavior). */
206export async function isPreviewThemeStale(key: string, expectedSalt: string): Promise<boolean> {
207 try {
208 const variants = await invoke<CacheVariant[]>('db_list_asset_cache_variants', { sourceId: key });
209 const p = variants.find((v) => v.variant === 'preview');
210 if (!p || !p.source_hash) return false;
211 const i = p.source_hash.indexOf(PREVIEW_SALT_SEP);
212 if (i < 0) return false; // pre-prefix format — can't compare, keep the cached preview
213 return p.source_hash.slice(0, i) !== expectedSalt;
214 } catch {
215 return false;
216 }
