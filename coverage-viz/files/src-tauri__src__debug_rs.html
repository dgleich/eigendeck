1//! Debug-only surface area, gated behind the `--debug` CLI flag.
2//!
3//! HARD INVARIANT (per project policy): the only reads of the debug flag in
4//! the entire codebase are inside THIS module and inside `src/debug/` on the
5//! frontend. Nothing else may branch on it — no `if debug_enabled` sprinkled
6//! through business code. To enforce this:
7//! - `parse_debug_flag()` is the ONLY place that touches `std::env::args`
8//! for the flag.
9//! - `DebugFlag` is the ONLY Tauri state carrying it.
10//! - `debug_enabled()` is the single command the frontend uses to gate
11//! `<DebugMenu />` mounting.
12//! - `attach_submenu_if_enabled()` is the only entry point lib.rs uses to
13//! conditionally append the Debug submenu; it returns `Ok(None)` when
14//! the flag is off, so lib.rs never sees the bool.
15//! - Every `debug_*` command self-gates with `require()` before doing
16//! anything (defense in depth — if the frontend somehow invoked one
17//! without `--debug`, it would refuse).
18//!
19//! The menu emits standard `menu-event` events with `debug-*` ids; the
20//! frontend's `<DebugMenu />` filters on that prefix.
21

28/// Detect the debug flag once at startup. THE ONLY env read for this flag.
29///
30/// DEBUG BUILDS ONLY. The entire debug subsystem — this call, `DebugFlag`, the Debug
31/// submenu, `debug_enabled`, and `read_dir` — is `#[cfg(debug_assertions)]`, so a
32/// shipped release build has NO debug tooling and `--debug` / `EIGENDECK_DEBUG` do
33/// nothing there (audit C-3). Within a debug build, either enables it:
34/// - `--debug` in argv (a debug binary run from the terminal), or
35/// - `EIGENDECK_DEBUG=1` env var — needed for `tauri dev`, because the Tauri CLI's
36/// argv passthrough places args BEFORE cargo's `--` separator, so they reach cargo
37/// (which rejects them) instead of the binary. tools/mac-build.sh translates its
38/// own `--debug` into the env var so `bash tools/mac-build.sh --debug` is all the
39/// user types.
40pub fn parse_debug_flag() -> bool {
41 if std::env::args().any(|a| a == "--debug") {
42 return true;
43 }
44 matches!(std::env::var("EIGENDECK_DEBUG").as_deref(), Ok(v) if !v.is_empty() && v != "0")
45}

53/// Self-gate helper for any future debug-only *command that must still exist in a
54/// debug build but refuse without `--debug`*. Kept (with `#[allow(dead_code)]`) so the
55/// pattern is in place. (`read_dir` doesn't need it — it's `#[cfg(debug_assertions)]`,
56/// so it's compiled out of release entirely rather than gated at runtime.)
57#[allow(dead_code)]
58fn require(flag: &DebugFlag) -> Result<(), String> {
59 if flag.0 {
60 Ok(())
61 } else {
62 Err("debug commands disabled (launch with --debug to enable)".into())
63 }
64}

66/// Build the Debug submenu. Called from lib.rs::build_app_menu; returns
67/// `Ok(None)` when the flag is off so lib.rs never has to read the bool.
68pub fn attach_submenu_if_enabled(app: &AppHandle) -> Result<Option<Submenu<Wry>>, String> {
69 let flag = app.state::<DebugFlag>();
70 if !flag.0 {
71 return Ok(None);
72 }
73 let batch_html = MenuItemBuilder::new("Batch HTML Export…")
74 .id("debug-batch-html")
75 .build(app)
76 .map_err(|e| e.to_string())?;
77 let batch_roundtrip = MenuItemBuilder::new("Batch Round-trip Save Test…")
78 .id("debug-batch-roundtrip")
79 .build(app)
80 .map_err(|e| e.to_string())?;
81 let batch_cache = MenuItemBuilder::new("Batch Math Cache Audit…")
82 .id("debug-batch-cache-audit")
83 .build(app)
84 .map_err(|e| e.to_string())?;
85 let batch_warm = MenuItemBuilder::new("Batch Warm Math Cache (write in place)…")
86 .id("debug-batch-warm-cache")
87 .build(app)
88 .map_err(|e| e.to_string())?;
89 let batch_strip = MenuItemBuilder::new("Batch Strip History (write in place)…")
90 .id("debug-batch-strip-history")
91 .build(app)
92 .map_err(|e| e.to_string())?;
93 let dump_pb = MenuItemBuilder::new("Dump Pasteboard Types")
94 .id("dump-pasteboard")
95 .build(app)
96 .map_err(|e| e.to_string())?;
97 let sub = SubmenuBuilder::new(app, "Debug")
98 .item(&batch_html)
99 .item(&batch_roundtrip)
100 .item(&batch_cache)
101 .separator()
102 .item(&batch_warm)
103 .item(&batch_strip)
104 .separator()
105 .item(&dump_pb)
106 .build()
107 .map_err(|e| e.to_string())?;
108 Ok(Some(sub))
109}
