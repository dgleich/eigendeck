14export type KernelStatus = 'disconnected' | 'connecting' | 'idle' | 'busy' | 'error' | 'dead' | 'no-server';
15 
16export interface UseKernelResult {
17 status: KernelStatus;
18 /** Human-readable error from the last failed connect / execute. */
19 error: string | null;
20 /** Start kernel + connect WS. Idempotent. Returns a Promise so the
21 * caller can await first-run delays before sending execute. */
22 connect(): Promise<void>;
23 /** Send execute_request. Auto-connects if not yet connected.
24 * Returns the handle (msgId, done promise, interrupt). */
25 runCell(code: string, callbacks?: ExecuteCallbacks): Promise<ExecuteHandle>;
26 /** Stop kernel + close WS. Used by "Restart Kernel" button. */
27 stop(): Promise<void>;
28}

30export function useKernel(resolved: ResolvedExternal): UseKernelResult {
31 const clientRef = useRef<JupyterClient | null>(null);
32 // 'no-server' when the registry produced no match for the requested
33 // kernel. We never attempt to connect in that state — connect() and
34 // runCell() reject early with a clear message.
35 const noServer = resolved.server == null;
36 const [status, setStatus] = useState<KernelStatus>(noServer ? 'no-server' : 'disconnected');
37 const [error, setError] = useState<string | null>(noServer
38 ? `No registered server advertises a "${resolved.kernelName}" kernel. Add one in Settings → Jupyter servers.`
39 : null);
40 
41 const serverBaseUrl = resolved.server?.baseUrl ?? '';
42 const serverToken = resolved.server?.token ?? '';
43 
44 // Recreate client when connection params change. Stops any in-flight
45 // kernel under the old config first.
46 useEffect(() => {
47 return () => {
48 const c = clientRef.current;
49 clientRef.current = null;
50 if (c) void c.stopKernel().catch(() => {});
51 };
52 }, [serverBaseUrl, serverToken, resolved.kernelName]);
53 
54 // When the registry update changes the matched server while mounted
55 // (e.g. user just saved a new server in Settings), sync our status
56 // out of 'no-server' so the next connect attempt is allowed.
57 useEffect(() => {
58 if (noServer) {
59 setStatus('no-server');
60 setError(`No registered server advertises a "${resolved.kernelName}" kernel. Add one in Settings → Jupyter servers.`);
61 } else if (status === 'no-server') {
62 setStatus('disconnected');
63 setError(null);
64 }
65 // status intentionally NOT in deps — we only want to transition
66 // out of no-server, not loop on every status change.
67 // eslint-disable-next-line react-hooks/exhaustive-deps
68 }, [noServer, resolved.kernelName]);
69 
70 const connect = useCallback(async () => {
71 if (noServer) {
72 const msg = `No registered server advertises a "${resolved.kernelName}" kernel.`;
73 setError(msg);
74 throw new Error(msg);
75 }
76 if (clientRef.current?.isConnected) return;
77 setStatus('connecting'); setError(null);
78 const client = new JupyterClient({
79 baseUrl: serverBaseUrl,
80 token: serverToken,
81 });
82 try {
83 await client.startKernel(resolved.kernelName);
84 clientRef.current = client;
85 setStatus('idle');
86 } catch (e) {
87 setStatus('error');
88 setError(e instanceof Error ? e.message : String(e));
89 throw e;
90 }
91 }, [noServer, serverBaseUrl, serverToken, resolved.kernelName]);
92 
93 const runCell = useCallback(async (code: string, callbacks?: ExecuteCallbacks) => {
94 if (!clientRef.current?.isConnected) await connect();
95 const client = clientRef.current;
96 if (!client) throw new Error('useKernel: connect did not produce a client');
97 const wrapped: ExecuteCallbacks = {
98 ...callbacks,
99 onStatus: (s) => {
100 if (s === 'busy' || s === 'idle' || s === 'dead') setStatus(s);
101 callbacks?.onStatus?.(s);
102 },
103 };
104 return client.execute(code, wrapped);
105 }, [connect]);
106 
107 const stop = useCallback(async () => {
108 const c = clientRef.current;
109 clientRef.current = null;
110 setStatus(noServer ? 'no-server' : 'disconnected');
111 if (c) await c.stopKernel().catch(() => {});
112 }, [noServer]);
113 
114 return { status, error, connect, runCell, stop };
115}
