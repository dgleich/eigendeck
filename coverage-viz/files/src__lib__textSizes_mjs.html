1// SINGLE SOURCE for the text type-scale: the named sizes, each preset's named
2// size, and the "effective size" cascade. Pure `.mjs` so BOTH the TS data model
3// (types/presentation.ts) and the headless CLI export (exportCore.mjs) share one
4// copy across the .mjs/.ts boundary — the same split themeColors.mjs uses.
5//
6// Why this exists: exportCore's legacy text path reimplemented size resolution as
7// `el.fontSize || ps.fontSize`, which ignored BOTH element.fontSizeName and the
8// deck's `config.textSizes` override. So CLI-exported text using a named size, or
9// a deck with a customized type scale, came out at the wrong (preset-default)
10// size — the editor/app-export (effectiveFontSize) and the PDF path got it right.
11// Routing the CLI through effectiveFontSize here closes that drift at the root.
12 
13/** Built-in defaults for the type scale (px). */
14export const DEFAULT_TEXT_SIZES = {
15 footnote: 24,
16 note: 32,
17 body: 48,
18 title: 72,
19 hype: 48,
20};
21 
22/** Text preset → its named size in the type scale. Mirrors the `sizeName` field
23 * of TEXT_PRESET_STYLES (guard-tested to stay in sync). */
24export const PRESET_SIZE_NAME = {
25 title: 'title',
26 body: 'body',
27 textbox: 'body',
28 annotation: 'note',
29 footnote: 'footnote',
30 hype: 'hype',
31};
32 
33/** Resolve a named size against the deck override + defaults. */
34export function resolveNamedSize(name, config) {
35 return config?.textSizes?.[name] ?? DEFAULT_TEXT_SIZES[name];
36}
37 
38/** Effective px size for a text preset, honoring the deck's textSizes override. */
39export function effectiveTextPresetSize(preset, config) {
40 return resolveNamedSize(PRESET_SIZE_NAME[preset], config);
41}
42 
43/** Single resolver for "what px size should this element render at?"
44 * Cascade: explicit numeric fontSize > named size via textSizes > preset default
45 * (text) / 'note' (notebook). */
46export function effectiveFontSize(element, config) {
47 if (element.fontSize != null) return element.fontSize;
48 if (element.fontSizeName) return resolveNamedSize(element.fontSizeName, config);
49 if (element.type === 'text') return effectiveTextPresetSize(element.preset, config);
50 return resolveNamedSize('note', config);
51}
