1//! Copy/paste support for asset-bearing slide elements.
2//!
3//! Two channels, both cross-platform, with the asset BYTES kept in Rust (never
4//! shuttled through JS):
5//!
6//! 1. **Internal clip** — a process-global the frontend uses to round-trip a
7//! copied element with FULL fidelity (element JSON + the asset bytes + the
8//! source deck id) across windows / presentations. This is what makes
9//! "copy an image in deck A, paste into deck B" carry the bytes: paste
10//! stores them into the DESTINATION deck, instead of leaving the pasted
11//! element pointing at an asset that only exists in deck A.
12//!
13//! 2. **System clipboard image** — `clip_copy_asset` also puts the real raster
14//! on the OS clipboard via `arboard` (cross-platform) so the element pastes
15//! into OTHER apps too; macOS additionally offers SVG under its native UTI.

53/// Copy an asset-bearing element: read the asset bytes from the open deck into
54/// the internal clip (with the element JSON `payload`) AND put the real image on
55/// the system clipboard for other apps. Bytes never cross to JS.
56#[tauri::command]
57pub fn clip_copy_asset(
58 app: tauri::AppHandle,
59 asset_id: String,
60 payload: String,
61 mime: String,
62) -> Result<(), String> {
63 let bytes = storage::db_get_asset_bytes_by_id(asset_id)?;
64 // System clipboard: put MULTIPLE representations on it (the native format +
65 // a PNG raster) so each target app pastes the format it understands —
66 // PDF/SVG into Keynote, PNG into apps that only take images. Best effort.
67 let reps = build_reps(&app, &bytes, &mime);
68 if let Err(e) = write_system(&app, &reps) {
69 eprintln!("[clip] system clipboard write failed: {e}");
70 }
71 // Capture the clipboard generation AFTER our write, so a later foreign copy
72 // is detectable (peek returns stale).
73 let generation = clipboard_generation(&app);
74 let mut g = INTERNAL_CLIP.lock().map_err(|e| e.to_string())?;
75 *g = Some(InternalClip { payload, bytes, mime, generation });
76 Ok(())
77}

79#[tauri::command]
80pub fn clip_peek_internal(app: tauri::AppHandle) -> Result<Option<InternalClipMeta>, String> {
81 let live = clipboard_generation(&app);
82 let mut g = INTERNAL_CLIP.lock().map_err(|e| e.to_string())?;
83 if let Some(c) = g.as_ref() {
84 // Stale: the system clipboard changed since our copy (a foreign app
85 // copied something). Drop the internal clip so the foreign content wins.
86 if c.generation >= 0 && live >= 0 && live != c.generation {
87 *g = None;
88 return Ok(None);
89 }
90 }
91 Ok(g.as_ref().map(|c| InternalClipMeta {
92 payload: c.payload.clone(),
93 mime: c.mime.clone(),
94 has_bytes: !c.bytes.is_empty(),
95 }))
96}

98/// Paste the internal clip's asset INTO the current deck: store the bytes as a
99/// new asset at `path` and return its fresh asset_id + the element payload.
100/// None when the internal clip is empty / has no bytes.
101#[tauri::command]
102pub fn clip_paste_asset(path: String) -> Result<Option<PastedAsset>, String> {
103 let (bytes, payload, mime) = {
104 let g = INTERNAL_CLIP.lock().map_err(|e| e.to_string())?;
105 match g.as_ref() {
106 Some(c) if !c.bytes.is_empty() => (c.bytes.clone(), c.payload.clone(), c.mime.clone()),
107 _ => return Ok(None),
108 }
109 };
110 // Dedupe by content hash: pasting the same image repeatedly reuses one
111 // asset instead of piling up identical copies in the deck.
112 let asset_id = storage::store_asset_deduped(path, bytes, mime.clone())?;
113 Ok(Some(PastedAsset { asset_id, payload, mime }))
114}

123/// Read a raster image off the SYSTEM clipboard as PNG bytes (or None if there
124/// isn't one). The Linux image-paste fallback (#94): WebKitGTK's sync
125/// `clipboardData` and async `navigator.clipboard.read()` don't reliably surface
126/// a screenshot's `image/png`, so the web paste paths miss. arboard reads the
127/// X11/Wayland clipboard image directly, so a screenshot pastes even when the
128/// web APIs expose nothing. macOS keeps using the unfiltered `pasteboard_*`
129/// path, so this is a no-op there.
130#[tauri::command]
131pub fn clip_read_system_image() -> Result<Option<Vec<u8>>, String> {
132 #[cfg(not(target_os = "macos"))]
133 {
134 let mut cb = match arboard::Clipboard::new() {
135 Ok(c) => c,
136 Err(_) => return Ok(None),
137 };
138 let img = match cb.get_image() {
139 Ok(img) => img,
140 Err(_) => return Ok(None), // no image on the clipboard
141 };
142 let (w, h) = (img.width as u32, img.height as u32);
143 let raw = img.bytes.into_owned();
144 let buf = image::RgbaImage::from_raw(w, h, raw)
145 .ok_or_else(|| "clipboard image: byte length doesn't match dimensions".to_string())?;
146 let mut png = Vec::new();
147 image::DynamicImage::ImageRgba8(buf)
148 .write_to(&mut std::io::Cursor::new(&mut png), image::ImageFormat::Png)
149 .map_err(|e| e.to_string())?;
150 Ok(Some(png))
151 }
152 #[cfg(target_os = "macos")]
153 {
154 Ok(None)
155 }
156}

158/// The system clipboard's monotonic generation counter, so we can tell whether
159/// the clipboard changed since our copy. macOS: NSPasteboard.changeCount.
160/// Other platforms: -1 (no cheap counter; the internal clip is treated as fresh
161/// — the rare "copied in eigendeck, then copied elsewhere, then pasted" case is
162/// a macOS-correct, best-effort-elsewhere edge).
163#[allow(unused_variables)]
164fn clipboard_generation(app: &tauri::AppHandle) -> i64 {
165 #[cfg(target_os = "macos")]
166 {
167 use std::sync::mpsc;
168 let (tx, rx) = mpsc::channel::<i64>();
169 let _ = app.run_on_main_thread(move || {
170 use objc2_app_kit::NSPasteboard;
171 let pb = NSPasteboard::generalPasteboard();
172 let _ = tx.send(pb.changeCount() as i64);
173 });
174 rx.recv().unwrap_or(-1)
175 }
176 #[cfg(not(target_os = "macos"))]
177 {
178 system_clipboard_generation()
179 }
180}

182/// Linux/Windows generation: a HASH of the current system clipboard content.
183/// These platforms have no cheap monotonic counter like NSPasteboard's
184/// changeCount, so a foreign copy (screenshot, text, or an image from another
185/// app) changes the bytes, the hash changes, and the internal clip reads stale so
186/// the foreign content wins. (#94: on Linux the old sentinel value made the
187/// internal clip ALWAYS fresh, so an in-app image copy permanently shadowed later
188/// screenshots.) Called only on copy and paste (user actions), so the one
189/// clipboard read is affordable. Returns a non-negative hash (top bit cleared),
190/// because the staleness check requires a value at or above zero; a negative
191/// value is returned only when the clipboard is empty or unreadable.
192#[cfg(not(target_os = "macos"))]
193fn system_clipboard_generation() -> i64 {
194 use std::hash::{Hash, Hasher};
195 let mut cb = match arboard::Clipboard::new() {
196 Ok(c) => c,
197 Err(_) => return -1,
198 };
199 let mut h = std::collections::hash_map::DefaultHasher::new();
200 // Prefer the image target (screenshots / our own raster), fall back to text.
201 // A distinct tag per kind keeps an image and same-length text apart.
202 if let Ok(img) = cb.get_image() {
203 b'I'.hash(&mut h);
204 img.width.hash(&mut h);
205 img.height.hash(&mut h);
206 img.bytes.as_ref().hash(&mut h);
207 } else if let Ok(txt) = cb.get_text() {
208 if txt.is_empty() {
209 return -1;
210 }
211 b'T'.hash(&mut h);
212 txt.hash(&mut h);
213 } else {
214 return -1; // empty / unreadable clipboard
215 }
216 (h.finish() >> 1) as i64
217}

276/// Build the clipboard representations (uti, bytes) for an asset: the native
277/// format plus a PNG raster where we can produce one in Rust.
278/// - PDF → com.adobe.pdf + public.png (pdfium raster)
279/// - SVG → public.svg-image (vector apps; no Rust SVG rasterizer, so no PNG)
280/// - PNG → public.png
281/// - other raster → public.png (decoded)
282fn build_reps(app: &tauri::AppHandle, bytes: &[u8], mime: &str) -> Vec<(String, Vec<u8>)> {
283 let mut reps: Vec<(String, Vec<u8>)> = Vec::new();
284 match mime {
285 "application/pdf" => {
286 reps.push(("com.adobe.pdf".to_string(), bytes.to_vec()));
287 match crate::pdf::render_first_page_png(app, bytes, 1600, 1200) {
288 Ok(png) => reps.push(("public.png".to_string(), png)),
289 Err(e) => eprintln!("[clip] pdf raster for clipboard failed: {e}"),
290 }
291 }
292 "image/svg+xml" => reps.push(("public.svg-image".to_string(), bytes.to_vec())),
293 "image/png" => reps.push(("public.png".to_string(), bytes.to_vec())),
294 _ => match to_png(bytes) {
295 Ok(png) => reps.push(("public.png".to_string(), png)),
296 Err(_) => reps.push(("public.png".to_string(), bytes.to_vec())),
297 },
298 }
299 reps
300}

302/// Re-encode arbitrary raster bytes (jpeg/gif/webp/...) to PNG.
303fn to_png(bytes: &[u8]) -> Result<Vec<u8>, String> {
304 let img = image::load_from_memory(bytes).map_err(|e| e.to_string())?;
305 let mut out = std::io::Cursor::new(Vec::new());
306 img.write_to(&mut out, image::ImageFormat::Png).map_err(|e| e.to_string())?;
307 Ok(out.into_inner())
308}

310/// Write the representations to the SYSTEM clipboard. macOS: all of them, as one
311/// multi-type pasteboard write (so apps pick PDF/SVG/PNG as they prefer).
312/// Other platforms: arboard sets the single PNG raster (true multi-format needs
313/// per-OS native code — deferred; PNG covers the common case).
314fn write_system(_app: &tauri::AppHandle, reps: &[(String, Vec<u8>)]) -> Result<(), String> {
315 if reps.is_empty() {
316 return Ok(());
317 }
318 #[cfg(target_os = "macos")]
319 {
320 return mac_write_multi(_app, reps.to_vec());
321 }
322 #[cfg(not(target_os = "macos"))]
323 {
324 if let Some((_, png)) = reps.iter().find(|(u, _)| u == "public.png") {
325 let img = image::load_from_memory(png).map_err(|e| e.to_string())?.to_rgba8();
326 let (w, h) = img.dimensions();
327 let mut cb = arboard::Clipboard::new().map_err(|e| e.to_string())?;
328 cb.set_image(arboard::ImageData {
329 width: w as usize,
330 height: h as usize,
331 bytes: std::borrow::Cow::Owned(img.into_raw()),
332 })
333 .map_err(|e| e.to_string())?;
334 }
335 Ok(())
336 }
337}

339/// macOS multi-type pasteboard write. declareTypes FIRST, in preference order
340/// (most-preferred UTI first — `reps` is already ordered, e.g. com.adobe.pdf
341/// before public.png), so all representations are registered and a paste target
342/// gets its preferred one. Then setData for each. (declareTypes is the
343/// documented way to put multiple types on the pasteboard — a bare setData loop
344/// doesn't reliably register more than one.)
345#[cfg(target_os = "macos")]
346fn mac_write_multi(app: &tauri::AppHandle, reps: Vec<(String, Vec<u8>)>) -> Result<(), String> {
347 use std::sync::mpsc;
348 let (tx, rx) = mpsc::channel::<Result<(), String>>();
349 let _ = app.run_on_main_thread(move || {
350 use objc2_app_kit::NSPasteboard;
351 use objc2_foundation::{NSArray, NSData, NSString};
352 let pb = NSPasteboard::generalPasteboard();
353 pb.clearContents();
354 let types: Vec<objc2::rc::Retained<NSString>> =
355 reps.iter().map(|(u, _)| NSString::from_str(u)).collect();
356 let arr = NSArray::from_retained_slice(&types);
357 // declareTypes_owner is marked unsafe in objc2 (owner-param contract);
358 // we pass None for the owner, which is safe.
359 let _ = unsafe { pb.declareTypes_owner(&arr, None) };
360 let mut ok = true;
361 for ((_, bytes), ty) in reps.iter().zip(types.iter()) {
362 let data = NSData::with_bytes(bytes);
363 if !pb.setData_forType(Some(&data), ty) {
364 ok = false;
365 }
366 }
367 let _ = tx.send(if ok { Ok(()) } else { Err("setData failed".into()) });
368 });
369 rx.recv().unwrap_or_else(|_| Err("main-thread dispatch failed".into()))
370}
