1// Topbar pill that aggregates the connection health of registered
2// Jupyter servers against the kernels the open deck actually needs.
3//
4// Design notes (per the user's directive "no UX in the notebook
5// element"):
6// - The pill is the ONLY surface that surfaces server status to
7// the presenter. Notebook elements stay quiet.
8// - During PresentMode the pill hides entirely — the presenter has
9// already verified green; cluttering the present view isn't
10// helpful.
11// - Hidden when the deck has zero notebook elements (no reason to
12// advertise server status if none are needed).
13//
14// Health levels:
15// green: every kernel the deck needs has a matching server that
16// was reached recently (≤ 30 min).
17// yellow: kernels matched but staleness > 30 min — server might
18// or might not still be up. Click "Test all" to refresh.
19// red: at least one kernel needed by the deck has no matching
20// server in the registry.
21// hidden: no notebook elements OR PresentMode active.

58 // Unique kernel names the deck's notebooks need, then a registry lookup per
59 // kernel so we can flag missing matches. Per-element resolution (parsing each
60 // notebook for its declared kernel) happens lazily inside NotebookContent;
61 // here we use the element-level value, then the deck default, then 'python3'.
62 const deckKernels = useMemo(() => deckExternalKernels(slides, config), [slides, config]);

64 const requirements: DeckRequirement[] = useMemo(() => {
65 return deckKernels.map((kernelName) => ({
66 kernelName,
67 matched: findServerForKernel(kernelName, servers),
68 }));
69 }, [deckKernels, servers]);

71 // Hide entirely when not relevant.
72 if (isPresenting) return null;

75 // Aggregate health.
76 const missing = requirements.filter((r) => r.matched == null);
77 const needed = new Set(requirements.map((r) => r.matched).filter(Boolean) as JupyterServerEntry[]);
78 const serverChecks: ServerCheck[] = [...needed].map((entry) => ({
79 entry, health: serverHealth(entry),
80 }));

82 let health: Health;
83 if (missing.length > 0) {
84 health = 'red';
85 } else if (serverChecks.every((c) => c.health === 'green')) {
86 health = 'green';
87 } else {
88 health = 'yellow';

100 const refreshAll = async () => {
101 const updated = await Promise.all(servers.map(async (s) => {
102 try {
103 const url = s.baseUrl.replace(/\/$/, '');
104 const q = s.token ? `?token=${encodeURIComponent(s.token)}` : '';
105 const r = await fetch(`${url}/api/kernelspecs${q}`, {
106 headers: s.token ? { Authorization: `token ${s.token}` } : {},
107 });
108 if (!r.ok) return s;
109 const data = await r.json();
110 const kernels = Object.keys(data.kernelspecs ?? {});
111 return { ...s, availableKernels: kernels, lastSeenAt: Date.now() };
112 } catch {
113 return s;
114 }
115 }));
116 setServers(updated);
117 };

249function serverHealth(s: JupyterServerEntry): 'green' | 'yellow' | 'gray' {
250 if (!s.lastSeenAt) return 'gray';
251 const age = Date.now() - s.lastSeenAt;
252 if (age < STALE_MS) return 'green';
253 return 'yellow';
254}

277function countLabel(checks: ServerCheck[], missingCount: number): string {
278 const green = checks.filter((c) => c.health === 'green').length;
279 const total = checks.length;
280 if (missingCount > 0) return `${green}/${total + missingCount}`;
281 return `${green}/${total}`;
282}

284function timeAgo(ms: number): string {
285 const diff = Date.now() - ms;
286 if (diff < 60_000) return 'just now';
287 if (diff < 3_600_000) return `${Math.round(diff / 60_000)}m ago`;
288 if (diff < 86_400_000) return `${Math.round(diff / 3_600_000)}h ago`;
289 return `${Math.round(diff / 86_400_000)}d ago`;
