23export function PasteAsModal({ reps, onPick, onCancel }: Props) {
24 const firstRef = useRef<HTMLButtonElement>(null);
25 
26 useEffect(() => { firstRef.current?.focus(); }, []);
27 
28 useEffect(() => {
29 const onKey = (e: KeyboardEvent) => {
30 if (e.key === 'Escape') { e.preventDefault(); onCancel(); return; }
31 // Number keys 1..N pick the nth representation.
32 const n = parseInt(e.key, 10);
33 if (!Number.isNaN(n) && n >= 1 && n <= reps.length) {
34 e.preventDefault();
35 onPick(reps[n - 1].kind);
36 }
37 };
38 window.addEventListener('keydown', onKey, true);
39 return () => window.removeEventListener('keydown', onKey, true);
40 }, [reps, onPick, onCancel]);

42 return (

48 display: 'flex', alignItems: 'flex-start', justifyContent: 'center',
49 zIndex: 100000, paddingTop: 60,
50 }}
51 onClick={(e) => { if (e.target === e.currentTarget) onCancel(); }}
52 >
53 <div style={{
54 background: '#fff', borderRadius: 8, boxShadow: '0 12px 48px rgba(0,0,0,0.25)',
