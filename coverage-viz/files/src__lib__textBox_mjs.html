1// Inner-box layout (line spacing + padding) for a text preset's rendered HTML —
2// the SINGLE SOURCE shared by the live editor / SVG render (TextElementSvg, via
3// the types/presentation re-export) and the static HTML exports (exportCore.mjs
4// for the CLI + printSlideHtml.ts for PDF). Pure `.mjs` so it crosses the
5// .mjs/.ts boundary like themeColors.mjs / textSizes.mjs.
6//
7// Footnote renders TIGHT (lineHeight 1, no padding) so a one-line 24px footnote
8// sits flush on the grid; every other preset gets the comfortable 1.3 line-height
9// + 8/12px padding. The export paths previously hardcoded `line-height:1.3` and
10// `8/12` padding, so they lost footnote-tightness and per-element padding
11// overrides vs the editor — keeping one copy stops that WYSIWYG drift.
12 
13export function textPresetBoxCss(preset) {
14 if (preset === 'footnote') return { lineHeight: 1, padY: 0, padX: 0 };
15 return { lineHeight: 1.3, padY: 8, padX: 12 };
16}
17 
18/** Effective inner padding as a CSS px shorthand ("8px 12px" or per-side), honoring
19 * the element's `padding` override else the preset default. */
20export function textPaddingCss(el, preset) {
21 const p = el && el.padding;
22 if (p) return `${p.top}px ${p.right}px ${p.bottom}px ${p.left}px`;
23 const box = textPresetBoxCss(preset);
24 return `${box.padY}px ${box.padX}px`;
25}
