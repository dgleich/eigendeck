42/** Check if a theme is "dark" (for UI decisions like icon colors) */
43export function isDarkTheme(theme: ThemeColors): boolean {
44 // Simple luminance check on background
45 const hex = theme.background.replace('#', '');
46 const r = parseInt(hex.slice(0, 2), 16);
47 const g = parseInt(hex.slice(2, 4), 16);
48 const b = parseInt(hex.slice(4, 6), 16);
49 return (r * 299 + g * 587 + b * 114) / 1000 < 128;
50}
