1// Background auto-discovery of available kernels on every registered
2// Jupyter server. Runs once on app start (so the topbar pill shows
3// fresh status without the user having to click "Test connection"
4// or "Refresh all" in Settings) and one-shot whenever a new server
5// is saved (handled inline in SettingsModal's row).
6//
7// We DON'T poll continuously. Doing so would write to the user's
8// terminal logs (each Jupyter request is a line) and add network
9// noise during talks. The presenter can manually refresh from the
10// pill dropdown when they want fresh status.
11//
12// Result of a successful ping: updates the entry's availableKernels
13// + lastSeenAt and writes the registry back. A failed ping leaves
14// the entry alone — the pill's staleness derives from lastSeenAt,
15// so users can see "haven't been able to reach this in a while."
16

21/** Probe one server. On success, returns the updated entry (with
22 * fresh availableKernels + lastSeenAt). On failure, returns the
23 * original entry untouched. Never throws. */
24async function probeServer(s: JupyterServerEntry): Promise<JupyterServerEntry> {
25 try {
26 const url = s.baseUrl.replace(/\/$/, '');
27 const q = s.token ? `?token=${encodeURIComponent(s.token)}` : '';
28 const r = await fetch(`${url}/api/kernelspecs${q}`, {
29 headers: s.token ? { Authorization: `token ${s.token}` } : {},
30 });
31 if (!r.ok) return s;
32 const data = await r.json();
33 const kernels = Object.keys(data.kernelspecs ?? {});
34 return { ...s, availableKernels: kernels, lastSeenAt: Date.now() };
35 } catch {
36 return s;
37 }
38}

40/** Probe every registered server in parallel, then write the new
41 * registry. Silent — no UI feedback. The topbar pill picks up the
42 * change via its usePreference subscription. */
43export async function discoverAllServers(): Promise<void> {
44 const servers = getPreference('jupyterServers');
45 if (servers.length === 0) return;
46 const updated = await Promise.all(servers.map(probeServer));
47 // Only write if anything actually changed — avoids a no-op
48 // pref-changed event that triggers re-renders for nothing.
49 const changed = updated.some((u, i) =>
50 u.lastSeenAt !== servers[i].lastSeenAt
51 || JSON.stringify(u.availableKernels) !== JSON.stringify(servers[i].availableKernels)
52 );
53 if (changed) setPreference('jupyterServers', updated);
54}
