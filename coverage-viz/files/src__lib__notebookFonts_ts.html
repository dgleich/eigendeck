1// Resolve the font packages used to render a notebook element.
2//
3// Two roles:
4// - 'prose': markdown cells, plain-text outputs. Inherits the
5// slide/presentation body font (default 'ptsans').
6// - 'code': code cell source + stream/error outputs. Falls back
7// to PresentationConfig.defaultMonoFont, then 'source-code'.
8//
9// Default-setting cascade per DESIGN_DECISIONS.md "Preferences cascade".
10// Prose font follows the EXISTING body-font cascade (slide → deck →
11// 'ptsans') so a notebook on a styled slide picks up the slide's
12// typography without per-element wiring.

40/** List the font packages eligible to be used as the deck-level
41 * monospace default. Reads from MONO_FONT_PACKAGES — its own
42 * registry, distinct from the text font list. Source Code Pro
43 * sorts first as the canonical default; other mono fonts (when
44 * added to MONO_FONT_PACKAGES) follow alphabetically. */
45export function listMonoEligible(): MonoFontPackage[] {
46 return [...MONO_FONT_PACKAGES].sort((a, b) => {
47 if (a.id === DEFAULT_MONO_ID) return -1;
48 if (b.id === DEFAULT_MONO_ID) return 1;
49 return a.label.localeCompare(b.label);
50 });
