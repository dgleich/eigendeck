56function addEntry(level: LogEntry['level'], args: any[]) {
57 const message = args.map((a) =>
58 typeof a === 'object' ? JSON.stringify(a, null, 2) : String(a)
59 ).join(' ');
60 const time = new Date().toLocaleTimeString();
61 const entry: LogEntry = { level, message, time };
62 globalLogs.push(entry);
63 if (globalLogs.length > MAX_LOG_ENTRIES) {
64 globalLogs.splice(0, globalLogs.length - MAX_LOG_ENTRIES);
65 }
66 if (AUTO_WRITE) {
67 pendingWrite.push(entry);
68 scheduleFlush();
69 }
70 listeners.forEach((fn) => fn());

73function interceptConsole() {
74 if (intercepted) return;
75 intercepted = true;
76 const origLog = console.log;
77 const origWarn = console.warn;
78 const origError = console.error;
79 console.log = (...args) => { origLog.apply(console, args); addEntry('log', args); };
80 console.warn = (...args) => { origWarn.apply(console, args); addEntry('warn', args); };
81 console.error = (...args) => { origError.apply(console, args); addEntry('error', args); };
82 window.addEventListener('error', (e) => addEntry('error', [`Unhandled: ${e.message} at ${e.filename}:${e.lineno}`]));
83 window.addEventListener('unhandledrejection', (e) => addEntry('error', [`Unhandled rejection: ${e.reason}`]));
84}
