1// Pure helpers for the editor alignment grid (snap-to-grid).
2//
3// Kept free of any store/preference reads so the rounding is trivially
4// unit-testable. The component layer (DraggableBox) supplies the live
5// spacing + on/off + bypass state and calls snapToGrid().
6 
7/**
8 * Round a slide-space coordinate to the nearest multiple of `spacing`.
9 *
10 * A spacing below 2px (or non-finite) is treated as "no grid" and returns
11 * the value unchanged — guards against a 0/1px grid snapping everything to
12 * the origin or to every pixel.
13 */
14export function snapToGrid(value: number, spacing: number): number {
15 if (!Number.isFinite(spacing) || spacing < 2) return value;
16 return Math.round(value / spacing) * spacing;
17}

19/**
20 * New size for a bottom-right (SE) resize that snaps the moving EDGE to the grid
21 * rather than the size itself (#97).
22 *
23 * Snapping width/height directly leaves the far edge off-grid whenever the
24 * element's `origin` (x or y) isn't grid-aligned — e.g. origin 130, width
25 * snapped to 200 → right edge 330, not on a 30px grid. Instead we snap the edge
26 * (`origin + rawSize`) and derive the size back from the fixed origin, so the
27 * edge lands on a gridline and only the size absorbs the origin's offset.
28 *
29 * `spacing < 2` (or non-finite) means "no grid" — pass 0 when snap is off or
30 * bypassed and the raw size is returned (clamped/rounded). Result is clamped to
31 * `minSize` and rounded to whole px.
32 */
33export function resizeEdgeToGrid(
34 origin: number,
35 rawSize: number,
36 spacing: number,
37 minSize: number,
38): number {
39 const edge = snapToGrid(origin + rawSize, spacing);
40 return Math.max(minSize, Math.round(edge - origin));
41}

43// Grid overlay marker color (light slate) — every tier (dots, thin/thick "+"
44// crosses, the dead-center "+") uses this SAME light grey; they're told apart by
45// shape and stroke weight, never by colour, so the grid stays unobtrusive.
46const MARK = 'rgba(100,116,139,0.55)';
47 
48// A single "+" SVG path centered at (x,y) with arm half-length a.
49const crossPath = (x: number, y: number, a: number) =>
50 `M${x} ${y - a}V${y + a}M${x - a} ${y}H${x + a}`;
51 
52/**
53 * The complete editor alignment-grid overlay (#89) as ONE inline-SVG markup
54 * string, sized to the whole slide (width × height). Three tiers of marker, all
55 * in the SAME light grey — distinguished by SHAPE/WEIGHT, not colour:
56 *
57 * 1. a FINE dot at every grid point (0, g, 2g…);
58 * 2. a thin "+" CROSS every 4th cell (offset 2 cells in → on the grid points
59 * 2g, 6g, 10g…), a coarse reference that reads at a glance;
60 * 3. on a FINE grid only (spacing < 30, e.g. 12/15/20/24, where the every-4th
61 * crosses get dense), a slightly THICKER "+" every 16th cell (every 4th
62 * cross), so you can count the "big chunks" without counting every cross;
63 *
64 * plus a single, larger "+" marking the DEAD CENTER of the slide.
65 *
66 * Why one full-slide SVG and not a repeating CSS background tile + a separate
67 * center element: the center mark kept drifting ~1px off the dot grid on
68 * Retina/fractional-scale (#89). A CSS background and an absolutely-positioned
69 * element rasterize in DIFFERENT compositing layers, each rounded to device
70 * pixels independently under the canvas' `transform: scale()`. Drawing the
71 * repeating tiers (via a <pattern>) AND the center cross in ONE <svg> puts them
72 * in one user-space and one raster, so nothing can drift relative to anything
73 * else.
74 *
75 * Dots are drawn only at the pattern tile's leading/interior corners ([0, span)
76 * in each axis); the <pattern> clips overflow and the tiling reproduces the
77 * trailing-edge dots, yielding one full dot at every grid line with no
78 * double-painting. No shape-rendering='crispEdges' on the crosses — at a fine
79 * spacing the canvas is downscaled enough that crisp-snapping a 1px stroke drops
80 * a whole arm (a "+" becomes "—" or "|"); plain anti-aliasing keeps both arms.
81 * Pure → unit-tested.
82 */
83export function gridOverlaySvg(spacing: number, width: number, height: number): string {
84 const g = spacing;
85 const fine = g < 30; // fine grid → add the every-16th "big chunk" cross
86 const span = fine ? 16 : 4; // cells per pattern tile (holds the every-16 mark)
87 const tile = span * g;
88 const r = 1.5; // fine-dot radius (slide px) — a crisp disc reads bold
89 // Crosses live at 2g, 6g, 10g… (every 4th line, offset 2). The one at (2g,2g)
90 // is the every-16th anchor → thick on a fine grid; the rest stay thin.
91 let thin = '', thick = '';
92 for (let x = 2 * g; x < tile; x += 4 * g) {
93 for (let y = 2 * g; y < tile; y += 4 * g) {
94 if (fine && x === 2 * g && y === 2 * g) thick += crossPath(x, y, 8);
95 else thin += crossPath(x, y, 6);
96 }
97 }
98 let dots = '';
99 for (let i = 0; i < span; i++) for (let j = 0; j < span; j++) {
100 dots += `<circle cx='${i * g}' cy='${j * g}' r='${r}'/>`;
101 }
102 // The dead-center mark: a larger "+" at the true slide center. At the default
103 // 30px spacing the center (960,540) is itself a grid dot, so it reads as a
104 // bullseye; at other spacings it just marks center.
105 const center = crossPath(width / 2, height / 2, 12);
106 return (
107 `<svg xmlns='http://www.w3.org/2000/svg' width='${width}' height='${height}' ` +
108 `viewBox='0 0 ${width} ${height}'>` +
109 `<defs><pattern id='eigendeck-grid' width='${tile}' height='${tile}' patternUnits='userSpaceOnUse'>` +
110 `<path d='${thin}' stroke='${MARK}' stroke-width='1' fill='none'/>` +
111 (thick ? `<path d='${thick}' stroke='${MARK}' stroke-width='2' fill='none'/>` : '') +
112 `<g fill='${MARK}'>${dots}</g>` +
113 `</pattern></defs>` +
114 `<rect width='${width}' height='${height}' fill='url(#eigendeck-grid)'/>` +
115 `<path d='${center}' stroke='${MARK}' stroke-width='1.5' fill='none'/>` +
116 `</svg>`
117 );
118}
