1// Lightweight non-modal toast/snackbar system.
2//
3// Mount <ToastHost /> once at app root. Anywhere in the codebase:
4//
5// showToast({
6// message: 'Save the presentation to enable file watching.',
7// kind: 'warning',
8// action: { label: 'Save', onClick: () => saveProject() },
9// });
10//
11// Toasts auto-dismiss after `ttl` ms (default 6000; 0 = sticky).
12// Users can dismiss manually via the × button. Stacks at the bottom-
13// center of the window.
14//
15// Simple subscribe pattern (no Zustand dep) — toast lifetime is short
16// and the surface area doesn't justify integrating with the store.
17 
18export type ToastKind = 'info' | 'warning' | 'error' | 'success';
19 
20export interface ToastAction {
21 label: string;
22 onClick: () => void;
23}

25export interface Toast {
26 id: string;
27 message: string;
28 kind: ToastKind;
29 action?: ToastAction;
30}
31 
32export interface ToastOptions {
33 message: string;
34 kind?: ToastKind;
35 action?: ToastAction;
36 /** Auto-dismiss after N ms. 0 = sticky (no auto-dismiss). Default 6000. */
37 ttl?: number;
38 /** Optional dedup key. Subsequent showToast() calls with the same key
39 * replace the existing toast in place instead of stacking. */
40 key?: string;
41}
42 
43type Listener = (toasts: Toast[]) => void;
44const listeners = new Set<Listener>();
45const toasts: Toast[] = [];
46const timers = new Map<string, ReturnType<typeof setTimeout>>();
47 
48function notify(): void {
49 const snapshot = [...toasts];
50 for (const l of listeners) l(snapshot);
51}
52 
53export function subscribeToasts(l: Listener): () => void {
54 listeners.add(l);
55 l([...toasts]);
56 return () => { listeners.delete(l); };
57}
58 
59export function showToast(opts: ToastOptions): string {
60 const id = opts.key ?? crypto.randomUUID();
61 const ttl = opts.ttl ?? 6000;
62 const toast: Toast = {
63 id,
64 message: opts.message,
65 kind: opts.kind ?? 'info',
66 action: opts.action,
67 };
68 
69 // Dedup by key/id — replace in place rather than stack a duplicate.
70 const existingIdx = toasts.findIndex((t) => t.id === id);
71 if (existingIdx >= 0) {
72 toasts[existingIdx] = toast;
73 } else {
74 toasts.push(toast);
75 }
76 
77 // Reset any pending timer for this id and schedule a new dismissal.
78 const existingTimer = timers.get(id);
79 if (existingTimer) clearTimeout(existingTimer);
80 if (ttl > 0) {
81 timers.set(id, setTimeout(() => dismissToast(id), ttl));
82 }
83 
84 notify();
85 return id;
86}
87 
88export function dismissToast(id: string): void {
89 const i = toasts.findIndex((t) => t.id === id);
90 if (i < 0) return;
91 toasts.splice(i, 1);
92 const t = timers.get(id);
93 if (t) { clearTimeout(t); timers.delete(id); }
94 notify();
95}
