20/** Stable JSON canonical form: recursively sort object keys. */
21function canonicalize(v: unknown): string {
22 if (v === null || typeof v !== 'object') return JSON.stringify(v);
23 if (Array.isArray(v)) return '[' + v.map(canonicalize).join(',') + ']';
24 const obj = v as Record<string, unknown>;
25 const keys = Object.keys(obj).sort();
26 return '{' + keys.map((k) => JSON.stringify(k) + ':' + canonicalize(obj[k])).join(',') + '}';
27}

29/** Find up to N field-paths that differ between two JSON values. */
30function findDiffs(a: unknown, b: unknown, path: string, out: string[], max: number): void {
31 if (out.length >= max) return;
32 if (canonicalize(a) === canonicalize(b)) return;
33 if (a === null || b === null || typeof a !== 'object' || typeof b !== 'object') {
34 out.push(`${path}: ${JSON.stringify(a)} -> ${JSON.stringify(b)}`);
35 return;
36 }
37 if (Array.isArray(a) || Array.isArray(b)) {
38 const aa = (a as unknown[]) || [];
39 const bb = (b as unknown[]) || [];
40 if (aa.length !== bb.length) out.push(`${path}.length: ${aa.length} -> ${bb.length}`);
41 const n = Math.min(aa.length, bb.length);
42 for (let i = 0; i < n && out.length < max; i++) findDiffs(aa[i], bb[i], `${path}[${i}]`, out, max);
43 return;
44 }
45 const ao = a as Record<string, unknown>;
46 const bo = b as Record<string, unknown>;
47 const keys = new Set([...Object.keys(ao), ...Object.keys(bo)]);
48 for (const k of [...keys].sort()) {
49 if (out.length >= max) break;
50 findDiffs(ao[k], bo[k], path ? `${path}.${k}` : k, out, max);
51 }
52}

54async function roundtripOne(input: string, tempDir: string): Promise<RoundtripFileReport> {
55 const start = performance.now();
56 const tempPath = `${tempDir}/${basename(input)}`;
57 try {
58 await invoke('db_open', { path: input });
59 const jsonBefore = await invoke<string>('db_export_json');
60 const before = JSON.parse(jsonBefore);
61 
62 // Save-as a fresh copy on disk. db_save_to_file commits the current
63 // SQLite handle's content to a new path.
64 await invoke('db_save_to_file', { path: tempPath });
65 
66 // Reopen the copy and re-export.
67 await invoke('db_open', { path: tempPath });
68 const jsonAfter = await invoke<string>('db_export_json');
69 const after = JSON.parse(jsonAfter);
70 
71 const slides = before.slides || [];
72 const slidesWithConfig = slides.filter((s: Record<string, unknown>) =>
73 s.theme || s.titleFont || s.bodyFont || s.hypeFont).length;
74 const elementCount = slides.reduce(
75 (n: number, s: { elements?: unknown[] }) => n + (s.elements?.length || 0), 0);
76 
77 const diffs: string[] = [];
78 findDiffs(before, after, '', diffs, 20);
79 
80 return {
81 input, ok: diffs.length === 0,
82 error: diffs.length > 0 ? `${diffs.length} field(s) differ after round-trip` : undefined,
83 slideCount: slides.length,
84 slidesWithConfig,
85 elementCount,
86 diffs,
87 elapsedMs: performance.now() - start,
88 };
89 } catch (e) {
90 return {
91 input, ok: false, error: e instanceof Error ? e.message : String(e),
92 slideCount: 0, slidesWithConfig: 0, elementCount: 0, diffs: [],
93 elapsedMs: performance.now() - start,
94 };
95 }
96}

98export async function runBatchRoundtrip(): Promise<void> {
99 try {
100 const picked = await pickDirectoryWithEigendecks('Pick a directory of .eigendeck files to round-trip');
101 if (!picked) return;
102 const { dir, files } = picked;
103 if (files.length === 0) {
104 await showBatchError('batch-roundtrip', 'No .eigendeck files in the chosen directory.');
105 return;
106 }
107 const originalPath = usePresentationStore.getState().projectPath;
108 const tempDir = `${dir}/_debug-roundtrip`;
109 try { await mkdirNative(tempDir); } catch { /* exists */ }
110 
111 const startWall = performance.now();
112 const reports: RoundtripFileReport[] = [];
113 for (const f of files) reports.push(await roundtripOne(f, tempDir));
114 const elapsedSeconds = (performance.now() - startWall) / 1000;
115 
116 const meta: RunMeta = {
117 action: 'batch-roundtrip',
118 startedAt: new Date().toISOString(),
119 directory: dir,
120 totalFiles: files.length,
121 elapsedSeconds,
122 passed: reports.filter((r) => r.ok).length,
123 failed: reports.filter((r) => !r.ok).length,
124 };
125 const report: RoundtripReport = { meta, files: reports };
126 await writeReportAndAlert(dir, meta, report);
127 
128 if (originalPath) {
129 try { await openSqliteProject(originalPath); } catch { /* manual reopen */ }
130 } else {
131 try { await invoke('db_close'); } catch { /* nothing open */ }
132 }
133 } catch (e) {
134 await showBatchError('batch-roundtrip', e);
135 }
