13/**
14 * The canonical pref schema. Every pref the app reads goes here so
15 * defaults are centralized and TypeScript catches typos.
16 */
17export interface PrefSchema {
18 /** Master switch for the asset file watcher. When false, NO asset is
19 * auto-reloaded on disk change unless the asset itself has
20 * auto_reload='on' (explicit per-asset opt-in overrides the global
21 * off). When true, assets are auto-reloaded UNLESS auto_reload='off'.
22 * See effectiveAutoReload() for the resolution. */
23 autoReloadAssets: boolean;
24 /** Default LaTeX preamble used as a template for new presentations and
25 * as the source for the per-presentation preamble's "Insert global" /
26 * "Replace with global" buttons. Render path only uses the
27 * per-presentation preamble; this is editing-only. */
28 mathPreamble: string;
29 /** Default deck-level type scale overrides for NEW presentations.
30 * When a new deck is created, its PresentationConfig.textSizes is
31 * seeded from this. Any keys absent here fall back to
32 * DEFAULT_TEXT_SIZES at render time. Empty object = use built-in
33 * defaults across the board. Existing decks are NOT affected — they
34 * keep whatever textSizes they were saved with. */
35 textSizes: Partial<Record<'footnote' | 'note' | 'body' | 'title' | 'hype', number>>;
36 /** Per-machine registry of Jupyter kernel servers. Notebook elements
37 * store ONLY the kernel name they need (`python3`, `julia-1.10`,
38 * ...) — the resolver finds the first registry server whose
39 * `availableKernels` contains the requested name and dials its
40 * baseUrl with its token. Tokens stay on this machine; decks never
41 * carry them. */
42 jupyterServers: JupyterServerEntry[];
43 /** Default editability for notebook elements. When a notebook's
44 * own `editable` field is unset, this is the fallback. Default
45 * false — the common "canned demo" case is read-only. A presenter
46 * who lives in live-typing mode can flip this so new notebooks
47 * start editable. (Per-element override still wins.) Turning a
48 * notebook editable disables file-watching for its asset — see
49 * NotebookElement.editable. */
50 defaultNotebookEditable: boolean;
51 /** Show the explanatory help text under inspector controls (the grey
52 * paragraphs that explain what a toggle does). On by default; experienced
53 * users can turn it off for a denser inspector. */
54 showHelpText: boolean;
55 /** Whether the inspector (properties panel) is open. On by default — there's a
56 * lot of essential functionality there. Persisted so ⌘I / the View toggle is
57 * remembered across launches (the store seeds `showProperties` from this and
58 * writes it back on toggle). */
59 showInspector: boolean;
60 /** Spacing (in slide-space px) of the editor alignment grid. The single
61 * global knob for snap-to-grid; whether snapping/the overlay are ON is
62 * per-session UI state (View menu), not persisted. Default 40. */
63 gridSpacing: number;
64 /** Insert-action ids hidden from the editor toolbar (see
65 * src/lib/insertItems.ts for the id list). A *hidden* list, not a
66 * visible one, so element types added in future releases show up on
67 * the toolbar by default. The native "Insert" menu ignores this — it
68 * always lists every action. Default = DEFAULT_HIDDEN_TOOLBAR_ITEMS
69 * (currently ['demo'] — a power-user action re-addable in Settings). */
70 hiddenToolbarItems: string[];
71 /** Whether Present Mode (F5) attempts the dual-monitor "projector mode":
72 * detect a second display and open the live slide there fullscreen with
73 * the speaker view on the main screen. When false, Present Mode always
74 * presents single-window on the current screen (no second-display
75 * detection). Default true. The "Present in This Window" and "Screen
76 * Share Presentation" menu items are explicit and ignore this. */
77 tryProjectorMode: boolean;
78 /** Master switch: may embedded demos use the internet at all. ON by default
79 * (demos are sandboxed web widgets — they can't touch your files; letting them
80 * reach the network is like a browser tab). When OFF, this TRUMPS everything —
81 * no demo in any deck can reach the network, regardless of the per-deck setting.
82 * (Enforcement — a CSP connect-src lockdown + WebRTC neuter on the demo docs —
83 * is wired separately; this is the setting it reads.) */
84 demoInternetAccess: boolean;
85 /** macOS only (native-toolbar build): compact toolbar view — hides the button
86 * labels and shrinks the icons to reclaim vertical space. Default false
87 * (labels shown). Ignored on non-mac / HTML-toolbar builds. Applied live via
88 * the set_toolbar_compact command; no restart needed. */
89 compactToolbar: boolean;
90}

126const KEY_PREFIX = 'eigendeck:pref:';
127const CHANGE_EVENT = 'eigendeck:pref-changed';
128 
129export function getPreference<K extends keyof PrefSchema>(key: K): PrefSchema[K] {
130 try {
131 const v = localStorage.getItem(KEY_PREFIX + key);
132 if (v === null) return DEFAULTS[key];
133 return JSON.parse(v) as PrefSchema[K];
134 } catch {
135 return DEFAULTS[key];
136 }
137}

139// Cross-window Tauri event: localStorage is shared across same-origin webview
140// windows, but the DOM CHANGE_EVENT is per-document — so a pref written in the
141// Settings window won't notify usePreference() in the main window without this
142// bridge. setPreference emits it; initPrefSync() (runtime.ts) re-dispatches the
143// local CHANGE_EVENT in every window. Guarded so non-Tauri contexts (tests) no-op.
144export const PREF_SYNC_EVENT = 'eigendeck:pref-sync';
145 
146export function setPreference<K extends keyof PrefSchema>(key: K, value: PrefSchema[K]): void {
147 try {
148 localStorage.setItem(KEY_PREFIX + key, JSON.stringify(value));
149 window.dispatchEvent(new CustomEvent(CHANGE_EVENT, { detail: { key } }));
150 // Notify OTHER webview windows (best-effort; no-op outside Tauri). Tag the
151 // origin window so it can ignore its own echo — Tauri's emit() delivers back
152 // to the sender too, and we already dispatched CHANGE_EVENT locally above.
153 void import('@tauri-apps/api/window')
154 .then(({ getCurrentWindow }) => {
155 const origin = getCurrentWindow().label;
156 return import('@tauri-apps/api/event').then(({ emit }) => emit(PREF_SYNC_EVENT, { key, origin }));
157 })
158 .catch(() => {});
159 } catch (e) {
160 console.warn(`[prefs] setPreference(${key}) failed:`, e);
161 }
162}

164/** Re-dispatch the local CHANGE_EVENT when ANOTHER window changes a pref.
165 * Called once per window from initRuntime(). Guarded for non-Tauri contexts. */
166export function initPrefSync(): void {
167 void import('@tauri-apps/api/window')
168 .then(({ getCurrentWindow }) => {
169 const self = getCurrentWindow().label;
170 return import('@tauri-apps/api/event').then(({ listen }) =>
171 listen<{ key?: string; origin?: string }>(PREF_SYNC_EVENT, (e) => {
172 // Ignore our own echo — setPreference already dispatched CHANGE_EVENT here.
173 if (e.payload?.origin === self) return;
174 window.dispatchEvent(new CustomEvent(CHANGE_EVENT, { detail: { key: e.payload?.key } }));
175 }));
176 })
177 .catch(() => {});
178}

180/**
181 * Reactive read of a preference. Re-renders the component when the
182 * preference changes (via setPreference from anywhere in the app).
183 * Returns [value, setter] like useState.
184 */
185export function usePreference<K extends keyof PrefSchema>(
186 key: K,
187): [PrefSchema[K], (v: PrefSchema[K]) => void] {
188 const [value, setValue] = useState<PrefSchema[K]>(() => getPreference(key));
189 useEffect(() => {
190 const handler = (e: Event) => {
191 const detail = (e as CustomEvent).detail as { key?: string } | undefined;
192 if (detail?.key === key) setValue(getPreference(key));
193 };
194 window.addEventListener(CHANGE_EVENT, handler);
195 return () => window.removeEventListener(CHANGE_EVENT, handler);
196 }, [key]);
197 const setter = useCallback((v: PrefSchema[K]) => setPreference(key, v), [key]);
198 return [value, setter];
199}

201/**
202 * Resolve the effective auto_reload state for an asset.
203 *
204 * Cascade is **downward-only**: any layer can refuse, no layer overrides a
205 * refusal above it. Pass null/undefined for layers without an explicit
206 * opt-out. For "would a NEW asset in this presentation auto-reload by
207 * default", pass null for perAsset.
208 *
209 * global must be true (else: off everywhere)
210 * per-presentation must not be 'off' (else: off in this deck)
211 * per-asset must not be 'off' (else: off for this asset)
212 *
213 * Per-asset 'on' is NOT a thing — an asset can opt out but can't opt in
214 * beyond what the presentation/global allow. Legacy 'on' values in the
215 * database (from earlier 3-state UI) are treated as if NULL.
216 */
217export function effectiveAutoReload(
218 perAsset: string | null | undefined,
219 perPresentation: string | null | undefined,
220 globalDefault: boolean,
221): boolean {
222 return globalDefault
223 && perPresentation !== 'off'
224 && perAsset !== 'off';
