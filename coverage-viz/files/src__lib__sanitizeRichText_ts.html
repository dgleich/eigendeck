1// Strict sanitizer for text-element `html` (#: rich-text validation).
2//
3// Policy (decided): a text box may only contain what the format toolbar can
4// author. Anything else — hand-written JSON, pasted foreign markup, a crafted
5// `.eigendeck` — is reduced to that allowlist on every ingest (deck open, paste,
6// edit commit). This is BOTH a safety fix (text html is rendered via
7// dangerouslySetInnerHTML in a privileged Tauri webview, so `onerror=`/`onload=`
8// handlers and `javascript:` URLs would otherwise execute when a shared deck is
9// opened) AND an authoring-consistency guarantee (no styles you can't reproduce
10// or edit in the UI — notably no inline font-size or background, which the
11// toolbar doesn't offer).
12//
13// The allowlist mirrors TextFormatToolbar.tsx:
14// bold/italic/strike (b/strong/i/em/s/strike or font-weight/style + text-
15// decoration), foreColor (color, or <font color>), uppercase+tracking
16// (text-transform + letter-spacing), bullet list (ul/ol/li), justify
17// (text-align), plus the structural span/div/br/p the editor emits on edit.
18// Truly arbitrary HTML/JS belongs in the (separate) sandboxed HTML widget.
19 
20const ALLOWED_TAGS = new Set([
21 'b', 'strong', 'i', 'em', 'u', 's', 'strike', 'span', 'div', 'p', 'br', 'ul', 'ol', 'li', 'font', 'code',
22]);

24// Tags whose ENTIRE subtree is dropped (never just unwrapped) — scripts, embeds,
25// media and form controls have no place in a text box and are the dangerous set.
26const DANGEROUS_TAGS = new Set([
27 'script', 'style', 'iframe', 'object', 'embed', 'img', 'svg', 'math',
28 'link', 'meta', 'base', 'input', 'textarea', 'button', 'select', 'option', 'form', 'audio', 'video', 'canvas',
29]);

31// CSS properties the toolbar can produce. Everything else (font-size, background,
32// margin, padding, border*, font-family, line-height, position, …) is dropped.
33const ALLOWED_STYLE_PROPS = new Set([
34 'color', 'font-weight', 'font-style', 'text-decoration', 'text-decoration-line', 'text-transform', 'letter-spacing', 'text-align',
35]);

37const UNSAFE_VALUE = /url\s*\(|expression\s*\(|javascript:|@import/i;

39/** Filter an inline style string to the allowed presentational props. */
40function cleanStyle(style: string): string {
41 const out: string[] = [];
42 for (const decl of style.split(';')) {
43 const i = decl.indexOf(':');
44 if (i < 0) continue;
45 const prop = decl.slice(0, i).trim().toLowerCase();
46 const value = decl.slice(i + 1).trim();
47 if (!ALLOWED_STYLE_PROPS.has(prop)) continue;
48 if (!value || UNSAFE_VALUE.test(value)) continue;
49 out.push(`${prop}: ${value}`);
50 }
51 return out.join('; ');
52}

54/** Recursively copy only allowed nodes/attrs from `src` into `dst`. */
55function cleanInto(src: Node, dst: Node, doc: Document): void {
56 for (const child of Array.from(src.childNodes)) {
57 if (child.nodeType === Node.TEXT_NODE) {
58 // Normalize non-breaking spaces to regular, breakable spaces. WebKit's
59 // contentEditable inserts &nbsp; as you type; in edit mode it renders them
60 // breakable (-webkit-nbsp-mode:space), but every OUTPUT path (SVG
61 // foreignObject / HTML export / PDF) honors them as non-breaking and wraps
62 // raggedly — a WYSIWYG divergence (#159). Since these are editor artifacts,
63 // not intentional non-breaking spaces, fold them here on every ingest.
64 dst.appendChild(doc.createTextNode((child.nodeValue || '').replace(/ /g, ' ')));
65 continue;
66 }
67 if (child.nodeType !== Node.ELEMENT_NODE) continue; // drop comments etc.
68 const el = child as Element;
69 const tag = el.tagName.toLowerCase();
70 if (DANGEROUS_TAGS.has(tag)) continue; // drop subtree entirely
71 if (!ALLOWED_TAGS.has(tag)) {
72 cleanInto(el, dst, doc); // unwrap: keep children, drop the tag
73 continue;
74 }
75 const clean = doc.createElement(tag);
76 const style = cleanStyle(el.getAttribute('style') || '');
77 if (style) clean.setAttribute('style', style);
78 if (tag === 'font') {
79 const color = el.getAttribute('color');
80 if (color && !UNSAFE_VALUE.test(color)) clean.setAttribute('color', color);
81 }
82 cleanInto(el, clean, doc);
83 dst.appendChild(clean);
84 }
85}

87/**
88 * Reduce a text element's `html` to the toolbar allowlist. Idempotent; returns
89 * '' for empty/whitespace input. Requires a DOM (browser or jsdom).
90 *
91 * NOTE: pass the RAW authored html (with `$…$` / `$$…$$` math as plain text) —
92 * sanitize BEFORE math is rendered to SVG, since the rendered <svg> is trusted,
93 * app-generated markup that this allowlist would otherwise strip.
94 */
95export function sanitizeRichText(html: string | undefined | null): string {
96 if (!html) return '';
97 const doc = new DOMParser().parseFromString(`<div id="r">${html}</div>`, 'text/html');
98 const root = doc.getElementById('r');
99 if (!root) return '';
100 const out = doc.createElement('div');
101 cleanInto(root, out, doc);
102 return out.innerHTML;
103}

105/**
106 * Sanitize every text element's html in a presentation IN PLACE (like the
107 * notebook-token migration). Returns true if anything changed. Call on deck
108 * load/import so JSON-authored or foreign decks are normalized to the allowlist.
109 */
110export function sanitizePresentationHtml(presentation: {
111 slides?: Array<{ elements?: Array<{ type?: string; html?: string }> }>;
112}): boolean {
113 let changed = false;
114 for (const slide of presentation.slides || []) {
115 for (const el of slide.elements || []) {
116 if (el.type !== 'text' || typeof el.html !== 'string') continue;
117 const clean = sanitizeRichText(el.html);
118 if (clean !== el.html) { el.html = clean; changed = true; }
119 }
120 }
121 return changed;
122}
