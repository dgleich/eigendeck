16// A real YouTube video id is exactly 11 chars from [A-Za-z0-9_-]. We validate to
17// this canonical shape so a shared/opened deck can't smuggle an arbitrary string
18// (raw `?v=` was previously taken unchecked) into the embed URL — and, once the
19// packaged-app loopback shim interpolates the id into HTML, into an injection sink.
20// See docs/youtube-embed-shim.md. A non-conforming id yields null (no embed).
21const YOUTUBE_ID = /^[A-Za-z0-9_-]{11}$/;
22 
23export function detectVideoProvider(raw) {
24 let u;
25 try { u = new URL(String(raw).trim()); } catch { return null; }
26 const host = u.hostname.replace(/^www\./, '');
27 
28 if (host === 'youtube.com' || host === 'm.youtube.com' || host === 'youtube-nocookie.com') {
29 const v = u.searchParams.get('v');
30 if (v) return YOUTUBE_ID.test(v) ? { provider: 'youtube', id: v } : null;
31 const m = u.pathname.match(/^\/(?:embed|shorts|live)\/([^/?#]+)/);
32 if (m) return YOUTUBE_ID.test(m[1]) ? { provider: 'youtube', id: m[1] } : null;
33 }
34 if (host === 'youtu.be') {
35 const id = u.pathname.slice(1).split('/')[0];
36 if (id) return YOUTUBE_ID.test(id) ? { provider: 'youtube', id } : null;
37 }
38 if (host === 'vimeo.com' || host === 'player.vimeo.com') {
39 const m = u.pathname.match(/(\d+)/);
40 if (m) return { provider: 'vimeo', id: m[1] };
41 }
42 // PeerTube (federated): /w/<id>, /videos/watch/<uuid>, /videos/embed/<uuid>.
43 const pt = u.pathname.match(/\/(?:w|videos\/(?:watch|embed))\/([\w-]+)/);
44 if (pt) return { provider: 'peertube', id: pt[1], origin: u.origin };
45 return null;
46}

48/**
49 * Build the iframe `src` for an embed-kind video, applying the element's options
50 * (loop/autoplay/muted/controls/captions as URL params; playback SPEED needs the
51 * provider JS API, not a URL param). Returns null when the URL isn't a recognized
52 * provider. `opts.jsApi` (default true) adds enablejsapi (YouTube) / api=1
53 * (PeerTube) so postMessage can drive playback rate — the live app wants it; the
54 * static HTML export passes `{ jsApi: false }` (no JS to drive it).
55 *
56 * showControls: hiding controls is only safe when autoplay starts the video —
57 * with autoplay off, controls=0 leaves no way to play (and on PeerTube/video.js
58 * also kills click-to-play). So controls show unless autoplay is on.
59 */
60export function buildEmbedSrc(el, opts = {}) {
61 const jsApi = opts.jsApi !== false;
62 if (!el || !el.url) return null;
63 const parsed = detectVideoProvider(el.url);
64 if (!parsed) return null;
65 const { provider, id, origin } = parsed;
66 const p = new URLSearchParams();
67 const showControls = !!el.controls || !el.autoplay;
68 
69 if (provider === 'youtube') {
70 if (el.autoplay) p.set('autoplay', '1');
71 if (el.muted) p.set('mute', '1');
72 if (el.loop) { p.set('loop', '1'); p.set('playlist', id); } // single-video loop needs playlist=id
73 p.set('controls', showControls ? '1' : '0');
74 if (el.captions) p.set('cc_load_policy', '1');
75 if (jsApi) p.set('enablejsapi', '1'); // postMessage control (setPlaybackRate)
76 p.set('rel', '0');
77 return `https://www.youtube-nocookie.com/embed/${id}?${p.toString()}`;
78 }
79 if (provider === 'vimeo') {
80 if (el.autoplay) p.set('autoplay', '1');
81 if (el.muted) p.set('muted', '1');
82 if (el.loop) p.set('loop', '1');
83 if (!showControls) p.set('controls', '0');
84 if (el.captions) p.set('texttrack', 'en');
85 return `https://player.vimeo.com/video/${id}?${p.toString()}`;
86 }
87 // PeerTube
88 const base = origin ?? (() => { try { return new URL(el.url).origin; } catch { return ''; } })();
89 if (!base) return null;
90 if (el.autoplay) p.set('autoplay', '1');
91 if (el.muted) p.set('muted', '1');
92 if (el.loop) p.set('loop', '1');
93 if (!showControls) p.set('controls', '0');
94 if (el.captions) p.set('subtitle', 'en');
95 if (jsApi) p.set('api', '1'); // PeerTube PlayerAPI (postMessage setPlaybackRate)
96 return `${base}/videos/embed/${id}?${p.toString()}`;
97}
