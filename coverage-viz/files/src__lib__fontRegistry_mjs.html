307/**
308 * Resolve which font package applies to a given preset on a slide.
309 * Priority: slide override > presentation default > 'ptsans'.
310 * - 'title' preset uses the title font
311 * - 'hype' preset uses the hype font
312 * - everything else uses the body font
313 */
314export function fontForPreset(preset, slide, presentationDefaults) {
315 const role = preset === 'title' ? 'title'
316 : preset === 'hype' ? 'hype'
317 : 'body';
318 const slideKey = role === 'title' ? 'titleFont' : role === 'hype' ? 'hypeFont' : 'bodyFont';
319 const presKey = role === 'title' ? 'defaultTitleFont' : role === 'hype' ? 'defaultHypeFont' : 'defaultBodyFont';
320 // Hype (sticky note) defaults to Shantell Sans when nothing overrides it.
321 const fallback = role === 'hype' ? HYPE_DEFAULT_FONT_ID : undefined;
322 return resolveFontPackage((slide && slide[slideKey]) ?? (presentationDefaults && presentationDefaults[presKey]) ?? fallback);
323}

347/**
348 * Generate @font-face declarations for a single package.
349 * URLs reference /fonts/<id>/<filename>.
350 */
351export function fontFaceCSSForPackage(pkg) {
352 const name = bareFamily(pkg.family);
353 const dir = `/fonts/${pkg.id}`;
354 const lines = [];
355 
356 if (pkg.files.kind === 'variable') {
357 const f = pkg.files;
358 const [w0, w1] = f.weightRange;
359 const fmt = "format('truetype-variations'), format('truetype')";
360 lines.push(
361 `@font-face { font-family: '${name}'; src: url('${dir}/${f.upright}') ${fmt}; ` +
362 `font-weight: ${w0} ${w1}; font-style: normal; font-display: swap; }`
363 );
364 if (f.italic) {
365 lines.push(
366 `@font-face { font-family: '${name}'; src: url('${dir}/${f.italic}') ${fmt}; ` +
367 `font-weight: ${w0} ${w1}; font-style: italic; font-display: swap; }`
368 );
369 }
370 } else {
371 const f = pkg.files;
372 const fmt = f.ext === 'otf' ? "format('opentype')" : "format('truetype')";
373 const decl = (file, weight, style) =>
374 `@font-face { font-family: '${name}'; src: url('${dir}/${file}') ${fmt}; ` +
375 `font-weight: ${weight}; font-style: ${style}; font-display: swap; }`;
376 
377 lines.push(decl(f.regular, 400, 'normal'));
378 if (f.bold) lines.push(decl(f.bold, 700, 'normal'));
379 if (f.italic) lines.push(decl(f.italic, 400, 'italic'));
380 if (f.boldItalic) lines.push(decl(f.boldItalic, 700, 'italic'));
381 
382 if (pkg.narrowFamily && f.narrowRegular) {
383 const narrowName = bareFamily(pkg.narrowFamily);
384 lines.push(
385 `@font-face { font-family: '${narrowName}'; src: url('${dir}/${f.narrowRegular}') ${fmt}; ` +
386 `font-weight: 400; font-style: normal; font-display: swap; }`
387 );
388 if (f.narrowBold) {
389 lines.push(
390 `@font-face { font-family: '${narrowName}'; src: url('${dir}/${f.narrowBold}') ${fmt}; ` +
391 `font-weight: 700; font-style: normal; font-display: swap; }`
392 );
393 }
394 }
395 }
396 return lines.join('\n');
397}

399/** Generate the full @font-face block for all packages — text font

409/**
410 * List all font package ids actually used in a presentation
411 * (presentation defaults + per-slide overrides + the DEFAULT_FONT_ID fallback).
412 */
413export function collectUsedFontIds(presentation) {
414 const ids = new Set([DEFAULT_FONT_ID]);
415 const cfg = presentation.config || {};
416 if (cfg.defaultTitleFont) ids.add(cfg.defaultTitleFont);
417 if (cfg.defaultBodyFont) ids.add(cfg.defaultBodyFont);
418 if (cfg.defaultHypeFont) ids.add(cfg.defaultHypeFont);
419 if (cfg.defaultMonoFont) ids.add(cfg.defaultMonoFont); // <code> runs / notebooks
420 for (const s of presentation.slides || []) {
421 if (s.titleFont) ids.add(s.titleFont);
422 if (s.bodyFont) ids.add(s.bodyFont);
423 if (s.hypeFont) ids.add(s.hypeFont);
424 }
425 // keep ids that resolve in EITHER the main or the mono registry
426 return [...ids].filter((id) => FONT_PACKAGE_MAP[id] || MONO_FONT_PACKAGE_MAP[id]);
427}

429/**
430 * For a package, list the font file paths (relative to /fonts/<id>/) plus the
431 * @font-face attrs each needs.
432 */
433export function fontFilesForPackage(pkg) {
434 const out = [];
435 if (pkg.files.kind === 'variable') {
436 const f = pkg.files;
437 const [w0, w1] = f.weightRange;
438 out.push({ filename: f.upright, cssAttrs: { weight: `${w0} ${w1}`, style: 'normal' } });
439 if (f.italic) out.push({ filename: f.italic, cssAttrs: { weight: `${w0} ${w1}`, style: 'italic' } });
440 } else {
441 const f = pkg.files;
442 out.push({ filename: f.regular, cssAttrs: { weight: '400', style: 'normal' } });
443 if (f.bold) out.push({ filename: f.bold, cssAttrs: { weight: '700', style: 'normal' } });
444 if (f.italic) out.push({ filename: f.italic, cssAttrs: { weight: '400', style: 'italic' } });
445 if (f.boldItalic) out.push({ filename: f.boldItalic, cssAttrs: { weight: '700', style: 'italic' } });
446 if (f.narrowRegular) out.push({ filename: f.narrowRegular, cssAttrs: { weight: '400', style: 'normal', isNarrow: true } });
447 if (f.narrowBold) out.push({ filename: f.narrowBold, cssAttrs: { weight: '700', style: 'normal', isNarrow: true } });
448 }
449 return out;
450}

452/**
453 * Stable cache key for (tex, bundle, display, preamble) — FNV-1a. Must stay
454 * byte-identical to mathCacheKey() in src/lib/mathjaxRenderer.ts so the CLI
455 * can look up SVGs the editor's iframe pool already produced.
456 */
457export function mathCacheKey(tex, bundle, display, preamble) {
458 let h = 0x811c9dc5;
459 const s = `${bundle}\x1f${display ? 'd' : 'i'}\x1f${preamble}\x1f${tex}`;
460 for (let i = 0; i < s.length; i++) {
461 h ^= s.charCodeAt(i);
462 h = Math.imul(h, 0x01000193);
463 }
464 return (h >>> 0).toString(16).padStart(8, '0');
