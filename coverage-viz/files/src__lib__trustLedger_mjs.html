1// The trust ledger — app-side record of which decks are trusted and which resolved
2// paths are approved for watching. See docs/ASSETS-SECURITY.md ("Trust store" +
3// "State model"). This module is PURE state + logic: the current time is injected
4// (`now` ms epoch) and persistence is external, so it is fully unit-testable and
5// has no clock/DOM/fs dependency. A thin Tauri-fs layer (elsewhere) loads/saves it.
6//
7// Keyed by `deck-token` (a random id stamped into the deck at File → New; a received
8// deck's token isn't in the ledger). Per deck we keep: whether it was ever trusted,
9// the last-open time (for the TTL), and the approved *resolved* paths — keyed by the
10// linked asset's id (`{ [assetId]: resolvedPath }`).
11//
12// Why key approvals by asset, not by path: an approval belongs to a linked asset, and
13// the asset id is STABLE across relocate (only the path changes). So `approve` REPLACES
14// an asset's entry in place — a relocate never orphans the old path — and `reconcile`
15// can drop approvals for assets the deck no longer references (delete / re-link) with a
16// plain id set-diff, no filesystem access, keeping a temporarily-missing-but-still-
17// referenced asset's approval intact. The READ gate still authorizes by resolved
18// target (symlink defense): isApproved checks whether ANY asset's resolved matches.
19// See docs/ASSETS-SECURITY.md ("Ledger hygiene").
20//
21// Three nested levels live here only for the DECK/TRUST axis: whether a deck is
22// trusted and whether a path is approved. Watching on/off and the asset-type gate
23// live elsewhere (preferences + assetTypes.mjs).

25export const TRUST_TTL_DAYS = 30;
26export const TRUST_TTL_MS = TRUST_TTL_DAYS * 24 * 60 * 60 * 1000;

42/** Deck-level status derived from an entry + the current time. */
43// 'untrusted-new' — never trusted (or revoked): no retained approvals.
44// 'untrusted-ttl' — was trusted, lapsed by the TTL: approvals retained but inactive.
45// 'trusted' — trusted and within the TTL: approvals active.

47export class TrustLedger {
48 /** @param {Record<string, DeckEntry>} [entries] */
49 constructor(entries = {}) {
50 /** @type {Map<string, DeckEntry>} */
51 this.decks = new Map(Object.entries(entries).map(([k, v]) => [k, normalizeEntry(v)]));
52 }

54 /** Deck status right now: { status, approvals, lapsed }. approvals are RETURNED
55 * even when lapsed (retained-but-inactive), so callers can offer "restore". */
56 deckState(token, now) {
57 const e = this.decks.get(token);
58 if (!e || !e.trusted) return { status: 'untrusted-new', approvals: [], lapsed: false, trustedAt: null, trustReason: null };
59 const lapsed = now - e.lastOpenMs > TRUST_TTL_MS;
60 const approvals = uniq(Object.values(e.approvals).map((a) => a.resolved)); // resolved paths (deduped)
61 return {
62 status: lapsed ? 'untrusted-ttl' : 'trusted',
63 approvals,
64 lapsed,
65 trustedAt: e.trustedAt,
66 trustReason: e.trustReason,
67 };
68 }

70 /** Provenance for one approved resolved path: { at, reason } or null. Trusted-only
71 * (a lapsed deck's retained approvals don't authorize, but their provenance is still
72 * useful in the re-confirm UI, so this returns them regardless of TTL). */
73 approvalDetail(token, resolvedPath) {
74 const e = this.decks.get(token);
75 if (!e || !e.trusted) return null;
76 for (const a of Object.values(e.approvals)) {
77 if (a.resolved === resolvedPath) return { at: a.at, reason: a.reason };
78 }
79 return null;
80 }

82 /** Is this deck effectively trusted right now (trusted AND not TTL-lapsed)? */
83 isTrusted(token, now) {
84 return this.deckState(token, now).status === 'trusted';
85 }
86 
87 /** May this resolved path be read/watched now? Requires effective trust AND an
88 * ACTIVE approval. A lapsed deck's retained approvals do NOT authorize reads. */
89 isApproved(token, resolvedPath, now) {
90 const s = this.deckState(token, now);
91 return s.status === 'trusted' && s.approvals.includes(resolvedPath);
92 }

94 /** Is this deck's demo internet access blocked (per-deck local choice)?
95 * Independent of trust — settable even on an untrusted deck. */
96 isInternetBlocked(token) {
97 const e = this.decks.get(token);
98 return !!(e && e.blockInternet);
99 }

101 /** Set the per-deck internet block. Creates a minimal (untrusted) entry if the
102 * deck has none yet, so the choice persists for any deck. Mutating; caller persists. */
103 setInternetBlocked(token, blocked, now) {
104 let e = this.decks.get(token);
105 if (!e) {
106 if (!blocked) return this; // nothing to store
107 e = { trusted: false, trustedAt: 0, trustReason: 'trusted', lastOpenMs: now || 0, approvals: {}, seenEligible: [], blockInternet: true };
108 this.decks.set(token, e);
109 } else {
110 e.blockInternet = !!blocked;
111 }
112 return this;
113 }

115 /** Is THIS demo (by assetId) blocked from the internet in this deck? Per-demo
116 * local choice, layered under the deck + global switches (any one blocks).
117 * Default false = a declared demo may reach its manifest hosts. */
118 isDemoBlocked(token, assetId) {
119 const e = this.decks.get(token);
120 return !!(e && e.blockedDemos && e.blockedDemos.includes(assetId));
121 }
122 
123 /** Allow/deny one demo's internet by assetId. Creates a minimal (untrusted)
124 * entry if the deck has none. Mutating; caller persists. */
125 setDemoBlocked(token, assetId, blocked, now) {
126 let e = this.decks.get(token);
127 if (!e) {
128 if (!blocked) return this; // nothing to store
129 e = { trusted: false, trustedAt: 0, trustReason: 'trusted', lastOpenMs: now || 0, approvals: {}, seenEligible: [], blockInternet: false, blockedDemos: [] };
130 this.decks.set(token, e);
131 }
132 if (!Array.isArray(e.blockedDemos)) e.blockedDemos = [];
133 const has = e.blockedDemos.includes(assetId);
134 if (blocked && !has) e.blockedDemos.push(assetId);
135 else if (!blocked && has) e.blockedDemos = e.blockedDemos.filter((a) => a !== assetId);
136 return this;
137 }

141 /** File → New (or explicit trust): stamp a fresh trusted deck with no approvals yet.
142 * `reason` records HOW it was trusted ('file-new' by default, or 'trusted'). */
143 createTrusted(token, now, reason = 'file-new') {
144 // Preserve the viewer's per-deck / per-demo internet block across the trust
145 // transition — those are independent of trust and must NOT be reset by trusting.
146 const prev = this.decks.get(token);
147 this.decks.set(token, { trusted: true, trustedAt: now, trustReason: reason, lastOpenMs: now, approvals: {}, seenEligible: [],
148 blockInternet: !!(prev && prev.blockInternet), blockedDemos: [...((prev && prev.blockedDemos) || [])] });
149 return this;
150 }

152 /** Explicitly trust a (received) deck, approving the reviewed assets. `approvals`
153 * is an `{ assetId: resolvedPath }` map; each is stamped now + reason 'trusted'.
154 * Replaces any prior approval set. */
155 trust(token, approvals, now) {
156 const stamped = {};
157 for (const [assetId, resolved] of Object.entries(approvals || {})) {
158 if (typeof assetId === 'string' && typeof resolved === 'string') {
159 stamped[assetId] = { resolved, at: now, reason: 'trusted' };
160 }
161 }
162 const prev = this.decks.get(token);
163 this.decks.set(token, { trusted: true, trustedAt: now, trustReason: 'trusted', lastOpenMs: now, approvals: stamped, seenEligible: [],
164 blockInternet: !!(prev && prev.blockInternet), blockedDemos: [...((prev && prev.blockedDemos) || [])] });
165 return this;
166 }

168 /** Record the deck's currently-ELIGIBLE (unapproved, in-policy) resolved paths that
169 * the on-open review nudge surfaced, and report how many are NEW since last time.
170 * Lets the toast scope its wording — a trusted deck newly pointing at unreviewed
171 * content is the risk to call out ("N NEW files"), vs links you've already seen
172 * ("N files still…"). Mutating; caller persists. No-op unless effectively trusted. */
173 noteEligible(token, resolvedPaths, now) {
174 if (!this.isTrusted(token, now)) return { total: 0, newCount: 0 };
175 const e = this.decks.get(token);
176 const cur = uniq(resolvedPaths);
177 const seen = new Set(e.seenEligible || []);
178 const newCount = cur.reduce((n, p) => n + (seen.has(p) ? 0 : 1), 0);
179 e.seenEligible = cur; // remember exactly what we surfaced this open
180 return { total: cur.length, newCount };
181 }

183 /** Approve (or re-point) ONE asset's resolved target on an already-trusted deck.
184 * REPLACES any prior entry for that asset — so a relocate updates in place and the
185 * old path is dropped atomically, never orphaned. `reason` records HOW (add |
186 * relocate | relocate-folder | approve | approve-folder | trust-all). No-op (returns
187 * false) if the deck isn't effectively trusted right now. */
188 approve(token, assetId, resolvedPath, reason, now) {
189 if (!this.isTrusted(token, now)) return false;
190 if (typeof assetId !== 'string' || typeof resolvedPath !== 'string') return false;
191 this.decks.get(token).approvals[assetId] = { resolved: resolvedPath, at: now, reason: reason || 'approve' };
192 return true;
193 }
194 
195 /** Revoke ONE asset's approval (the per-file "Revoke approval" action) without
196 * touching deck trust or the other approvals. Returns true if an approval was
197 * removed. No-op unless effectively trusted. */
198 unapprove(token, assetId, now) {
199 if (!this.isTrusted(token, now)) return false;
200 const e = this.decks.get(token);
201 if (!e || !(assetId in e.approvals)) return false;
202 delete e.approvals[assetId];
203 return true;
204 }

206 /** Ledger hygiene: drop approvals for assets the deck no longer references.
207 * `keepAssetIds` = the deck's CURRENT linked asset ids. Returns the count removed.
208 * No-op unless effectively trusted — a TTL-lapsed deck's retained approvals are
209 * preserved for one-click re-confirm (the sole retain exception). A still-referenced
210 * but currently-missing asset keeps its approval (its id is still in keepAssetIds).
211 * See docs/ASSETS-SECURITY.md ("Ledger hygiene"). */
212 reconcile(token, keepAssetIds, now) {
213 if (!this.isTrusted(token, now)) return 0;
214 const keep = new Set(keepAssetIds);
215 const e = this.decks.get(token);
216 let removed = 0;
217 for (const id of Object.keys(e.approvals)) {
218 if (!keep.has(id)) { delete e.approvals[id]; removed++; }
219 }
220 return removed;
221 }
222 
223 /** Opening a trusted (non-lapsed) deck refreshes the TTL clock. A lapsed deck is
224 * NOT refreshed here (it stays untrusted-ttl until an explicit re-confirm), and
225 * an untrusted deck is a no-op. Returns the resulting status. */
226 open(token, now) {
227 const s = this.deckState(token, now);
228 if (s.status === 'trusted') this.decks.get(token).lastOpenMs = now;
229 return s.status;
230 }
231 
232 /** Re-confirm after a TTL lapse: restore the retained approvals by refreshing the
233 * clock. The caller has shown the user the retained paths; excluding any that no
234 * longer pass the gate is the caller's job (it re-checks at read anyway). */
235 reconfirm(token, now) {
236 const e = this.decks.get(token);
237 if (!e) return false;
238 e.trusted = true;
239 e.lastOpenMs = now;
240 return true;
241 }
242 
243 /** Explicit revoke: drop the deck entirely — untrust AND remove approvals. Next
244 * open is untrusted-new (re-trust = approve from scratch). */
245 revoke(token) {
246 return this.decks.delete(token);
247 }

251 /** Plain-object form for JSON persistence. */
252 serialize() {
253 /** @type {Record<string, DeckEntry>} */
254 const out = {};
255 for (const [k, v] of this.decks) {
256 const approvals = {};
257 for (const [id, a] of Object.entries(v.approvals)) approvals[id] = { resolved: a.resolved, at: a.at, reason: a.reason };
258 out[k] = { trusted: v.trusted, trustedAt: v.trustedAt, trustReason: v.trustReason, lastOpenMs: v.lastOpenMs, approvals, seenEligible: [...v.seenEligible], blockInternet: !!v.blockInternet, blockedDemos: [...(v.blockedDemos || [])] };
259 }
260 return out;
261 }

269function uniq(arr) {
270 return Array.isArray(arr) ? [...new Set(arr.filter((x) => typeof x === 'string'))] : [];
271}

273/** Coerce a persisted/handed-in approvals value to a clean `{assetId: {resolved, at,
274 * reason}}` map. Accepts the current object form; also tolerates a bare `{assetId:
275 * resolvedString}` (older shape) → stamped with at:0, reason:'unknown'. Junk dropped. */
276function normApprovals(a) {
277 const out = {};
278 if (a && typeof a === 'object' && !Array.isArray(a)) {
279 for (const [k, v] of Object.entries(a)) {
280 if (typeof k !== 'string') continue;
281 if (typeof v === 'string') {
282 out[k] = { resolved: v, at: 0, reason: 'unknown' };
283 } else if (v && typeof v === 'object' && typeof v.resolved === 'string') {
284 out[k] = {
285 resolved: v.resolved,
286 at: Number.isFinite(v.at) ? v.at : 0,
287 reason: typeof v.reason === 'string' ? v.reason : 'unknown',
288 };
289 }
290 }
291 }
292 return out;
293}

295function normalizeEntry(v) {
296 return {
297 trusted: !!(v && v.trusted),
298 trustedAt: v && Number.isFinite(v.trustedAt) ? v.trustedAt : (v && Number.isFinite(v.lastOpenMs) ? v.lastOpenMs : 0),
299 trustReason: v && typeof v.trustReason === 'string' ? v.trustReason : 'trusted',
300 lastOpenMs: v && Number.isFinite(v.lastOpenMs) ? v.lastOpenMs : 0,
301 approvals: normApprovals(v && v.approvals),
302 seenEligible: uniq(v && v.seenEligible),
303 // Per-deck: block this deck's demos from the internet (independent of trust —
304 // you can block a deck you don't otherwise trust). Default false (allowed).
305 blockInternet: !!(v && v.blockInternet),
306 // Per-demo: assetIds of demos in this deck denied internet individually
307 // (layered under the deck + global switches). Default [] (all declared allowed).
308 blockedDemos: uniq(v && v.blockedDemos),
309 };
