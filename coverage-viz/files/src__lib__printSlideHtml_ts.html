26const W = 1920, H = 1080;
27const S = 11 / 1920; // inches per pixel (11in-wide letter-landscape slide)
28const px2in = (px: number) => (px * S).toFixed(4) + 'in';
29const px2pt = (px: number) => (px * S * 72).toFixed(1) + 'pt'; // for font sizes

31// Absolute px lengths that live INSIDE a text element's rich-text inline styles
32// (letter-spacing / word-spacing on pasted or tracked runs — e.g. an uppercase
33// tracked title) don't get scaled when the print path converts the font-size to
34// points and the box to inches: only the caller-supplied font-size + box go
35// through px2pt/px2in, the inner span styles pass through verbatim. So a title
36// that fits on one line in the editor / HTML-export SVG (where EVERYTHING renders
37// in native px and scales uniformly) gets a font shrunk to print scale but the
38// SAME 3.84px of tracking — the tracking's share of the em roughly doubles, the
39// title widens, and it wraps in the print / PDF output only (#174). Scale those
40// two lengths by the same S the box uses (px -> in) so print matches the editor.
41const scaleTrackingForPrint = (html: string): string =>
42 html.replace(
43 /(letter-spacing|word-spacing)\s*:\s*(-?[\d.]+)px/gi,
44 (_m, prop: string, n: string) => `${prop}:${(parseFloat(n) * S).toFixed(4)}in`,
45 );

52/** The letter-landscape @page + `.slide` print metrics, scoped under `scope`
53 * (e.g. '.eig-print-layer ' for the interactive export's print layer; '' for the
54 * standalone Print doc). One source of truth for the print geometry. */
55export function printPageCss(scope = ''): string {
56 return `@page { size: letter landscape; margin: 0; }
57${scope}.slide { width: 11in; height: 6.1875in; position: relative; overflow: hidden; box-sizing: border-box; break-after: page; margin-top: 1.15625in; }
58${scope}.slide:last-child { break-after: auto; }`;
59}

74/**
75 * One slide as a print `<div class="slide">…</div>`, all positions in inches.
76 * @param imageCache assetId → data-URL for raster/pdf images
77 * @param demoScreenshots `${slide.id}:${el.id}` → data-URL for baked live elements
78 * @param mathHtmlByKey `${slide.id}:${el.id}` → math-rendered text HTML (inline
79 * SVG). The caller pre-renders math (async, via the iframe pool) since this
80 * builder is pure/sync. Keyed by slide+element (not just element id) because one
81 * element can appear on several slides with a different font each; falls back to
82 * el.html for elements with no math.
83 */
84export function buildPrintSlideHtml(
85 slide: Slide,
86 presentation: Presentation,
87 imageCache: Map<string, string>,
88 demoScreenshots: Map<string, string>,
89 mathHtmlByKey?: Map<string, string>,
90 slideNumber?: number,
91): string {
92 const theme = resolveTheme(presentation.theme, slide.theme);
93 let inner = '';
94 for (const el of slide.elements) {
95 const p = el.position;
96 if (el.type === 'text') {
97 // Same shared box/style assembly as the HTML export — only the units differ
98 // (inches for lengths via px2in, points for font-size via px2pt).
99 const presetFontFamily = fontFamilyForPreset(fontForPreset(el.preset, slide, presentation.config), el.preset);
100 inner += textElementHtml(el, {
101 color: resolveColor(el.color, theme, themeColorForPreset(theme, el.preset)),
102 fontFamily: el.fontFamily || presetFontFamily,
103 fontSize: effectiveFontSize(el, presentation.config),
104 content: markAsEigendeck(scaleTrackingForPrint(mathHtmlByKey?.get(`${slide.id}:${el.id}`) ?? el.html ?? '')),
105 len: px2in,
106 fsize: px2pt,
107 theme,
108 });
109 } else if (el.type === 'image') {
110 const src = imageCache.get(el.assetId);
111 // Same shared visual styles as the HTML export (one drop-shadow + radius);
112 // only the box position is inch-scaled (px2in).
113 if (src) inner += imageHtml(src, el, px2in);
114 } else if (el.type === 'arrow') {
115 // viewBox maps the px arrow coords into the inch-scaled container.
116 inner += arrowSvgHtml(el, { viewBox: `0 0 ${W} ${H}`, theme });
117 } else if (el.type === 'cover') {
118 inner += coverHtml(el, theme.background, px2in, theme);
119 } else if (el.type === 'html') {
120 // Static + locked (no script/network) → the srcdoc iframe renders in the
121 // browser's print output directly, no screenshot bake needed.
122 //
123 // The iframe CONTENT is authored in CSS px (slide-px == CSS-px in the
124 // editor). Print positions elements in INCHES with NO slide-level transform,
125 // and an iframe's document always renders at 96 CSS-px/in regardless of the
126 // parent — so a slide-px is only S*96 CSS-px here. Without compensating, the
127 // content renders ~1/(S*96) ≈ 1.8× too big for its shrunken box. So ALWAYS
128 // size the iframe in CSS px (its design size) and scale it DOWN to the inch
129 // box. (HTML export scales the whole slide via a CSS transform, so its
130 // iframes scale for free — this only bites the inch-positioned print path.)
131 // • scaleMode → design = the content's natural size (contain-fit + the shrink);
132 // • otherwise → design = the box itself (content fills it, then shrinks).
133 const designW = htmlIsScaled(el) ? el.scaleW! : p.width;
134 const designH = htmlIsScaled(el) ? el.scaleH! : p.height;
135 const L = htmlScaleLayout(p.width * S * 96, p.height * S * 96, designW, designH);
136 const box = `position:absolute;left:${px2in(p.x)};top:${px2in(p.y)};width:${px2in(p.width)};height:${px2in(p.height)}`;
137 inner += htmlElementScaledIframeHtml(el, box, L, 'px', undefined, theme);
138 } else if (isLiveElement(el.type)) {
139 // P0-2: notebook joins demo/demo-piece/video as a baked screenshot.
140 const screenshot = demoScreenshots.get(`${slide.id}:${el.id}`);
141 if (screenshot) {
142 inner += `<img src="${screenshot}" style="position:absolute;left:${px2in(p.x)};top:${px2in(p.y)};width:${px2in(p.width)};height:${px2in(p.height)};" />`;
143 } else {
144 const label = el.type === 'notebook' ? 'Notebook' : el.type === 'video' ? 'Video' : 'Interactive Demo';
145 inner += `<div style="position:absolute;left:${px2in(p.x)};top:${px2in(p.y)};width:${px2in(p.width)};height:${px2in(p.height)};background:#f8f8f8;border:1px dashed #ccc;display:flex;align-items:center;justify-content:center;color:#999;font-size:${px2pt(24)};font-family:system-ui;">${label}</div>`;
146 }
147 }
148 }
149 // Slide footer (author · venue + slide number), mirroring the HTML export.
150 const meta = [presentation.config?.author, presentation.config?.venue].filter(Boolean).join(' · ');
151 if (showFooter(slide) && (meta || slideNumber != null)) {
152 inner += `<div class="slide-footer" style="position:absolute;bottom:${px2in(20)};right:${px2in(40)};display:flex;align-items:baseline;gap:${px2in(16)};font-family:${footerFontFamily(presentation.config)};color:#888;font-size:${px2pt(18)};">` +
153 `<span class="slide-footer-meta">${meta}</span><span class="slide-footer-number" style="font-size:${px2pt(24)};">${slideNumber ?? ''}</span></div>`;
154 }
155 return `<div class="slide" style="background:${theme.background};">${inner}</div>`;
156}
