4/** Arrow-key nudge delta for a selected element. 1px, or 10px with Shift
5 * (Keynote/Sketch convention). Returns null for non-arrow keys. */
6export function nudgeDelta(key: string, shiftKey: boolean): { dx: number; dy: number } | null {
7 const step = shiftKey ? 10 : 1;
8 switch (key) {
9 case 'ArrowLeft': return { dx: -step, dy: 0 };
10 case 'ArrowRight': return { dx: step, dy: 0 };
11 case 'ArrowUp': return { dx: 0, dy: -step };
12 case 'ArrowDown': return { dx: 0, dy: step };
13 default: return null;
14 }
15}

17/** Z-order direction for Cmd+] / Cmd+[ (Shift → all the way to front/back).
18 * Returns null for other keys. */
19export function zOrderDirection(key: string, shiftKey: boolean): 'top' | 'up' | 'down' | 'bottom' | null {
20 if (key === ']') return shiftKey ? 'top' : 'up';
21 if (key === '[') return shiftKey ? 'bottom' : 'down';
22 return null;
