19// modern-screenshot minimizes serialized styles by diffing against defaults
20// computed in a NESTED sandbox iframe (`getSandBox` -> `contentWindow.document`).
21// Inside our opaque-sandboxed demo that nested iframe is also opaque, so that
22// access throws "Blocked a frame ... cross-origin". Neuter the sandbox lookup so
23// getDefaultStyle takes its graceful empty-Map path (no minimization, still
24// correct). See docs/DEMO-PLATFORM.md §8 and issue #117.
25//
26// The match is a MINIFIED token, so modern-screenshot is PINNED (exact 4.7.0 in

28// error instead of a silent capture regression — a deliberate version bump must

58export interface DemoBridgeOpts {
59 /** editor-only: inline modern-screenshot + a capture handler for thumbnails. */
60 capture?: boolean;
61 /** notebook output: report content height to the parent so the host iframe
62 * grows to fit (a demo owns a fixed box; an output does not). */
63 reportSize?: boolean;
64 /** network policy for this demo. 'block' = no internet at all (default-src 'none';
65 * only the demo's own inline scripts/styles + data:/blob: assets render). { hosts }
66 * = internet SCOPED to those declared hosts (the manifest allowlist) for scripts,
67 * styles, fetch, images, media, fonts; anything else is blocked. Either way the
68 * demo still renders from its own content, and WebRTC is neutered (CSP blind spot).
69 * Omitted → treated as 'block' (fail closed): a demo is never mounted un-gated. */
70 net?: 'block' | { hosts: string[] };
71}

73// Injected FIRST in <head>. The CSP <meta> must precede any resource load to be
74// honored. It sets `default-src 'none'` and re-opens ONLY what a demo needs to
75// render: its own INLINE scripts/styles ('unsafe-inline'), data:/blob: assets, and
76// blob: frames/workers — plus the demo's DECLARED hosts (scoped connect, and the
77// same hosts opened for script/style/img/media/font so a declared CDN works). A
78// "block" demo declares no hosts, so every REMOTE resource — scripts, styles,
79// fetches, images, media, fonts, frames — is refused: no internet means no internet.
80// Inline scripts still run, so the demo renders from its own content. WebRTC bypasses
81// CSP, so delete its constructors before any demo code runs (robust: the demo is
82// opaque-origin, so it can't steal a fresh copy from a nested frame).
83function netBlockMeta(net: 'block' | { hosts: string[] }): string {
84 const hostSrc = net === 'block' ? '' : hostsToCspSources(net.hosts);
85 const s = hostSrc ? ` ${hostSrc}` : ''; // declared hosts appended to a directive
86 const connect = hostSrc || "'none'";
87 const form = hostSrc || "'none'";
88 const csp = [
89 "default-src 'none'",
90 // 'wasm-unsafe-eval' permits WebAssembly compilation ONLY (NOT JS eval/new
91 // Function — that's the separate, dangerous 'unsafe-eval'). Without it, script-src
92 // being set blocks WebAssembly.instantiate, so no Pyodide/GeoGebra/Emscripten
93 // (Rust/Go/C→wasm) demo can run — including in exported decks. Demos are already
94 // opaque-origin sandboxed + connect-src-gated, so wasm can't escape any further
95 // than the JS we already allow. See docs/CSP-AND-EGRESS.md.
96 `script-src 'unsafe-inline' 'wasm-unsafe-eval'${s}`,
97 `style-src 'unsafe-inline'${s}`,
98 `img-src data: blob:${s}`,
99 `media-src data: blob:${s}`,
100 `font-src data:${s}`,
101 `connect-src ${connect}`,
102 'frame-src blob: data:',
103 'child-src blob:',
104 'worker-src blob:',
105 `form-action ${form}`,
106 "base-uri 'none'",
107 ].join('; ');
108 return `<meta http-equiv="Content-Security-Policy" content="${csp}">`
109 + `<script>try{delete window.RTCPeerConnection;delete window.webkitRTCPeerConnection;delete window.RTCDataChannel;delete window.WebTransport;}catch(e){}</script>`;
110}

125 * "#piece=graph" (or ""), `channelKey` namespaces the relay per demo instance. */
126export function injectDemoBridge(html: string, hash: string, channelKey: string, opts: DemoBridgeOpts = {}): string {
127 const hashParams: Record<string, string> = {};
128 if (hash) {
129 const qs = hash.startsWith('#') ? hash.slice(1) : hash;
130 for (const part of qs.split('&')) {
131 const [k, v] = part.split('=');
132 if (k) hashParams[k] = v || '';
133 }
134 }
135 const bootstrap = `<script>
136(function(){
137 var __ch = ${JSON.stringify(channelKey)};
138 var __hp = ${JSON.stringify(hashParams)};
139 // Forward demo errors to the parent (opaque origin hides the demo console).
140 function __report(msg, src){ try{ window.parent.postMessage({__eigendeck:1,type:'demo-error',message:String(msg),src:src}, '*'); }catch(_){} }
141 window.addEventListener('error', function(e){ __report(e.message || e.error, 'error'); });
142 window.addEventListener('unhandledrejection', function(e){ __report(e.reason, 'rejection'); });
143 // Forward slide-navigation keys to the parent (#155): a focused demo otherwise
144 // swallows Space/arrows in present mode so the deck can't advance. Bubble phase
145 // + !defaultPrevented + not-a-form-field means a genuinely interactive demo (a
146 // stepper that uses arrows) keeps them by calling preventDefault; the parent
147 // only acts on this in present mode (the editor ignores it).
148 var __NAVK = {' ':1,'ArrowLeft':1,'ArrowRight':1,'ArrowUp':1,'ArrowDown':1,'PageUp':1,'PageDown':1,'Home':1,'End':1};
149 window.addEventListener('keydown', function(e){
150 if(!__NAVK[e.key] || e.defaultPrevented) return;
151 var t=e.target;
152 if(t && (t.tagName==='INPUT'||t.tagName==='TEXTAREA'||t.tagName==='SELECT'||t.isContentEditable)) return;
153 try{ window.parent.postMessage({__eigendeck:1,type:'nav-key',key:e.key}, '*'); }catch(_){}
154 }, false);
155 var _ce=console.error, _cw=console.warn;
156 console.error=function(){ __report([].slice.call(arguments).map(String).join(' '), 'console.error'); return _ce.apply(console,arguments); };
157 console.warn=function(){ __report([].slice.call(arguments).map(String).join(' '), 'console.warn'); return _cw.apply(console,arguments); };
158 try { window.location.hash = ${JSON.stringify(hash || '')}; } catch(e) {}
159 var _USP = window.URLSearchParams;
160 window.URLSearchParams = function(init) {
161 var inst = new _USP(init);
162 if (!init || init === '' || init === '#') { for (var k in __hp) inst.set(k, __hp[k]); }
163 return inst;
164 };
165 window.URLSearchParams.prototype = _USP.prototype;
166 // Opaque origins can't share a BroadcastChannel; relay via the parent.
167 window.BroadcastChannel = function(name) {
168 this._name = __ch + ':' + name;
169 this.onmessage = null;
170 var self = this;
171 window.addEventListener('message', function(e) {
172 if (!e.data || !self.onmessage) return;
173 if (e.data.__bc === self._name) { self.onmessage({ data: e.data.payload }); }
174 if (e.data.type === 'request-state' && !e.data.__bc) { self.onmessage({ data: e.data }); }
175 });
176 };
177 window.BroadcastChannel.prototype.postMessage = function(msg) {
178 try { window.parent.postMessage({ __bc: this._name, payload: msg }, '*'); } catch(e) {}
179 };

182 // Drive requestAnimationFrame from the parent's UN-throttled 60fps loop. WebKit
183 // throttles rAF in cross-origin (opaque) frames to 30fps until interaction; the
184 // parent top document isn't throttled, so it pumps ticks and we fire the demo's
185 // rAF callbacks on each. Falls back to native rAF if no tick arrives (e.g. no
186 // parent pump, as in a plain browser). See docs/DEMO-PLATFORM.md §16.
187 var __nativeRaf = window.requestAnimationFrame.bind(window);
188 var __nativeCancel = window.cancelAnimationFrame.bind(window);
189 var __rafMap = Object.create(null), __rafNextId = 1, __seenTick = false, __native = false;
190 // Fire callbacks queued for this frame. Delete-before-call so cancelAnimationFrame
191 // during a flush (deletes from the map) is honored, and re-scheduled ids (added
192 // during the flush) fire on the NEXT tick, matching native rAF semantics.
193 function __flush(t){ var ids = Object.keys(__rafMap); for (var i = 0; i < ids.length; i++){ var id = ids[i], cb = __rafMap[id]; if (cb){ delete __rafMap[id]; try { cb(t); } catch(e){} } } }
194 function __pumpMode(){ __native = false;
195 window.requestAnimationFrame = function(cb){ var id = __rafNextId++; __rafMap[id] = cb; return id; };
196 window.cancelAnimationFrame = function(id){ delete __rafMap[id]; }; }
197 function __nativeMode(){ __native = true; var due = __rafMap; __rafMap = Object.create(null);
198 window.requestAnimationFrame = __nativeRaf; window.cancelAnimationFrame = __nativeCancel;
199 for (var id in due) __nativeRaf(due[id]); }
200 __pumpMode();
201 window.addEventListener('message', function(e){
202 if (e.data && e.data.__eigendeck === 1 && e.data.type === 'raf-tick'){
203 __seenTick = true;
204 if (__native) __pumpMode(); // a slow/returning pump re-takes the clock (re-armable)
205 __flush(e.data.t);
206 }
207 });
208 // If no parent tick arrives within a generous margin, assume there's no pump
209 // (plain browser / export) and use native rAF. Re-armable above if a tick shows up.
210 setTimeout(function(){ if (!__seenTick) __nativeMode(); }, 1500);
211})();
212</script>`;
213 let scripts = opts.capture ? bootstrap + `<script>${MS_UMD}</script>` + CAPTURE_HANDLER : bootstrap;
214 if (opts.reportSize) scripts += SIZE_REPORTER;
215 // The network policy (CSP meta + WebRTC neuter) must be FIRST in <head>. Default
216 // to 'block' (fail CLOSED): a demo built without an explicit net policy gets the
217 // offline lockdown, never an un-gated document — so a future caller that forgets
218 // to pass `net` can't accidentally mount a demo with no CSP at all.
219 scripts = netBlockMeta(opts.net ?? 'block') + scripts;
220 if (html.includes('<head>')) return html.replace('<head>', '<head>' + scripts);
221 return scripts + html;
