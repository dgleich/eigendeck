1// The ONE color picker, shared by the inline text-format toolbar and the inspector
2// (text color / background / arrow / cover). Purely presentational: it renders the
3// swatch row consistently and emits SEMANTIC events — onNone / onColor(hex) /
4// onTint(base). Each call site wires those to its exact field writes (or the
5// toolbar's execCommand), so the data model stays per-site while the UI is unified.
6//
7// Row order (all wrapping): [None chip] [deck palette] [tint swatches] [fixed
8// palette] [custom-hex]. A tint's swatch previews its RESOLVED look against the
9// current theme; a corner mark flags it as "themed" (adapts per theme) vs a fixed
10// color. Only render the pieces a site opts into.

17// The 135° corner triangle that marks a swatch as a themed tint.

20export interface ColorControlProps {
21 /** Current literal color (hex), for fixed/custom active state. `undefined` = none. */
22 value?: string;
23 /** Current tint base (`el.boxTint`, or `'accent'` when a foreground uses the token),
24 * for tint active state. */
25 activeTint?: string;
26 /** Fixed swatches. */
27 palette: readonly Swatch[];
28 /** Deck brand colors (rendered first, ringed). */
29 customPalette?: readonly string[];
30 allowNone?: boolean;
31 /** Label for the clear chip — 'Auto' | 'None' | 'Match'. */
32 noneLabel?: string;
33 allowCustom?: boolean;
34 /** Enables the tint swatches. `fill` = the full wash palette (text bg / cover);
35 * `accent` = just the theme-accent solid (foreground text / arrow). */
36 tint?: { kind: 'fill' | 'accent'; theme: ThemeColors };
37 onNone?: () => void;
38 onColor: (hex: string) => void;
39 onTint?: (base: string) => void;
40}

45/** Preview color for a tint base against the current theme. */
46function tintPreview(base: string, kind: 'fill' | 'accent', theme: ThemeColors): string {
47 if (kind === 'accent') return theme.accent;
48 return textBackgroundResolved({ boxTint: base }, theme) || theme.background;
49}

51export function ColorControl({
52 value, activeTint, palette, customPalette, allowNone, noneLabel = 'None',
53 allowCustom, tint, onNone, onColor, onTint,
54}: ColorControlProps) {
55 const tintSwatches = tint ? (tint.kind === 'accent' ? ACCENT_TINT : TINT_SWATCHES) : [];
56 return (
57 <div className="prop-color-row">
58 {allowNone && (
59 <button
60 className={`prop-zbtn ${!value && !activeTint ? 'active' : ''}`}
61 style={{ fontSize: 11, width: 'auto', padding: '2px 6px' }}
62 onClick={onNone}
63 >
64 {noneLabel}
65 </button>
66 )}
67 {(customPalette ?? []).map((c) => (
68 <button
69 key={`cp-${c}`}
70 className={`${SWATCH} ${value === c && !activeTint ? 'active' : ''}`}
71 style={{ background: c, boxShadow: '0 0 0 1px #94a3b8 inset' }}
72 title={`Deck palette ${c}`}
73 onClick={() => onColor(c)}
74 />
75 ))}
76 {tintSwatches.map(({ base, title }) => (
77 <button
78 key={`tint-${base}`}
79 className={`${SWATCH} ${activeTint === base ? 'active' : ''}`}
80 title={title}
81 style={{ background: tintPreview(base, tint!.kind, tint!.theme), backgroundImage: TINT_MARK }}
82 onClick={() => onTint?.(base)}
83 />
84 ))}
85 {palette.map(({ color, label }) => (
86 <button
87 key={color}
88 className={`${SWATCH} ${value === color && !activeTint ? 'active' : ''}`}
89 title={label}
90 style={{ background: color, border: color === '#ffffff' ? '1px solid #ccc' : undefined }}
91 onClick={() => onColor(color)}
92 />
93 ))}
94 {allowCustom && (
95 <input
96 type="color"
97 title="Custom colour"
98 value={value && /^#[0-9a-fA-F]{6}$/.test(value) ? value : '#000000'}
99 onChange={(e) => onColor(e.target.value)}
100 style={CUSTOM_STYLE}
101 />
102 )}
103 </div>
104 );
