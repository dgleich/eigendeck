1// Lazy DOMPurify wrappers for UNTRUSTED strings that get innerHTML'd into the
2// privileged window from a shared .eigendeck:
3// - sanitizeSvg: the math-render cache (audit C-4) and static notebook
4// image/svg+xml output (matplotlib).
5// - sanitizeHtml: static notebook text/html output (pandas tables, styled
6// divs) and rendered markdown (audit C-1/C-2/C-5).
7// - outputHasExecutable: routes a notebook output — static content is
8// sanitized inline (here); executable content (Plotly etc.) is instead
9// mounted in an opaque-origin iframe (docs/NOTEBOOK-ISOLATION.md), never
10// sanitized-then-inlined, so it stays interactive AND contained.
11//
12// DOMPurify strips <script>, event handlers, and javascript: refs while keeping
13// tables/headings/links/img and the <svg>/<path>/<use xlink:href="#…">/<defs> a
14// real MathJax render needs. Lazy-imported (decks without math/notebooks don't
15// pay for it) and needs a DOM (the app webview / jsdom).

17type AttrHookData = { attrName: string; attrValue: string; keepAttr: boolean };
18type Purify = {
19 sanitize: (s: string, cfg?: Record<string, unknown>) => string;
20 addHook?: (entry: string, cb: (node: unknown, data: AttrHookData) => void) => void;
21};
22let _purify: Promise<Purify | null> | null = null;

24function getPurify(): Promise<Purify | null> {
25 if (!_purify) {
26 _purify = import('dompurify')
27 .then((m) => {
28 const p = (m.default ?? m) as Purify;
29 // DOMPurify needs a DOM to filter (it's an inert stub without one). Gate on
30 // the DOM directly rather than probing sanitize()'s return — a build that
31 // changed the stub's return value could otherwise flip us fail-open.
32 if (typeof p?.sanitize !== 'function' || typeof document === 'undefined') return null;
33 // MathJax references glyph <path>s via <use>, which the svg profile drops
34 // as an external-content vector. Re-allow it (see SVG_CFG) but constrain
35 // its href to in-document fragment refs (#id) — that's all MathJax emits,
36 // and it can't reach external/data: SVG (the actual <use> XSS).
37 p.addHook?.('uponSanitizeAttribute', (node, data) => {
38 if ((data.attrName === 'href' || data.attrName === 'xlink:href') && !data.attrValue.startsWith('#')) {
39 // SVG elements that PULL external content (<use>/<image>/<feImage>) may
40 // only reference in-document fragments (#id) — an external/data: ref is
41 // the SVG-include XSS class. A normal <a href="https://…"> in html output
42 // is fine, so constrain only the content-pulling elements.
43 const tag = (node as Element)?.tagName?.toLowerCase?.();
44 if (tag === 'use' || tag === 'image' || tag === 'feimage') data.keepAttr = false;
45 }
46 });
47 // Any link that survives opens externally with no window.opener handle.
48 p.addHook?.('afterSanitizeAttributes', (node) => {
49 const el = node as Element;
50 if (el?.tagName?.toLowerCase?.() === 'a' && el.getAttribute('href')) {
51 el.setAttribute('target', '_blank');
52 el.setAttribute('rel', 'noopener noreferrer');
53 }
54 });
55 return p;
56 })
57 .catch(() => null);
58 }
59 return _purify;
60}

115// Tags that carry (or load) executable content, or CSS we won't inline into the
116// privileged frame. <style>/<link> are here because a <style> block scopes an
117// output's own CSS (pandas Styler / df.style is exactly this) — DOMPurify would
118// strip it inline (unstyled table), and its CSS can @import/url()-exfil, so such
119// output is routed to the opaque iframe where the CSS renders fully and harmlessly.
120const EXEC_TAGS = new Set(['script', 'iframe', 'object', 'embed', 'link', 'meta', 'base', 'template', 'style']);
121 
122/**
123 * Does this notebook `text/html` output carry executable content? If so it is
124 * routed to an opaque-origin iframe (kept interactive + contained) instead of
125 * being sanitized inline. False positives are harmless (an iframe still renders);
126 * false negatives are impossible — anything a sanitizer would strip is detected.
127 * Sync (plain DOM parse, no DOMPurify) so the render path can branch immediately.
128 */
129export function outputHasExecutable(html: string | undefined | null): boolean {
130 if (!html || html.indexOf('<') < 0) return false;
131 let doc: Document;
132 try { doc = new DOMParser().parseFromString(html, 'text/html'); } catch { return true; }
133 // Scan the WHOLE tree, not just body: a fragment that starts with <script>
134 // (or <link>/<meta>) is hoisted into <head> by the parser, so a body-only scan
135 // would miss it and wrongly route executable output to the inline sanitizer.
136 for (const el of Array.from(doc.documentElement?.getElementsByTagName('*') || [])) {
137 if (EXEC_TAGS.has(el.tagName.toLowerCase())) return true;
138 for (const attr of Array.from(el.attributes)) {
139 const name = attr.name.toLowerCase();
140 if (name.startsWith('on')) return true;
141 if ((name === 'href' || name === 'src' || name === 'xlink:href')
142 && /^\s*(javascript:|data:text\/html)/i.test(attr.value)) return true;
143 }
144 }
145 return false;
146}
