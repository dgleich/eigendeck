1// Pure delta helpers for the two element relationships:
2// - sync (syncId) — "the same element shown on several slides"
3// - link (linkId) — "animate between two elements across slides"
4// plus the remembered-group fields (_syncId/_linkId) used to re-join after a
5// temporary free/unlink.
6//
7// These return a Partial<SlideElement> describing ONLY the four
8// sync/link/identity fields. They are the single source of truth for the
9// remember/restore dance, replacing ~10 hand-built `{ syncId, _syncId, ... }`
10// literals scattered across components (which had diverged — see the
11// link-asymmetry bug #30 and the PropertiesPanel "Unlink" that forgot to
12// remember _linkId). No store, no side effects: trivially unit-testable.

55/** After elements are removed (a slide or element is deleted), a sync or link
56 * group can be left with a SINGLE member — there's nothing to sync or animate
57 * with anymore. That lone id is an orphan: it still lights the "synced" border
58 * and the inspector's sync UI even though the element is effectively standalone
59 * (e.g. duplicate a slide, then delete the copy — the original kept its syncId).
60 *
61 * This strips the orphaned id — sync and link independently — from every lone
62 * group across the deck, leaving genuine multi-member groups untouched. A group
63 * is counted by its identity (`syncId ?? _syncId`, `linkId ?? _linkId`) so a
64 * member that's only *remembering* the group still keeps a partner alive — the
65 * SAME membership the sync badge uses, so border + badge stay consistent.
66 *
67 * Pure; returns a NEW slides array only when something changed (so the store's
68 * referential-equality checks don't see spurious updates). */
69export function pruneOrphanedGroups(slides: Slide[]): Slide[] {
70 const syncCount = new Map<string, number>();
71 const linkCount = new Map<string, number>();
72 for (const s of slides) for (const el of s.elements) {
73 const sid = el.syncId ?? el._syncId;
74 if (sid) syncCount.set(sid, (syncCount.get(sid) ?? 0) + 1);
75 const lid = el.linkId ?? el._linkId;
76 if (lid) linkCount.set(lid, (linkCount.get(lid) ?? 0) + 1);
77 }
78 let changed = false;
79 const next = slides.map((s) => {
80 let slideChanged = false;
81 const elements = s.elements.map((el) => {
82 const sid = el.syncId ?? el._syncId;
83 const lid = el.linkId ?? el._linkId;
84 const dropSync = sid != null && syncCount.get(sid) === 1;
85 const dropLink = lid != null && linkCount.get(lid) === 1;
86 if (!dropSync && !dropLink) return el;
87 slideChanged = true;
88 const copy: SlideElement = { ...el };
89 if (dropSync) { delete copy.syncId; delete copy._syncId; }
90 if (dropLink) { delete copy.linkId; delete copy._linkId; }
91 return copy;
92 });
93 if (slideChanged) { changed = true; return { ...s, elements }; }
94 return s;
95 });
96 return changed ? next : slides;
97}

99/** Decide how a pasted copy of `el` joins (or doesn't join) a group, by where
100 * it lands:
101 * - SAME slide → independent copy (detached); you can't animate within a slide.
102 * - DIFFERENT slide, source is SYNCED → JOIN the sync group: keep its
103 * syncId/linkId, drop only the remembered (_*) ids.
104 * - DIFFERENT slide, source NOT synced → detached now, then LINK to the source
105 * (the caller does the actual cross-element link); `link` flags that.
106 * Returns the sync/link delta to merge onto the new element + whether to link. */
107export function pasteElementDelta(
108 el: Pick<SlideElement, 'syncId'>, sameSlide: boolean,
109): { delta: SyncLinkDelta; link: boolean } {
110 if (!sameSlide && el.syncId) {
111 return { delta: { _syncId: undefined, _linkId: undefined }, link: false };
112 }
113 return { delta: detachDelta(), link: !sameSlide };
114}

116/** Shared linkId + the symmetric per-side delta for an ANIMATION link between a
117 * source element and a target on another slide. Link-only and NON-destructive:
118 * it sets a shared linkId (and clears _linkId) on BOTH sides and does NOT touch
119 * syncId — the elements stay SEPARATE (own position, content, recording) so the
120 * presenter animates between them. Sync (collapsing two elements into one
121 * entry) is a different, destructive operation that only the junction model
122 * (duplicate) produces cleanly; "L" must never auto-sync.
123 *
124 * The shared linkId prefers the SOURCE's existing/remembered group, then the
125 * target's. This is what makes a link directional+groupable: repeatedly linking
126 * FROM one anchor element to several slides builds ONE group (each new target
127 * joins the anchor's linkId) instead of minting a fresh id that strands the
128 * anchor's earlier partner (#S9). `mergeIds` lists the OTHER live group ids
129 * pulled in (so the caller migrates every member). (#30: both sides cleared
130 * symmetrically.) */
131export function linkPairDeltas(
132 source: Pick<SlideElement, 'linkId' | '_linkId'>,
133 target: Pick<SlideElement, 'linkId' | '_linkId'>,
134 newId: () => string = () => crypto.randomUUID(),
135): { sharedLinkId: string; delta: SyncLinkDelta; mergeIds: string[] } {
136 const sharedLinkId = source.linkId || source._linkId || target.linkId || target._linkId || newId();
137 // Distinct live/remembered group ids being merged INTO sharedLinkId — every
138 // element carrying one must be re-pointed so no prior partner is stranded.
139 const mergeIds = [...new Set(
140 [source.linkId, source._linkId, target.linkId, target._linkId]
141 .filter((id): id is string => !!id && id !== sharedLinkId),
142 )];
143 return { sharedLinkId, delta: { linkId: sharedLinkId, _linkId: undefined }, mergeIds };
144}
