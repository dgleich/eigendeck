51export function NotebookCells({
52 merged, language, highlight, dark, baseSize, showLineNumbers,

54}: {
55 merged: MergedCell[];
56 language: string | null;
57 highlight: boolean;
58 dark: boolean;
59 baseSize: number;
60 showLineNumbers?: boolean;
61 hideHeader: boolean;
62 kernelDisplayName: string | null;
63 /** Pre-resolved syntax highlight (export/static path), keyed by the
64 * same cell key NotebookCells uses (`i<index>` / `a<id>`). When a
65 * key is present its HTML is passed to CodeCell as highlightedHtml. */
66 highlights?: Map<string, string>;
67 /** Pre-rendered markdown HTML (export/static path), keyed by the same
68 * cell key. Markdown's async `marked` render can't run under
69 * renderToStaticMarkup, so the export resolves it up front. */
70 markdowns?: Map<string, string>;
71 /** Present = interactive (live view). Absent = read-only (export). */
72 live?: NotebookCellsLive;
73 /** Live-only status rows. */
74 loading?: boolean;
75 error?: Error | null;
76}) {
77 // Pointer-events gate. In the LIVE editor the body must NOT capture events
78 // until interacting (after double-click), so the DraggableBox can drag — see
79 // CLAUDE.md rule #8. In the EXPORT (no live, no DraggableBox) the body must
80 // capture so the notebook stays scrollable/selectable. So: live → gate on
81 // live.interactive; export → always 'auto'.
82 const captures = live ? live.interactive : true;
83 return (
84 <>
85 {!hideHeader && (
86 <div className="nb-header">
87 <span className="nb-kernel-label">
88 {kernelDisplayName}
89 </span>
90 </div>
91 )}
92 <div className="nb-body" style={{ pointerEvents: captures ? 'auto' : 'none' }}>
93 {loading && <div className="nb-status">Loading…</div>}
94 {error && <div className="nb-status nb-error">Parse error: {error.message}</div>}
95 {merged.map((m) => {
96 if (m.origin === 'ipynb') {
97 const c = m.cell;
98 if (c.kind === 'markdown') return <MarkdownCell key={`i${c.index}`} cell={{ ...c, source: m.source }} prerenderedHtml={markdowns?.get(`i${c.index}`)} />;
99 if (c.kind === 'raw') return <RawCell key={`i${c.index}`} cell={{ ...c, source: m.source }} />;
100 const key = `i${c.index}`;
101 if (live) {
102 const run = live.running.get(key);
103 const wsrc = live.working.get(key);
104 return (
105 <CodeCell key={key}
106 cell={c as CodeCellT}
107 source={wsrc ?? m.source}
108 liveOutputs={m.outputs}
109 liveExecutionCount={run?.count ?? m.executionCount}
110 running={run?.running ?? false}
111 onRun={() => live.execute(key, live.working.get(key) ?? m.source, (o, cnt) => live.ov.recordOutput(c.index, o, cnt))}
112 language={language}
113 highlight={highlight}
114 dark={dark}
115 editable={live.editable}
116 showLineNumbers={showLineNumbers}
117 fontSize={baseSize}
118 onEdit={(next) => live.setWorking((p) => new Map(p).set(key, next))}
119 onCommit={() => { const w = live.working.get(key); if (w !== undefined) live.ov.setEdit(c.index, w, c.source); }}
120 edited={m.edited}
121 onRevert={() => { live.setWorking((p) => { const n = new Map(p); n.delete(key); return n; }); live.ov.revertEdit(c.index); }}
122 />
123 );
124 }
125 // read-only (export): outputs/exec-count from the merged model,
126 // no handlers, pre-resolved highlight.
127 return (
128 <CodeCell key={key}
129 cell={c as CodeCellT}
130 source={m.source}
131 liveOutputs={m.outputs}
132 liveExecutionCount={m.executionCount}
133 running={false}
134 language={language}
135 highlight={highlight}
136 dark={dark}
137 editable={false}
138 showLineNumbers={showLineNumbers}
139 fontSize={baseSize}
140 highlightedHtml={highlights?.get(key)}
141 />
142 );
143 }
144 // appended cell
145 const a = m.appended;
146 const key = `a${a.id}`;
147 if (a.cellType === 'markdown') {
148 const src = live ? (live.working.get(key) ?? a.source) : a.source;
149 return <MarkdownCell key={key} cell={{ kind: 'markdown', index: -1, source: src }}
150 prerenderedHtml={live ? undefined : markdowns?.get(key)} />;
151 }
152 const synth: CodeCellT = {
153 kind: 'code', index: -1,
154 source: a.source,
155 executionCount: a.executionCount ?? null,
156 outputs: a.outputs ?? [],
157 };
158 if (live) {
159 const run = live.running.get(key);
160 return (
161 <CodeCell key={key}
162 cell={synth}
163 source={live.working.get(key) ?? a.source}
164 liveOutputs={a.outputs ?? []}
165 liveExecutionCount={run?.count ?? a.executionCount ?? null}
166 running={run?.running ?? false}
167 onRun={() => live.execute(key, live.working.get(key) ?? a.source, (o, cnt) => live.ov.recordAppendedOutput(a.id, o, cnt))}
168 language={language}
169 highlight={highlight}
170 dark={dark}
171 editable={live.editable}
172 showLineNumbers={showLineNumbers}
173 fontSize={baseSize}
174 added
175 onEdit={(next) => live.setWorking((p) => new Map(p).set(key, next))}
176 onCommit={() => { const w = live.working.get(key); if (w !== undefined) live.ov.setAppendedSource(a.id, w); }}
177 onRevert={() => live.ov.removeAppended(a.id)}
178 />
179 );
180 }
181 // read-only (export): appended cell, no handlers.
182 return (
183 <CodeCell key={key}
184 cell={synth}
185 source={a.source}
186 liveOutputs={a.outputs ?? []}
187 liveExecutionCount={a.executionCount ?? null}
188 running={false}
189 language={language}
190 highlight={highlight}
191 dark={dark}
192 editable={false}
193 showLineNumbers={showLineNumbers}
194 fontSize={baseSize}
195 added
196 highlightedHtml={highlights?.get(key)}
197 />
198 );
199 })}
200 {/* "+ Cell" lives at the END of the body (not the header) so it's

203 <button className="nb-add-cell" title="Add a code cell at the end"
204 onClick={() => live.ov.addAppended(lastIpynbIndex(merged), 'code')}>
205 + []
206 </button>
207 )}
208 </div>
209 </>

213/** Highest .ipynb cell index in the merged list, for anchoring a new
214 * appended cell at the end. null when there are no .ipynb cells. */
215export function lastIpynbIndex(merged: MergedCell[]): number | null {
216 let last: number | null = null;
217 for (const m of merged) if (m.origin === 'ipynb') last = m.cell.index;
218 return last;
