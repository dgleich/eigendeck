7/** Halo color for the glow effect: dark text → light halo, light → dark. */
8function haloFor(color) {
9 const hex = (color || '').replace('#', '');
10 if (!/^[0-9a-fA-F]{6}$/.test(hex)) return '#ffffff';
11 const r = parseInt(hex.slice(0, 2), 16), g = parseInt(hex.slice(2, 4), 16), b = parseInt(hex.slice(4, 6), 16);
12 const luma = 0.299 * r + 0.587 * g + 0.114 * b; // Rec. 601
13 return luma < 140 ? '#ffffff' : '#000000';
14}

16/** Effective text-element background (color + opacity → rgba), or undefined when none. */
17export function textBackgroundCss(el) {
18 if (!el || !el.backgroundColor) return undefined;
19 const a = el.backgroundOpacity ?? 1;
20 if (a >= 1) return el.backgroundColor;
21 const hex = el.backgroundColor.replace('#', '');
22 if (/^[0-9a-fA-F]{6}$/.test(hex)) {
23 const r = parseInt(hex.slice(0, 2), 16), g = parseInt(hex.slice(2, 4), 16), b = parseInt(hex.slice(4, 6), 16);
24 return `rgba(${r}, ${g}, ${b}, ${a})`;
25 }
26 return el.backgroundColor; // non-hex color: opacity not applied
27}

29/** Mix two #rrggbb colors: result = a*(1-t) + b*t. Falls back to `a` on bad input. */
30export function mixHex(a, b, t) {
31 const pa = (a || '').replace('#', ''), pb = (b || '').replace('#', '');
32 if (!/^[0-9a-fA-F]{6}$/.test(pa) || !/^[0-9a-fA-F]{6}$/.test(pb)) return a;
33 const ch = (h, i) => parseInt(h.slice(i, i + 2), 16);
34 const m = (i) => Math.round(ch(pa, i) * (1 - t) + ch(pb, i) * t);
35 return '#' + [m(0), m(2), m(4)].map((v) => v.toString(16).padStart(2, '0')).join('');
36}

48/** Rec.601 luminance (0–255) of a #rrggbb color; 0 on bad input. */
49function luma601(hex) {
50 const h = (hex || '').replace('#', '');
51 if (!/^[0-9a-fA-F]{6}$/.test(h)) return 0;
52 return 0.299 * parseInt(h.slice(0, 2), 16) + 0.587 * parseInt(h.slice(2, 4), 16) + 0.114 * parseInt(h.slice(4, 6), 16);
53}

55/** Effective text-element background, honoring a themed tint (#132 "card"): when
56 * `el.boxTint` is set, tint RELATIVE TO THE SLIDE THEME so the fill stays colored
57 * AND contrasting on any theme. On LIGHT themes we wash the background toward the
58 * base color (a pale pastel). On DARK themes we mix the base in far more strongly
59 * and keep it SATURATED (no white-wash) so the panel reads as a real colored
60 * surface — an earlier white-wash desaturated it into a muddy grey slate. Otherwise
61 * the fixed backgroundColor. `theme` is the resolved ThemeColors ({ background, accent }). */
62export function textBackgroundResolved(el, theme) {
63 if (el && el.boxTint && theme) {
64 const bg = theme.background || '#ffffff';
65 const base = el.boxTint === 'accent' ? (theme.accent || '#3b82f6') : el.boxTint;
66 return mixHex(bg, base, luma601(bg) < 100 ? TINT_STRENGTH_DARK : TINT_STRENGTH);
67 }
68 return textBackgroundCss(el);
69}

71/** Resolve an element's foreground color, honoring the theme-relative `'accent'`
72 * token (#132 follow-up): `undefined` → the caller's per-preset theme fallback;
73 * `'accent'` → the slide theme's accent (so it re-adapts per theme, like a tint,
74 * but as a solid foreground); any other value is a literal color. `theme` is the
75 * resolved ThemeColors ({ accent, ... }); `fallback` is the preset/default color. */
76export function resolveColor(color, theme, fallback) {
77 if (!color) return fallback;
78 if (color === 'accent') return (theme && theme.accent) || fallback;
79 return color;
80}

82/** Text legibility effect (#73): drop shadow or high-contrast glow. */
83export function textEffectCss(effect, color) {
84 if (effect === 'shadow') return '0 2px 4px rgba(0,0,0,0.45)';
85 if (effect === 'glow') {

88 }

102/** How far (px, per side) a card's box shadow paints OUTSIDE its box — so a cover
103 * mask sized to the card can be grown to hide the shadow too. Derived from the
104 * actual shadow (offset-x, offset-y, blur), so it tracks if the shadow changes.
105 * All zero when the element has no box shadow. A shadow at offset (ox,oy) blurred
106 * by `b` reaches `b∓ox` left/right and `b∓oy` top/bottom past the box edge. */
107export function boxShadowExtents(el) {
108 const sh = textBoxShadowCss(el);
109 if (!sh) return { left: 0, right: 0, top: 0, bottom: 0 };
110 const [ox = 0, oy = 0, b = 0] = sh.split(/\s+/).map((t) => parseFloat(t) || 0);
111 return {
112 left: Math.max(0, b - ox), right: Math.max(0, b + ox),
113 top: Math.max(0, b - oy), bottom: Math.max(0, b + oy),
114 };
115}

117/** Give `<code>` runs the deck's mono family by splicing font-family into each
118 * code tag's style. No-op when `mono` is empty. Shared by the live render
119 * (TextElementSvg) and the HTML export (exportCore). */
120export function applyCodeFont(html, mono) {
121 if (!mono || !html) return html || '';
122 return html.replace(/<code\b([^>]*)>/gi, (_m, attrs) =>
123 /\bstyle\s*=/.test(attrs)
124 ? `<code${attrs.replace(/style\s*=\s*"([^"]*)"/i, (_s, c) => `style="${c};font-family:${mono}"`)}>`
125 : `<code${attrs} style="font-family:${mono}">`);
