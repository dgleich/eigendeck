1// CodeMirror 6 editor for a single notebook code cell. Lazy-loads
2// the CodeMirror modules + the language grammar on first mount so
3// decks without editable notebooks don't pay the cost.
4//
5// Language support:
6// - python via @codemirror/lang-python (full grammar + indent)
7// - julia / r / sql / shell / etc. via @codemirror/legacy-modes
8// (StreamLanguage wrappers over the CM5 modes)
9// - anything else: plain text editing (no grammar)
10//
11// Shift-Enter runs the cell (onRun); the keymap intercepts it before
12// CodeMirror inserts a newline. Edits flow out via onChange; the
13// parent decides when to commit them (we commit on blur, not per
14// keystroke — see NotebookContent).

18export interface NotebookCellEditorProps {
19 value: string;
20 /** Kernel language for the grammar (e.g. 'python', 'julia', 'r'). */
21 language: string | null;
22 /** Base font size in slide-pixels (matches the cell's --nb-base-size). */
23 fontSize: number;
24 /** Apply syntax highlighting in the editor. Default true. */
25 highlight?: boolean;
26 /** Dark slide theme — picks the GitHub-Dark highlight palette to match
27 * the static view. Default false (GitHub Light). */
28 dark?: boolean;
29 /** Show a line-number gutter. Default false (opt-in). */
30 showLineNumbers?: boolean;
31 onChange: (next: string) => void;
32 /** Shift-Enter handler. */
33 onRun: () => void;
34 /** Fired when the editor loses focus — parent commits the edit. */
35 onBlur?: () => void;
36}

38// Module-level singletons so the heavy imports resolve once.
39type CM = typeof import('@codemirror/view') & {
40 state: typeof import('@codemirror/state');
41 commands: typeof import('@codemirror/commands');
42};
43let cmPromise: Promise<CM> | null = null;
44function loadCM(): Promise<CM> {
45 if (!cmPromise) {
46 cmPromise = Promise.all([
47 import('@codemirror/view'),
48 import('@codemirror/state'),
49 import('@codemirror/commands'),
50 ]).then(([view, state, commands]) =>
51 Object.assign({}, view, { state, commands }) as CM
52 );
53 }
54 return cmPromise;
55}

57// Resolve a CodeMirror language extension for the kernel language.
58// Returns null for plaintext. Lazy per-language so we only pull the
59// grammar the deck actually uses.
60async function languageExtension(language: string | null): Promise<unknown | null> {
61 if (!language) return null;
62 const l = language.toLowerCase();
63 try {
64 if (l === 'python' || l.startsWith('python')) {
65 const m = await import('@codemirror/lang-python');
66 return m.python();
67 }
68 // legacy-modes: StreamLanguage wrappers.
69 const { StreamLanguage } = await import('@codemirror/language');
70 if (l === 'julia') {
71 const m = await import('@codemirror/legacy-modes/mode/julia');
72 return StreamLanguage.define(m.julia);
73 }
74 if (l === 'r' || l === 'ir') {
75 const m = await import('@codemirror/legacy-modes/mode/r');
76 return StreamLanguage.define(m.r);
77 }
78 if (l === 'sql') {
79 const m = await import('@codemirror/legacy-modes/mode/sql');
80 return StreamLanguage.define(m.standardSQL);
81 }
82 if (l === 'shell' || l === 'bash' || l === 'sh') {
83 const m = await import('@codemirror/legacy-modes/mode/shell');
84 return StreamLanguage.define(m.shell);
85 }
86 } catch {
87 // Grammar module missing → plaintext editing.
88 }
89 return null;
90}

92// CM6 needs a highlight STYLE separate from the language parser. We mirror
93// the static view's highlight.js palette (GitHub Light / Dark, see
94// .el-notebook .hljs-* in App.css) so colors DON'T shift when you double-
95// click a cell to edit. `dark` picks the matching palette.
96async function highlightExtension(dark: boolean): Promise<unknown> {
97 const { syntaxHighlighting, HighlightStyle } = await import('@codemirror/language');
98 const { tags: t } = await import('@lezer/highlight');
99 const c = dark
100 ? { kw: '#ff7b72', fn: '#d2a8ff', lit: '#79c0ff', str: '#a5d6ff',
101 builtin: '#ffa657', comment: '#8b949e', name: '#7ee787' }
102 : { kw: '#d73a49', fn: '#6f42c1', lit: '#005cc5', str: '#032f62',
103 builtin: '#e36209', comment: '#6a737d', name: '#22863a' };
104 const style = HighlightStyle.define([
105 { tag: [t.keyword, t.controlKeyword, t.operatorKeyword, t.moduleKeyword,
106 t.definitionKeyword, t.modifier, t.typeName, t.self], color: c.kw },
107 { tag: [t.function(t.variableName), t.function(t.definition(t.variableName)),
108 t.definition(t.function(t.variableName)), t.className,
109 t.definition(t.className)], color: c.fn },
110 { tag: [t.number, t.bool, t.atom, t.operator, t.attributeName], color: c.lit },
111 { tag: [t.string, t.special(t.string), t.regexp, t.docString], color: c.str },
112 { tag: [t.standard(t.name), t.macroName], color: c.builtin },
113 { tag: [t.comment, t.lineComment, t.blockComment, t.docComment],
114 color: c.comment, fontStyle: 'italic' },
115 { tag: [t.tagName, t.quote], color: c.name },
116 ]);
117 return syntaxHighlighting(style);
118}

120// CodeMirror renders a trailing newline as an empty final line — which the
121// static <pre> view doesn't show. Strip ONE trailing newline so entering
122// edit mode doesn't add a spurious blank line at the bottom of the cell.
123function editorDoc(value: string): string {
124 return value.endsWith('\n') ? value.slice(0, -1) : value;
125}

127export function NotebookCellEditor({
128 value, language, fontSize, highlight = true, dark = false,
129 showLineNumbers = false, onChange, onRun, onBlur,
130}: NotebookCellEditorProps) {
131 const hostRef = useRef<HTMLDivElement>(null);
132 // Keep the latest callbacks in refs so the CodeMirror instance
133 // (created once) always calls the current handler without rebuild.
134 const onChangeRef = useRef(onChange); onChangeRef.current = onChange;
135 const onRunRef = useRef(onRun); onRunRef.current = onRun;
136 const onBlurRef = useRef(onBlur); onBlurRef.current = onBlur;
137 // Editor view, created async. Stored to apply external value
138 // updates + teardown.
139 const viewRef = useRef<import('@codemirror/view').EditorView | null>(null);
140 
141 useEffect(() => {
142 let cancelled = false;
143 let view: import('@codemirror/view').EditorView | null = null;
144 
145 (async () => {
146 const cm = await loadCM();
147 const langExt = await languageExtension(language);
148 const hlExt = highlight ? await highlightExtension(dark) : null;
149 if (cancelled || !hostRef.current) return;
150 
151 const { EditorView, keymap, lineNumbers } = cm;
152 const { EditorState } = cm.state;
153 const { defaultKeymap, history, historyKeymap, indentWithTab } = cm.commands;
154 
155 const runKeymap = keymap.of([{
156 key: 'Shift-Enter',
157 run: () => { onRunRef.current(); return true; },
158 }]);
159 
160 const updateListener = EditorView.updateListener.of((u) => {
161 if (u.docChanged) onChangeRef.current(u.state.doc.toString());
162 });
163 const blurListener = EditorView.domEventHandlers({
164 blur: () => { onBlurRef.current?.(); return false; },
165 });
166 
167 // Font size matches the cell's code size so the editor doesn't
168 // jump when switching between static + editing. Colors come from
169 // the notebook's theme CSS variables (set on .nb-frame) so the
170 // editor integrates with light/dark/custom slide themes instead
171 // of CodeMirror's default black-on-white.
172 // Base text color: when highlighting, match the static .hljs base
173 // (GitHub Light/Dark) so plain identifiers don't shift on edit;
174 // otherwise follow the slide theme's foreground.
175 const baseColor = highlight ? (dark ? '#c9d1d9' : '#24292e') : 'var(--nb-fg, #111827)';
176 const theme = EditorView.theme({
177 '&': {
178 fontSize: `${fontSize}px`,
179 backgroundColor: 'transparent',
180 color: baseColor,
181 },
182 '.cm-content': {
183 fontFamily: 'var(--nb-mono-family, ui-monospace, Menlo, monospace)',
184 caretColor: 'var(--nb-fg, #111827)',
185 },
186 '.cm-cursor, .cm-dropCursor': { borderLeftColor: 'var(--nb-fg, #111827)' },
187 '.cm-gutters': {
188 backgroundColor: 'transparent', border: 'none',
189 color: 'var(--nb-muted, #cbd5e1)',
190 },
191 // Line numbers: smaller + subtle than the code (matches the static
192 // gutter). CM aligns gutter elements per-line regardless of font.
193 '.cm-lineNumbers .cm-gutterElement': { fontSize: '0.72em', opacity: '0.7' },
194 '.cm-activeLine, .cm-activeLineGutter': { backgroundColor: 'transparent' },
195 '&.cm-focused': { outline: 'none' },
196 '.cm-selectionBackground, ::selection': { backgroundColor: 'var(--nb-code-bg, rgba(0,0,0,0.08))' },
197 });
198 
199 const extensions = [
200 history(),
201 keymap.of([...defaultKeymap, ...historyKeymap, indentWithTab]),
202 runKeymap,
203 updateListener,
204 blurListener,
205 theme,
206 ];
207 // With line numbers on, behave like a code editor: no wrap, gutter,
208 // horizontal scroll (consistent with the static gutter view). Without
209 // them, wrap long lines so code stays readable on a slide.
210 if (showLineNumbers) extensions.unshift(lineNumbers());
211 else extensions.push(EditorView.lineWrapping);
212 if (langExt) extensions.push(langExt as never);
213 if (hlExt) extensions.push(hlExt as never);
214 
215 view = new EditorView({
216 state: EditorState.create({ doc: editorDoc(value), extensions }),
217 parent: hostRef.current,
218 });
219 viewRef.current = view;
220 })();
221 
222 return () => {
223 cancelled = true;
224 view?.destroy();
225 viewRef.current = null;
226 };
227 // Rebuild when language / fontSize / highlight / line-number options
228 // change — value updates are applied imperatively below to avoid losing
229 // cursor position.
230 // eslint-disable-next-line react-hooks/exhaustive-deps
231 }, [language, fontSize, highlight, dark, showLineNumbers]);
232 
233 // Apply external value changes (e.g. revert, file-watcher reload)
234 // without rebuilding the editor — only when they differ from the
235 // current doc so user typing isn't clobbered.
236 useEffect(() => {
237 const view = viewRef.current;
238 if (!view) return;
239 const next = editorDoc(value);
240 const current = view.state.doc.toString();
241 if (current !== next) {
242 view.dispatch({ changes: { from: 0, to: current.length, insert: next } });
243 }
244 }, [value]);
245 
246 return <div ref={hostRef} className="nb-cell-editor" />;
247}
