193/** Leading-edge debounce: first call fires immediately, subsequent
194 * calls within `ms` are dropped, then the gate resets after `ms`
195 * of idle. The right semantic for "collapse a burst of state
196 * changes into one undo step recording the PRE-burst state." */
197function debounceUndoSnapshot<F extends (...args: never[]) => void>(fn: F, ms: number): F {
198 let lastFire = 0;
199 return ((...args: Parameters<F>) => {
200 const now = performance.now();
201 if (now - lastFire >= ms) {
202 lastFire = now;
203 fn(...args);
204 }
