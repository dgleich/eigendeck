19/** The :root{--eigendeck-*} block for a slide's resolved theme + fonts. */
20export function demoVarsCssForSlide(config: PresentationConfig, theme: string, slide: Slide): string {
21 const colors = resolveTheme(theme, slide.theme);
22 const bodyPkg = fontForPreset('body', slide, config);
23 const monoPkg = resolveMonoFontPackage(config?.defaultMonoFont);
24 // Mono pkg carries a full CSS stack ("'Source Code Pro', monospace"); take the
25 // first quoted family for the bare --eigendeck-mono name.
26 const monoFamily = monoPkg?.family?.match(/'([^']+)'/)?.[1] || undefined;
27 // Only PT Sans ships a real narrow variant. For every other font, fall back to
28 // the body font itself (NOT a clashing 'PT Sans Narrow') so a demo using
29 // var(--eigendeck-narrow) stays in the deck's typeface — same as the footnote
30 // preset's cascade.
31 const narrowFamily = bareNarrowFamilyName(bodyPkg) || bareFamilyName(bodyPkg);
32 return demoThemeVarsCss(colors, {
33 font: bareFamilyName(bodyPkg),
34 narrow: narrowFamily,
35 mono: monoFamily,
36 baseSize: effectiveTextPresetSize('body', config),
37 });
