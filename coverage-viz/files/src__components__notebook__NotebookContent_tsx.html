1// The notebook's interactive body — header + scrollable cell area
2// with optional Run buttons. Used by both NotebookBox (editor,
3// wrapped in DraggableBox) and PresentNotebook (PresentMode,
4// wrapped in an absolute-positioned div).
5//
6// Display = the pristine .ipynb merged with the element's "overlay"
7// (source edits, recorded outputs, live-authored appended cells) via
8// mergeNotebook. The overlay is an owner-tagged asset managed by
9// useOverlay — the .ipynb is never mutated. See
10// .claude/notes/notebook-recording-decisions.md.
11//
12// `interactive: false` disables the pointer-events gate so cells
13// accept clicks immediately in PresentMode.

54 // BUG-2: a notebook that's EFFECTIVELY editable — whether via the per-element
55 // toggle or the global default — must not be file-watched, else a disk reload
56 // clobbers in-deck edits. The toggle already sets auto_reload='off'; this also
57 // covers the default-on case (element.editable left undefined). Editor only,
58 // so we never write during a presentation. One-directional BY DESIGN: making
59 // a notebook editable stops watching, and un-editing LEAVES it stopped —
60 // taking control is sticky; the user re-enables Watch explicitly (Asset
61 // section) when ready. So nothing here re-enables on editable=false.
62 useEffect(() => {
63 if (mode !== 'editor' || !editable || !element.assetId) return;
64 const assetId = element.assetId;
65 void import('@tauri-apps/api/core')
66 .then(({ invoke }) => invoke('db_set_asset_auto_reload', { assetId, value: 'off' }))
67 // Fire asset-changed so the file watcher re-evaluates the cascade and
68 // UNSUBSCRIBES — without this the watcher keeps firing and a disk change
69 // clobbers the in-deck edits despite "taking control" (matches what
70 // AssetSection.setAutoReload does).
71 .then(() => window.dispatchEvent(new CustomEvent('eigendeck:asset-changed', { detail: { assetId } })))
72 .catch(() => {});
73 }, [editable, element.assetId, mode]);

169/** Apply the element's visibleCells whitelist + hideMarkdown filter to
170 * a plain Cell[] (lite path). */
171function filterCells(cells: Cell[], element: NotebookElement): Cell[] {
172 let out = (element.visibleCells && element.visibleCells.length > 0)
173 ? cells.filter((c) => element.visibleCells!.includes(c.index))
174 : cells;
175 if (element.hideMarkdown) out = out.filter((c) => c.kind !== 'markdown');
176 return out;
177}

179/** Apply the same filters to the merged render list (external path). */
180export function filterMerged(merged: MergedCell[], element: NotebookElement): MergedCell[] {
181 return merged.filter((m) => {
182 if (m.origin === 'ipynb') {
183 if (element.visibleCells && element.visibleCells.length > 0
184 && !element.visibleCells.includes(m.cell.index)) return false;
185 if (element.hideMarkdown && m.cell.kind === 'markdown') return false;
186 return true;
187 }
188 // appended cells are always shown (user-authored), except markdown
189 // when hideMarkdown.
190 if (element.hideMarkdown && m.appended.cellType === 'markdown') return false;
191 return true;
192 });
193}

196 element, mode, notebook, loading, error, interactive, editable, resolved, preamble, autoRun,
197 highlight, dark, language, baseSize, hideHeader, kernelDisplayName, themeSalt, previewBg,
198}: {
199 element: NotebookElement;
200 mode: 'editor' | 'present';
201 notebook: Notebook | null;
202 loading: boolean; error: Error | null;
203 interactive: boolean;
204 editable: boolean;
205 resolved: ResolvedExternal;
206 preamble: string | undefined;
207 autoRun: boolean;
208 highlight: boolean;
209 dark: boolean;
210 language: string | null;

212 hideHeader: boolean;
213 kernelDisplayName: string | null;
214 themeSalt: string;

217 const kernel = useKernel(resolved);
218 // Overlay identity is the element's SYNC identity, not its per-slide id:
219 // synced instances (the same notebook shown on multiple slides) are the
220 // SAME thing and must share ONE overlay. syncId is set to the original
221 // element's id on first duplicate, so the whole group resolves here to a
222 // single stable key — consistent across Save and Save As. A lone notebook
223 // has no syncId and keys by its own id (unchanged).
224 const ov = useOverlay(element.syncId ?? element.id);
225 const updateElement = usePresentationStore((s) => s.updateElement);

227 // Keep the cached PNG preview fresh when the RECORDING changes (cell edits,
228 // recorded outputs, appended cells). Those live in the overlay, not the
229 // element, so the element-keyed capture in NotebookContent misses them and the
230 // sidebar/picker thumbnail would go stale. Debounced + editor-only; the
231 // element comes via a ref so this fires only on overlay changes (the
232 // NotebookContent effect already covers element/prop changes).
233 const elForCapture = useRef(element);
234 elForCapture.current = element;
235 const ovSig = serializeOverlay(ov.overlay);
236 useEffect(() => {
237 if (mode !== 'editor') return;
238 const t = setTimeout(() => { void capturePreview(elForCapture.current, '.nb-frame', themeSalt, previewBg); }, 700);
239 return () => clearTimeout(t);
240 }, [ovSig, mode, themeSalt, previewBg]);

242 // Migrate legacy element.cellEdits (pre-overlay) into the overlay
243 // once, then strip the field. cellEdits is being retired in favor of
244 // the overlay asset.
245 const migratedRef = useRef(false);
246 useEffect(() => {
247 if (migratedRef.current) return;
248 const legacy = element.cellEdits;
249 if (legacy && Object.keys(legacy).length > 0) {
250 migratedRef.current = true;
251 for (const [k, v] of Object.entries(legacy)) {
252 ov.setEdit(Number(k), v, ''); // savedSource '' → always set
253 }
254 updateElement(element.id, { cellEdits: undefined } as Partial<NotebookElement>);
255 }

258 // Drop the overlay ONLY when the .ipynb source genuinely changes
259 // (deliberate reload-from-disk / restore-version) — detected by the
260 // parsed notebook's CONTENT changing, not by the generic
261 // `eigendeck:asset-changed` event. That event fires for many incidental
262 // reasons (preview/cache invalidation, watcher re-eval, auto-reload
263 // toggle); reacting to it wiped recorded outputs + edits during plain
264 // editing. Same bytes re-parse to identical content, so a spurious event
265 // is a no-op here; a real reload re-parses to different content → clear.
266 const clearOverlay = ov.clear;
267 const nbSigRef = useRef<{ id: string; sig: string } | null>(null);
268 useEffect(() => {
269 if (!notebook) return;
270 const sig = notebookSourceSignature(notebook);
271 const prev = nbSigRef.current;
272 nbSigRef.current = { id: element.id, sig };
273 if (overlaySourceChanged(prev, element.id, sig)) clearOverlay();
274 }, [notebook, element.id, clearOverlay]);

276 // Trust-independent "Discard Changes" (#121): the asset inspector fires this
277 // to drop the run-cell overlay and re-render the notebook as stored in the
278 // deck. No disk read, no trust gate (unlike Reload Now) — ov.clear() re-merges
279 // against the embedded .ipynb bytes every untrusted deck already renders.
280 useEffect(() => {
281 if (!element.assetId) return;
282 const assetId = element.assetId;
283 const handler = (e: Event) => {
284 if ((e as CustomEvent).detail?.assetId === assetId) clearOverlay();
285 };
286 window.addEventListener('eigendeck:discard-overlay', handler);
287 return () => window.removeEventListener('eigendeck:discard-overlay', handler);
288 }, [element.assetId, clearOverlay]);

290 // Transient running status, keyed by cell key.
291 const [running, setRunning] = useState<Map<string, RunningState>>(new Map());
292 const setRun = useCallback((key: string, s: RunningState | null) => {

296 return next;

308 // Execute `source`; stream outputs into the overlay live. `record`
309 // writes the accumulating outputs (ipynb index or appended id).
310 const execute = useCallback(async (
311 key: string,
312 source: string,
313 record: (outs: CellOutput[], count: number | null) => void,
314 ) => {
315 setRun(key, { running: true, count: '*' });
316 let outs: CellOutput[] = [];
317 let count: number | null = null;
318 const push = (o: CellOutput) => { outs = [...outs, o]; record(outs, count); };
319 try {
320 if (preamble && !preambleFiredRef.current) {
321 preambleFiredRef.current = true;
322 const ph = await kernel.runCell(preamble, {});
323 await ph.done;
324 }
325 record([], null); // clear prior outputs at run start
326 const handle = await kernel.runCell(source, {
327 onStream: (s) => {
328 const last = outs[outs.length - 1];
329 if (last && last.kind === 'stream' && last.name === s.name) {
330 outs = [...outs.slice(0, -1), { ...last, text: last.text + s.text }];
331 record(outs, count);
332 } else { push({ kind: 'stream', ...s }); }
333 },
334 onDisplayData: (d) => push({ kind: 'display_data', data: d.data }),
335 onExecuteResult: (r) => { count = r.executionCount; push({ kind: 'execute_result', data: r.data, executionCount: r.executionCount }); },
336 onError: (e) => push({ kind: 'error', ename: e.ename, evalue: e.evalue, traceback: e.traceback }),
337 // Authoritative `[N]` count — fires for every cell, including ones with
338 // no output, so an assignment / print-only cell still updates its prompt.
339 onReply: (r) => { if (r.executionCount != null) { count = r.executionCount; record(outs, count); } },
340 });
341 await handle.done;
342 record(outs, count);
343 } catch (e) {
344 record([...outs, { kind: 'error', ename: 'KernelError', evalue: e instanceof Error ? e.message : String(e), traceback: [] }], count);
345 } finally {
346 setRun(key, null);
347 }
348 }, [kernel, preamble, setRun]);
349 
350 const merged = filterMerged(mergeNotebook(notebook, ov.overlay), element);
351 
352 // autoRun: fire all visible code cells in order (present mode).
353 useEffect(() => {
354 if (!autoRun) return;
355 let cancelled = false;
356 (async () => {
357 for (const m of merged) {
358 if (cancelled) break;
359 if (m.origin === 'ipynb' && m.cell.kind === 'code') {
360 await execute(`i${m.cell.index}`, m.source, (o, c) => ov.recordOutput(m.cell.index, o, c));
361 } else if (m.origin === 'appended' && m.appended.cellType === 'code') {
362 const a = m.appended;
363 await execute(`a${a.id}`, a.source, (o, c) => ov.recordAppendedOutput(a.id, o, c));
364 }
365 }
366 })();
367 return () => { cancelled = true; };
368 // eslint-disable-next-line react-hooks/exhaustive-deps
369 }, [autoRun]);

391/** Human-readable status, used only as the dot's hover title. */
392function labelForStatus(s: KernelStatus): string {
393 switch (s) {
394 case 'disconnected': return 'not connected';
395 case 'connecting': return 'connecting…';
396 case 'idle': return 'idle';
397 case 'busy': return 'running';
398 case 'error': return 'error';
399 case 'dead': return 'kernel died';
400 case 'no-server': return 'no matching server';
401 }
