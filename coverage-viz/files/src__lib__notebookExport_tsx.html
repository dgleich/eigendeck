42/**
43 * Build the inner HTML (a positioned <iframe>) for a notebook element in
44 * the interactive HTML export. Returns null on failure so the caller can
45 * fall back to the preview PNG / placeholder.
46 *
47 * The returned iframe is positioned by the CALLER (exportCore wraps it /
48 * sets left/top/width/height) — here we only emit the iframe tag with the
49 * srcdoc; exportCore adds the absolute positioning, consistent with how
50 * the demo iframe is handled.
51 */
52export async function renderNotebookElementHtml(
53 element: NotebookElement,
54 slide: Slide,
55 presentation: Presentation,
56 getAssetBytes: AssetBytes,
57): Promise<string> {
58 // 1. Load + parse the .ipynb.
59 const nb = parseNotebookBytes(toUint8(await getAssetBytes(element.assetId)));
60 
61 // 2. Load the overlay if present (owned asset keyed by the element's
62 // sync identity, mirroring useOverlay), else empty.
63 const overlay = await loadOverlay(element, getAssetBytes);
64 
65 // 3. Merge + apply the element's cell filters (same as the live view).
66 const merged = filterMerged(mergeNotebook(nb, overlay), element);
67 
68 // 4. Pre-resolve syntax highlight + markdown (async paths that
69 // renderToStaticMarkup can't run).
70 const highlight = element.syntaxHighlight !== false;
71 const language = nb.language ?? null;
72 const highlights = new Map<string, string>();
73 const markdowns = new Map<string, string>();
74 const md = await loadMarked();
75 // Cell OUTPUTS (html/svg) are sanitized SYNCHRONOUSLY during renderToStaticMarkup
76 // via StaticExportContext, so preload DOMPurify once here first. Markdown is
77 // pre-rendered below and also sanitized (untrusted deck content).
78 await preloadSanitizer();
79 const mdSafe = async (src: string) => sanitizeHtml(await md(src));
80 for (const m of merged) {
81 if (m.origin === 'ipynb') {
82 const c = m.cell;
83 if (c.kind === 'code') {
84 if (highlight) {
85 // Mirror CodeCell's display: strip one trailing newline.
86 const display = m.source.replace(/\n$/, '');
87 highlights.set(`i${c.index}`, await highlightCode(display, language));
88 }
89 } else if (c.kind === 'markdown') {
90 markdowns.set(`i${c.index}`, await mdSafe(m.source));
91 }
92 } else {
93 const a = m.appended;
94 const key = `a${a.id}`;
95 if (a.cellType === 'code') {
96 if (highlight) {
97 highlights.set(key, await highlightCode(a.source.replace(/\n$/, ''), language));
98 }
99 } else {
100 markdowns.set(key, await mdSafe(a.source));
101 }
102 }
103 }
104 
105 // 5. Render the read-only cells (no `live`).
106 const baseSize = effectiveFontSize(element, presentation.config);
107 const theme = resolveTheme(presentation.theme ?? 'white', slide.theme);
108 const dark = isDarkTheme(theme);
109 const kernelDisplayName = nb.kernelDisplayName ?? nb.kernelspecName ?? null;
110 const body = renderToStaticMarkup(
111 <StaticExportContext.Provider value={true}>
112 <NotebookCells
113 merged={merged}
114 language={language}
115 highlight={highlight}
116 dark={dark}
117 baseSize={baseSize}
118 showLineNumbers={element.showLineNumbers}
119 hideHeader={element.hideHeader === true}
120 kernelDisplayName={kernelDisplayName}
121 highlights={highlights}
122 markdowns={markdowns}
123 />
124 </StaticExportContext.Provider>,
125 );
126 
127 // 6. Compute the same --nb-* CSS variables + frame class NotebookContent
128 // sets, so the export matches the live typography/theme.
129 const proseFont = fontForNotebookProse(slide, presentation.config);
130 const codeFont = fontForNotebookCode(presentation.config);
131 const tint = dark ? 'rgba(255,255,255,0.07)' : 'rgba(0,0,0,0.045)';
132 const borderColor = dark ? 'rgba(255,255,255,0.18)' : 'rgba(0,0,0,0.12)';
133 const vars: Record<string, string> = {
134 '--nb-prose-family': proseFont.family,
135 '--nb-mono-family': codeFont.family,
136 '--nb-base-size': `${baseSize}px`,
137 '--nb-fg': theme.text,
138 '--nb-bg': theme.background,
139 '--nb-code-bg': tint,
140 '--nb-muted': theme.muted,
141 '--nb-accent': theme.accent,
142 '--nb-border': borderColor,
143 };
144 const styleStr = Object.entries(vars).map(([k, v]) => `${k}:${v}`).join(';');
145 const frameClass = [
146 'nb-frame',
147 dark ? 'nb-theme-dark' : 'nb-theme-light',
148 element.showBorder ? 'nb-frame--bordered' : '',
149 ].filter(Boolean).join(' ');
150 
151 // 7. Assemble the iframe document. `.el-notebook` wraps the frame so the
152 // scoped `.el-notebook .nb-*` rules apply (verbatim from the app).
153 // html/body sized to fill so the .nb-body scrolls.
154 const doc = `<!doctype html><html><head><meta charset="utf-8">`
155 + `<style>${nbCss}\n`
156 // print-color-adjust:exact so the notebook's theme background/colours survive
157 // when the exported HTML is printed — the parent's does NOT cascade into this
158 // sandboxed srcdoc iframe, so without it dark-theme output is unreadable (light
159 // text on a stripped-to-white background). Same fix as the html element (#137).
160 + `html,body{margin:0;padding:0;height:100%;-webkit-print-color-adjust:exact;print-color-adjust:exact;}`
161 + `.el-notebook{height:100%;}`
162 + `.el-notebook .nb-frame{height:100%;}`
163 + `</style></head><body>`
164 + `<div class="el-notebook">`
165 + `<div class="${frameClass}" style="${styleStr}">${body}</div>`
166 + `</div></body></html>`;
167 
168 return `<iframe srcdoc="${htmlEscapeForSrcdoc(doc)}" sandbox="allow-same-origin" `
169 + `style="border:none;width:100%;height:100%;"></iframe>`;
170}

172/** Load the element's overlay asset (owner-tagged) for the export. The
173 * app fetches the owned asset id via db_get_owned_asset_id; the export
174 * callback in fileOps resolves overlay bytes the same way and passes
175 * them through getAssetBytes by the overlay's own asset id when known.
176 * Here we look it up via the same convention useOverlay uses; if the
177 * caller can't supply it, we fall back to empty (no recorded session). */
178async function loadOverlay(
179 element: NotebookElement,
180 getAssetBytes: AssetBytes,
181): Promise<Overlay> {
182 const key = element.syncId ?? element.id;
183 try {
184 // useOverlay persists overlays under a deterministic asset id
185 // `overlay-<key>` (see useOverlay.flushNow / writeOverlayFor). Try
186 // that id directly via getAssetBytes; a missing asset throws → empty.
187 const bytes = await getAssetBytes(`overlay-${key}`);
188 return parseOverlay(toUint8(bytes));
189 } catch {
190 return emptyOverlay();
191 }
192}
