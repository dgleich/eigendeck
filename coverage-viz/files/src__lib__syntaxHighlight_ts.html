1// Lazy-loaded syntax highlighter for notebook code cells.
2//
3// highlight.js is imported via its 'common' subset (~75 KB
4// minified), which covers Python / Julia / R / JS / TS / C / C++ /
5// Rust / Bash / Go / SQL / Java / Kotlin / Swift / etc. — every
6// kernel a CS/HPC professor is realistically going to demo. Niche
7// languages render as plain text rather than fail.
8//
9// The module-level promise is shared across every notebook cell in
10// the app so the lib loads exactly once on the first highlight call.
11 
12type HighlightFn = (code: string, lang: string) => string;

14let highlighterPromise: Promise<HighlightFn> | null = null;

16function loadHighlighter(): Promise<HighlightFn> {
17 if (!highlighterPromise) {
18 highlighterPromise = import('highlight.js/lib/common').then((mod) => {
19 const hljs = mod.default;
20 return (code: string, lang: string): string => {
21 const language = mapKernelLanguage(lang);
22 if (language && hljs.getLanguage(language)) {
23 try {
24 return hljs.highlight(code, { language, ignoreIllegals: true }).value;
25 } catch {
26 // fall through to plain text on highlighter errors
27 }
28 }
29 // No known grammar — escape and return as-is so the cell
30 // still renders, just unhighlighted.
31 return escapeHtml(code);
32 };
33 });
34 }
35 return highlighterPromise;
36}

38/** Map a kernelspec language to a highlight.js grammar id. */
39function mapKernelLanguage(lang: string): string | null {
40 if (!lang) return null;
41 const l = lang.toLowerCase();
42 // Direct hits — highlight.js's id matches the kernelspec field.
43 if (['python', 'julia', 'r', 'javascript', 'typescript', 'java',
44 'kotlin', 'swift', 'rust', 'go', 'cpp', 'c', 'sql', 'bash',
45 'json', 'yaml', 'xml', 'css', 'scss', 'php', 'ruby'].includes(l)) {
46 return l;
47 }
48 // Common alternate names.
49 if (l === 'shell' || l === 'sh') return 'bash';
50 if (l === 'c++' || l === 'cxx') return 'cpp';
51 if (l === 'js') return 'javascript';
52 if (l === 'ts') return 'typescript';
53 // R kernels sometimes report 'R' uppercase or 'ir' (IRkernel).
54 if (l === 'ir') return 'r';
55 // Julia kernels are usually 'julia'; nothing extra to do.
56 return null;
57}

59function escapeHtml(s: string): string {
60 return s
61 .replace(/&/g, '&amp;')
62 .replace(/</g, '&lt;')
63 .replace(/>/g, '&gt;');
64}

66/** Highlight `code` for the given kernel language. Lazy-loads the
67 * highlighter on first call. Returns a Promise of HTML; render
68 * with dangerouslySetInnerHTML. */
69export async function highlightCode(code: string, kernelLanguage: string | null): Promise<string> {
70 if (!kernelLanguage) return escapeHtml(code);
71 const fn = await loadHighlighter();
72 return fn(code, kernelLanguage);
