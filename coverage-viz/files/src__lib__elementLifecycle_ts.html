1// Per-element-type lifecycle hooks for the sync/link transitions. Lets a
2// specific element type (notably `notebook`) customize what happens when it is
3// freed, re-synced, or merged with another element — without the store needing
4// to import that type's code. Types register at app boot
5// (registerNotebookLifecycle in components/notebook/notebookLifecycle.ts); the
6// store calls the type-agnostic run* dispatchers.
7//
8// Hooks may be async and are fired BEFORE the store flips the relevant fields,
9// so they can seed in-memory caches synchronously (the clone-on-unsync
10// contract: cloneOverlay seeds its cache before its first await, so a
11// void-fired hook shows no empty flash on the next render).

15export interface MergeContext {
16 /** Element on the current slide being linked. */
17 source: SlideElement;
18 /** Element on another slide it links to. */
19 target: SlideElement;
20 /** The shared syncId the merged element will live under. */
21 sharedSyncId: string;
22 /** Which side's type-specific state to keep: 'auto' decides by content,
23 * 'source'/'target' is an explicit (e.g. user-chosen) winner. */
24 keep: 'auto' | 'source' | 'target';
25}

27export interface ElementLifecycleHooks {
28 /** A synced element is being freed; `freedId` becomes its standalone key. */
29 onFree?(el: SlideElement, freedId: string): void | Promise<void>;
30 /** A freed element is rejoining its remembered group. */
31 onResync?(el: SlideElement): void | Promise<void>;
32 /** Two elements are merging under one group. */
33 onMerge?(ctx: MergeContext): void | Promise<void>;
34 /** `copy` was just created from `source` (duplicate / paste). Carry
35 * type-specific state across — e.g. clone a notebook's recording — so a copy
36 * keeps everything intact. Should no-op when the copy SHARES the source's
37 * group (same overlay key), which it can detect via the keys. */
38 onCopy?(source: SlideElement, copy: SlideElement): void | Promise<void>;
39}

41const registry = new Map<string, ElementLifecycleHooks>();

47/** Test/teardown helper — drop all registrations. */

52export function runFreeHook(el: SlideElement, freedId: string): void | Promise<void> {
53 return registry.get(el.type)?.onFree?.(el, freedId);

56export function runResyncHook(el: SlideElement): void | Promise<void> {

60export function runCopyHook(source: SlideElement, copy: SlideElement): void | Promise<void> {
61 return registry.get(source.type)?.onCopy?.(source, copy);

64/** Run the merge hook for each DISTINCT type among source/target (so a hook
65 * that reconciles both sides — e.g. notebook overlays — runs exactly once). */
66export async function runMergeHook(ctx: MergeContext): Promise<void> {
67 const types = new Set([ctx.source.type, ctx.target.type]);
68 for (const t of types) {
69 await registry.get(t)?.onMerge?.(ctx);
70 }
