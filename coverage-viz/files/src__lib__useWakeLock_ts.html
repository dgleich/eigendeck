3/**
4 * Hold a **Screen Wake Lock** while `active` (i.e. during Present mode) so the
5 * display doesn't sleep and the screensaver doesn't start mid-talk (#179).
6 *
7 * Best-effort + feature-detected: silently no-ops where the API is missing (older
8 * WebViews). The OS releases the lock whenever the document becomes hidden (the
9 * user switches apps), so we re-acquire on `visibilitychange` while still active.
10 * A rejected request (not visible / low battery) is ignored — this is a nicety,
11 * never a hard failure.
12 */
13export function useScreenWakeLock(active: boolean): void {
14 useEffect(() => {
15 if (!active || typeof navigator === 'undefined' || !navigator.wakeLock) return;
16 
17 let sentinel: WakeLockSentinel | null = null;
18 let cancelled = false;
19 
20 const acquire = async (): Promise<void> => {
21 if (cancelled || sentinel || document.visibilityState !== 'visible') return;
22 try {
23 const s = await navigator.wakeLock.request('screen');
24 if (cancelled) { void s.release().catch(() => {}); return; }
25 sentinel = s;
26 // The OS can drop the lock on its own — clear our handle so the next
27 // visibilitychange re-acquires it.
28 s.addEventListener('release', () => { if (sentinel === s) sentinel = null; });
29 } catch {
30 /* request rejected — best-effort, ignore */
31 }
32 };
33 const onVisibility = (): void => { if (document.visibilityState === 'visible') void acquire(); };
34 
35 void acquire();
36 document.addEventListener('visibilitychange', onVisibility);
37 
38 return () => {
39 cancelled = true;
40 document.removeEventListener('visibilitychange', onVisibility);
41 void sentinel?.release().catch(() => {});
42 sentinel = null;
43 };
44 }, [active]);
45}
