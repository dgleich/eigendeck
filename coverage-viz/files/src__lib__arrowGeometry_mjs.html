12export function arrowGeometry(x1, y1, x2, y2, headSize, heads, c1x, c1y, c2x, c2y, points) {
13 const ha = ARROW_HA;
14 // Cubic-Bézier control points (#129): c1 off the start, c2 off the end. When all
15 // four are present the arrow curves; otherwise it's a straight line (unchanged).
16 const curved = c1x != null && c1y != null && c2x != null && c2y != null;
17 // Head angle at each tip = the curve TANGENT there (straight: the line dir).
18 // end head points along c2→end; start head points OUTWARD along c1→start.
19 const endAng = curved ? Math.atan2(y2 - c2y, x2 - c2x) : Math.atan2(y2 - y1, x2 - x1);
20 const startAng = curved ? Math.atan2(y1 - c1y, x1 - c1x) : Math.atan2(y1 - y2, x1 - x2);
21 const atEnd = heads !== 'start' && heads !== 'none'; // default/'end'/'both'
22 const atStart = heads === 'start' || heads === 'both';
23 // tip → base-centre distance. For an arrow shorter than the combined insets,
24 // pulling both endpoints in would make them cross; clamp by the straight-line
25 // length so at worst they meet (never reverse). The curved arc is longer, so
26 // this stays conservative there.
27 const len = Math.hypot(x2 - x1, y2 - y1);
28 const nHeads = (atStart ? 1 : 0) + (atEnd ? 1 : 0);
29 const inset = nHeads > 0
30 ? Math.min(headSize * Math.cos(ha), len / nHeads)
31 : headSize * Math.cos(ha);
32 const tri = (tx, ty, a) => [
33 [tx, ty],
34 [tx - headSize * Math.cos(a - ha), ty - headSize * Math.sin(a - ha)],
35 [tx - headSize * Math.cos(a + ha), ty - headSize * Math.sin(a + ha)],
36 ];
37 const triangles = [];
38 if (atEnd) triangles.push(tri(x2, y2, endAng));
39 if (atStart) triangles.push(tri(x1, y1, startAng));
40 // Pull each headed endpoint back to the head base ALONG its tangent — the stroke
41 // meets the head cleanly on the curve too (keeps the control points fixed).
42 const ex = atEnd ? x2 - Math.cos(endAng) * inset : x2;
43 const ey = atEnd ? y2 - Math.sin(endAng) * inset : y2;
44 const sx = atStart ? x1 - Math.cos(startAng) * inset : x1;
45 const sy = atStart ? y1 - Math.sin(startAng) * inset : y1;
46 if (curved) {
47 const pts = Array.isArray(points) ? points.filter((p) => p && p.x != null && p.y != null) : [];
48 if (pts.length === 0) {
49 return { curved: true, path: `M ${sx} ${sy} C ${c1x} ${c1y} ${c2x} ${c2y} ${ex} ${ey}`, triangles };
50 }
51 // Interior interpolation points (#129): the curve passes THROUGH each point,
52 // smoothly. The two END tangents keep coming from the user handles (c1/c2, so
53 // the heads orient exactly as before); interior knots get automatic
54 // Catmull-Rom tangents (T = (next − prev)/6). No handles on interior points.
55 const K = [{ x: sx, y: sy }, ...pts, { x: ex, y: ey }];
56 const outC = new Array(K.length);
57 const inC = new Array(K.length);
58 outC[0] = { x: c1x, y: c1y };
59 inC[K.length - 1] = { x: c2x, y: c2y };
60 for (let i = 1; i < K.length - 1; i++) {
61 const tx = (K[i + 1].x - K[i - 1].x) / 6, ty = (K[i + 1].y - K[i - 1].y) / 6;
62 outC[i] = { x: K[i].x + tx, y: K[i].y + ty };
63 inC[i] = { x: K[i].x - tx, y: K[i].y - ty };
64 }
65 let d = `M ${sx} ${sy}`;
66 for (let i = 0; i < K.length - 1; i++) {
67 d += ` C ${outC[i].x} ${outC[i].y} ${inC[i + 1].x} ${inC[i + 1].y} ${K[i + 1].x} ${K[i + 1].y}`;
68 }
69 return { curved: true, path: d, triangles };
70 }
71 return {
72 line: { x1: sx, y1: sy, x2: ex, y2: ey },
73 triangles, // each: [[x,y],[x,y],[x,y]]
74 };
75}

77/** Insert a new interior waypoint on the current curve, at the MIDPOINT of its
78 * longest segment, then let the arrow re-smooth. The endpoint handles c1/c2 are
79 * left UNCHANGED — the curve is allowed to change (no shape preservation, no
80 * handle scaling). The new knot lands ON the existing curve (segment t = 0.5) so
81 * it doesn't jump; interior knots get Catmull-Rom auto-tangents. Returns
82 * `{ points, index }`, or null if the arrow isn't curved (the caller materialises
83 * c1/c2 first for a straight arrow).
84 */
85export function arrowInsertPoint(x1, y1, x2, y2, c1x, c1y, c2x, c2y, points) {
86 if (c1x == null || c1y == null || c2x == null || c2y == null) return null;
87 const pts = Array.isArray(points) ? points.filter((p) => p && p.x != null && p.y != null) : [];
88 const K = [{ x: x1, y: y1 }, ...pts, { x: x2, y: y2 }];
89 const n = K.length;
90 // Segment controls exactly as arrowGeometry builds them: user handles at the
91 // ends, Catmull-Rom auto-tangents at the interior knots.
92 const outC = new Array(n), inC = new Array(n);
93 outC[0] = { x: c1x, y: c1y };
94 inC[n - 1] = { x: c2x, y: c2y };
95 for (let i = 1; i < n - 1; i++) {
96 const tx = (K[i + 1].x - K[i - 1].x) / 6, ty = (K[i + 1].y - K[i - 1].y) / 6;
97 outC[i] = { x: K[i].x + tx, y: K[i].y + ty };
98 inC[i] = { x: K[i].x - tx, y: K[i].y - ty };
99 }
100 // Longest segment (chord length) hosts the new knot, at its midpoint on the curve.
101 let seg = 0, segLen = -1;
102 for (let i = 0; i < n - 1; i++) {
103 const L = Math.hypot(K[i + 1].x - K[i].x, K[i + 1].y - K[i].y);
104 if (L > segLen) { segLen = L; seg = i; }
105 }
106 const P0 = K[seg], P1 = outC[seg], P2 = inC[seg + 1], P3 = K[seg + 1];
107 // Cubic at t = 0.5 → Bézier weights (1,3,3,1)/8.
108 const np = {
109 x: Math.round((P0.x + 3 * P1.x + 3 * P2.x + P3.x) / 8),
110 y: Math.round((P0.y + 3 * P1.y + 3 * P2.y + P3.y) / 8),
111 };
112 const newPts = [...pts];
113 newPts.splice(seg, 0, np);
114 return { points: newPts, index: seg };
115}

122/** SVG-STRING inner `<g>` for an arrow — the inset line + head polygon(s) wrapped
123 * in `<g [opacity]>`. Shared by the HTML/string render targets (exportCore HTML
124 * export + App.tsx print-to-PDF), which built this identically. The caller
125 * supplies its own `<svg>` wrapper. */
126export function arrowSvgInner(geo, color, strokeWidth, opacity) {
127 // Escape values spliced into SVG attributes — a crafted color/strokeWidth reaching
128 // the exported artifact must not create a new attribute/tag (audit C-2 export). geo
129 // path/coords are computed numerics (a crafted coord becomes NaN, not markup), but
130 // escaping is uniform and cheap.
131 const c = escAttr(color), sw = escAttr(strokeWidth);
132 const op = opacity != null && opacity < 1 ? ` opacity="${escAttr(opacity)}"` : '';
133 const stroke = geo.path
134 ? `<path d="${escAttr(geo.path)}" fill="none" stroke="${c}" stroke-width="${sw}"/>`
135 : `<line x1="${escAttr(geo.line.x1)}" y1="${escAttr(geo.line.y1)}" x2="${escAttr(geo.line.x2)}" y2="${escAttr(geo.line.y2)}" stroke="${c}" stroke-width="${sw}"/>`;
136 const heads = geo.triangles.map((t) => `<polygon points="${escAttr(triPoints(t))}" fill="${c}"/>`).join('');
137 return `<g${op}>${stroke}${heads}</g>`;
138}

140/** Bounding box over endpoints + head corners (editor hit-area), padded. Includes
141 * the Bézier control points when present — the curve stays within the convex hull
142 * of {endpoints, control points}, so this bounds it. */
143export function arrowBBox(x1, y1, x2, y2, headSize, heads, pad, c1x, c1y, c2x, c2y, points) {
144 const g = arrowGeometry(x1, y1, x2, y2, headSize, heads, c1x, c1y, c2x, c2y, points);
145 const xs = [x1, x2], ys = [y1, y2];
146 if (c1x != null && c1y != null && c2x != null && c2y != null) { xs.push(c1x, c2x); ys.push(c1y, c2y); }
147 if (Array.isArray(points)) for (const p of points) { if (p && p.x != null) { xs.push(p.x); ys.push(p.y); } }
148 for (const t of g.triangles) for (const p of t) { xs.push(p[0]); ys.push(p[1]); }
149 pad = pad || 0;
150 return {
151 minX: Math.min(...xs) - pad, minY: Math.min(...ys) - pad,
152 maxX: Math.max(...xs) + pad, maxY: Math.max(...ys) + pad,
153 };
