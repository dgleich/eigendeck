1//! Native NSPasteboard access for Mac UTIs the webview filters out.
2//!
3//! WebKit's clipboardData / navigator.clipboard.read() only expose a
4//! small allowlist of MIME types (text/plain, text/html, image/png).
5//! Microsoft Office puts SVG on the clipboard as
6//! com.microsoft.image-svg-xml; Adobe puts PDF as com.adobe.pdf;
7//! Apple's own SVG UTI is public.svg-image. None of those reach JS.
8//!
9//! These two Tauri commands read the raw NSPasteboard via objc2-app-kit
10//! so the paste handler can see and grab those formats. Linux/Windows
11//! builds return empty/None — we only need this on macOS.
12//!
13//! Pasteboard access is dispatched to the main thread (objc2-app-kit
14//! API requires MainThreadMarker) using the same Tauri
15//! run_on_main_thread + mpsc::channel pattern as show_unsaved_dialog
16//! in lib.rs.
17 
18/// List all UTIs currently on the general pasteboard. Empty Vec if
19/// nothing's there. Used by the paste handler to decide whether to
20/// fetch via this native path or fall through to the web clipboard.
21#[tauri::command]
22pub fn pasteboard_list_types(_app: tauri::AppHandle) -> Result<Vec<String>, String> {
23 #[cfg(target_os = "macos")]
24 {
25 use std::sync::mpsc;
26 let (tx, rx) = mpsc::channel::<Result<Vec<String>, String>>();
27 let _ = _app.run_on_main_thread(move || {
28 let _ = tx.send(mac_pasteboard_list_types());
29 });
30 rx.recv().unwrap_or_else(|_| Err("main-thread dispatch failed".into()))
31 }
32 #[cfg(not(target_os = "macos"))]
33 {
34 Ok(Vec::new())
35 }
36}

143#[cfg(target_os = "macos")]
144fn mac_pasteboard_list_types() -> Result<Vec<String>, String> {
145 use objc2_app_kit::NSPasteboard;
146 
147 let pb = NSPasteboard::generalPasteboard();
148 match pb.types() {
149 Some(arr) => {
150 let mut out = Vec::with_capacity(arr.len());
151 for t in arr.iter() {
152 out.push(t.to_string());
153 }
154 Ok(out)
155 }
156 None => Ok(Vec::new()),
157 }
158}

171#[cfg(all(target_os = "macos", debug_assertions))]
172fn mac_pasteboard_dump() -> String {
173 use objc2_app_kit::NSPasteboard;
174 let pb = NSPasteboard::generalPasteboard();
175 let mut lines = vec![String::from("=== general pasteboard types ===")];
176 match pb.types() {
177 Some(arr) => {
178 for t in arr.iter() {
179 let uti = t.to_string();
180 let detail = match mac_pasteboard_read_type(&uti) {
181 Ok(Some(bytes)) => {
182 let len = bytes.len();
183 // Show a text preview for text-ish payloads; binary (png/tiff)
184 // just reports its length (the lossy preview would be noise).
185 let printable = bytes
186 .iter()
187 .take(64)
188 .filter(|&&b| b == b'\n' || b == b'\t' || (0x20..=0x7e).contains(&b))
189 .count();
190 if printable * 100 >= bytes.len().min(64) * 80 {
191 let text = String::from_utf8_lossy(&bytes);
192 let snippet: String = text.chars().take(300).collect();
193 format!("{len} bytes | {}", snippet.replace('\n', "\\n"))
194 } else {
195 format!("{len} bytes | <binary>")
196 }
197 }
198 Ok(None) => "no data".into(),
199 Err(e) => format!("read error: {e}"),
200 };
201 lines.push(format!(" {uti}\n {detail}"));
202 }
203 }
204 None => lines.push(" (no types)".into()),
205 }
206 lines.join("\n")
207}
