306/** Build a minimal PDF from JPEG images (one per page) */
307/** @internal — kept for future direct PDF export */
308export function buildPdf(images: Uint8Array[], pageW: number, pageH: number): Uint8Array {
309 // PDF uses points (72 per inch). Scale 1920x1080 to fit standard proportions.
310 const ptW = 1920 * 0.5; // 960pt = ~13.3in
311 const ptH = 1080 * 0.5; // 540pt = ~7.5in
312 
313 const encoder = new TextEncoder();
314 const parts: (Uint8Array | string)[] = [];
315 const offsets: number[] = [];
316 let pos = 0;
317 
318 function write(s: string) { parts.push(s); pos += encoder.encode(s).length; }
319 function writeBin(b: Uint8Array) { parts.push(b); pos += b.length; }
320 function objStart(id: number) { offsets[id] = pos; write(`${id} 0 obj\n`); }
321 
322 write('%PDF-1.4\n%\xFF\xFF\xFF\xFF\n');
323 
324 // Object 1: Catalog
325 objStart(1);
326 write('<< /Type /Catalog /Pages 2 0 R >>\nendobj\n');
327 
328 // Object 2: Pages (references all page objects)
329 const pageObjStart = 3;
330 const firstImageObj = pageObjStart + images.length;
331 objStart(2);
332 const kids = images.map((_, i) => `${pageObjStart + i} 0 R`).join(' ');
333 write(`<< /Type /Pages /Kids [${kids}] /Count ${images.length} >>\nendobj\n`);
334 
335 // Page objects and image objects
336 for (let i = 0; i < images.length; i++) {
337 const pageObj = pageObjStart + i;
338 const imgObj = firstImageObj + i;
339 const imgBytes = images[i];
340 
341 // Page object
342 objStart(pageObj);
343 if (imgBytes.length > 0) {
344 // Page with image
345 const contentsObj = firstImageObj + images.length + i;
346 write(`<< /Type /Page /Parent 2 0 R /MediaBox [0 0 ${ptW} ${ptH}] ` +
347 `/Resources << /XObject << /Img${i} ${imgObj} 0 R >> >> ` +
348 `/Contents ${contentsObj} 0 R >>\nendobj\n`);
349 } else {
350 // Blank page (failed capture)
351 write(`<< /Type /Page /Parent 2 0 R /MediaBox [0 0 ${ptW} ${ptH}] >>\nendobj\n`);
352 }
353 }
354 
355 // Image XObjects
356 for (let i = 0; i < images.length; i++) {
357 const imgObj = firstImageObj + i;
358 const imgBytes = images[i];
359 if (imgBytes.length === 0) { objStart(imgObj); write('<< >>\nendobj\n'); continue; }
360 
361 objStart(imgObj);
362 write(`<< /Type /XObject /Subtype /Image /Width ${pageW} /Height ${pageH} ` +
363 `/ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /DCTDecode /Length ${imgBytes.length} >>\nstream\n`);
364 writeBin(imgBytes);
365 write('\nendstream\nendobj\n');
366 }
367 
368 // Content streams (draw image on each page)
369 for (let i = 0; i < images.length; i++) {
370 const contentsObj = firstImageObj + images.length + i;
371 if (images[i].length === 0) { objStart(contentsObj); write('<< /Length 0 >>\nstream\nendstream\nendobj\n'); continue; }
372 const stream = `q ${ptW} 0 0 ${ptH} 0 0 cm /Img${i} Do Q`;
373 objStart(contentsObj);
374 write(`<< /Length ${stream.length} >>\nstream\n${stream}\nendstream\nendobj\n`);
375 }
376 
377 // Cross-reference table
378 const xrefPos = pos;
379 const totalObjs = firstImageObj + images.length * 2;
380 write(`xref\n0 ${totalObjs + 1}\n`);
381 write('0000000000 65535 f \n');
382 for (let i = 1; i <= totalObjs; i++) {
383 const off = offsets[i] || 0;
384 write(`${String(off).padStart(10, '0')} 00000 n \n`);
385 }
386 
387 write(`trailer\n<< /Size ${totalObjs + 1} /Root 1 0 R >>\n`);
388 write(`startxref\n${xrefPos}\n%%EOF\n`);
389 
390 // Concatenate all parts
391 let totalLen = 0;
392 for (const p of parts) totalLen += typeof p === 'string' ? encoder.encode(p).length : p.length;
393 const result = new Uint8Array(totalLen);
394 let offset = 0;
395 for (const p of parts) {
396 const bytes = typeof p === 'string' ? encoder.encode(p) : p;
397 result.set(bytes, offset);
398 offset += bytes.length;
399 }
400 return result;
401}

403/** Compute relative path from eigendeck's directory to a file */
404export function relPath(projectPath: string | null, fullPath: string): string {
405 if (!projectPath) return fullPath.split('/').pop() || 'file';
406 const dir = projectPath.replace(/\/[^/]+$/, '');
407 const dirParts = dir.split('/');
408 const fileParts = fullPath.split('/');
409 // Find common prefix
410 let common = 0;
411 while (common < dirParts.length && common < fileParts.length && dirParts[common] === fileParts[common]) {
412 common++;
413 }
414 // Build relative path: ../ for each remaining dir part, then remaining file parts
415 const ups = dirParts.length - common;
416 const rel = [...Array(ups).fill('..'), ...fileParts.slice(common)].join('/');
417 return rel || fullPath.split('/').pop() || 'file';
418}
