9/** Decode a `file://` URL (or a bare path) to an absolute filesystem path.
10 * Percent-decodes, strips the scheme + host, and normalizes. Returns null for a
11 * non-file scheme (http:, data:, …) so remote refs never reach the disk read. */
12export function fileUrlToPath(ref: string): string | null {
13 const s = (ref || '').trim();
14 if (!s) return null;
15 // A Windows drive path (C:\… or C:/…) looks like a scheme — accept it as a bare
16 // path before the scheme check below claims `C:` as a scheme.
17 if (/^[a-zA-Z]:[\\/]/.test(s)) return s;
18 if (/^[a-z][a-z0-9+.-]*:/i.test(s)) {
19 // Has a scheme — must be file:.
20 const m = s.match(/^file:\/\/([^/]*)(\/.*)$/i);
21 if (!m) return null;
22 // m[1] is an (ignored) host/authority; m[2] is the absolute path.
23 try { return decodeURIComponent(m[2]); } catch { return null; }
24 }
25 // No scheme → treat as a bare absolute path (NSFilenames / CF_HDROP give these).
26 return s.startsWith('/') || /^[a-zA-Z]:[\\/]/.test(s) ? s : null;
27}

29function decodeXmlEntities(s: string): string {
30 return s
31 .replace(/&lt;/g, '<').replace(/&gt;/g, '>')
32 .replace(/&quot;/g, '"').replace(/&apos;/g, "'")
33 .replace(/&#(\d+);/g, (_, d) => String.fromCodePoint(Number(d)))
34 .replace(/&amp;/g, '&'); // last, so &amp;lt; doesn't double-decode
35}

37/** Parse macOS `NSFilenamesPboardType` (a plist `<array>` of `<string>` POSIX
38 * paths or file:// URLs) into file paths. This is what Finder populates with the
39 * REAL path on a file copy — `public.file-url` there is a /.file/id= reference
40 * URL with no usable extension. Being an array, this also covers multi-file. */
41export function parseNSFilenames(plistXml: string): string[] {
42 const out: string[] = [];
43 const re = /<string>([\s\S]*?)<\/string>/g;
44 let m: RegExpExecArray | null;
45 while ((m = re.exec(plistXml || ''))) {
46 const val = decodeXmlEntities(m[1].trim());
47 if (!val) continue;
48 const p = fileUrlToPath(val);
49 if (p) out.push(p);
50 }
51 return out;
52}

54/** Parse a `text/uri-list` payload (RFC 2483: CRLF-separated URIs, `#` comment
55 * lines ignored) into file paths. Non-file URIs are dropped. */
56export function parseUriList(text: string): string[] {
57 return (text || '')
58 .split(/\r?\n/)
59 .map((l) => l.trim())
60 .filter((l) => l && !l.startsWith('#'))
61 .map(fileUrlToPath)
62 .filter((p): p is string => !!p);
63}

65/** Parse GNOME's `x-special/gnome-copied-files` (first line is `copy`/`cut`, then
66 * file:// URIs) into file paths. */
67export function parseGnomeCopiedFiles(text: string): string[] {
68 const lines = (text || '').split(/\r?\n/);
69 // Drop a leading `copy`/`cut` op line if present.
70 const start = /^(copy|cut)$/i.test((lines[0] || '').trim()) ? 1 : 0;
71 return lines.slice(start)
72 .map((l) => l.trim())
73 .filter(Boolean)
74 .map(fileUrlToPath)
75 .filter((p): p is string => !!p);
76}
