215/// The single external-file read primitive for asset security (docs/ASSETS-SECURITY.md).
216/// Resolves `path` to its REAL on-disk target — following every symlink and
217/// normalizing `.`/`..` via `canonicalize` — then reads that resolved target and
218/// returns its canonical path alongside the bytes. Callers gate on the RESOLVED
219/// path + bytes (so a symlink `a.png -> ~/.ssh/id_rsa` is judged as `id_rsa`), and
220/// re-run the gate on every read. Rejects non-regular files (dirs, fifos, devices)
221/// and anything over a sane size cap. Errors (missing/too-big/not-a-file) are
222/// returned so the caller can treat them as "unreadable → not watchable".
223///
224/// `max_bytes`: read at most this many bytes (a bounded "sniff" read). None reads the
225/// whole file (the read path, which needs the bytes). The whole-file SIZE is still
226/// checked against the cap first — so an oversized file is rejected identically whether
227/// the caller wants a sniff or the full bytes, keeping the report's verdict in lockstep
228/// with the read gate. The type gate inspects at most a small prefix (except .ipynb,
229/// which the caller re-reads in full), so a bounded read yields the same verdict.
230/// Command wrapper: restrict the arbitrary-path read to the windows that legitimately
231/// resolve deck-referenced files — the main editor, the presenter (both render assets;
232/// the CLI export runs in the main window), and the SECURITY window (securityReport.ts
233/// → resolveAndGate resolves every linked file to show its gate decision). Settings is
234/// excluded (it renders no assets). NOTE (audit C-3, read-side): the TRUST decision on
235/// the resolved path is still made in JS (assetGate.ts) AFTER the bytes cross into the
236/// webview; this window allowlist is blast-radius reduction, not a Rust-side trust gate.
237/// Moving the trust decision into Rust (or passing assets by db id / app-issued grant)
238/// is deferred read-side hardening.
239#[tauri::command]
240fn resolve_and_read(
241 window: tauri::WebviewWindow,
242 path: String,
243 max_bytes: Option<u64>,
244) -> Result<ResolvedRead, String> {
245 fscmds::require_windows(&window, &["main", "presenter", "security"])?;
246 resolve_and_read_impl(path, max_bytes)
247}

249fn resolve_and_read_impl(path: String, max_bytes: Option<u64>) -> Result<ResolvedRead, String> {
250 use std::io::Read;
251 const MAX_BYTES: u64 = 512 * 1024 * 1024;
252 let canonical = std::fs::canonicalize(&path)
253 .map_err(|e| format!("cannot resolve {}: {}", path, e))?;
254 let meta = std::fs::metadata(&canonical).map_err(|e| e.to_string())?;
255 if !meta.is_file() {
256 return Err(format!("not a regular file: {}", canonical.display()));
257 }
258 let size = meta.len();
259 if size > MAX_BYTES {
260 return Err(format!("file too large ({} bytes): {}", size, canonical.display()));
261 }
262 let to_read = max_bytes.map(|m| m.min(size)).unwrap_or(size);
263 let mut bytes = Vec::with_capacity(to_read as usize);
264 std::fs::File::open(&canonical)
265 .map_err(|e| e.to_string())?
266 .take(to_read)
267 .read_to_end(&mut bytes)
268 .map_err(|e| e.to_string())?;
269 Ok(ResolvedRead {
270 canonical_path: canonical.to_string_lossy().to_string(),
271 bytes,
272 size,
273 })
274}

611 }

634 }

636 Ok(true)
637 }
638 }

640 #[cfg(not(target_os = "macos"))]
