18export function coverHtml(el, resolvedSlideBg, len, theme) {
19 const d = describeCover(el, resolvedSlideBg, theme);

22}

24/** arrow — inset line + head triangle(s) as an absolute 100% SVG overlay.
25 * `opts.viewBox` is set by inch-scaled targets (the PDF export) so the px arrow
26 * coordinates map into the inch container; px targets (HTML export) omit it.
27 * pointer-events:none so the overlay never blocks the elements beneath it. */
28export function arrowSvgHtml(el, opts = {}) {
29 const a = describeArrow(el, opts.theme);
30 const vb = opts.viewBox ? `viewBox="${escAttr(opts.viewBox)}" ` : '';
31 return `<svg ${vb}style="position:absolute;top:0;left:0;width:100%;height:100%;pointer-events:none;overflow:visible;">${arrowSvgInner(a.geo, a.color, a.strokeWidth, a.opacity)}</svg>`;
32}

34/** image — a positioned <img> with the shared visual styles (drop-shadow / corner
35 * radius / opacity / rotation from imageVisuals — IDENTICAL across targets). The
36 * `src` is already resolved (data URL / cached PNG); `len` formats the box. */
37export function imageHtml(src, el, len, extraAttrs = '') {
38 const p = el.position;
39 const iv = imageVisuals(el);
40 const styles = [
41 `position:absolute`, `left:${escAttr(len(p.x))}`, `top:${escAttr(len(p.y))}`,
42 `width:${escAttr(len(p.width))}`, `height:${escAttr(len(p.height))}`, `object-fit:contain`,
43 ];
44 if (iv.shadow) styles.push(`filter:${escAttr(iv.shadow)}`);
45 if (iv.borderRadius) styles.push(`border-radius:${escAttr(iv.borderRadius)}px`);
46 if (iv.opacity != null) styles.push(`opacity:${escAttr(iv.opacity)}`);
47 if (iv.transform) styles.push(`transform:${escAttr(iv.transform)}`);
48 // `src` may be omitted for the single-store export path: the image is emitted as
49 // a `data-asset-id` placeholder (via extraAttrs) and painted by the in-body
50 // loader from the embedded #eigendeck-deck block, so its bytes aren't inlined twice.
51 // (extraAttrs is escaped at its source — see exportCore's data-asset-id.)
52 const srcAttr = src ? ` src="${escAttr(src)}"` : '';
53 const extra = extraAttrs ? ` ${extraAttrs}` : '';
54 return `<img${srcAttr}${extra} style="${styles.join(';')};" />`;
55}
