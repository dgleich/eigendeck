1// Single source of truth for the "insertable element" actions.
2//
3// Both the editor toolbar (the "+ Title / + Body / …" buttons) and the
4// native macOS "Insert" menu are driven from this list. The toolbar
5// hides any item whose id is in the `hiddenToolbarItems` preference; the
6// Insert menu ALWAYS shows everything (that's the whole point — toolbar
7// customization never removes an action, it just declutters the toolbar).
8//
9// The actual handlers live in App.tsx (`runInsert`) because they need
10// component context (store, file dialogs, the video modal). This module
11// is metadata only: stable ids, labels, tooltips, and toolbar grouping.
12 
13export type InsertGroup = 'text' | 'objects' | 'embeds';
14 
15export interface InsertItem {
16 /** Stable id. Used as the toolbar-visibility key AND, prefixed with
17 * `insert-`, as the native menu item id. Never rename without a
18 * migration — it's persisted in the hiddenToolbarItems preference. */
19 id: string;
20 /** Toolbar button text (rendered as "+ {label}"). */
21 label: string;
22 /** Button tooltip / accessibility title. */
23 tooltip: string;
24 /** Toolbar group — items render in groups separated by a divider. */
25 group: InsertGroup;
26 /** Hidden from the editor toolbar by default (still always in the Insert
27 * menu, and re-addable via Settings → Toolbar buttons). Seeds the default
28 * of the `hiddenToolbarItems` preference. Used for power-user actions most
29 * users shouldn't see out of the box — currently just `demo`, which needs
30 * infrastructure most users don't have. */
31 defaultHidden?: boolean;
32}
