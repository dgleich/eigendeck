1// HTML-escape a string for embedding in an iframe `srcdoc` attribute. Security-
2// relevant (it's what keeps demo/notebook HTML from breaking out of the srcdoc),
3// so it lives in ONE place — shared by the HTML export (exportCore.mjs) and the
4// notebook export (notebookExport.tsx), which each used to keep their own copy.
5export function htmlEscapeForSrcdoc(s) {
6 return s
7 .replace(/&/g, '&amp;')
8 .replace(/"/g, '&quot;')
9 .replace(/</g, '&lt;')
10 .replace(/>/g, '&gt;');
11}

20/** URL policy for an exported href/src. Attribute-escaping does NOT stop a
21 * `javascript:`/`vbscript:` URL (no breakout chars), so restrict schemes: only
22 * http(s) — and `data:` for images when `allowData` — pass; anything else returns
23 * '' so the exported link/media is inert. Protocol-relative `//host` → https. */
24export function safeExportUrl(url, { allowData = false } = {}) {
25 if (typeof url !== 'string') return '';
26 const s = url.trim();
27 if (/^https?:\/\//i.test(s)) return s;
28 if (/^\/\//.test(s)) return 'https:' + s;
29 if (allowData && /^data:/i.test(s)) return s;
30 return '';
31}
