111// The blob-cache identity for ONE mount. The `prefix` identifies the mount
112// (asset + role/piece + capture + net-block); the trailing theme/font hashes vary
113// so a theme switch / async-fonts-resolving / internet-toggle supersedes the prior
114// blob for the same prefix. ONE source of truth for the async builder AND the sync
115// peek so they can't drift.
116function demoDocCacheKeyParts(assetId: string, opts: DemoMountOpts): { prefix: string; key: string } {
117 const { hash = '', channelKey, varsCss = '', fontFacesCss = '', capture = false, blockInternet = false } = opts;
118 const prefix = `${assetId} ${hash} ${channelKey} ${capture ? 'C' : ''}${blockInternet ? 'B' : ''} `;
119 return { prefix, key: prefix + `${hashString(varsCss)} ${hashString(fontFacesCss)}` };
120}

122/** Synchronous read of an ALREADY-BUILT demo blob URL, or undefined on a miss.
123 * Lets a re-mounted demo set its iframe `src` on the FIRST render instead of
124 * waiting an async tick for getDemoDocumentUrl to resolve the same cached blob —
125 * the synchronous availability the old same-origin useDemoUrl had. Positive hits

127export function peekDemoDocumentUrl(assetId: string | undefined, opts: DemoMountOpts): string | undefined {
128 if (!assetId || blockedCache.has(assetId)) return undefined;
129 return docBlobCache.get(demoDocCacheKeyParts(assetId, opts).key);

132/** Build (or reuse) the opaque-origin demo document blob URL. Returns null when
133 * the bytes aren't a marked demo, undefined while loading. */
134export async function getDemoDocumentUrl(assetId: string | undefined, opts: DemoMountOpts): Promise<string | null> {
135 if (!assetId) return null;
136 if (blockedCache.has(assetId)) return null;
137 const { hash = '', channelKey, varsCss = '', fontFacesCss = '', capture = false, blockInternet = false } = opts;
138 // The prefix identifies ONE mount (asset+role/piece+capture+net-block); only the
139 // trailing theme/font hashes vary. So a theme switch, the async fonts resolving
140 // from '', OR the internet toggle flipping supersedes the prior blob for the same
141 // prefix — revoke it (else it leaks, with the inlined modern-screenshot bytes when
142 // capture is on). Other pieces of the same demo use a different `hash` → different
143 // prefix → untouched.
144 const { prefix, key } = demoDocCacheKeyParts(assetId, opts);
145 const existing = docBlobCache.get(key);
146 if (existing) return existing;
147 for (const [k, u] of docBlobCache) {
148 if (k !== key && k.startsWith(prefix)) { URL.revokeObjectURL(u); docBlobCache.delete(k); }
149 }
150 const raw = await fetchRawDemo(assetId);
151 if (raw == null) return null;
152 // Only splice the (megabyte-scale, base64) @font-face faces into demos that
153 // actually name a deck font — otherwise every mount re-parses fonts the demo
154 // can't use. The theme vars still go in (cheap, and a demo may use the colors).
155 const effFontFaces = demoReferencesFonts(raw, fontFacesCss) ? fontFacesCss : '';
156 const withBridge = injectDemoBridge(raw, hash ? `#${hash}` : '', channelKey, { capture, net: demoNet(blockInternet, raw) });
157 const doc = injectDemoThemeIntoHtml(withBridge, effFontFaces, varsCss);
158 const url = URL.createObjectURL(new Blob([doc], { type: 'text/html' }));
159 docBlobCache.set(key, url);
160 return url;
161}

172export function buildIsolatedOutputUrl(html: string, opts: {
173 channelKey: string; varsCss?: string; fontFacesCss?: string; blockInternet?: boolean;
174}): string {
175 const { channelKey, varsCss = '', fontFacesCss = '', blockInternet = false } = opts;
176 const contentSig = hashString(html);
177 const key = `${contentSig} ${channelKey} ${blockInternet ? 'B' : ''} ${hashString(varsCss)} ${hashString(fontFacesCss)}`;
178 const existing = outputBlobCache.get(key);
179 if (existing) return existing;
180 const prefix = `${contentSig} ${channelKey} `;
181 for (const [k, u] of outputBlobCache) {
182 if (k !== key && k.startsWith(prefix)) { URL.revokeObjectURL(u); outputBlobCache.delete(k); }
183 }
184 const page = `<!DOCTYPE html><html><head><meta charset="utf-8"><style>html,body{margin:0;padding:0;background:transparent;}</style></head><body>${html}</body></html>`;
185 const withBridge = injectDemoBridge(page, '', channelKey, { reportSize: true, net: demoNet(blockInternet, html) });
186 const doc = injectDemoThemeIntoHtml(withBridge, fontFacesCss, varsCss);
187 const url = URL.createObjectURL(new Blob([doc], { type: 'text/html' }));
188 outputBlobCache.set(key, url);
189 return url;
190}

201/** React hook: the opaque-origin demo document URL, rebuilt when inputs change. */
202export function useDemoDoc(assetId: string | undefined, opts: DemoMountOpts): string | null | undefined {
203 const { hash, channelKey, varsCss, fontFacesCss, capture } = opts;
204 const blockInternet = useDemoInternetBlocked(assetId);
205 // Seed synchronously from an already-built blob (re-mount / same demo on another
206 // slide) so the iframe `src` is present on the first render — no async gap. Cold
207 // mounts still resolve via the effect below.
208 const [url, setUrl] = useState<string | null | undefined>(() =>
209 peekDemoDocumentUrl(assetId, { hash, channelKey, varsCss, fontFacesCss, capture, blockInternet }));
210 const [refresh, setRefresh] = useState(0);
211 // Reload when the underlying asset bytes change on disk (file-watch / "Reload
212 // from disk now" fire eigendeck:asset-changed). invalidateDemoDoc drops the
213 // cached raw bytes + blobs so the refetch below rebuilds from the new content —
214 // the behavior the old same-origin useDemoUrl had.
215 useEffect(() => {
216 if (!assetId) return;
217 const onChanged = (e: Event) => {
218 if ((e as CustomEvent).detail?.assetId === assetId) { invalidateDemoDoc(assetId); setRefresh((k) => k + 1); }
219 };
220 window.addEventListener('eigendeck:asset-changed', onChanged);
221 return () => window.removeEventListener('eigendeck:asset-changed', onChanged);
222 }, [assetId]);
223 useEffect(() => {
224 if (!assetId) { setUrl(undefined); return; }
225 let alive = true;
226 getDemoDocumentUrl(assetId, { hash, channelKey, varsCss, fontFacesCss, capture, blockInternet })
227 .then((r) => { if (alive) setUrl(r); });
228 return () => { alive = false; };
229 }, [assetId, hash, channelKey, varsCss, fontFacesCss, capture, blockInternet, refresh]);
230 // Poke the rAF pump once the src is set: this effect runs after the commit that
231 // mounts the iframe, so the frame is in the DOM and the pump resumes on the next
232 // frame instead of waiting out its idle backstop (kills the "controls, then the
233 // animation" pop for demos that draw in rAF).
234 useEffect(() => { if (url) wakeDemoRafPump(); }, [url]);
235 return url;
236}

245 slides?: Array<{ titleFont?: string; bodyFont?: string; hypeFont?: string }>;

247function fontDeckKey(presentation: FontDeck): string {
248 return JSON.stringify({
249 c: presentation.config,
250 s: (presentation.slides || []).map((s) => [s.titleFont, s.bodyFont, s.hypeFont]),

253export async function deckFontFacesCss(presentation: FontDeck): Promise<string> {
254 const key = fontDeckKey(presentation);
255 if (fontCssCache && fontCssCache.key === key) return fontCssCache.css;
256 const css = await buildEmbeddedFontFacesCSS(presentation);
257 fontCssCache = { key, css };
258 return css;

292// Opaque origin blocks the parent from reaching a demo's DOM, so the parent asks
293// the in-demo bridge to rasterize itself and post the PNG back (docs/DEMO-PLATFORM.md §8).
294let captureSeq = 0;
295export function requestDemoCapture(
296 iframe: HTMLIFrameElement,
297 opts: { width: number; height: number; backgroundColor?: string },
298 timeoutMs = 5000,
299): Promise<string | null> {
300 const cw = iframe.contentWindow;
301 if (!cw) return Promise.resolve(null);
302 const id = `cap${++captureSeq}`;
303 return new Promise((resolve) => {
304 const onMsg = (e: MessageEvent) => {

309 };

312 window.addEventListener('message', onMsg);

317// --- demo host infrastructure (shared by every window that renders demos) ----
318// Both the relay and the rAF pump are per-window infra. Rather than wire them into
319// each window entry (App, the presenter window, …) — which is exactly how the
320// presenter kept missing demo features — any component that HOSTS demos calls
321// useDemoHost(). Ref-counted so overlapping hosts share one install and it tears
322// down when the last unmounts. This is the single source of truth for "a window
323// is showing demos, so it needs the relay + pump".
324let hostRefs = 0;
325let hostCleanup: (() => void) | null = null;
326export function useDemoHost(): void {
327 useEffect(() => {
328 hostRefs++;
329 if (hostRefs === 1) {
330 const relay = installDemoRelay();
331 const pump = installRafPump();
332 hostCleanup = () => { relay(); pump(); };
333 }
334 return () => {
335 hostRefs--;
336 if (hostRefs === 0 && hostCleanup) { hostCleanup(); hostCleanup = null; }
337 };
338 }, []);
339}

341// --- parent-side rAF pump ----------------------------------------------------
342// WebKit throttles rAF to 30fps in cross-origin (opaque) demo frames until they
343// see a trusted interaction (docs/DEMO-PLATFORM.md §16). The parent top document
344// is NOT throttled, so run one 60fps rAF loop and post a tick to every demo frame
345// each frame; the bridge fires the demo's rAF callbacks on each tick.
346//
347// When no demo is on screen the loop must NOT spin at 60fps (a text-only slide
348// would burn the main thread). It used to nap on a fixed 100ms setTimeout, which
349// delayed a freshly-mounted demo's first tick by up to 100ms (its rAF-driven
350// content "popped in" AFTER its synchronously-rendered controls). Now a demo mount
351// EXPLICITLY pokes the pump (wakeDemoRafPump, called from useDemoDoc once the src
352// is set + committed), so it resumes within a frame of the mount. The idle loop is
353// just a cheap backstop poll — deliberately NOT a MutationObserver, which fired on
354// every DOM mutation and measurably slowed initial render / thumbnail building on
355// element-heavy decks. Single-install is guaranteed by useDemoHost's ref-count.
356let wakePumpFn: (() => void) | null = null;
357/** A demo mounted — resume the pump now instead of waiting out the idle poll. */
358export function wakeDemoRafPump(): void { wakePumpFn?.(); }
359function installRafPump(): () => void {
360 if (typeof window === 'undefined') return () => {};
361 let running = true;
362 let idle: ReturnType<typeof setTimeout> | undefined;
363 let scheduled = false;
364 const schedule = () => { if (scheduled || !running) return; scheduled = true; requestAnimationFrame(pump); };
365 const pump = (t: number) => {
366 scheduled = false;
367 if (!running) return;
368 const frames = document.querySelectorAll('iframe.el-demo-frame');
369 if (frames.length === 0) { idle = setTimeout(schedule, 250); return; } // backstop only
370 for (const f of frames) {
371 const cw = (f as HTMLIFrameElement).contentWindow;
372 if (cw) { try { cw.postMessage({ __eigendeck: 1, type: 'raf-tick', t }, '*'); } catch { /* opaque */ } }
373 }
374 schedule();
375 };
376 wakePumpFn = () => { if (!running) return; if (idle) { clearTimeout(idle); idle = undefined; } schedule(); };
377 schedule();
378 return () => { running = false; if (idle) clearTimeout(idle); wakePumpFn = null; };
379}

381// --- parent-side relay -------------------------------------------------------
382// Opaque-origin demos can't share a BroadcastChannel, so the injected bootstrap
383// posts {__bc, payload} to window.parent; this relay fans each message out to
384// every OTHER demo iframe (sender excluded). Receivers filter by their own
385// channel name, so per-demo/per-instance keying happens on the receive side
386// (same as the export relay). Single-install is guaranteed by useDemoHost.
387function installDemoRelay(): () => void {
388 if (typeof window === 'undefined') return () => {};
389 const onMessage = (e: MessageEvent) => {
390 const d = e.data as { __bc?: string } | undefined;
391 if (!d || !d.__bc) return;
392 const frames = document.querySelectorAll('iframe.el-demo-frame');
393 frames.forEach((f) => {
394 const cw = (f as HTMLIFrameElement).contentWindow;
395 if (cw && cw !== e.source) { try { cw.postMessage(d, '*'); } catch { /* opaque; ignore */ } }
396 });
397 };
398 window.addEventListener('message', onMessage);
399 return () => window.removeEventListener('message', onMessage);
