18// A clipboard clip is untrusted: any app can put HTML on the clipboard carrying the
19// public, unauthenticated eigendeck marker (clipboardModel), so decoded elements can
20// hold arbitrary html AND arbitrary properties (fontFamily/color/geometry). Run the
21// same normalize boundary as deck open / undo seeding / history restore (audit H-1 +
22// C-2): sanitize html and DROP any element with an out-of-safe-shape property. Mutates
23// html in place; returns only the elements safe to install.
24function normalizeClipElements(els: readonly SlideElement[] | undefined): SlideElement[] {
25 return (els || []).filter((el) => normalizeUntrustedElement(el) !== null);
26}

28/** Create the cross-slide animation LINK between a freshly-pasted element and its
29 * source (shared linkId), IF the source still exists. Shared by the element
30 * paste (pasteInternalClip) and the asset (image) paste. No-op when the source
31 * slide/element is gone. */
32export function linkPastedToSource(pastedId: string, fromSlideId: string | undefined, sourceId: string): void {
33 const state = usePresentationStore.getState();
34 const srcSlideIdx = state.presentation.slides.findIndex((s) => s.id === fromSlideId);
35 if (srcSlideIdx < 0) return;
36 if (!state.presentation.slides[srcSlideIdx].elements.some((s) => s.id === sourceId)) return;
37 state.linkElements(pastedId, srcSlideIdx, sourceId);
38}

40export function pasteInternalClip(clip: EigendeckClip): void {
41 const state = usePresentationStore.getState();
42 
43 if (clip.kind === 'slide') {
44 // Paste the COPIED slide (clip.slide) as a new, independent slide after the
45 // current slide/group — NOT a duplicate of the current slide. Same-deck this
46 // is "copy slide 2, paste it"; cross-deck it inserts the slide (assets that
47 // aren't in the target deck come in broken — the clip is JSON, not bytes; #167).
48 const s = clip.slide as { elements?: SlideElement[] } | undefined;
49 if (s?.elements) s.elements = normalizeClipElements(s.elements); // drop unsafe els before install
50 state.pasteSlide(clip.slide);
51 return;
52 }
53 
54 // Normalize (sanitize html + validate properties) and drop any unsafe element
55 // before the loop clones/installs (audit H-1 + C-2).
56 const els = normalizeClipElements((clip.elements || []) as SlideElement[]);
57 if (!els.length) return;
58 
59 const targetSlide = state.presentation.slides[state.currentSlideIndex];
60 // Same slide if pasting back onto the slide we copied from (by id, so slide
61 // reordering doesn't fool it).
62 const sameSlide = targetSlide?.id === clip.fromSlideId;
63 
64 const newIds: string[] = [];
65 const toLink: Array<{ pastedId: string; sourceId: string }> = [];
66 for (const el of els) {
67 // Same slide → independent copy; cross-slide → join the source's sync group
68 // (if synced) else link to the source (animation).
69 const { delta, link } = pasteElementDelta(el, sameSlide);
70 const newEl = { ...JSON.parse(JSON.stringify(el)), id: crypto.randomUUID(), ...delta } as SlideElement;
71 if (sameSlide) offsetElement(newEl, 40, 40);
72 state.addElement(newEl);
73 newIds.push(newEl.id);
74 // Carry type-specific state (e.g. a notebook's recording) to the copy.
75 void runCopyHook(el, newEl);
76 if (link) toLink.push({ pastedId: newEl.id, sourceId: el.id });
77 }
78 for (const { pastedId, sourceId } of toLink) {
79 linkPastedToSource(pastedId, clip.fromSlideId, sourceId);
80 }
81 if (newIds.length === 1) state.selectObject({ type: 'element', id: newIds[0] });
82 else if (newIds.length > 1) state.selectObject({ type: 'multi', ids: newIds });
83}
