1// General "paste HTML as a picture" support.
2//
3// When rich HTML is pasted onto the canvas (a table, list, formatted block —
4// e.g. from Google Sheets/Docs, a browser, Word), we render it in an offscreen
5// DOM node styled with the deck font, then SCREENSHOT that node to a PNG via
6// modern-screenshot (the same rasterizer capturePreview uses for notebooks),
7// and insert the PNG as a normal image. The browser does the layout, so this
8// handles arbitrary HTML — far more robust than reverse-engineering one app's
9// table markup. The result is a static image: thumbnails / present / export
10// all work for free.
11//
12// Deck font as default: we strip the source font-family so the deck body font
13// applies (keeps bold/italic/color/background/borders/alignment — just unifies
14// the typeface so pastes look on-brand). Scripts / event handlers / javascript:
15// URLs are stripped before rendering.

17/** Does this pasted HTML contain structure a text box CAN'T represent — a real
18 * TABLE, an image, or preformatted/SVG/MathML content? Those go to the screenshot
19 * path. Everything else (paragraphs, <div>/<span> wrappers, bold/italic/color,
20 * headings, lists) is representable as an editable text element via
21 * sanitizeRichText's allowlist, so it must NOT screenshot. This is deliberately
22 * narrow: Word and browsers wrap even a one-line styled sentence in <p>/<div>,
23 * which should still paste as editable text, not an image (#161). */
24export function htmlNeedsScreenshot(html: string | null | undefined): boolean {
25 if (!html) return false;
26 // Structure a text box genuinely can't hold.
27 if (/<(table|thead|tbody|tfoot|tr|td|th|svg|figure|pre|math)[\s>]/i.test(html)) return true;
28 // An EMBEDDED (data:) image can't live in a text box either. A REMOTE <img> is
29 // NOT a screenshot trigger: sanitizeForCapture strips it (the capture is
30 // network-free), so a browser copy of text + a remote image pastes as editable
31 // TEXT instead of a rasterized (and formerly ~60s-hang-prone) screenshot. Use
32 // "Paste as… → Simple Image" to force a rasterization.
33 return /<img\b[^>]*\bsrc\s*=\s*["']?\s*data:/i.test(html);
34}

36/** Pull the FIRST embedded data-URL `<img>` out of pasted HTML (#158). Google
37 * Slides (and similar) put no image on the clipboard — only `text/html` with an
38 * `<img src="data:image/…;base64,…">` inside a `<b docs-internal-guid>` wrapper,
39 * so the clipboard image-item paths all miss. Returns the decoded bytes + mime,
40 * or null when there's no data-URL image (leave it to the rich-HTML fallback).
41 * Pure string/base64 work — no DOM — so it runs anywhere and is unit-testable. */
42export function extractPastedDataUrlImage(
43 html: string | null | undefined,
44): { mime: string; bytes: Uint8Array } | null {
45 if (!html) return null;
46 const m = html.match(/<img\b[^>]*\bsrc\s*=\s*["']data:(image\/[a-zA-Z0-9.+-]+);base64,([A-Za-z0-9+/=\s]+)["']/i);
47 if (!m) return null;
48 const mime = m[1];
49 const b64 = m[2].replace(/\s+/g, '');
50 try {
51 const bin = atob(b64);
52 const bytes = new Uint8Array(bin.length);
53 for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
54 return bytes.length ? { mime, bytes } : null;
55 } catch {
56 return null; // malformed base64
57 }
58}

60/** Strip scripts / handlers / js: URLs and the source font-family, returning
61 * body innerHTML safe to drop into an offscreen render node. DOM required. */
62export function sanitizeForCapture(html: string): string {
63 const doc = new DOMParser().parseFromString(html, 'text/html');
64 // Drop <style> too — it can carry @font-face / @import with remote URLs the
65 // capture would otherwise try (and hang) to fetch.
66 doc.querySelectorAll('script, link, meta, title, base, iframe, object, embed, style').forEach((n) => n.remove());
67 // Remove REMOTE (non-data:) <img>: the capture is network-free, and a remote
68 // src makes modern-screenshot stall on a connect timeout (~30s ×2). Keep data:
69 // images (they render offline).
70 doc.querySelectorAll('img').forEach((img) => {
71 if (!/^\s*data:/i.test(img.getAttribute('src') || '')) img.remove();
72 });
73 doc.querySelectorAll('*').forEach((el) => {
74 for (const attr of Array.from(el.attributes)) {
75 const name = attr.name.toLowerCase();
76 if (name.startsWith('on')) el.removeAttribute(attr.name);
77 else if ((name === 'href' || name === 'src' || name === 'xlink:href') && /^\s*javascript:/i.test(attr.value)) {
78 el.removeAttribute(attr.name);
79 }
80 }
81 const style = (el as HTMLElement).style;
82 if (style) {
83 if (style.fontFamily) style.fontFamily = ''; // deck font becomes the default
84 // Strip any inline style property referencing a remote url() (background
85 // image, mask, border-image, …) — same network-hang reason.
86 for (const prop of Array.from(style)) {
87 if (/url\(\s*["']?\s*https?:/i.test(style.getPropertyValue(prop))) style.removeProperty(prop);
88 }
89 }
90 });
91 return doc.body.innerHTML;
92}

94function dataUrlToBytes(dataUrl: string): Uint8Array {
95 const comma = dataUrl.indexOf(',');
96 const b64 = comma >= 0 ? dataUrl.slice(comma + 1) : dataUrl;
97 const bin = atob(b64);
98 const out = new Uint8Array(bin.length);
99 for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
100 return out;
101}

110/**
111 * Render sanitized HTML offscreen in the deck font and screenshot it to a PNG.
112 * Returns null if nothing renders. Requires a browser/Tauri DOM + canvas.
113 */
114export async function captureHtmlToPng(
115 html: string,
116 opts: { fontFamily: string; scale?: number; maxWidth?: number } = { fontFamily: 'sans-serif' },
117): Promise<CapturedHtml | null> {
118 if (typeof document === 'undefined') return null;
119 const container = document.createElement('div');
120 container.style.cssText =
121 `position:fixed; left:-99999px; top:0; z-index:-1; display:inline-block; ` +
122 `max-width:${opts.maxWidth ?? 1400}px; background:#ffffff; color:#1a1a1a; ` +
123 `font-family:${opts.fontFamily}; line-height:1.3;`;
124 container.innerHTML = sanitizeForCapture(html);
125 document.body.appendChild(container);
126 try {
127 // Time-box font readiness — an injected remote @font-face could otherwise
128 // stall this await.
129 if (document.fonts?.ready) {
130 try { await Promise.race([document.fonts.ready, new Promise((r) => setTimeout(r, 800))]); } catch { /* ignore */ }
131 }
132 await new Promise((r) => requestAnimationFrame(() => r(null)));
133 const rect = container.getBoundingClientRect();
134 if (rect.width < 1 || rect.height < 1) return null;
135 const { domToDataUrl } = await import('modern-screenshot');
136 // NETWORK-FREE capture: never fetch remote resources (fetchFn → false) and
137 // cap any residual media wait (timeout), so a stray remote URL can't hang the
138 // paste for ~60s (two stacked 30s modern-screenshot timeouts). The deck font
139 // is applied via the container style, so skip webfont embedding (font:false).
140 const dataUrl = await domToDataUrl(container, {
141 scale: opts.scale ?? 4,
142 backgroundColor: '#ffffff',
143 timeout: 2500,
144 // Block only REMOTE image fetches (modern-screenshot calls fetchFn just for
145 // requestType==='image'). Do NOT disable font embedding: the deck fonts are
146 // data-URL @font-face already in <head> (zero network), so they embed for
147 // free — `font:false` would drop them and the capture would fall back to a
148 // serif. sanitizeForCapture + timeout cover the hang; this covers images.
149 fetchFn: async (): Promise<string | false> => false,
150 });
151 return { bytes: dataUrlToBytes(dataUrl), width: Math.round(rect.width), height: Math.round(rect.height) };
152 } catch (e) {
153 console.warn('captureHtmlToPng failed:', e);
154 return null;
155 } finally {
156 container.remove();
157 }
