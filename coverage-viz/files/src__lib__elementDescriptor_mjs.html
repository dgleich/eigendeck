1// Per-type element DESCRIPTOR — the single place that knows what each element
2// TYPE is (its box + resolved visual values + content spec), independent of HOW
3// it's painted. Per-target ADAPTERS consume a descriptor and specialize the
4// emit form (React node / HTML string), the wrapper (editor DraggableBox /
5// bare / absolute div), and how heavy types degrade (live iframe / cached
6// preview / placeholder).
7//
8// Goal: ONE rendering path with specializations, not k parallel switches — so a
9// new element type or a render tweak happens in one place and can't drift across
10// the editor / present / thumbnail / HTML-export / print targets.
11//
12// Pure + `.mjs` (no TS, no React) so BOTH the HTML export (exportCore.mjs, shared
13// with the CLI) and the React renderers can import it. The resolved slide
14// background is passed IN, because each target obtains it differently (editor:
15// a prop; others: resolveTheme()/themeBackground()).
16//
17// Migration: types are moved onto this path incrementally (cover, image, arrow).

22/** cover — a reveal mask filled with the slide background; an explicit color wins.
23 * A themed tint (`boxTint`, #132) resolves against `theme` (a wash relative to the
24 * slide background) so a colored mask stays on-theme; when no theme is supplied
25 * (e.g. the link overlay) a tinted cover falls back to the plain slide background. */
26export function describeCover(el, resolvedSlideBg, theme) {
27 const background = el.boxTint
28 ? (textBackgroundResolved(el, theme) || resolvedSlideBg)
29 : (el.color || resolvedSlideBg);
30 return { kind: 'cover', box: el.position, background };
31}

33/**
34 * image visual styles — the optional shadow / corner-radius / opacity / rotation
35 * an image carries. The single source of the four predicates + the magic shadow
36 * string; each target maps the present values into its own form (a React style
37 * object via imageVisualStyle, or CSS-string fragments in the HTML export).
38 * Absent props are `undefined` so adapters skip them. `borderRadius` stays a raw
39 * px number; `transform` is the ready-to-use CSS `rotate(...)` value.
40 */
41export function imageVisuals(el) {
42 return {
43 shadow: el.shadow ? 'drop-shadow(4px 8px 16px rgba(0,0,0,0.3))' : undefined,
44 borderRadius: el.borderRadius || undefined,
45 opacity: (el.opacity != null && el.opacity < 1) ? el.opacity : undefined,
46 transform: el.rotation ? `rotate(${el.rotation}deg)` : undefined,
47 };
48}

50/**
51 * arrow — endpoints + resolved style + computed geometry. Owns the ONE canonical
52 * default (matching SlideEditor's creation defaults: #2563eb, strokeWidth 4,
53 * headSize 16) so a color-omitted arrow can't render differently per target —
54 * closes the latent #105 divergence where app paths defaulted red (#e53e3e) and
55 * export paths blue (#2563eb). Geometry (inset line + head triangles) is computed
56 * once here; each target renders geo/color/strokeWidth/opacity its own way (React
57 * ArrowGlyph, the SVG-string arrowSvgInner, or LinkOverlay's hit-target).
58 */
59export function describeArrow(el, theme) {
60 const { x1, y1, x2, y2, strokeWidth = 4, headSize = 16, heads, opacity,
61 c1x, c1y, c2x, c2y, points } = el;
62 // `'accent'` follows the slide theme (resolveColor); default stays #2563eb.
63 const color = resolveColor(el.color, theme, '#2563eb');
64 return {
65 kind: 'arrow', x1, y1, x2, y2, color, strokeWidth, headSize, heads, opacity,
66 c1x, c1y, c2x, c2y, points,
67 geo: arrowGeometry(x1, y1, x2, y2, headSize, heads, c1x, c1y, c2x, c2y, points),
68 };
