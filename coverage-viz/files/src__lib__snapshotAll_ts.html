1// "Generate Missing Snapshots" / "Refresh All Snapshots" (#109-adjacent).
2//
3// Live elements (demo / demo-piece / video / notebook) render to a static
4// snapshot only when their slide is mounted — so a deck you open and export
5// without visiting every slide has holes (placeholders in print/export/
6// thumbnails). This walks the deck, visiting each slide with a live element so
7// that element's OWN capture effect (SlideElementRenderer / NotebookContent —
8// which already knows the right per-type args: 'iframe'/'.nb-frame'/'video', the
9// theme salt, the background) fires and persists. We don't re-implement per-type
10// capture; we just make sure every element gets a turn on screen.
11//
12// - Generate Missing: capturePreview's own skip means only missing/stale ones
13// actually re-render; current ones are no-ops.
14// - Refresh All (force): clear each live element's cached preview first, so the
15// visit re-renders everything.

25/** Per-slide dwell: long enough for a freshly-mounted element's capture effect
26 * (a ~700–900ms debounce) to fire AND its async rasterize to grab the DOM before
27 * we flip away. Generous on purpose — a slow demo/notebook otherwise captures blank. */
28const DWELL_MS = 1600;

30export interface SnapshotProgress { current: number; total: number }

32/** One live element's capture status, resolved from the cache. */
33export interface LiveEntry { slideIdx: number; present: boolean; themeStale: boolean }
34 
35/** PURE decision: given each live element's cache status, which slides to visit
36 * and how many elements will be captured. An element is captured when `force`,
37 * or it has no current preview (missing → export placeholder), or it's theme-
38 * stale. Slides are the deduped, ascending indices of the captured elements.
39 * (Unit-tested — this is the logic that had the "runs every time" idempotency bug.) */
40export function planSnapshotCapture(entries: readonly LiveEntry[], force: boolean): { slidesToVisit: number[]; captured: number } {
41 const needy = entries.filter((e) => force || !e.present || e.themeStale);
42 const slidesToVisit = [...new Set(needy.map((e) => e.slideIdx))].sort((a, b) => a - b);
43 return { slidesToVisit, captured: needy.length };
44}
45 
46/** Resolve one live element's cache status: present (has a current preview — else
47 * the export shows a placeholder) and theme-stale. (For a force run, clearPreview
48 * already ran, so present reads false and it's captured regardless. Content-drift
49 * staleness isn't detectable without rendering; those re-cache on view / Refresh All.) */
50async function liveEntry(el: SlideElement, presentation: Presentation, slideIdx: number): Promise<LiveEntry> {
51 const key = previewKey(el);
52 const present = !!(await loadPreviewDataUrl(key));
53 const salt = previewThemeSalt(resolveTheme(presentation.theme, presentation.slides[slideIdx].theme));
54 const themeStale = present ? await isPreviewThemeStale(key, salt) : false;
55 return { slideIdx, present, themeStale };
56}

58/**
59 * (Re)capture live-element snapshots. Only the slides that actually have a
60 * missing/stale element are visited — so a second "Generate Missing" run with
61 * nothing missing is a no-op (visits 0 slides, captures 0). `force` (Refresh All)
62 * clears every live element's preview first, so all are re-rendered. Restores the
63 * original slide + selection. Best-effort. Returns { slidesVisited, captured,
64 * totalLive }.
65 */
66export async function captureAllSnapshots(
67 presentation: Presentation,
68 opts: { force?: boolean; onProgress?: (p: SnapshotProgress) => void } = {},
69): Promise<{ slidesVisited: number; captured: number; totalLive: number }> {
70 const force = !!opts.force;
71 // All live elements (for totalLive) + which slides have at least one that needs work.
72 const liveEls: SlideElement[] = [];
73 presentation.slides.forEach((s) => liveEls.push(...s.elements.filter((e) => isLive(e.type))));
74 if (!liveEls.length) return { slidesVisited: 0, captured: 0, totalLive: 0 };
75 
76 // Refresh All: clear first so every one is "missing" → gets re-rendered.
77 if (force) { for (const el of liveEls) await clearPreview(previewKey(el)); }
78 
79 // Resolve each live element's cache status, then decide (pure) what to visit.
80 const entries: LiveEntry[] = [];
81 for (let i = 0; i < presentation.slides.length; i++) {
82 for (const el of presentation.slides[i].elements) {
83 if (!isLive(el.type)) continue;
84 entries.push(await liveEntry(el, presentation, i));
85 }
86 }
87 const { slidesToVisit, captured } = planSnapshotCapture(entries, force);
88 if (!slidesToVisit.length) return { slidesVisited: 0, captured: 0, totalLive: liveEls.length };
89 
90 const store = usePresentationStore.getState();
91 const originalIdx = store.currentSlideIndex;
92 const originalSel = store.selectedObject;
93 usePresentationStore.getState().selectObject({ type: 'slide' });
94 try {
95 for (let n = 0; n < slidesToVisit.length; n++) {
96 opts.onProgress?.({ current: n + 1, total: slidesToVisit.length });
97 usePresentationStore.getState().selectSlide(slidesToVisit[n]);
98 await new Promise((r) => setTimeout(r, DWELL_MS));
99 }
100 } finally {
101 usePresentationStore.getState().selectSlide(originalIdx);
102 usePresentationStore.getState().selectObject(originalSel);
103 }
104 return { slidesVisited: slidesToVisit.length, captured, totalLive: liveEls.length };
105}
