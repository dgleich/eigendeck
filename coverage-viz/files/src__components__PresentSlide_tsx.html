32/** One presented element. Prop-driven so it renders the same in any window. */
33export function PresentElement({ element: el, zIndex, style, ctx }: {
34 element: SlideElement; zIndex: number; style?: React.CSSProperties; ctx: PresentCtx;
35}) {
36 const pos = el.position;
37 switch (el.type) {
38 case 'text':
39 return <PresentTextElement element={el} zIndex={zIndex} style={style} ctx={ctx} />;
40 case 'image':
41 return <PresentImage element={el} zIndex={zIndex} style={style} />;
42 case 'demo':
43 return <PresentDemoIframe assetId={el.assetId} pos={pos} zIndex={zIndex} style={style} ctx={ctx} />;
44 case 'demo-piece':
45 return <PresentDemoIframe assetId={el.assetId} hash={`piece=${el.piece}`} title={`demo-piece: ${el.piece}`} pos={pos} zIndex={zIndex} style={style} ctx={ctx} />;
46 case 'video':
47 return <PresentVideo element={el} zIndex={zIndex} style={style} />;
48 case 'notebook':
49 return (
50 <div className="el-notebook" style={{
51 position: 'absolute', left: pos.x, top: pos.y, width: pos.width, height: pos.height, zIndex, ...style,
52 }}>
53 <NotebookContent element={el} interactive={true} mode="present" />
54 </div>
55 );
56 case 'cover': {
57 // Reveal mask filled with the slide background (explicit color wins).
58 const t = resolveTheme(ctx.presentationTheme, ctx.slide.theme);
59 const d = describeCover(el, t.background, t);
60 return <CoverView box={d.box} background={d.background} extraStyle={{ zIndex, ...style }} />;
61 }
62 case 'arrow': {
63 const a = describeArrow(el, resolveTheme(ctx.presentationTheme, ctx.slide.theme));
64 return (
65 <svg style={{ position: 'absolute', top: 0, left: 0, width: '100%', height: '100%', pointerEvents: 'none', overflow: 'visible', zIndex, ...style }}>
66 <ArrowGlyph geo={a.geo} color={a.color} strokeWidth={a.strokeWidth} opacity={a.opacity} />
67 </svg>
68 );
69 }
70 case 'html': {
71 // Locked (no-script, no-network, no same-origin). `interactive` elements
72 // receive clicks so native controls (range/radio/details/:hover) work live;
73 // static ones stay pass-through so they never block the slide.
74 const htmlTheme = resolveTheme(ctx.presentationTheme, ctx.slide.theme);
75 // Scale mode: content laid out at its design size, contain-scaled into the box.
76 if (htmlIsScaled(el)) {
77 const L = htmlScaleLayout(pos.width, pos.height, el.scaleW!, el.scaleH!);
78 return (
79 <div style={{
80 position: 'absolute', left: pos.x, top: pos.y, width: pos.width, height: pos.height,
81 overflow: 'hidden', zIndex, ...style,
82 }}>
83 <iframe title="HTML element" srcDoc={htmlElementSrcdoc(el.html, el.background, el.vars, htmlTheme)}
84 sandbox={HTML_SANDBOX_LOCKED} style={{
85 position: 'absolute', left: 0, top: 0, width: L.designW, height: L.designH,
86 border: 'none', background: 'transparent',
87 transform: `translate(${L.offsetX}px, ${L.offsetY}px) scale(${L.scale})`,
88 transformOrigin: 'top left',
89 pointerEvents: el.interactive ? 'auto' : 'none',
90 }} />
91 </div>
92 );
93 }
94 return (
95 <iframe title="HTML element" srcDoc={htmlElementSrcdoc(el.html, el.background, el.vars, htmlTheme)}
96 sandbox={HTML_SANDBOX_LOCKED} style={{
97 position: 'absolute', left: pos.x, top: pos.y, width: pos.width, height: pos.height,
98 border: 'none', background: 'transparent',
99 pointerEvents: el.interactive ? 'auto' : 'none', zIndex, ...style,
100 }} />
101 );
102 }
103 default:
104 return null;
105 }
106}

123function PresentImage({ element: el, zIndex, style }: {
124 element: Extract<SlideElement, { type: 'image' }>; zIndex: number; style?: React.CSSProperties;
125}) {
126 const pos = el.position;
127 const src = useImageSrc(el.assetId, el.kind, {
128 displayWidth: el.position.width, displayHeight: el.position.height, snapshotVariant: el.snapshotVariant,
129 });
130 if (!src) return null;
131 const iv = imageVisualStyle(el);
132 // The transition wrapper `style` carries the fade opacity; if spread last it would
133 // clobber the element's authored opacity. Combine them (fade × element opacity) so
134 // both survive — matching editor/export where the element opacity always applies.
135 const elOpacity = typeof iv.opacity === 'number' ? iv.opacity : 1;
136 const fadeOpacity = typeof style?.opacity === 'number' ? style.opacity : 1;
137 return (
138 <img src={src} alt="" style={{
139 position: 'absolute', left: pos.x, top: pos.y, width: pos.width, height: pos.height, objectFit: 'contain', zIndex,
140 ...iv,
141 ...style,
142 opacity: elOpacity * fadeOpacity,
143 }} />
144 );
145}

147function PresentVideo({ element: el, zIndex, style }: {
148 element: Extract<SlideElement, { type: 'video' }>; zIndex: number; style?: React.CSSProperties;
149}) {
150 const pos = el.position;
151 const ref = useRef<HTMLVideoElement>(null);
152 const embedRef = useRef<HTMLIFrameElement>(null);
153 const src = useAssetUrl(el.assetId); // video FILE — a plain asset,
154 const captionsSrc = useAssetUrl(el.captionsAssetId); // not a demo (no demo-marker gate)
155 const shimBase = useYoutubeShimBase();
156 const embedSrc = el.kind === 'embed' ? liveEmbedSrc(el, shimBase) : null;
157 usePlaybackRate(ref, el.playbackRate ?? 1, src);
158 usePingPong(ref, !!el.pingPong, el.playbackRate ?? 1, src);
159 useEmbedSpeed(embedRef, el.provider, el.playbackRate ?? 1, embedSrc);
160 const box: React.CSSProperties = {
161 position: 'absolute', left: pos.x, top: pos.y, width: pos.width, height: pos.height, objectFit: 'contain', background: '#000', zIndex, ...style,
162 };
163 if (el.kind === 'embed') {
164 if (!embedSrc) return null;
165 return <iframe key={embedSrc} ref={embedRef} src={embedSrc} title="video" allow={VIDEO_EMBED_ALLOW} style={{ ...box, border: 'none' }} />;
166 }
167 if (!src) return null;
168 return (
169 <video ref={ref} src={src} playsInline loop={!!el.loop && !el.pingPong} muted={!!el.muted}
170 autoPlay={!!el.autoplay} controls={!!el.controls}
171 onClick={el.controls ? undefined : () => togglePlay(ref.current)}
172 style={el.controls ? box : { ...box, cursor: 'pointer' }}>
173 {el.captions && captionsSrc && (
174 <track kind="captions" src={captionsSrc} srcLang="en" label={el.captionsLabel || 'Captions'} default />
175 )}
176 </video>
177 );
178}

180function PresentDemoIframe({ assetId, hash, title, pos, zIndex, style, ctx }: {
181 assetId: string; hash?: string; title?: string;
182 pos: { x: number; y: number; width: number; height: number }; zIndex: number;
183 style?: React.CSSProperties; ctx?: PresentCtx;
184}) {
185 // Opaque-origin mount (docs/DEMO-PLATFORM.md): theme vars + data-URL fonts
186 // spliced at build; comm over the parent relay.
187 const fontFacesCss = useDeckFontFacesCss();
188 const varsCss = ctx?.slide
189 ? demoVarsCssForSlide(ctx.presentationConfig as any, ctx.presentationTheme || 'white', ctx.slide)
190 : '';
191 const src = useDemoDoc(assetId, {
192 hash: hash || '',
193 channelKey: assetId,
194 varsCss,
195 fontFacesCss,
196 });
197 if (!src) return null;
198 return (
199 <iframe src={src} sandbox="allow-scripts" className="el-demo-frame" title={title || 'demo'} style={{
200 position: 'absolute', left: pos.x, top: pos.y, width: pos.width, height: pos.height, border: 'none', zIndex, ...style,
201 }} />
202 );
203}
