1// The private Eigendeck clipboard "flavor".
2//
3// Per docs/copy-and-paste.md: we stop keeping a separate authoritative in-memory
4// buffer (clipboardRef) and instead carry the copied element(s)/slide as JSON on
5// the OS clipboard, base64-embedded in the text/html representation behind our
6// marker. On paste we decode it FIRST — so an internal copy round-trips with full
7// fidelity (references included) straight from the clipboard, with nothing to go
8// stale. (Figma's technique; when copy later routes through Rust NSPasteboard we
9// can promote this to a real custom UTI.)
10//
11// The codec is pure + unit-tested. Encoding/reading the actual clipboard, and
12// re-resolving links/sync on paste, live in the copy/paste handlers.

20/** The copied objects. `elements` for one-or-more elements (their JSON carries

23export interface EigendeckClip {
24 v: number;

32// UTF-8-safe base64 (non-ASCII in text content would corrupt a plain btoa/atob).
33function b64encode(s: string): string {
34 return btoa(unescape(encodeURIComponent(s)));
35}
36function b64decode(b: string): string {
37 return decodeURIComponent(escape(atob(b)));
38}

40/**
41 * Wrap `visibleHtml` (what foreign apps see + the styled fallback) with the
42 * Eigendeck marker AND the base64 clip payload. The marker attribute keeps the
43 * existing hasEigendeckMarker() detection working; the json attribute is the
44 * high-fidelity private flavor.
45 */
46export function encodeClipHtml(clip: Omit<EigendeckClip, 'v'>, visibleHtml: string): string {
47 const json = b64encode(JSON.stringify({ v: CLIP_VERSION, ...clip }));
48 return `<div ${MARKER_ATTR}="v1" ${JSON_ATTR}="${json}">${visibleHtml}</div>`;
49}

51/**
52 * Decode the Eigendeck clip from pasted HTML (from the native pasteboard or
53 * clipboardData). Returns null when the payload is absent or malformed — e.g.
54 * foreign HTML, or an Eigendeck *text-run* copy that carries the marker but no
55 * element/slide payload (that pastes as a new text box, not objects).
56 */
57export function decodeClipHtml(html: string | null | undefined): EigendeckClip | null {
58 if (!html) return null;
59 // Accept single OR double quotes: a pasteboard round-trip (macOS
60 // public.html re-serialization) can re-quote attributes, and dropping to
61 // single quotes must not silently downgrade an internal paste to the
62 // screenshot/text fallback.
63 const m = html.match(new RegExp(`${JSON_ATTR}=["']([A-Za-z0-9+/=]+)["']`));
64 if (!m) return null;
65 try {
66 const clip = JSON.parse(b64decode(m[1])) as EigendeckClip;
67 if (!clip || (clip.kind !== 'elements' && clip.kind !== 'slide')) return null;
68 if (clip.kind === 'elements' && !Array.isArray(clip.elements)) return null;
69 return clip;
70 } catch {
71 return null;
72 }
