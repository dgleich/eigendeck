1// Combinatorial test-case generation over a set of named parameters, each with a
2// list of candidate values. Used to drive the export/print style matrix at three
3// coverage levels (see exportMatrixGenerated.test.mjs):
4// • single — each value of each parameter appears at least once (tier 1)
5// • allPairs — every PAIR of values from any two parameters appears (tier 2)
6// • full — the full cartesian product (tier 3, expanded env only)
7//
8// `params` is an object `{ name: [v0, v1, …] }`. Each generator returns an array
9// of cases, where a case is `{ name: value }` for every parameter.
10 
11const names = (params) => Object.keys(params);
12 
13/** Each parameter value appears in ≥1 case; other params take their first value.
14 * Case count = max domain size. */
15export function single(params) {
16 const ns = names(params);
17 const max = Math.max(0, ...ns.map((n) => params[n].length));
18 const cases = [];
19 for (let i = 0; i < max; i++) {
20 const c = {};
21 for (const n of ns) c[n] = params[n][Math.min(i, params[n].length - 1)];
22 cases.push(c);
23 }
24 return cases;
25}

27/** All-pairs (pairwise): every (paramA=a, paramB=b) combination appears in ≥1
28 * case. Greedy set-cover over value INDICES (values may be any type). */
29export function allPairs(params) {
30 const ns = names(params);
31 if (ns.length < 2) return single(params);
32 const key = (i, a, j, b) => `${i}=${a}:${j}=${b}`; // param index = value index
33 const uncovered = new Set();
34 for (let i = 0; i < ns.length; i++) {
35 for (let j = i + 1; j < ns.length; j++) {
36 for (let a = 0; a < params[ns[i]].length; a++) {
37 for (let b = 0; b < params[ns[j]].length; b++) uncovered.add(key(i, a, j, b));
38 }
39 }
40 }
41 const cases = [];
42 while (uncovered.size) {
43 const idx = new Array(ns.length).fill(-1);
44 // Seed with any still-uncovered pair.
45 const [seed] = uncovered;
46 const m = seed.match(/^(\d+)=(\d+):(\d+)=(\d+)$/);
47 idx[+m[1]] = +m[2]; idx[+m[3]] = +m[4];
48 // Fill the rest greedily — pick the value covering the most uncovered pairs
49 // against the already-chosen params.
50 for (let i = 0; i < ns.length; i++) {
51 if (idx[i] >= 0) continue;
52 let best = 0, bestCov = -1;
53 for (let a = 0; a < params[ns[i]].length; a++) {
54 let cov = 0;
55 for (let j = 0; j < ns.length; j++) {
56 if (j === i || idx[j] < 0) continue;
57 const [lo, la, hi, hb] = i < j ? [i, a, j, idx[j]] : [j, idx[j], i, a];
58 if (uncovered.has(key(lo, la, hi, hb))) cov++;
59 }
60 if (cov > bestCov) { bestCov = cov; best = a; }
61 }
62 idx[i] = best;
63 }
64 for (let i = 0; i < ns.length; i++) {
65 for (let j = i + 1; j < ns.length; j++) uncovered.delete(key(i, idx[i], j, idx[j]));
66 }
67 cases.push(Object.fromEntries(ns.map((n, k) => [n, params[n][idx[k]]])));
68 }
69 return cases;
70}

72/** Full cartesian product. Case count = product of all domain sizes (can be huge). */
73export function full(params) {
74 const ns = names(params);
75 let cases = [{}];
76 for (const n of ns) {
77 const next = [];
78 for (const c of cases) for (const v of params[n]) next.push({ ...c, [n]: v });
79 cases = next;
80 }
81 return cases;
82}
