33export async function detectProjector(): Promise<MonitorInfo | null> {
34 try {
35 const monitors = await availableMonitors();
36 console.log(`[multi-monitor] Found ${monitors.length} monitor(s):`);
37 for (const m of monitors) {
38 console.log(` - "${m.name}" ${m.size.width}x${m.size.height} at (${m.position.x}, ${m.position.y}) scale=${m.scaleFactor}`);
39 }
40 if (monitors.length <= 1) {
41 console.log('[multi-monitor] Only one monitor, using single-window mode');
42 return null;
43 }
44 
45 const primary = await currentMonitor();
46 const primaryName = primary?.name || '';
47 console.log(`[multi-monitor] Primary monitor: "${primaryName}"`);
48 
49 // Find the non-primary monitor — prefer one with "projector" or "external" in name
50 for (const m of monitors) {
51 console.log(`[multi-monitor] Checking "${m.name}" vs primary "${primaryName}"`);
52 if (m.name !== primaryName) {
53 const nameLower = (m.name || '').toLowerCase();
54 if (nameLower.includes('projector') || nameLower.includes('external')) {
55 console.log(`[multi-monitor] Found projector: "${m.name}"`);
56 return {
57 name: m.name || 'External',
58 width: m.size.width,
59 height: m.size.height,
60 x: m.position.x,
61 y: m.position.y,
62 scaleFactor: m.scaleFactor,
63 isPrimary: false,
64 };
65 }
66 }
67 }
68 
69 // Fall back to any non-primary monitor
70 for (const m of monitors) {
71 if (m.name !== primaryName) {
72 console.log(`[multi-monitor] Using non-primary: "${m.name}"`);
73 return {
74 name: m.name || 'External',
75 width: m.size.width,
76 height: m.size.height,
77 x: m.position.x,
78 y: m.position.y,
79 scaleFactor: m.scaleFactor,
80 isPrimary: false,
81 };
82 }
83 }
84 } catch (e) {
85 console.error('Monitor detection failed:', e);
86 }
87 return null;
88}

90/**
91 * Open the presenter window on the given monitor (or detected projector).
92 * Returns true if a second window was opened, false if single-monitor fallback.
93 */
94export async function openPresenterWindow(
95 presentation: Presentation,
96 currentIndex: number,
97 projectPath: string | null,
98 opts?: { windowed?: boolean }
99): Promise<boolean> {
100 // WINDOWED MODE (screen-share presentation): open the presenter as a normal,
101 // chromeless, non-fullscreen window on the CURRENT monitor — so it can be
102 // shared as a single window over Zoom/Meet without going fullscreen and taking
103 // over the whole display. The main window keeps the speaker view. Skips mirror
104 // handling, projector detection, and the above-the-menubar fullscreen that the
105 // real dual-monitor path uses.
106 const windowed = !!opts?.windowed;
107 
108 // Check if displays are mirrored — if so, disable mirroring first
109 if (!windowed) try {
110 const mirrorInfo = await invoke<{ isMirrored: boolean; displayCount: number }>('check_display_mirroring');
111 console.log('[multi-monitor] Mirror info:', mirrorInfo);
112 
113 if (mirrorInfo.isMirrored) {
114 console.log('[multi-monitor] Displays are mirrored, disabling mirroring...');
115 const disabled = await invoke<boolean>('disable_display_mirroring');
116 if (disabled) {
117 wasMirrored = true;
118 console.log('[multi-monitor] Mirroring disabled, waiting for displays to reconfigure...');
119 // Wait for macOS to reconfigure displays
120 await new Promise((r) => setTimeout(r, 1500));
121 }
122 }
123 } catch (e) {
124 console.warn('[multi-monitor] Mirror check failed:', e);
125 }
126 
127 const projector = windowed ? null : await detectProjector();
128 
129 if (!windowed && !projector) {
130 // If we disabled mirroring but still can't find a second monitor, re-enable
131 if (wasMirrored) {
132 try { await invoke('enable_display_mirroring'); wasMirrored = false; } catch { /* ignore */ }
133 }
134 return false; // Single monitor — caller should use in-window presenter
135 }
136 
137 try {
138 // Close existing presenter window if any
139 await closePresenterWindow();
140 
141 if (windowed) {
142 // Size the window to the slide's aspect ratio so the shared window shows
143 // the slide edge-to-edge with no letterboxing. Chromeless (decorations:
144 // false) so what you share over Zoom is just the slide — no title bar.
145 const aw = presentation.config.width || 1280;
146 const ah = presentation.config.height || 720;
147 const winW = 1280;
148 const winH = Math.round(winW * (ah / aw));
149 console.log(`[multi-monitor] SCREEN-SHARE MODE — windowed chromeless presenter ${winW}x${winH} on the current monitor`);
150 presenterWindow = new WebviewWindow('presenter', {
151 url: '/presenter.html',
152 // Title shows in the OS/Zoom "share a window" picker — keep it findable.
153 title: 'Eigendeck Presentation',
154 x: 120, y: 120, width: winW, height: winH,
155 // Not fullscreen, no chrome, not pinned on top (so Zoom's share toolbar
156 // stays reachable); focus it so it's visible to grab for sharing.
157 fullscreen: false, decorations: false, alwaysOnTop: false, focus: true,
158 });
159 } else {
160 // Create presenter window on the secondary monitor
161 // Tauri window position uses logical pixels; monitor API returns physical pixels
162 const s = projector!.scaleFactor || 1;
163 const logX = Math.round(projector!.x / s);
164 const logY = Math.round(projector!.y / s);
165 const logW = Math.round(projector!.width / s);
166 const logH = Math.round(projector!.height / s);
167 console.log(`[multi-monitor] Opening presenter on "${projector!.name}" physical=(${projector!.x}, ${projector!.y}) ${projector!.width}x${projector!.height} scale=${s} logical=(${logX}, ${logY}) ${logW}x${logH}`);
168 presenterWindow = new WebviewWindow('presenter', {
169 url: '/presenter.html',
170 title: 'Eigendeck Presenter',
171 x: logX,
172 y: logY,
173 width: logW,
174 height: logH,
175 fullscreen: false, // Position first, fullscreen after
176 decorations: false,
177 alwaysOnTop: true,
178 focus: false,
179 });
180 }
181 
182 // Wait for the presenter window to signal ready
183 const readyPromise = new Promise<void>((resolve) => {
184 listen('presenter:ready', () => resolve()).then((unlisten) => {
185 // Clean up after ready
186 setTimeout(unlisten, 5000);
187 });
188 });
189 
190 // Wait for window creation + ready signal (with timeout)
191 await Promise.race([
192 readyPromise,
193 new Promise((_, reject) => setTimeout(() => reject(new Error('Presenter window timeout')), 5000)),
194 ]);
195 
196 // Set window level above the menu bar so it covers the secondary monitor fully.
197 // This is how Keynote/PowerPoint do it — no fullscreen API, just a high window level.
198 // Skipped for the windowed screen-share presenter (it's a normal window).
199 if (!windowed) {
200 console.log('[multi-monitor] Window ready, setting window level above menu bar');
201 try {
202 await invoke('set_window_above_menubar', { label: 'presenter' });
203 } catch (e) {
204 console.warn('[multi-monitor] Could not set window level:', e);
205 }
206 }
207 
208 // Send presentation data. `windowed` tells the projector webview it's the
209 // chromeless screen-share window, so it can show a drag handle (a chromeless
210 // window has no title bar to grab).
211 await emitTo('presenter', 'presenter:init', {
212 presentation,
213 currentIndex,
214 projectPath,
215 windowed,
216 });
217 
218 return true;
219 } catch (e) {
220 console.error('Failed to open presenter window:', e);
221 await closePresenterWindow();
222 return false;
223 }
224}

252/** Does this monitor's (physical) bounds contain the given physical point? */
253function monitorContains(mon: Monitor, x: number, y: number): boolean {
254 return x >= mon.position.x && x < mon.position.x + mon.size.width &&
255 y >= mon.position.y && y < mon.position.y + mon.size.height;
256}

258/**
259 * Swap which physical display shows the live slides vs the speaker view
260 * (Keynote-style "Swap Displays"). Moves the projector (live) window onto the
261 * monitor the main/speaker window is on, and the main window onto the monitor
262 * the projector was on; re-asserts the projector's above-the-menubar level on
263 * its new display. No-op unless the dual-monitor projector window is open and
264 * there are at least two monitors.
265 *
266 * Everything is done in PHYSICAL coordinates (monitor.position/size and
267 * window.outerPosition are physical) — no logical/scale conversion, which is
268 * where cross-display moves usually go wrong. Monitors are matched by GEOMETRY,
269 * not by name (names are often empty or duplicated on real hardware).
270 */
271export async function swapPresenterDisplay(): Promise<void> {
272 if (!presenterWindow) { console.warn('[multi-monitor] swap: no presenter window open'); return; }
273 try {
274 const monitors = await availableMonitors();
275 if (monitors.length < 2) { console.warn('[multi-monitor] swap: need 2+ monitors, have', monitors.length); return; }
276 
277 const mainWin = getCurrentWindow();
278 const mainPos = await mainWin.outerPosition(); // physical, top-left of main window
279 const mainMon = monitors.find((m) => monitorContains(m, mainPos.x, mainPos.y)) || monitors[0];
280 const otherMon = monitors.find((m) => m !== mainMon);
281 if (!otherMon) { console.warn('[multi-monitor] swap: could not find a second monitor'); return; }
282 
283 console.log('[multi-monitor] swap: speaker on', mainMon.name, '-> slides here; speaker to', otherMon.name);
284 // Move the main (speaker) window onto the other display.
285 await mainWin.setPosition(new PhysicalPosition(otherMon.position.x, otherMon.position.y));
286 // Move + size the projector window onto the (now vacated) main display.
287 await presenterWindow.setPosition(new PhysicalPosition(mainMon.position.x, mainMon.position.y));
288 await presenterWindow.setSize(new PhysicalSize(mainMon.size.width, mainMon.size.height));
289 // Re-assert the above-the-menubar level so it covers the new display fully.
290 try { await invoke('set_window_above_menubar', { label: 'presenter' }); } catch { /* best effort */ }
291 } catch (e) {
292 console.error('[multi-monitor] swap displays failed:', e);
293 }
294}
