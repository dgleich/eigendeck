1// Reflect this webview window's focus onto a body class so the app chrome can
2// render "subdued and visually farther away" while the window is inactive — the
3// macOS convention (see the mac-assed checklist). Cross-platform via Tauri's
4// focus event; native AppKit already de-emphasizes native controls, this covers
5// the webview content. Runs in every window (main / settings / security) via
6// initRuntime.
7import { getCurrentWindow } from '@tauri-apps/api/window';
8 
9/** Reflect focus state onto the document body. Pure; testable in jsdom. */
10export function applyWindowFocus(focused: boolean): void {
11 document.body.classList.toggle('window-inactive', !focused);
12}

14/** Wire this window's focus changes to the body class. Tauri's onFocusChanged is
15 * authoritative; DOM focus/blur is the fallback (covers plain-browser / dev). */
16export function initWindowFocus(): void {
17 window.addEventListener('focus', () => applyWindowFocus(true));
18 window.addEventListener('blur', () => applyWindowFocus(false));
19 applyWindowFocus(typeof document !== 'undefined' ? document.hasFocus() : true);
20 try {
21 const w = getCurrentWindow();
22 void w.isFocused().then(applyWindowFocus).catch(() => {});
23 void w.onFocusChanged(({ payload }) => applyWindowFocus(payload));
24 } catch {
25 /* not running under Tauri (tests / plain browser) */
26 }
