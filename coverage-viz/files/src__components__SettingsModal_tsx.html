16export type SettingsTab = 'general' | 'security' | 'ui' | 'servers';
17type Tab = SettingsTab;
18const VALID_TABS: Tab[] = ['general', 'security', 'ui', 'servers'];

20// Deep-link support: the Settings window can open on a specific tab. A fresh
21// window carries the tab in its URL hash (#ui); an already-open window is told
22// via the `eigendeck:settings-tab` window event (settings.tsx bridges the Tauri
23// event to it). "Customize Toolbar…" (View menu) uses this to jump to `ui`.
24function tabFromHash(): Tab {
25 const h = (typeof location !== 'undefined' ? location.hash.replace(/^#/, '') : '') as Tab;
26 return VALID_TABS.includes(h) ? h : 'general';
27}

212function ServerRow({ entry, isFirst, onChange, onRemove, onMoveUp }: {
213 entry: JupyterServerEntry;
214 isFirst: boolean;
215 onChange: (patch: Partial<JupyterServerEntry>) => void;
216 onRemove: () => void;

218}) {
219 // Connection-test state. Result lives in component-local state so
220 // the user sees feedback without persisting test-only flags.
221 // Successful tests DO persist availableKernels + lastSeenAt to the
222 // entry (those are useful for matching + the topbar pill).
223 const [testing, setTesting] = useState(false);
224 const [testResult, setTestResult] = useState<{ ok: boolean; msg: string } | null>(null);
225 
226 const test = async () => {
227 setTesting(true);
228 setTestResult(null);
229 try {
230 const url = entry.baseUrl.replace(/\/$/, '');
231 const q = entry.token ? `?token=${encodeURIComponent(entry.token)}` : '';
232 const r = await fetch(`${url}/api/kernelspecs${q}`, {
233 headers: entry.token ? { Authorization: `token ${entry.token}` } : {},
234 });
235 if (!r.ok) {
236 setTestResult({ ok: false, msg: `${r.status} ${r.statusText}` });
237 } else {
238 const data = await r.json();
239 const kernels = Object.keys(data.kernelspecs ?? {});
240 onChange({ availableKernels: kernels, lastSeenAt: Date.now() });
241 setTestResult({ ok: true, msg: kernels.length
242 ? `Kernels: ${kernels.join(', ')}`
243 : 'Connected, but server reports no kernels' });
244 }
245 } catch (e) {
246 setTestResult({ ok: false, msg: e instanceof Error ? e.message : String(e) });
247 } finally {
248 setTesting(false);
249 }
250 };

336function timeAgo(ms: number): string {
337 const diff = Date.now() - ms;
338 if (diff < 60_000) return 'just now';
339 if (diff < 3_600_000) return `${Math.round(diff / 60_000)}m ago`;
340 if (diff < 86_400_000) return `${Math.round(diff / 3_600_000)}h ago`;
341 return `${Math.round(diff / 86_400_000)}d ago`;
342}
