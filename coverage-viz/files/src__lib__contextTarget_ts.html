1// The element currently targeted by an open context menu (Mac convention: a
2// right-click targets + highlights an element WITHOUT changing selection). Kept
3// in a tiny external store rather than the Zustand deck store so it's not part
4// of undo history and DraggableBox can self-subscribe without prop threading
5// through every element-type branch. Cleared when the menu closes.
6import { useSyncExternalStore } from 'react';

8let target: string | null = null;
9const listeners = new Set<() => void>();
10 
11export function setContextTarget(id: string | null): void {
12 if (target === id) return;
13 target = id;
14 listeners.forEach((l) => l());
15}
16 
17export function getContextTarget(): string | null {
18 return target;
19}
20 
21function subscribe(l: () => void): () => void {
22 listeners.add(l);
23 return () => { listeners.delete(l); };
24}
25 
26/** Re-renders the caller when the context-target id changes. */
27export function useContextTarget(): string | null {
28 return useSyncExternalStore(subscribe, getContextTarget, getContextTarget);
