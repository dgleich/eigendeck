1// Migration: extract legacy `kernel.baseUrl` + `kernel.token` from
2// notebook elements in an old deck (pre-server-registry schema) and
3// fold them into the per-machine PrefSchema.jupyterServers registry.
4//
5// Runs on deck load. Mutates the presentation in place (strips the
6// legacy fields) so the next save no longer carries them. Idempotent:
7// running on a clean deck does nothing.
8//
9// Heuristic for de-duplication: an existing registry entry "matches"
10// a legacy element when its baseUrl is byte-identical to the
11// element's. If the tokens differ, we keep the existing entry as-is
12// (the user's saved registry wins; we're not in the business of
13// overwriting their token mid-load). If no match exists, we add a
14// new entry labeled by the URL host:port.

35/** Returns true if anything was changed. Caller should `setPresentation`
36 * with the (mutated) input either way; only the return value is
37 * needed for logging. */
38export function migrateLegacyNotebookTokens(pres: Presentation): boolean {
39 let mutated = false;
40 const registry = [...getPreference('jupyterServers')];
41 let registryChanged = false;
42 
43 // Helper: strip legacy fields from a kernel object in place.
44 const cleanKernel = (k: LegacyNotebookKernel): void => {
45 const obj = k as unknown as Record<string, unknown>;
46 if ('baseUrl' in obj) { delete obj.baseUrl; mutated = true; }
47 if ('token' in obj) { delete obj.token; mutated = true; }
48 };
49 
50 // Harvest from deck-level config.notebookKernel + every element.
51 const harvest = (k: LegacyExternal | undefined) => {
52 if (!k || k.kind !== 'external') return;
53 const baseUrl = k.baseUrl?.trim();
54 if (!baseUrl) return;
55 const existing = registry.find((s) => s.baseUrl === baseUrl);
56 if (existing) {
57 // If existing entry has no token but the deck did, fill it in;
58 // otherwise the registry's token wins.
59 if (!existing.token && k.token) {
60 existing.token = k.token;
61 registryChanged = true;
62 }
63 } else {
64 const newEntry: JupyterServerEntry = {
65 label: `Migrated ${labelFromUrl(baseUrl)}`,
66 baseUrl,
67 token: k.token ?? '',
68 };
69 registry.push(newEntry);
70 registryChanged = true;
71 }
72 };
73 
74 // Deck-level default kernel.
75 const cfgKernel = (pres.config as { notebookKernel?: LegacyNotebookKernel }).notebookKernel;
76 if (cfgKernel && cfgKernel.kind === 'external') {
77 harvest(cfgKernel as LegacyExternal);
78 cleanKernel(cfgKernel);
79 }
80 
81 // Per-element overrides.
82 for (const slide of pres.slides) {
83 for (const el of slide.elements) {
84 if (el.type !== 'notebook') continue;
85 const k = (el as { kernel?: LegacyNotebookKernel }).kernel;
86 if (!k) continue;
87 if (k.kind === 'external') {
88 harvest(k as LegacyExternal);
89 cleanKernel(k);
90 }
91 }
92 }
93 
94 if (registryChanged) {
95 setPreference('jupyterServers', registry);
96 mutated = true;
97 }
98 
99 return mutated;
100}

102function labelFromUrl(url: string): string {
103 try {
104 const u = new URL(url);
105 return `${u.hostname}:${u.port || (u.protocol === 'https:' ? '443' : '80')}`;
106 } catch {
107 return url;
108 }
