1//! "Install LLM Tools" — writes a self-contained kit to a user-chosen folder
2//! so AI coding agents (Claude Code, etc.) can edit `.eigendeck` decks and
3//! author HTML demos.
4//!
5//! The kit is shipped as a bundled resource (`resources/llm-tools/`): the
6//! committed AGENTS.md / CLAUDE.md / demo-starter.html, the distilled `skills/`
7//! (the front door), and `reference/` — the 3 authoring docs build.rs copies
8//! from docs/ (LLM-EDITING.md, DEMO_AUTHORING.md, DEMO_SPEC.md). The install
9//! copies the kit tree RECURSIVELY (preserving `skills/`/`reference/` subdirs)
10//! into `<target>/eigendeck-llm-tools/` and substitutes the real absolute path
11//! to the app's `eigendeck-cli` for the `__EIGENDECK_CLI_PATH__` placeholder in
12//! AGENTS.md.

22/// Locate the bundled `llm-tools` resource directory. Mirrors
23/// `pdf.rs::resolve_pdfium_dylib`: `resource_dir()/resources/llm-tools`, with
24/// a flat fallback for tauri-dev modes that don't prefix `resources/`.
25///
26/// A candidate must contain the NESTED structure (a `skills/` subdir), not merely
27/// exist. Tauri's resource copy can FLATTEN a `**/*` glob (and never cleans the
28/// destination, so stale files linger) — a flattened/stale kit has all files at the
29/// top level and no `skills/`. Requiring `skills/` rejects such a kit so we fall back
30/// to the nested source (dev) or fail loudly rather than install a broken, flattened
31/// kit (#141).
32fn resolve_kit_dir(app: &tauri::AppHandle) -> Result<PathBuf, String> {
33 let resource_dir = app
34 .path()
35 .resource_dir()
36 .map_err(|e| format!("resource_dir() failed: {}", e))?;
37 
38 let is_valid_kit = |p: &Path| p.join("skills").is_dir();
39 
40 let candidate = resource_dir.join("resources").join("llm-tools");
41 if is_valid_kit(&candidate) {
42 return Ok(candidate);
43 }
44 let flat = resource_dir.join("llm-tools");
45 if is_valid_kit(&flat) {
46 return Ok(flat);
47 }
48 // Dev fallback: the source tree (only present in `cargo tauri dev`).
49 let dev = PathBuf::from(env!("CARGO_MANIFEST_DIR"))
50 .join("resources")
51 .join("llm-tools");
52 if is_valid_kit(&dev) {
53 return Ok(dev);
54 }
55 
56 Err(format!(
57 "LLM tools kit not found (no valid nested kit with a skills/ subdir). \
58 Checked: {}, {}, and {}. If a kit exists there but is flattened/stale, \
59 rebuild after clearing the stale resource dir. The kit ships as a bundled \
60 resource — installing only works in a correctly-built app.",
61 candidate.display(),
62 flat.display(),
63 dev.display(),
64 ))
65}

67/// Resolve the absolute path to the app's sibling `eigendeck-cli` binary.
68/// In a packaged macOS app both binaries live in `Contents/MacOS/`; on
69/// Windows/Linux the CLI sits next to the main executable.
70fn resolve_cli_path() -> Result<PathBuf, String> {
71 let exe = std::env::current_exe().map_err(|e| format!("current_exe() failed: {}", e))?;
72 let dir = exe
73 .parent()
74 .ok_or_else(|| "current_exe has no parent directory".to_string())?;
75 let name = if cfg!(target_os = "windows") {
76 "eigendeck-cli.exe"
77 } else {
78 "eigendeck-cli"
79 };
80 Ok(dir.join(name))
81}

83/// Install the LLM tools kit into `<target_dir>/eigendeck-llm-tools/`.
84/// Returns the created directory path so the frontend can reveal it.
85#[tauri::command]
86pub fn install_llm_tools(
87 window: tauri::WebviewWindow,
88 app: tauri::AppHandle,
89 target_dir: String,
90) -> Result<String, String> {
91 crate::fscmds::require_main(&window)?; // audit C-3: writes a tool kit into a caller-chosen directory
92 let target = Path::new(&target_dir);
93 if !target.is_dir() {
94 return Err(format!("Target folder does not exist: {}", target_dir));
95 }
96 
97 let kit_dir = resolve_kit_dir(&app)?;
98 let cli_path = resolve_cli_path()?;
99 let cli_str = cli_path.to_string_lossy().to_string();
100 
101 let out_dir = target.join("eigendeck-llm-tools");
102 // Refuse to overwrite: if the output already exists at all, stop before touching
103 // anything. The user removes it (or picks another folder) deliberately.
104 if out_dir.exists() {
105 return Err(format!(
106 "{} already exists — remove it (or choose another folder) first; \
107 the installer will not overwrite an existing kit.",
108 out_dir.display()
109 ));
110 }
111 
112 // Copy the whole kit tree — the kit has subdirs (skills/, reference/), so recurse.
113 // AGENTS.md gets the CLI-path substitution; everything else is a straight copy.
114 // copy_kit_tree refuses to overwrite any individual file (catches a mid-copy race);
115 // on ANY failure we remove the partial output so a retry starts clean.
116 if let Err(e) = copy_kit_tree(&kit_dir, &out_dir, &cli_str) {
117 let _ = std::fs::remove_dir_all(&out_dir);
118 return Err(e);
119 }
120 
121 Ok(out_dir.to_string_lossy().to_string())
122}

124/// Recursively copy `src` into `dst`, preserving the directory structure. The
125/// top-level `AGENTS.md` has `__EIGENDECK_CLI_PATH__` replaced with the resolved
126/// CLI path; all other files are copied verbatim.
127fn copy_kit_tree(src: &Path, dst: &Path, cli_str: &str) -> Result<(), String> {
128 std::fs::create_dir_all(dst)
129 .map_err(|e| format!("Could not create {}: {}", dst.display(), e))?;
130 let entries = std::fs::read_dir(src)
131 .map_err(|e| format!("Could not read {}: {}", src.display(), e))?;
132 for entry in entries {
133 let entry = entry.map_err(|e| e.to_string())?;
134 let path = entry.path();
135 let name = match path.file_name().and_then(|n| n.to_str()) {
136 Some(n) => n.to_string(),
137 None => continue,
138 };
139 let dest = dst.join(&name);
140 if path.is_dir() {
141 copy_kit_tree(&path, &dest, cli_str)?;
142 } else if path.is_file() {
143 // Never overwrite: if the destination file already exists (e.g. it
144 // appeared mid-copy), stop rather than clobber it.
145 if dest.exists() {
146 return Err(format!("Refusing to overwrite existing file: {}", dest.display()));
147 }
148 if name == "AGENTS.md" {
149 let contents = std::fs::read_to_string(&path)
150 .map_err(|e| format!("Could not read {}: {}", path.display(), e))?;
151 let contents = contents.replace(CLI_PLACEHOLDER, cli_str);
152 std::fs::write(&dest, contents)
153 .map_err(|e| format!("Could not write {}: {}", dest.display(), e))?;
154 } else {
155 std::fs::copy(&path, &dest)
156 .map_err(|e| format!("Could not copy {} → {}: {}", path.display(), dest.display(), e))?;
157 }
158 }
159 }
160 Ok(())
161}

174 #[test]
175 fn copy_kit_tree_preserves_nested_structure_and_substitutes_cli() {
176 let src = tmp("src1");
177 fs::create_dir_all(src.join("skills").join("eigendeck")).unwrap();
178 fs::write(src.join("AGENTS.md"), format!("cli={}", CLI_PLACEHOLDER)).unwrap();
179 fs::write(src.join("skills").join("eigendeck").join("SKILL.md"), "skill").unwrap();
180 let dst = tmp("dst1");
181 copy_kit_tree(&src, &dst, "/abs/cli").unwrap();
182 // Nested path preserved (not flattened) + CLI placeholder substituted.
183 assert_eq!(fs::read_to_string(dst.join("AGENTS.md")).unwrap(), "cli=/abs/cli");
184 assert_eq!(fs::read_to_string(dst.join("skills").join("eigendeck").join("SKILL.md")).unwrap(), "skill");
185 fs::remove_dir_all(&src).ok();
186 fs::remove_dir_all(&dst).ok();
187 }
