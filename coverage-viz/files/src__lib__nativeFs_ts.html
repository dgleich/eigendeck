1// Native filesystem access via Rust commands — the replacement for the JS
2// `@tauri-apps/plugin-fs` plugin, so the webview can drop the `fs` capability
3// entirely (no ambient disk access for any injected script). Signatures mirror
4// the plugin's so call sites are near-drop-in.
5//
6// Reads go through the gated `resolve_and_read` (realpath + regular-file +
7// 512 MB cap): a picked/deck-relative source is canonicalized and size-capped,
8// same gate the watcher uses. Writes/stat/mkdir/readDir are the plain fscmds
9// commands (paths come from dialogs / deck resolution, i.e. user-chosen).
10 
11import { invoke } from '@tauri-apps/api/core';
12 
13interface ResolvedRead { canonicalPath: string; bytes: number[]; size: number }
14 
15/** Read a file's bytes (optionally a bounded prefix). Follows symlinks and rejects
16 * non-regular / oversized (>512 MB) files, via resolve_and_read. */
17export async function readFileNative(path: string, maxBytes?: number): Promise<Uint8Array> {
18 const r = await invoke<ResolvedRead>('resolve_and_read', { path, maxBytes: maxBytes ?? null });
19 return new Uint8Array(r.bytes);
20}
21 
22/** Read a file as UTF-8 text. */
23export async function readTextFileNative(path: string): Promise<string> {
24 return new TextDecoder().decode(await readFileNative(path));
25}

27export async function writeFileNative(path: string, data: Uint8Array): Promise<void> {

29}

31export async function writeTextFileNative(path: string, text: string, opts?: { append?: boolean }): Promise<void> {

35export interface NativeStat { mtime: Date | null; size: number; isFile: boolean; isDir: boolean }
36 
37/** stat a path. REJECTS if the path is missing (callers rely on that). `mtime` is
38 * a Date (from epoch-ms), matching the plugin's FileInfo.mtime. */
39export async function statNative(path: string): Promise<NativeStat> {
40 const r = await invoke<{ mtimeMs: number | null; size: number; isFile: boolean; isDir: boolean }>(
41 'path_stat', { path },
42 );
43 return { mtime: r.mtimeMs != null ? new Date(r.mtimeMs) : null, size: r.size, isFile: r.isFile, isDir: r.isDir };
44}
