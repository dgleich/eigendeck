1// Transcode arbitrary image bytes to PNG via the WebView's own decoder.
2//
3// The clipboard/paste path stores PNG/JPEG/SVG/PDF directly, but macOS apps
4// (Preview especially) put HEIC / TIFF on the pasteboard for a copied image —
5// formats the deck can't render everywhere (a raw HEIC/TIFF asset would only
6// display in Mac WebKit, and would break export + non-Mac viewers). WebKit CAN
7// decode HEIC and TIFF into an <img>, so we round-trip through a canvas to get a
8// universally-renderable PNG (#178).
9//
10// Returns null if the running WebView can't decode the input (e.g. HEIC on a
11// platform without the system codec) — the caller then falls through to the next
12// paste candidate rather than storing an unrenderable blob.

14/** Decode `bytes` (of MIME `sourceMime`) and re-encode as PNG. null on failure. */
15export async function transcodeImageToPng(bytes: Uint8Array, sourceMime: string): Promise<Uint8Array | null> {
16 const url = URL.createObjectURL(new Blob([bytes as BlobPart], { type: sourceMime }));
17 try {
18 const img = await loadImage(url);
19 const w = img.naturalWidth, h = img.naturalHeight;
20 if (!w || !h) return null;
21 const canvas = document.createElement('canvas');
22 canvas.width = w; canvas.height = h;
23 const ctx = canvas.getContext('2d');
24 if (!ctx) return null;
25 ctx.drawImage(img, 0, 0);
26 const blob: Blob | null = await new Promise((resolve) =>
27 canvas.toBlob((b) => resolve(b), 'image/png'));
28 if (!blob) return null;
29 return new Uint8Array(await blob.arrayBuffer());
30 } catch {
31 return null; // WebView couldn't decode this format
32 } finally {
33 URL.revokeObjectURL(url);
34 }
35}

37function loadImage(src: string): Promise<HTMLImageElement> {
38 return new Promise((resolve, reject) => {
39 const img = new Image();
40 img.onload = () => resolve(img);
41 img.onerror = () => reject(new Error('decode failed'));
42 img.src = src;
