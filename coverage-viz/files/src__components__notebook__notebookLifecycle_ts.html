12export function registerNotebookLifecycle(): void {
13 registerElementLifecycle('notebook', {
14 // Freeing a synced notebook: keep the freed instance a private copy of the
15 // group's recording (the still-synced instances keep the original).
16 onFree(el, freedId) {
17 if (el.syncId) return cloneOverlay(el.syncId, freedId);
18 },
19 
20 // Re-syncing: the element rejoins its group, adopting the group's shared
21 // recording — drop its private fork so it doesn't shadow the group's.
22 onResync(el) {
23 if (el._syncId && el._syncId !== el.id) return discardOverlay(el.id);
24 },
25 
26 // Copy/duplicate carries the recording: clone the source's overlay to the
27 // copy's key so an independent or animation-linked copy keeps its own copy
28 // of the recording. cloneOverlay no-ops when the keys match — which is the
29 // join-a-sync-group case (same key → already shares the one overlay).
30 onCopy(source, copy) {
31 return cloneOverlay(source.syncId ?? source.id, copy.syncId ?? copy.id);
32 },
33 
34 // Merging two notebooks under one group: the merged element can keep only
35 // ONE recording. Honour an explicit user choice, else keep whichever side
36 // actually has one (source wins a tie). The losing recording is discarded.
37 async onMerge({ source, target, sharedSyncId, keep }) {
38 const sourceKey = source.syncId ?? source.id;
39 const targetKey = target.syncId ?? target.id;
40 let keepKey: string | null;
41 if (keep === 'source') keepKey = sourceKey;
42 else if (keep === 'target') keepKey = targetKey;
43 else {
44 const sHas = !isOverlayEmpty(await loadOverlayFor(sourceKey));
45 const tHas = !isOverlayEmpty(await loadOverlayFor(targetKey));
46 keepKey = sHas ? sourceKey : (tHas ? targetKey : null);
47 }
48 await applyLinkOverlay({ sourceKey, targetKey, newKey: sharedSyncId }, keepKey);
49 },
50 });
51}
