14// Every element type that carries an `assetId` binding (src/types/presentation.ts):
15// image, demo, demo-piece, notebook, and file-kind video. Omitting notebook/video made
16// their linked assets count as "unused" everywhere usage is shown (e.g. the Security
17// window) and under-counted the blast radius of asset-scoped actions (Restore, collision).
18function isAssetBearing(elType: string): boolean {
19 return elType === 'image' || elType === 'demo' || elType === 'demo-piece'
20 || elType === 'notebook' || elType === 'video';
21}

23function elementBoundToAsset(
24 el: { type: string } & AssetBearingElement,
25 assetId: string,

27 if (!isAssetBearing(el.type)) return false;
28 return el.assetId === assetId;

31export interface AssetUsage {
32 /** Total number of elements bound to the asset. The actual blast
33 * radius of asset-scoped actions (e.g. Restore). */
34 elementCount: number;
35 /** Number of distinct slides containing at least one bound element. */
36 slideCount: number;
37 /** 1-based slide numbers (per getSlideNumber) of bound slides,
38 * ascending. Useful when callers want to name specific slides
39 * (e.g. the collision dialog). */
40 slideNumbers: number[];
41}

43/** Walk every current element and tally bindings to the given asset.
44 * See AssetUsage for the three returned counts. */
45export function computeAssetUsage(
46 presentation: Presentation | null | undefined,
47 assetId: string,
48): AssetUsage {
49 if (!presentation) return { elementCount: 0, slideCount: 0, slideNumbers: [] };
50 let elementCount = 0;
51 const slideNumbers: number[] = [];
52 presentation.slides.forEach((slide: Slide, idx: number) => {
53 let slideHit = false;
54 for (const el of slide.elements) {
55 if (elementBoundToAsset(el as { type: string } & AssetBearingElement, assetId)) {
56 elementCount++;
57 slideHit = true;
58 }
59 }
60 if (slideHit) slideNumbers.push(getSlideNumber(presentation.slides, idx));
61 });
62 return { elementCount, slideCount: slideNumbers.length, slideNumbers };
63}
