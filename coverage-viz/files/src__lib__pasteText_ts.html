1// Build a text element's `html` from a plain/styled-text paste (#161).
2//
3// When a paste onto the canvas has no image, isn't an Eigendeck-internal paste,
4// and isn't block-structured HTML (tables/lists/formatted blocks — those go to
5// the screenshot path), we create an editable text element from the text. This
6// is the pure decision: given the clipboard's text/html and text/plain, produce
7// the element html, or null when there's nothing usable.
8//
9// Styling policy (docs/copy-and-paste.md): keep only what the format toolbar can
10// author — bold / italic / strikethrough, foreground color, uppercase + letter-
11// spacing, alignment, lists. sanitizeRichText already DROPS font-size + font-
12// family (pasted text adopts the target preset). On top of that, for a PASTE we:
13// - strip a color applied to the WHOLE string (a source default — Word's black,
14// WebKit's baked neutral black — invisible on themed slides), so it inherits
15// the deck theme; keep colors on SUB-RANGES (intentional highlights);
16// - drop underline (Eigendeck has none), keeping line-through (strikethrough).

20/** Escape plain text for use as element html, preserving line breaks. */
21function plainToHtml(plain: string): string {
22 return plain
23 .replace(/\r\n?/g, '\n')
24 .replace(/&/g, '&amp;')
25 .replace(/</g, '&lt;')
26 .replace(/>/g, '&gt;')
27 .replace(/\n/g, '<br>');
28}

30/** Post-sanitize paste normalization: drop a whole-string color + underline.
31 * (Element-copy paste preserves color via the private flavor — this only runs
32 * in the text-branch paste, foreign or edit-mode text-run.) */
33function normalizePastedStyles(html: string): string {
34 if (typeof DOMParser === 'undefined') return html;
35 const doc = new DOMParser().parseFromString(`<body>${html}</body>`, 'text/html');
36 const rootText = doc.body.textContent || '';
37 const hasText = rootText.trim().length > 0;
38 doc.body.querySelectorAll<HTMLElement>('[style]').forEach((el) => {
39 // A color on an element that wraps the ENTIRE text is a blanket default → drop.
40 if (el.style.color && hasText && (el.textContent || '') === rootText) el.style.removeProperty('color');
41 // Underline is not authorable in Eigendeck — strip it, keep line-through.
42 const td = el.style.textDecoration || el.style.getPropertyValue('text-decoration-line');
43 if (td && /underline/i.test(td)) {
44 const kept = td.replace(/underline/ig, '').replace(/\s+/g, ' ').trim();
45 el.style.removeProperty('text-decoration');
46 el.style.removeProperty('text-decoration-line');
47 if (kept) el.style.textDecoration = kept;
48 }
49 if (!(el.getAttribute('style') || '').trim()) el.removeAttribute('style');
50 });
51 // Multi-block uniform default: when several sibling blocks (Word/Docs
52 // paragraphs) all carry the SAME color and together cover ALL the text, that's
53 // a blanket default too (no single element equalled rootText, so the pass above
54 // missed it). Drop it so the deck theme's text color applies. A non-uniform set
55 // (different colors, or colors covering only part of the text) is an
56 // intentional highlight and is kept.
57 if (hasText) {
58 const colored = Array.from(doc.body.querySelectorAll<HTMLElement>('[style]')).filter((el) => el.style.color);
59 const topColored = colored.filter((el) => !colored.some((o) => o !== el && o.contains(el)));
60 const colorVals = new Set(topColored.map((el) => el.style.color.trim().toLowerCase()));
61 const norm = (s: string) => s.replace(/\s+/g, '');
62 const covered = norm(topColored.map((el) => el.textContent || '').join(''));
63 if (colorVals.size === 1 && covered === norm(rootText)) {
64 topColored.forEach((el) => {
65 el.style.removeProperty('color');
66 if (!(el.getAttribute('style') || '').trim()) el.removeAttribute('style');
67 });
68 }
69 }
70 doc.body.querySelectorAll('font[color]').forEach((el) => {
71 if (hasText && (el.textContent || '') === rootText) el.removeAttribute('color');
72 });
73 // <u> → unwrap (no underline element in Eigendeck).
74 doc.body.querySelectorAll('u').forEach((u) => {
75 const parent = u.parentNode;
76 if (!parent) return;
77 while (u.firstChild) parent.insertBefore(u.firstChild, u);
78 parent.removeChild(u);
79 });
80 return doc.body.innerHTML;
81}

83/**
84 * Prefer text/html (sanitized to the authorable allowlist); fall back to
85 * text/plain (escaped, newlines → <br>). Returns null when both are empty —
86 * including the case where the html sanitizes down to nothing (e.g. it was only
87 * an <img>), so the caller can still use the plain text.
88 */
89export function pasteTextToElementHtml(
90 html: string | null | undefined,
91 plain: string | null | undefined,
92): string | null {
93 const h = (html || '').trim();
94 if (h) {
95 const clean = normalizePastedStyles(sanitizeRichText(h)).trim();
96 if (clean) return clean;
97 }
98 const p = plain || '';
99 if (p.trim()) return plainToHtml(p);
100 return null;
