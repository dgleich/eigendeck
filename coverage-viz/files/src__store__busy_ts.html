1// Global "busy" indicator for operations that can take more than a moment —
2// importing a large video/image/PDF (reading the file + embedding it in the
3// deck), etc. A single message at a time; the most recent wins.
4//
5// Use `withBusy(message, fn)`: the overlay only appears after a short delay so
6// quick operations don't flash a spinner, and it's always cleared when the
7// operation settles (success OR throw). Pair with <BusyOverlay/> (mounted once
8// at the app root) to render it.
9import { create } from 'zustand';
10 
11interface BusyState {
12 message: string | null;
13 setMessage: (m: string | null) => void;
14}
15 
16export const useBusyStore = create<BusyState>((set) => ({
17 message: null,
18 setMessage: (message) => set({ message }),
19}));
20 
21/**
22 * Run `fn` behind the global busy overlay. The overlay appears only if `fn`
23 * is still running after `delayMs` (default 400ms) — so sub-second operations
24 * never flash a spinner — and is always cleared when `fn` settles.
25 */
26export async function withBusy<T>(
27 message: string,
28 fn: () => Promise<T>,
29 delayMs = 400,
30): Promise<T> {
31 const timer = setTimeout(() => useBusyStore.getState().setMessage(message), delayMs);
32 try {
33 return await fn();
34 } finally {
35 clearTimeout(timer);
36 useBusyStore.getState().setMessage(null);
37 }
38}
