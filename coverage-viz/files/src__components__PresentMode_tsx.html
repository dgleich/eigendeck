436// Effective cubic control points for an arrow — its own, or the 1/3 & 2/3 points
437// on the chord when absent — so a straight↔curved transition interpolates smoothly
438// (a straight arrow's implicit controls sit on the line).
439export function effControls(s: ArrowEl) {
440 return {
441 c1x: s.c1x ?? s.x1 + (s.x2 - s.x1) / 3, c1y: s.c1y ?? s.y1 + (s.y2 - s.y1) / 3,
442 c2x: s.c2x ?? s.x1 + 2 * (s.x2 - s.x1) / 3, c2y: s.c2y ?? s.y1 + 2 * (s.y2 - s.y1) / 3,
443 };
444}

446// Interpolate interior waypoints: lerp point-for-point when the counts match
447// (the common linked-arrow case); otherwise snap to the target set (a count change
448// can't be tweened smoothly).
449function lerpPoints(a: ArrowEl['points'], b: ArrowEl['points'], e: number): ArrowEl['points'] {
450 if (a && b && a.length === b.length) {
451 return a.map((p, i) => ({ x: p.x + (b[i].x - p.x) * e, y: p.y + (b[i].y - p.y) * e }));
452 }
453 return b ?? a;
454}
