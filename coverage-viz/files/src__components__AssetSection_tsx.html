53/** Strip the `-NNNNNNNN` monotonic counter suffix that storage's
54 * `chrono_lite_now()` appends after the ISO Z. JavaScript's Date
55 * doesn't parse that and silently returns NaN; without this strip
56 * the entire version-history list renders blank. */
57function parseTimestamp(iso: string | null | undefined): Date | null {
58 if (!iso) return null;
59 const stripped = iso.replace(/-\d{8}$/, '');
60 const d = new Date(stripped);
61 return isNaN(d.getTime()) ? null : d;
62}

64function fmtTime(iso: string | null | undefined): string {

71function relativeAgo(iso: string | null | undefined): string {
72 if (!iso) return '';
73 const d = parseTimestamp(iso);
74 if (!d) return '';
75 const sec = Math.floor((Date.now() - d.getTime()) / 1000);
76 if (sec < 10) return 'just now';
77 if (sec < 60) return `${sec} seconds ago`;
78 const min = Math.floor(sec / 60);
79 if (min < 60) return `${min} minute${min === 1 ? '' : 's'} ago`;
80 const hr = Math.floor(min / 60);
81 if (hr < 24) return `${hr} hour${hr === 1 ? '' : 's'} ago`;
82 const day = Math.floor(hr / 24);
83 if (day < 30) return `${day} day${day === 1 ? '' : 's'} ago`;
84 const mo = Math.floor(day / 30);
85 if (mo < 12) return `${mo} month${mo === 1 ? '' : 's'} ago`;
86 const yr = Math.floor(day / 365);
87 return `${yr} year${yr === 1 ? '' : 's'} ago`;
88}

90/** Guess a blob MIME type for a version's preview from its stored
91 * mime_type or path extension. PDFs and HTML demos render as a
92 * placeholder; everything else gets the raw blob. */
93function previewMimeFor(mimeType: string | null | undefined, path: string | null | undefined): string | null {
94 const m = (mimeType || '').toLowerCase();
95 if (m === 'image/svg+xml' || m.startsWith('image/')) return mimeType!;
96 if (m === 'application/pdf') return null; // pdfium not wired
97 if (m === 'text/html') return null; // demo HTML — not previewable as image
98 // Fallback: sniff from extension
99 const ext = (path || '').split('.').pop()?.toLowerCase() || '';
100 const guess: Record<string, string> = {
101 svg: 'image/svg+xml', png: 'image/png', jpg: 'image/jpeg',
102 jpeg: 'image/jpeg', gif: 'image/gif', webp: 'image/webp',
103 };
104 return guess[ext] ?? null;
105}

655/**
656 * One row in the version history list. Relative-time as the primary label
657 * with the full ISO timestamp on hover via `title`. On mouseEnter, lazy-
658 * fetches that specific version's bytes (db_get_asset_version) and shows
659 * a thumbnail in a floating popover so the user can see what the asset
660 * looked like at that point in time. Blob URL is revoked on mouseLeave /
661 * unmount.
662 */
663function VersionRow({
664 version: v, isFirst, mimeType, path, onRestore,
665}: {
666 version: AssetVersion;
667 isFirst: boolean;
668 mimeType: string | null;
669 path: string;
670 onRestore: () => void;
671}) {
672 const isCurrent = v.valid_to === null;
673 const [previewUrl, setPreviewUrl] = useState<string | null>(null);
674 const [previewError, setPreviewError] = useState(false);
675 const [hovered, setHovered] = useState(false);
676 const rowRef = useRef<HTMLDivElement | null>(null);
677 const [popoverPos, setPopoverPos] = useState<{ top: number; left: number } | null>(null);
678 const previewMime = previewMimeFor(mimeType, path);
679 
680 // Lazy-load on first hover. Caches per-row so subsequent hovers
681 // re-show the same blob URL without re-fetching.
682 const handleEnter = useCallback(async () => {
683 setHovered(true);
684 if (rowRef.current) {
685 const r = rowRef.current.getBoundingClientRect();
686 // Position the popover to the LEFT of the inspector row, vertically
687 // anchored to the row's top. 8px gap + 160px popover width.
688 setPopoverPos({ top: r.top, left: r.left - 168 });
689 }
690 if (previewUrl !== null || previewError || !previewMime) return;
691 try {
692 const data = await invoke<number[]>('db_get_asset_version', {
693 assetId: v.asset_id, validFrom: v.valid_from,
694 });
695 const blob = new Blob([new Uint8Array(data)], { type: previewMime });
696 setPreviewUrl(URL.createObjectURL(blob));
697 } catch {
698 setPreviewError(true);
699 }
700 }, [previewUrl, previewError, previewMime, v.asset_id, v.valid_from]);
701 
702 const handleLeave = useCallback(() => {
703 setHovered(false);
704 }, []);
705 
706 useEffect(() => {
707 return () => {
708 if (previewUrl) URL.revokeObjectURL(previewUrl);
709 };
710 }, [previewUrl]);
711 
712 return (
713 <div ref={rowRef}
714 onMouseEnter={handleEnter} onMouseLeave={handleLeave}
715 style={{
716 padding: '6px 8px',
717 borderTop: isFirst ? 'none' : '1px solid #f0f0f0',
718 background: isCurrent ? '#eff6ff' : '#fff',
719 display: 'flex', justifyContent: 'space-between', alignItems: 'center', gap: 8,
720 }}>
721 <div style={{ display: 'flex', flexDirection: 'column', minWidth: 0, flex: 1 }}>
722 <div title={fmtTime(v.valid_from)} style={{ color: '#222' }}>
723 {relativeAgo(v.valid_from)}{isCurrent && <span style={{ color: '#3b82f6', marginLeft: 6, fontWeight: 500 }}>current</span>}
724 </div>
725 <div style={{ color: '#888', fontSize: 10 }}>{fmtBytes(v.size)}</div>
726 </div>
727 {!isCurrent && (
728 <button onClick={onRestore}
729 style={{ padding: '2px 8px', fontSize: 11, border: '1px solid #ccc', borderRadius: 3, cursor: 'pointer' }}>
730 Restore
731 </button>
732 )}
733 {hovered && popoverPos && createPortal(
734 <div style={{
735 position: 'fixed', top: popoverPos.top, left: popoverPos.left,
736 width: 160,
737 background: '#fff', border: '1px solid #ccc', borderRadius: 4,
738 boxShadow: '0 4px 14px rgba(0,0,0,0.18)',
739 padding: 6, zIndex: 10000, pointerEvents: 'none',
740 }}>
741 {!previewMime ? (
742 <div style={{ fontSize: 10, color: '#888', textAlign: 'center', padding: '20px 0' }}>
743 No preview<br />({mimeType || 'unknown type'})
744 </div>
745 ) : previewError ? (
746 <div style={{ fontSize: 10, color: '#888', textAlign: 'center', padding: '20px 0' }}>
747 Preview failed
748 </div>
749 ) : previewUrl ? (
750 <img src={previewUrl} alt="" style={{
751 maxWidth: '100%', maxHeight: 160, display: 'block', margin: '0 auto',
752 imageRendering: 'auto',
753 }} />
754 ) : (
755 <div style={{ fontSize: 10, color: '#888', textAlign: 'center', padding: '20px 0' }}>
756 Loading…
757 </div>
758 )}
759 </div>,
760 document.body,
761 )}
762 </div>
763 );
764}
